   320→   320→  async loadChallenge(id: string): Promise<void> {
   321→   321→    this.stopTutorial(false);

   323→   323→    this.character = this.repo.getCharacter(this.challenge.characterId);
   324→   324→    this.profile = this.repo.getBodyProfile(this.character.bodyProfileId);
   325→   325→    this.wall = this.repo.getWall(this.challenge.wallId);
   326→   326→    this.holdPosById = new Map(this.wall.holds.map((h) => [h.id, h.position] as const));
   327→   327→    this.bio = new BiomechanicsSystem(this.character, this.profile);
   328→   328→    this.evaluator = new ChallengeEvaluator(this.challenge);
   329→   329→
   330→   330→    const strengthFactor = lerp(this.profile.biomech.strengthFactorMin, this.profile.biomech.strengthFactorMax, this.character.virtues.strength);
   331→   331→    this.anchorMaxForce = this.character.physical.baseGripForce * strengthFactor;
   332→   332→    this.followForce = this.character.physical.baseGripForce * this.character.control.gripAssist * this.character.control.targetFollowStrength;
   333→   333→
   334→   334→    this.solver = new PbdSolver({
   335→   335→      gravity: v3(...this.profile.sim.gravity),
   336→   336→      damping: this.profile.sim.damping,
   337→   337→      iterations: this.profile.sim.iterations,
   338→   338→      floorY: 0,
   339→   339→      wallFrontZ: this.wall.transform.position[2] - this.wall.dimensions.depth / 2,
   340→   340→      tremorAmplitude: this.profile.sim.tremorAmplitude,
   341→   341→      loadEstimatorGain: this.profile.biomech.loadEstimatorGain
   342→   342→    });
   343→   343→
   344→   344→    this.scene.buildWorld(this.wall);
   345→   345→    this.hud.setChallenge(this.challenge, CHALLENGE_META[id]?.label ?? CHALLENGE_LABELS[this.challenge.type]);
   346→   346→    // Chip permanente de campeonato (panel superior izquierdo) mientras se juega una etapa.
   347→   347→    const stage = stageOf(id);
   348→   348→    this.hud.setChampionshipStatus(
   349→   349→      stage ? `Campeonato · etapa ${stage.index}/5 · ${stage.name} (${stage.heightM} m)` : null
   350→   350→    );
   351→   351→    this.buildInteraction();
   352→   352→    this.spawn();
   353→   353→  }

   355→   355→  /**
   356→   356→   * Recreacion del estado inicial (10.2): libera constraints, destruye y recrea el
   357→   357→   * ragdoll, restaura fatiga/timers/progreso y espera resetSettleSeconds.
   358→   358→   * Idempotente: reconstruye todo desde datos; N reinicios no duplican recursos.
   359→   359→   */
   360→   360→  private spawn(): void {
   361→   361→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   362→   362→    this.interaction?.cancelDrag();
   363→   363→    this.interaction?.cancelAssistAll();
   364→   364→    this.keyboardLimb = null;
   365→   365→    this.grabOffset = null;
   366→   366→    this.scene.clearHoldHighlights();
   367→   367→    for (const hold of this.wall.holds) this.scene.setHoldOccupied(hold.id, false);
   368→   368→
   369→   369→    this.scene.despawnRagdollVisuals();
   370→   370→    this.rig = spawnRagdoll(this.solver, this.profile, this.character, this.wall, this.challenge, holdById, (limb) => this.anchorMaxForceFor(limb));
   371→   371→    this.scene.spawnRagdollVisuals(this.rig);
   372→   372→
   373→   373→    this.bio.reset();
   374→   374→    this.evaluator?.reset();
   375→   375→    this.gameTime = 0;
   376→   376→    this.accumulator = 0;
   377→   377→    this.guardActive = false;
   378→   378→    this.lastSnapshot = { stability: 0, com: { x: 0, y: 0, z: 0 }, supportCount: 0 };
   379→   379→
   380→   380→    // El combo de flujo se rompe con cada respawn (intento nuevo); el conteo de
   381→   381→    // caidas persiste para la puntuacion del desafio cargado.
   382→   382→    this.flowCount = 0;
