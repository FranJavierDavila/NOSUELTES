   224→  static async create(canvas: HTMLCanvasElement): Promise<GameScene> {
   225→    // WebGL2 primero: es la ruta verificada en E2E y la universalmente soportada.
   226→    // WebGPU queda como respaldo (y en iframes cross-origin suele estar bloqueado
   227→    // por Permissions Policy, por lo que intentar primero WebGPU hacía fracasar
   228→    // el render en el preview).
   229→    let engine: AbstractEngine;
   230→    let backend = "WebGL 2";
   231→    try {
   232→      engine = new Engine(canvas, true, { stencil: true, preserveDrawingBuffer: false }, true);
   233→    } catch {
   234→      const gpu = new WebGPUEngine(canvas, { antialias: true, stencil: true });
   235→      await gpu.initAsync();
   236→      engine = gpu;
   237→      backend = "WebGPU";
   238→    }
   239→    const scene = new Scene(engine);
   240→    scene.useRightHandedSystem = false;
   241→    return new GameScene(engine, scene, backend, canvas);
   242→  }

   246→  buildWorld(wall: WallDefinition): void {
   247→    for (const hold of this.holdMeshes.values()) hold.dispose();
   248→    this.holdMeshes.clear();
   249→    this.holdBaseColor.clear();
   250→    this.holdRadius.clear();
   251→    this.wallMesh?.dispose();
   252→
   253→    const d = wall.dimensions;
   254→    const t = wall.transform.position;
   255→    const wallMesh = MeshBuilder.CreateBox("wall", { width: d.width, height: d.height, depth: d.depth }, this.scene);
   256→    wallMesh.position = new Vector3(t[0], t[1], t[2]);
   257→    wallMesh.isPickable = true;
   258→    wallMesh.material = wallMaterial(this.scene);
   259→    wallMesh.receiveShadows = true;
   260→    this.wallMesh = wallMesh;
   261→
   262→    const floor = MeshBuilder.CreateGround("floor", { width: 12, height: 12 }, this.scene);
   263→    const floorMat = new StandardMaterial("floorMat", this.scene);
   264→    floorMat.diffuseColor = Color3.FromHexString("#23262d");
   265→    floorMat.specularColor = new Color3(0.05, 0.05, 0.05);
   266→    floor.material = floorMat;
   267→    floor.isPickable = true;
   268→    floor.receiveShadows = true;
   269→
   270→    for (const hold of wall.holds) {
   271→      this.addHoldMesh(hold);
   272→    }
   273→  }

   275→  private addHoldMesh(hold: HoldDefinition): void {
   276→    const mesh = MeshBuilder.CreateSphere(`hold_${hold.id}`, { diameter: hold.radius * 2, segments: 8 }, this.scene);
   277→    mesh.position = new Vector3(hold.position[0], hold.position[1], hold.position[2]);
   278→    mesh.scaling.z = 0.62;
   279→    mesh.isPickable = true;
   280→    mesh.metadata = { holdId: hold.id };
   281→    const color = Color3.FromHexString(HOLD_COLORS[hold.type] ?? "#8a97a8");
   282→    const mat = new StandardMaterial(`holdMat_${hold.id}`, this.scene);
   283→    mat.diffuseColor = color;
   284→    mat.specularColor = new Color3(0.15, 0.15, 0.15);
   285→    mesh.material = mat;
   286→    this.holdMeshes.set(hold.id, mesh);
   287→    this.holdBaseColor.set(hold.id, color);
   288→    this.holdRadius.set(hold.id, hold.radius);
   289→  }

   293→  spawnRagdollVisuals(rig: RagdollRig): void {
   294→    this.despawnRagdollVisuals();
   295→    this.currentRig = rig;
   296→    const scene = this.scene;
   297→    const bodyMat = sharedMaterial("bodyMat", "#4b5563", scene);
   298→    const headMat = sharedMaterial("headMat", "#9aa3af", scene);
   299→    const bagMat = sharedMaterial("bagMat", BAG_COLOR, scene);
   300→    const markMat = emissiveMaterial("markMat", Color3.FromHexString(MARKER_COLOR), scene);
   301→    // Materiales POR EXTREMIDAD: el tinte de zona (verde/ambar/rojo) se aplica
   302→    // de forma independiente a cada brazo y cada pierna.
   303→    const limbMats: Record<LimbId, StandardMaterial> = {
   304→      hand_l: sharedMaterial("limbMat_hand_l", SEG_BASE, scene),
   305→      hand_r: sharedMaterial("limbMat_hand_r", SEG_BASE, scene),
   306→      foot_l: sharedMaterial("limbMat_foot_l", SEG_BASE, scene),
   307→      foot_r: sharedMaterial("limbMat_foot_r", SEG_BASE, scene)
   308→    };
   309→    const segOfLimb: Record<LimbId, Mesh[]> = { hand_l: [], hand_r: [], foot_l: [], foot_r: [] };
   310→
   311→    const P = rig.particles;
   312→    const dist = (ia: number, ib: number): number =>
   313→      Vector3.Distance(
   314→        new Vector3(P[ia].pos.x, P[ia].pos.y, P[ia].pos.z),
   315→        new Vector3(P[ib].pos.x, P[ib].pos.y, P[ib].pos.z)
   316→      );
   317→    const seg = (name: string, ia: number, ib: number, radius: number, mat: StandardMaterial, limb?: LimbId): void => {
   318→      const rest = dist(ia, ib);
   319→      const mesh = MeshBuilder.CreateCapsule(name, { radius, height: rest + radius * 2, tessellation: 10, capSubdivisions: 6 }, scene);
   320→      mesh.material = mat;
   321→      mesh.isPickable = false;
   322→      mesh.rotationQuaternion = Quaternion.Identity();
   323→      this.boneMeshes.push({ a: ia, b: ib, mesh, rest });
   324→      this.ragdollRoot.push(mesh);
   325→      if (limb) segOfLimb[limb].push(mesh);
   326→    };
   327→
   328→    seg("torso", 1, 4, 0.155, bodyMat);
   329→    seg("pelvisBar", 5, 6, 0.095, bodyMat);
   330→    seg("shoulderBar", 2, 3, 0.075, bodyMat);
   331→    seg("neck", 1, 0, 0.055, bodyMat);
   332→    seg("upperArmL", 2, 7, 0.058, limbMats.hand_l, "hand_l");
   333→    seg("forearmL", 7, 9, 0.05, limbMats.hand_l, "hand_l");
   334→    seg("upperArmR", 3, 8, 0.058, limbMats.hand_r, "hand_r");
   335→    seg("forearmR", 8, 10, 0.05, limbMats.hand_r, "hand_r");
   336→    seg("thighL", 5, 11, 0.075, limbMats.foot_l, "foot_l");
   337→    seg("shinL", 11, 13, 0.062, limbMats.foot_l, "foot_l");
   338→    seg("thighR", 6, 12, 0.075, limbMats.foot_r, "foot_r");
   339→    seg("shinR", 12, 14, 0.062, limbMats.foot_r, "foot_r");
   340→
   341→    const head = MeshBuilder.CreateSphere("head", { diameter: 0.24, segments: 12 }, scene);
   342→    head.material = headMat;
   343→    head.isPickable = false;
   344→    this.boneMeshes.push({ a: 0, b: 0, mesh: head, rest: 1 });
   345→    this.ragdollRoot.push(head);
   346→
   347→    // --- Marcas de orientacion frente/espalda ---
   348→    // El cuerpo vive en un plano paralelo a la pared: espalda = -z (hacia la
   349→    // camara), frente = +z (hacia la pared). La mochila (grande, en -z) es la
   350→    // referencia de espalda; el visor facial y los discos de rodilla/codo (+z)
   351→    // marcan el frente. La sincronizacion aplica offZ/offX/offY en el mundo.
   352→    const torsoLen = dist(1, 4);
   353→    const bag = MeshBuilder.CreateBox("backpack", { width: 0.26, height: torsoLen * 0.84, depth: 0.105 }, scene);
   354→    bag.material = bagMat;
   355→    bag.isPickable = false;
   356→    bag.rotationQuaternion = Quaternion.Identity();
   357→    this.boneMeshes.push({ a: 1, b: 4, mesh: bag, rest: torsoLen, offZ: -0.225 });
   358→    this.ragdollRoot.push(bag);
   359→    // Banda vertical clara sobre la cara externa de la mochila (refleja el turquesa
   360→    // de los marcadores frontales: mismo color = lados opuestos del mismo eje).
   361→    const bagStripe = MeshBuilder.CreateBox("backpackStripe", { width: 0.06, height: torsoLen * 0.78, depth: 0.02 }, scene);
   362→    bagStripe.material = markMat;
   363→    bagStripe.isPickable = false;
   364→    bagStripe.rotationQuaternion = Quaternion.Identity();
   365→    this.boneMeshes.push({ a: 1, b: 4, mesh: bagStripe, rest: torsoLen, offZ: -0.283 });
   366→    this.ragdollRoot.push(bagStripe);
   367→    // Tirantes sobre los hombros: refuerzan la lectura "esto es la espalda" incluso
   368→    // en silueta (se ven desde detras, a diferencia de las marcas frontales).
   369→    const strapMat = sharedMaterial("strapMat", "#33281e", scene);
   370→    for (const [idx, nm] of [[2, "strapL"], [3, "strapR"]] as const) {
   371→      const strap = MeshBuilder.CreateBox(nm, { width: 0.07, height: 0.13, depth: 0.05 }, scene);
   372→      strap.material = strapMat;
   373→      strap.isPickable = false;
   374→      strap.rotationQuaternion = Quaternion.Identity();
   375→      this.boneMeshes.push({ a: idx, b: idx, mesh: strap, rest: 1, offZ: -0.1, offY: 0.015 });
   376→      this.ragdollRoot.push(strap);
   377→    }
   378→
   379→    const eyeMat = sharedMaterial("eyeMat", "#141821", scene);
   380→    const marker = (name: string, idx: number, offZ: number, d: number, zScale: number, offX = 0, offY = 0, mat: StandardMaterial = markMat): void => {
   381→      const disc = MeshBuilder.CreateSphere(name, { diameter: d, segments: 8 }, scene);
   382→      disc.material = mat;
   383→      disc.scaling.z = zScale;
   384→      disc.isPickable = false;
   385→      // Punto anclado (a === b): la sincronizacion solo copia posicion (+offsets).
   386→      this.boneMeshes.push({ a: idx, b: idx, mesh: disc, rest: 1, offZ, offX, offY });
   387→      this.ragdollRoot.push(disc);
   388→    };
   389→    // Cara: visor amplio + dos ojos oscuros. Al girar la cabeza (ragdoll) el frente
   390→    // se reconoce al instante; de frente a la pared, la mochila cubre el papel inverso.
   391→    marker("faceVisor", 0, 0.125, 0.19, 0.32);
   392→    marker("eyeL", 0, 0.13, 0.05, 0.5, -0.04, 0.025, eyeMat);
   393→    marker("eyeR", 0, 0.13, 0.05, 0.5, 0.04, 0.025, eyeMat);
   394→    marker("kneeMarkL", 11, 0.05, 0.09, 0.3);
   395→    marker("kneeMarkR", 12, 0.05, 0.09, 0.3);

   515→  /** Copia posiciones del solver a los meshes con interpolacion entre pasos. */
   516→  syncRagdoll(solver: PbdSolver, alpha: number): void {
   517→    void this.currentRig;
   518→    const tmpA = new Vector3();
   519→    const tmpB = new Vector3();
   520→    for (const bone of this.boneMeshes) {
   521→      const pa = solver.getParticle(bone.a);
   522→      const pb = solver.getParticle(bone.b);
   523→      if (!pa || !pb) continue;
   524→      if (bone.a === bone.b) {
   525→        tmpA.set(pa.pos.x, pa.pos.y, pa.pos.z);
   526→        bone.mesh.position.copyFrom(tmpA);
   527→        // Marcas puntuales (visor, ojos, etiquetas): desplazamiento pared-camara y lateral.
   528→        if (bone.offZ) bone.mesh.position.z += bone.offZ;
   529→        if (bone.offX) bone.mesh.position.x += bone.offX;
   530→        if (bone.offY) bone.mesh.position.y += bone.offY;
   531→        continue;
   532→      }
   533→      const ax = pa.prev.x + (pa.pos.x - pa.prev.x) * alpha;
   534→      const ay = pa.prev.y + (pa.pos.y - pa.prev.y) * alpha;
   535→      const az = pa.prev.z + (pa.pos.z - pa.prev.z) * alpha;
   536→      const bx = pb.prev.x + (pb.pos.x - pb.prev.x) * alpha;
   537→      const by = pb.prev.y + (pb.pos.y - pb.prev.y) * alpha;
   538→      const bz = pb.prev.z + (pb.pos.z - pb.prev.z) * alpha;
   539→      tmpA.set(ax, ay, az);
   540→      tmpB.set(bx, by, bz);
   541→      const mesh = bone.mesh;
   542→      mesh.position.set((ax + bx) / 2 + (bone.offX ?? 0), (ay + by) / 2 + (bone.offY ?? 0), (az + bz) / 2 + (bone.offZ ?? 0));
   543→      const dir = tmpB.subtract(tmpA);
   544→      const len = Math.max(dir.length(), 1e-5);
   545→      const d = dir.scale(1 / len);
   546→      const ref = Math.abs(d.y) < 0.95 ? new Vector3(0, 1, 0) : new Vector3(1, 0, 0);
   547→      const xAxis = Vector3.Cross(ref, d).normalize();
   548→      const zAxis = Vector3.Cross(xAxis, d).normalize();
   549→      if (!mesh.rotationQuaternion) mesh.rotationQuaternion = Quaternion.Identity();
   550→      Quaternion.RotationQuaternionFromAxisToRef(xAxis, d, zAxis, mesh.rotationQuaternion);
   551→      mesh.scaling.y = Math.min(1.35, len / bone.rest);
   552→    }
   553→  }

   661→  /** Tinte de zona por extremidad (9.5 + retroalimentacion del HUD): la
   662→   * extremidad fijada se tine con la paleta del panel lateral (verde normal,
   663→   * ambar tension, rojo sobrecarga con pulso); libre, color base neutro.
   664→   * Aplica al agarre (mano/pie) y a los dos segmentos de su cadena.
   665→   * urgency (0..1): presion de fallo del temporizador de sobrecarga — acelera
   666→   * el parpadeo y sube el brillo a medida que el agarre se acerca a ceder. */
   667→  updateLimbTint(limb: LimbId, anchored: boolean, zone: LoadZone, urgency = 0): void {
   668→    const visual = this.limbVisual[limb];
   669→    if (!visual?.hand || visual.hand.isDisposed()) return;
   670→    const zoneColor = ZONE_COLORS[zone] ?? ZONE_COLORS.NORMAL;
   671→    const overloading = zone === "OVERLOADED" || zone === "ERROR_GUARD";
   672→    const u = overloading ? Math.min(1, Math.max(0, urgency)) : 0;
   673→    const period = 700 - 380 * u;
   674→    const wave = 0.5 + 0.5 * Math.sin((Date.now() % period) / period * Math.PI * 2);
   675→    const glow = anchored ? (overloading ? 0.2 + 0.4 * wave + 0.3 * u : 0.18) : 0;
   676→    const tint = anchored ? Color3.Lerp(visual.base, zoneColor, 0.8) : visual.base;
   677→    const segTint = anchored ? Color3.Lerp(visual.segBase, zoneColor, 0.62) : visual.segBase;
   678→    const handMat = visual.hand.material as StandardMaterial;
   679→    handMat.diffuseColor = tint;
   680→    handMat.emissiveColor = zoneColor.scale(glow);
   681→    for (const segMesh of visual.segments) {
   682→      if (segMesh.isDisposed()) continue;
   683→      const m = segMesh.material as StandardMaterial;
   684→      m.diffuseColor = segTint;
   685→      m.emissiveColor = zoneColor.scale(glow * 0.6);
   686→    }
   687→  }

   689→  /** Anillo sobre la presa anclada de una extremidad: hereda su zona de carga
   690→   * (verde/ambar/rojo) con pulso en sobrecarga, para saber que apoyo esta
   691→   * comprometido sin mirar el panel lateral. La urgencia (0..1) acelera el
   692→   * pulso y lo agranda: la presa "late" mas fuerte cuanto antes cedera. */
   693→  setAnchorRing(limb: LimbId, holdId: string | null, zone: LoadZone, urgency = 0): void {
   694→    const ring = this.anchorRings.get(limb);
   695→    if (!ring) return;
   696→    const holdMesh = holdId ? this.holdMeshes.get(holdId) : undefined;
   697→    if (!holdId || !holdMesh) {
   698→      ring.setEnabled(false);
   699→      return;
   700→    }
   701→    const r = this.holdRadius.get(holdId) ?? 0.14;
   702→    const overloading = zone === "OVERLOADED" || zone === "ERROR_GUARD";
   703→    const u = overloading ? Math.min(1, Math.max(0, urgency)) : 0;
   704→    const period = 700 - 380 * u;
   705→    const wave = 0.5 + 0.5 * Math.sin((Date.now() % period) / period * Math.PI * 2);
   706→    const zoneColor = ZONE_COLORS[zone] ?? ZONE_COLORS.NORMAL;
   707→    const mat = ring.material as StandardMaterial;
   708→    if (overloading) {
   709→      mat.emissiveColor = Color3.Lerp(zoneColor, Color3.FromHexString("#ffffff"), 0.4 * wave + 0.3 * u);
   710→    } else {
   711→      mat.emissiveColor = zoneColor;
   712→    }
   713→    ring.scaling.setAll(((r + 0.05) / 0.18) * (overloading ? 1 + (0.1 + 0.2 * u) * wave : 1));
   714→    // La presa es una esfera aplanada en z (0.62): el anillo flota justo delante
   715→    // de su cara frontal; mano y pie apilados en la misma presa van a cotas distintas.
   716→    const zOff = r * 0.62 + 0.03 + (limb.startsWith("foot") ? 0.035 : 0);

   719→  }

   773→  private burst(
   774→    name: string,
   775→    emitter: Vector3,
   776→    count: number,
   777→    c1: Color4,
   778→    c2: Color4,
   779→    dir1: Vector3,
   780→    dir2: Vector3,
   781→    opts: { sizeMin: number; sizeMax: number; lifeMin: number; lifeMax: number; gravity?: Vector3; emitTime?: number; scaleGravity?: number }
   782→  ): void {
   783→    const ps = new ParticleSystem(name, Math.max(32, count + 16), this.scene);
   784→    ps.particleTexture = this.spriteTex;
   785→    ps.emitter = emitter.clone();
   786→    ps.minEmitBox = new Vector3(-0.03, -0.03, -0.03);
   787→    ps.maxEmitBox = new Vector3(0.03, 0.03, 0.03);
   788→    ps.color1 = c1;
   789→    ps.color2 = c2;
   790→    ps.colorDead = new Color4(c1.r, c1.g, c1.b, 0);
   791→    ps.minSize = opts.sizeMin;
   792→    ps.maxSize = opts.sizeMax;
   793→    ps.minLifeTime = opts.lifeMin;
   794→    ps.maxLifeTime = opts.lifeMax;
   795→    ps.emitRate = count / Math.max(0.04, opts.emitTime ?? 0.1);
   796→    ps.direction1 = dir1;
   797→    ps.direction2 = dir2;
   798→    if (opts.gravity) ps.gravity = opts.gravity;
   799→    ps.blendMode = ParticleSystem.BLENDMODE_STANDARD;
   800→    ps.minAngularSpeed = -Math.PI;
   801→    ps.maxAngularSpeed = Math.PI;
   802→    ps.updateSpeed = 0.012;
   803→    ps.targetStopDuration = opts.emitTime ?? 0.1;
   804→    ps.disposeOnStop = true;
   805→    ps.start();
   806→  }

   865→  updateCamera(dt: number, pelvis: { x: number; y: number; z: number }, recenter: boolean): void {
   866→    const deadzoneHalf = 0.55;
   867→    const target = this.camera.target;
   868→    if (recenter) {
   869→      target.copyFrom(this.camTargetBase);
   870→    } else if (pelvis) {
   871→      const desiredY = Math.min(4.2, Math.max(1.0, pelvis.y));
   872→      const dy = desiredY - target.y;
   873→      if (Math.abs(dy) > deadzoneHalf) {
   874→        target.y += (dy - Math.sign(dy) * deadzoneHalf) * Math.min(1, dt * 3);
   875→      }
   876→      const desiredX = Math.min(1.6, Math.max(-1.6, pelvis.x));
   877→      const dx = desiredX - target.x;
   878→      if (Math.abs(dx) > deadzoneHalf) {
   879→        target.x += (dx - Math.sign(dx) * deadzoneHalf) * Math.min(1, dt * 3);
   880→      }
   881→    }
   882→    target.z = this.camTargetBase.z;
   883→
   884→    // La mutacion de camera.target no reorienta a una FreeCamera: hay que llamar
   885→    // a setTarget. Ademas aplicamos la sacudida sobre el punto de mira.
   886→    if (this.shakeAmp > 0.0006) {
   887→      const t = performance.now() / 1000;
   888→      const look = new Vector3(
   889→        target.x + Math.sin(t * 47) * this.shakeAmp,
   890→        target.y + Math.cos(t * 39) * this.shakeAmp * 0.7,
   891→        target.z
   892→      );
   893→      this.camera.setTarget(look);
   894→      this.shakeAmp *= Math.exp(-dt * 5.2);
   895→    } else {
   896→      this.shakeAmp = 0;
   897→      this.camera.setTarget(new Vector3(target.x, target.y, target.z));
   898→    }
   899→
   900→    if (this.ortho) {
   901→      const aspect = this.engine.getRenderWidth() / Math.max(1, this.engine.getRenderHeight());
   902→      const halfH = 3.1;
   903→      this.camera.orthoLeft = -halfH * aspect;
   904→      this.camera.orthoRight = halfH * aspect;
   905→      this.camera.orthoBottom = -halfH;
   906→      this.camera.orthoTop = halfH;
   907→    }
   908→  }

   999→  /** Radio de iman en pixeles CSS por tipo de extremidad: los pies, mas grandes
  1000→   * (son los mas dificiles de clicar: queja del usuario de agilidad). */
  1001→  private static readonly PICK_MAGNET_PX: Record<string, number> = { hand: 46, foot: 60 };
  1002→
  1003→  /** Posicion de picking/visual de una extremidad: cuando mano y pie comparten presa
  1004→   * (maxOccupants 2) coinciden en el mundo y seria imposible elegir; el pie se marca
  1005→   * 7 cm por debajo y la mano 3 cm por encima (apilado legible y seleccionable). */
  1006→  limbPickPos(limb: LimbId, solver: PbdSolver): { x: number; y: number; z: number } {
  1007→    const p = solver.positionOfLimb(limb);
  1008→    const holdId = solver.getAnchor(limb)?.holdId;
  1009→    if (!holdId) return { x: p.x, y: p.y, z: p.z };
  1010→    const shared = LIMB_IDS.some((other) => other !== limb && solver.getAnchor(other)?.holdId === holdId);
  1011→    if (!shared) return { x: p.x, y: p.y, z: p.z };
  1012→    const dy = limb.startsWith("foot") ? -0.07 : 0.03;
  1013→    return { x: p.x, y: p.y + dy, z: p.z };
  1014→  }
  1015→
  1016→  /** Seleccion de extremidad (7): 1) raycast exacto contra el mesh; 2) iman en espacio
  1017→   * de pantalla: la extremidad proyectada mas cercana al cursor dentro de su radio.
  1018→   * La pared bloquea el raycast directo (AT-03); el iman solo alcanza extremidades
  1019→   * visibles delante del plano frontal. */
  1020→  pickLimb(x: number, y: number, solver?: PbdSolver): LimbId | null {
  1021→    const pick = this.scene.pick(
  1022→      x,
  1023→      y,
  1024→      (m) => !!(m.metadata as { limbId?: LimbId } | undefined)?.limbId,
  1025→      false,
  1026→      this.camera
  1027→    );
  1028→    if (pick.hit && pick.pickedMesh) {
  1029→      const wallPick = this.scene.pick(x, y, (m) => m.name === "wall", false, this.camera);
  1030→      if (!(wallPick.hit && wallPick.distance < pick.distance - 0.15)) {
  1031→        const meta = pick.pickedMesh.metadata as { limbId?: LimbId };
  1032→        if (meta.limbId) return meta.limbId;
  1033→      }
  1034→    }
  1035→    if (!solver) return null;
  1036→    // Iman: proyecta las 4 extremidades y elige la mas cercana al cursor (px CSS).
  1037→    let best: LimbId | null = null;
  1038→    let bestDist = Infinity;
  1039→    for (const limb of LIMB_IDS) {
  1040→      const pp = this.limbPickPos(limb, solver);
  1041→      const s = this.projectToScreenCss(pp.x, pp.y, pp.z);
  1042→      const kind = limb.startsWith("hand") ? "hand" : "foot";
  1043→      const radius = GameScene.PICK_MAGNET_PX[kind];
  1044→      const d = Math.hypot(s.x - x, s.y - y);
  1045→      if (d <= radius && d < bestDist) {
  1046→        bestDist = d;
  1047→        best = limb;
  1048→      }
  1049→    }
  1050→    return best;
  1051→  }

  1054→  wallPlanePoint(x: number, y: number, zPlane: number): Vector3 | null {

  1083→  private static circlePoints(radius: number, segments: number, center?: Vector3): Vector3[] {
  1084→    const pts: Vector3[] = [];
  1085→    const c = center ?? Vector3.Zero();
  1086→    for (let i = 0; i <= segments; i++) {
  1087→      const a = (i / segments) * Math.PI * 2;
  1088→      pts.push(new Vector3(c.x + Math.cos(a) * radius, c.y + Math.sin(a) * radius, c.z));
  1089→    }
  1090→    return pts;
  1091→  }

  1094→function wallMaterial(scene: Scene): StandardMaterial {
  1095→  const mat = new StandardMaterial("wallMat", scene);
  1096→  const tex = new DynamicTexture("wallTex", { width: 512, height: 512 }, scene, false);
  1097→  const ctx = tex.getContext() as CanvasRenderingContext2D;
  1098→  ctx.fillStyle = "#5d646e";
  1099→  ctx.fillRect(0, 0, 512, 512);
  1100→  ctx.strokeStyle = "rgba(30,34,40,0.35)";
  1101→  ctx.lineWidth = 2;
  1102→  for (let i = 0; i <= 8; i++) {
  1103→    const p = (i / 8) * 512;
  1104→    ctx.beginPath();
  1105→    ctx.moveTo(p, 0);
  1106→    ctx.lineTo(p, 512);
  1107→    ctx.stroke();
  1108→    ctx.beginPath();
  1109→    ctx.moveTo(0, p);
  1110→    ctx.lineTo(512, p);
  1111→    ctx.stroke();
  1112→  }
  1113→  ctx.fillStyle = "rgba(255,255,255,0.05)";
  1114→  for (let i = 0; i < 900; i++) {
  1115→    ctx.fillRect(Math.random() * 512, Math.random() * 512, 2, 2);
  1116→  }
  1117→  tex.update();
  1118→  mat.diffuseTexture = tex;
  1119→  mat.specularColor = new Color3(0.06, 0.06, 0.06);
  1120→  return mat;
  1121→}
