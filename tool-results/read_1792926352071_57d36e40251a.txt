   204→  static async create(canvas: HTMLCanvasElement): Promise<GameScene> {
   205→    // WebGL2 primero: es la ruta verificada en E2E y la universalmente soportada.
   206→    // WebGPU queda como respaldo (y en iframes cross-origin suele estar bloqueado
   207→    // por Permissions Policy, por lo que intentar primero WebGPU hacía fracasar
   208→    // el render en el preview).
   209→    let engine: AbstractEngine;
   210→    let backend = "WebGL 2";
   211→    try {
   212→      engine = new Engine(canvas, true, { stencil: true, preserveDrawingBuffer: false }, true);
   213→    } catch {
   214→      const gpu = new WebGPUEngine(canvas, { antialias: true, stencil: true });
   215→      await gpu.initAsync();
   216→      engine = gpu;
   217→      backend = "WebGPU";
   218→    }
   219→    const scene = new Scene(engine);
   220→    scene.useRightHandedSystem = false;
   221→    return new GameScene(engine, scene, backend, canvas);
   222→  }

   226→  buildWorld(wall: WallDefinition): void {
   227→    for (const hold of this.holdMeshes.values()) hold.dispose();
   228→    this.holdMeshes.clear();
   229→    this.holdBaseColor.clear();
   230→    this.holdRadius.clear();
   231→    this.wallMesh?.dispose();
   232→
   233→    const d = wall.dimensions;
   234→    const t = wall.transform.position;
   235→    const wallMesh = MeshBuilder.CreateBox("wall", { width: d.width, height: d.height, depth: d.depth }, this.scene);
   236→    wallMesh.position = new Vector3(t[0], t[1], t[2]);
   237→    wallMesh.isPickable = true;
   238→    wallMesh.material = wallMaterial(this.scene);
   239→    wallMesh.receiveShadows = true;
   240→    this.wallMesh = wallMesh;
   241→
   242→    const floor = MeshBuilder.CreateGround("floor", { width: 12, height: 12 }, this.scene);
   243→    const floorMat = new StandardMaterial("floorMat", this.scene);
   244→    floorMat.diffuseColor = Color3.FromHexString("#23262d");
   245→    floorMat.specularColor = new Color3(0.05, 0.05, 0.05);
   246→    floor.material = floorMat;
   247→    floor.isPickable = true;
   248→    floor.receiveShadows = true;
   249→
   250→    for (const hold of wall.holds) {
   251→      this.addHoldMesh(hold);
   252→    }
   253→  }

   255→  private addHoldMesh(hold: HoldDefinition): void {
   256→    const mesh = MeshBuilder.CreateSphere(`hold_${hold.id}`, { diameter: hold.radius * 2, segments: 8 }, this.scene);
   257→    mesh.position = new Vector3(hold.position[0], hold.position[1], hold.position[2]);
   258→    mesh.scaling.z = 0.62;
   259→    mesh.isPickable = true;
   260→    mesh.metadata = { holdId: hold.id };
   261→    const color = Color3.FromHexString(HOLD_COLORS[hold.type] ?? "#8a97a8");
   262→    const mat = new StandardMaterial(`holdMat_${hold.id}`, this.scene);
   263→    mat.diffuseColor = color;
   264→    mat.specularColor = new Color3(0.15, 0.15, 0.15);
   265→    mesh.material = mat;
   266→    this.holdMeshes.set(hold.id, mesh);
   267→    this.holdBaseColor.set(hold.id, color);
   268→    this.holdRadius.set(hold.id, hold.radius);
   269→  }

   273→  spawnRagdollVisuals(rig: RagdollRig): void {
   274→    this.despawnRagdollVisuals();
   275→    this.currentRig = rig;
   276→    const scene = this.scene;
   277→    const bodyMat = sharedMaterial("bodyMat", "#4b5563", scene);
   278→    const headMat = sharedMaterial("headMat", "#9aa3af", scene);
   279→    const bagMat = sharedMaterial("bagMat", BAG_COLOR, scene);
   280→    const markMat = emissiveMaterial("markMat", Color3.FromHexString(MARKER_COLOR), scene);
   281→    // Materiales POR EXTREMIDAD: el tinte de zona (verde/ambar/rojo) se aplica
   282→    // de forma independiente a cada brazo y cada pierna.
   283→    const limbMats: Record<LimbId, StandardMaterial> = {
   284→      hand_l: sharedMaterial("limbMat_hand_l", SEG_BASE, scene),
   285→      hand_r: sharedMaterial("limbMat_hand_r", SEG_BASE, scene),
   286→      foot_l: sharedMaterial("limbMat_foot_l", SEG_BASE, scene),
   287→      foot_r: sharedMaterial("limbMat_foot_r", SEG_BASE, scene)
   288→    };
   289→    const segOfLimb: Record<LimbId, Mesh[]> = { hand_l: [], hand_r: [], foot_l: [], foot_r: [] };
   290→
   291→    const P = rig.particles;
   292→    const dist = (ia: number, ib: number): number =>
   293→      Vector3.Distance(
   294→        new Vector3(P[ia].pos.x, P[ia].pos.y, P[ia].pos.z),
   295→        new Vector3(P[ib].pos.x, P[ib].pos.y, P[ib].pos.z)
   296→      );
   297→    const seg = (name: string, ia: number, ib: number, radius: number, mat: StandardMaterial, limb?: LimbId): void => {
   298→      const rest = dist(ia, ib);
   299→      const mesh = MeshBuilder.CreateCapsule(name, { radius, height: rest + radius * 2, tessellation: 10, capSubdivisions: 6 }, scene);
   300→      mesh.material = mat;
   301→      mesh.isPickable = false;
   302→      mesh.rotationQuaternion = Quaternion.Identity();
   303→      this.boneMeshes.push({ a: ia, b: ib, mesh, rest });
   304→      this.ragdollRoot.push(mesh);
   305→      if (limb) segOfLimb[limb].push(mesh);
   306→    };
   307→
   308→    seg("torso", 1, 4, 0.155, bodyMat);
   309→    seg("pelvisBar", 5, 6, 0.095, bodyMat);
   310→    seg("shoulderBar", 2, 3, 0.075, bodyMat);
   311→    seg("neck", 1, 0, 0.055, bodyMat);
   312→    seg("upperArmL", 2, 7, 0.058, limbMats.hand_l, "hand_l");
   313→    seg("forearmL", 7, 9, 0.05, limbMats.hand_l, "hand_l");
   314→    seg("upperArmR", 3, 8, 0.058, limbMats.hand_r, "hand_r");
   315→    seg("forearmR", 8, 10, 0.05, limbMats.hand_r, "hand_r");
   316→    seg("thighL", 5, 11, 0.075, limbMats.foot_l, "foot_l");
   317→    seg("shinL", 11, 13, 0.062, limbMats.foot_l, "foot_l");
   318→    seg("thighR", 6, 12, 0.075, limbMats.foot_r, "foot_r");
   319→    seg("shinR", 12, 14, 0.062, limbMats.foot_r, "foot_r");
   320→
   321→    const head = MeshBuilder.CreateSphere("head", { diameter: 0.24, segments: 12 }, scene);
   322→    head.material = headMat;
   323→    head.isPickable = false;
   324→    this.boneMeshes.push({ a: 0, b: 0, mesh: head, rest: 1 });
   325→    this.ragdollRoot.push(head);
   326→
   327→    // --- Marcas de orientacion frente/espalda ---
   328→    // El cuerpo vive en un plano paralelo a la pared: espalda = -z (hacia la
   329→    // camara), frente = +z (hacia la pared). La mochila (grande, en -z) es la
   330→    // referencia de espalda; el visor facial y los discos de rodilla/codo (+z)
   331→    // marcan el frente. La sincronizacion aplica offZ en z del mundo.
   332→    const torsoLen = dist(1, 4);
   333→    const bag = MeshBuilder.CreateBox("backpack", { width: 0.2, height: torsoLen * 0.78, depth: 0.09 }, scene);
   334→    bag.material = bagMat;
   335→    bag.isPickable = false;
   336→    bag.rotationQuaternion = Quaternion.Identity();
   337→    this.boneMeshes.push({ a: 1, b: 4, mesh: bag, rest: torsoLen, offZ: -0.21 });
   338→    this.ragdollRoot.push(bag);
   339→    // Banda vertical clara sobre la cara externa de la mochila.
   340→    const bagStripe = MeshBuilder.CreateBox("backpackStripe", { width: 0.045, height: torsoLen * 0.74, depth: 0.02 }, scene);
   341→    bagStripe.material = markMat;
   342→    bagStripe.isPickable = false;
   343→    bagStripe.rotationQuaternion = Quaternion.Identity();
   344→    this.boneMeshes.push({ a: 1, b: 4, mesh: bagStripe, rest: torsoLen, offZ: -0.262 });
   345→    this.ragdollRoot.push(bagStripe);
   346→
   347→    const marker = (name: string, idx: number, offZ: number, d: number, zScale: number): void => {
   348→      const disc = MeshBuilder.CreateSphere(name, { diameter: d, segments: 8 }, scene);
   349→      disc.material = markMat;
   350→      disc.scaling.z = zScale;
   351→      disc.isPickable = false;
   352→      // Punto anclado (a === b): la sincronizacion solo copia posicion (+offZ).
   353→      this.boneMeshes.push({ a: idx, b: idx, mesh: disc, rest: 1, offZ });
   354→      this.ragdollRoot.push(disc);
   355→    };
   356→    marker("faceVisor", 0, 0.115, 0.16, 0.35);
   357→    marker("kneeMarkL", 11, 0.05, 0.09, 0.3);
   358→    marker("kneeMarkR", 12, 0.05, 0.09, 0.3);
   359→    marker("elbowMarkL", 7, 0.055, 0.075, 0.35);
   360→    marker("elbowMarkR", 8, 0.055, 0.075, 0.35);
   361→
   362→    const grips: [LimbId, number, string, string][] = [
   363→      ["hand_l", 9, "handL", HAND_BASE],
   364→      ["hand_r", 10, "handR", HAND_BASE],
   365→      ["foot_l", 13, "footL", FOOT_BASE],
   366→      ["foot_r", 14, "footR", FOOT_BASE]
   367→    ];
   368→    for (const [limb, idx, name, color] of grips) {
   369→      const mesh = MeshBuilder.CreateSphere(name, { diameter: 0.15, segments: 8 }, scene);
   370→      mesh.material = sharedMaterial(`${name}Mat`, color, scene);
   371→      mesh.isPickable = true;
   372→      mesh.metadata = { limbId: limb };
   373→      mesh.rotationQuaternion = Quaternion.Identity();
   374→      // Sincronizada con el solver como articulacion de un solo punto (a === b).
   375→      this.boneMeshes.push({ a: idx, b: idx, mesh, rest: 1 });
   376→      this.limbVisual[limb] = {
   377→        hand: mesh,
   378→        segments: segOfLimb[limb],
   379→        base: Color3.FromHexString(color),
   380→        segBase: Color3.FromHexString(SEG_BASE)
   381→      };
   382→      this.ragdollRoot.push(mesh);
   383→    }
   384→
   385→    // Esferas de depuracion por particula.
   386→    this.jointSpheres = [];
   387→    for (let i = 0; i < P.length; i++) {
   388→      const s = MeshBuilder.CreateSphere(`joint_${i}`, { diameter: 0.07, segments: 6 }, scene);
   389→      s.material = emissiveMaterial(`jointMat_${i}`, Color3.FromHexString("#e05252"), scene);
   390→      s.isPickable = false;
   391→      s.setEnabled(false);
   392→      this.jointSpheres.push(s);
   393→    }
   394→  }
   395→
   396→  despawnRagdollVisuals(): void {
   397→    for (const b of this.boneMeshes) b.mesh.dispose();
   398→    for (const m of this.jointSpheres) m.dispose();
   399→    this.boneMeshes = [];
   400→    this.jointSpheres = [];
   401→    this.ragdollRoot = [];
   402→    this.currentRig = null;
   403→  }

   405→  /** Copia posiciones del solver a los meshes con interpolacion entre pasos. */
   406→  syncRagdoll(solver: PbdSolver, alpha: number): void {
   407→    void this.currentRig;
   408→    const tmpA = new Vector3();
   409→    const tmpB = new Vector3();
   410→    for (const bone of this.boneMeshes) {
   411→      const pa = solver.getParticle(bone.a);
   412→      const pb = solver.getParticle(bone.b);
   413→      if (!pa || !pb) continue;
   414→      if (bone.a === bone.b) {
   415→        tmpA.set(pa.pos.x, pa.pos.y, pa.pos.z);
   416→        bone.mesh.position.copyFrom(tmpA);
   417→        // Marcas puntuales (visor, discos): desplazamiento en el eje pared-camara.
   418→        if (bone.offZ) bone.mesh.position.z += bone.offZ;
   419→        continue;
   420→      }
   421→      const ax = pa.prev.x + (pa.pos.x - pa.prev.x) * alpha;
   422→      const ay = pa.prev.y + (pa.pos.y - pa.prev.y) * alpha;
   423→      const az = pa.prev.z + (pa.pos.z - pa.prev.z) * alpha;
   424→      const bx = pb.prev.x + (pb.pos.x - pb.prev.x) * alpha;
   425→      const by = pb.prev.y + (pb.pos.y - pb.prev.y) * alpha;
   426→      const bz = pb.prev.z + (pb.pos.z - pb.prev.z) * alpha;
   427→      tmpA.set(ax, ay, az);
   428→      tmpB.set(bx, by, bz);
   429→      const mesh = bone.mesh;
   430→      mesh.position.set((ax + bx) / 2, (ay + by) / 2, (az + bz) / 2 + (bone.offZ ?? 0));
   431→      const dir = tmpB.subtract(tmpA);
   432→      const len = Math.max(dir.length(), 1e-5);
   433→      const d = dir.scale(1 / len);
   434→      const ref = Math.abs(d.y) < 0.95 ? new Vector3(0, 1, 0) : new Vector3(1, 0, 0);
   435→      const xAxis = Vector3.Cross(ref, d).normalize();
   436→      const zAxis = Vector3.Cross(xAxis, d).normalize();
   437→      if (!mesh.rotationQuaternion) mesh.rotationQuaternion = Quaternion.Identity();
   438→      Quaternion.RotationQuaternionFromAxisToRef(xAxis, d, zAxis, mesh.rotationQuaternion);
   439→      mesh.scaling.y = Math.min(1.35, len / bone.rest);
   440→    }
   441→  }

   549→  /** Tinte de zona por extremidad (9.5 + retroalimentacion del HUD): la
   550→   * extremidad fijada se tine con la paleta del panel lateral (verde normal,
   551→   * ambar tension, rojo sobrecarga con pulso); libre, color base neutro.
   552→   * Aplica al agarre (mano/pie) y a los dos segmentos de su cadena.
   553→   * urgency (0..1): presion de fallo del temporizador de sobrecarga — acelera
   554→   * el parpadeo y sube el brillo a medida que el agarre se acerca a ceder. */
   555→  updateLimbTint(limb: LimbId, anchored: boolean, zone: LoadZone, urgency = 0): void {
   556→    const visual = this.limbVisual[limb];
   557→    if (!visual?.hand || visual.hand.isDisposed()) return;
   558→    const zoneColor = ZONE_COLORS[zone] ?? ZONE_COLORS.NORMAL;
   559→    const overloading = zone === "OVERLOADED" || zone === "ERROR_GUARD";
   560→    const u = overloading ? Math.min(1, Math.max(0, urgency)) : 0;
   561→    const period = 700 - 380 * u;
   562→    const wave = 0.5 + 0.5 * Math.sin((Date.now() % period) / period * Math.PI * 2);
   563→    const glow = anchored ? (overloading ? 0.2 + 0.4 * wave + 0.3 * u : 0.18) : 0;
   564→    const tint = anchored ? Color3.Lerp(visual.base, zoneColor, 0.8) : visual.base;
   565→    const segTint = anchored ? Color3.Lerp(visual.segBase, zoneColor, 0.62) : visual.segBase;
   566→    const handMat = visual.hand.material as StandardMaterial;
   567→    handMat.diffuseColor = tint;
   568→    handMat.emissiveColor = zoneColor.scale(glow);
   569→    for (const segMesh of visual.segments) {
   570→      if (segMesh.isDisposed()) continue;
   571→      const m = segMesh.material as StandardMaterial;
   572→      m.diffuseColor = segTint;
   573→      m.emissiveColor = zoneColor.scale(glow * 0.6);
   574→    }
   575→  }

   577→  /** Anillo sobre la presa anclada de una extremidad: hereda su zona de carga
   578→   * (verde/ambar/rojo) con pulso en sobrecarga, para saber que apoyo esta
   579→   * comprometido sin mirar el panel lateral. La urgencia (0..1) acelera el
   580→   * pulso y lo agranda: la presa "late" mas fuerte cuanto antes cedera. */
   581→  setAnchorRing(limb: LimbId, holdId: string | null, zone: LoadZone, urgency = 0): void {
   582→    const ring = this.anchorRings.get(limb);
   583→    if (!ring) return;
   584→    const holdMesh = holdId ? this.holdMeshes.get(holdId) : undefined;
   585→    if (!holdId || !holdMesh) {
   586→      ring.setEnabled(false);
   587→      return;
   588→    }
   589→    const r = this.holdRadius.get(holdId) ?? 0.14;
   590→    const overloading = zone === "OVERLOADED" || zone === "ERROR_GUARD";
   591→    const u = overloading ? Math.min(1, Math.max(0, urgency)) : 0;
   592→    const period = 700 - 380 * u;
   593→    const wave = 0.5 + 0.5 * Math.sin((Date.now() % period) / period * Math.PI * 2);
   594→    const zoneColor = ZONE_COLORS[zone] ?? ZONE_COLORS.NORMAL;
   595→    const mat = ring.material as StandardMaterial;
   596→    if (overloading) {
   597→      mat.emissiveColor = Color3.Lerp(zoneColor, Color3.FromHexString("#ffffff"), 0.4 * wave + 0.3 * u);
   598→    } else {
   599→      mat.emissiveColor = zoneColor;
   600→    }
   601→    ring.scaling.setAll(((r + 0.05) / 0.18) * (overloading ? 1 + (0.1 + 0.2 * u) * wave : 1));
   602→    // La presa es una esfera aplanada en z (0.62): el anillo flota justo delante
   603→    // de su cara frontal; mano y pie apilados en la misma presa van a cotas distintas.
   604→    const zOff = r * 0.62 + 0.03 + (limb.startsWith("foot") ? 0.035 : 0);
   605→    ring.setEnabled(true);
   606→    ring.position.set(holdMesh.position.x, holdMesh.position.y, holdMesh.position.z - zOff);
   607→  }

   666→    c2: Color4,
   667→    dir1: Vector3,

   682→    ps.maxLifeTime = opts.lifeMax;

   753→  updateCamera(dt: number, pelvis: { x: number; y: number; z: number }, recenter: boolean): void {
   754→    const deadzoneHalf = 0.55;
   755→    const target = this.camera.target;
   756→    if (recenter) {
   757→      target.copyFrom(this.camTargetBase);
   758→    } else if (pelvis) {
   759→      const desiredY = Math.min(4.2, Math.max(1.0, pelvis.y));
   760→      const dy = desiredY - target.y;
   761→      if (Math.abs(dy) > deadzoneHalf) {
   762→        target.y += (dy - Math.sign(dy) * deadzoneHalf) * Math.min(1, dt * 3);
   763→      }
   764→      const desiredX = Math.min(1.6, Math.max(-1.6, pelvis.x));
   765→      const dx = desiredX - target.x;
   766→      if (Math.abs(dx) > deadzoneHalf) {
   767→        target.x += (dx - Math.sign(dx) * deadzoneHalf) * Math.min(1, dt * 3);
   768→      }
   769→    }
   770→    target.z = this.camTargetBase.z;
   771→
   772→    // La mutacion de camera.target no reorienta a una FreeCamera: hay que llamar
   773→    // a setTarget. Ademas aplicamos la sacudida sobre el punto de mira.
   774→    if (this.shakeAmp > 0.0006) {
   775→      const t = performance.now() / 1000;
   776→      const look = new Vector3(
   777→        target.x + Math.sin(t * 47) * this.shakeAmp,
   778→        target.y + Math.cos(t * 39) * this.shakeAmp * 0.7,
   779→        target.z
   780→      );
   781→      this.camera.setTarget(look);
   782→      this.shakeAmp *= Math.exp(-dt * 5.2);
   783→    } else {
   784→      this.shakeAmp = 0;
   785→      this.camera.setTarget(new Vector3(target.x, target.y, target.z));
   786→    }
   787→
   788→    if (this.ortho) {
   789→      const aspect = this.engine.getRenderWidth() / Math.max(1, this.engine.getRenderHeight());
   790→      const halfH = 3.1;
   791→      this.camera.orthoLeft = -halfH * aspect;
   792→      this.camera.orthoRight = halfH * aspect;
   793→      this.camera.orthoBottom = -halfH;
   794→      this.camera.orthoTop = halfH;
   795→    }
   796→  }

   861→  /** Proyeccion de un punto del mundo a coordenadas de pantalla (diagnostico). */
   862→  projectToScreen(x: number, y: number, z: number): { x: number; y: number } {
   863→    const engine = this.scene.getEngine();
   864→    const p = Vector3.Project(
   865→      new Vector3(x, y, z),
   866→      Matrix.Identity(),
   867→      this.scene.getTransformMatrix(),
   868→      this.scene.activeCamera!.viewport.toGlobal(engine.getRenderWidth(), engine.getRenderHeight())
   869→    );
   870→    return { x: Math.round(p.x), y: Math.round(p.y) };
   871→  }

   873→  /**
   874→   * Proyeccion a pixeles CSS relativos al canvas, lista para overlays DOM/SVG
   875→   * (el motor renderiza a devicePixelRatio con adaptToDeviceRatio=true).
   876→   */
   877→  projectToScreenCss(x: number, y: number, z: number): { x: number; y: number } {
   878→    const p = this.projectToScreen(x, y, z);
   879→    const canvas = this.engine.getRenderingCanvas();
   880→    const renderW = this.engine.getRenderWidth();
   881→    const scale = canvas && renderW > 0 ? canvas.clientWidth / renderW : 1;
   882→    return { x: p.x * scale, y: p.y * scale };
   883→  }

   885→  // --- Picking ---

   887→  /** Radio de iman en pixeles CSS por tipo de extremidad: los pies, mas grandes
   888→   * (son los mas dificiles de clicar: queja del usuario de agilidad). */
   889→  private static readonly PICK_MAGNET_PX: Record<string, number> = { hand: 46, foot: 60 };
   890→
   891→  /** Posicion de picking/visual de una extremidad: cuando mano y pie comparten presa
   892→   * (maxOccupants 2) coinciden en el mundo y seria imposible elegir; el pie se marca
   893→   * 7 cm por debajo y la mano 3 cm por encima (apilado legible y seleccionable). */
   894→  limbPickPos(limb: LimbId, solver: PbdSolver): { x: number; y: number; z: number } {
   895→    const p = solver.positionOfLimb(limb);
   896→    const holdId = solver.getAnchor(limb)?.holdId;
   897→    if (!holdId) return { x: p.x, y: p.y, z: p.z };
   898→    const shared = LIMB_IDS.some((other) => other !== limb && solver.getAnchor(other)?.holdId === holdId);
   899→    if (!shared) return { x: p.x, y: p.y, z: p.z };
   900→    const dy = limb.startsWith("foot") ? -0.07 : 0.03;
   901→    return { x: p.x, y: p.y + dy, z: p.z };
   902→  }
   903→
   904→  /** Seleccion de extremidad (7): 1) raycast exacto contra el mesh; 2) iman en espacio
   905→   * de pantalla: la extremidad proyectada mas cercana al cursor dentro de su radio.
   906→   * La pared bloquea el raycast directo (AT-03); el iman solo alcanza extremidades
   907→   * visibles delante del plano frontal. */
   908→  pickLimb(x: number, y: number, solver?: PbdSolver): LimbId | null {
   909→    const pick = this.scene.pick(
   910→      x,
   911→      y,
   912→      (m) => !!(m.metadata as { limbId?: LimbId } | undefined)?.limbId,
   913→      false,
   914→      this.camera
   915→    );
   916→    if (pick.hit && pick.pickedMesh) {
   917→      const wallPick = this.scene.pick(x, y, (m) => m.name === "wall", false, this.camera);
   918→      if (!(wallPick.hit && wallPick.distance < pick.distance - 0.15)) {
   919→        const meta = pick.pickedMesh.metadata as { limbId?: LimbId };
   920→        if (meta.limbId) return meta.limbId;
   921→      }
   922→    }
   923→    if (!solver) return null;
   924→    // Iman: proyecta las 4 extremidades y elige la mas cercana al cursor (px CSS).
   925→    let best: LimbId | null = null;
   926→    let bestDist = Infinity;
   927→    for (const limb of LIMB_IDS) {
   928→      const pp = this.limbPickPos(limb, solver);
   929→      const s = this.projectToScreenCss(pp.x, pp.y, pp.z);
   930→      const kind = limb.startsWith("hand") ? "hand" : "foot";
   931→      const radius = GameScene.PICK_MAGNET_PX[kind];
   932→      const d = Math.hypot(s.x - x, s.y - y);
   933→      if (d <= radius && d < bestDist) {
   934→        bestDist = d;
   935→        best = limb;
   936→      }
   937→    }
   938→    return best;
   939→  }

   941→  /** Proyeccion del cursor sobre el plano paralelo a la pared a la profundidad de agarre (7.1). */
   942→  wallPlanePoint(x: number, y: number, zPlane: number): Vector3 | null {
   943→    const ray = this.scene.createPickingRay(x, y, Matrix.Identity(), this.camera);
   944→    if (Math.abs(ray.direction.z) < 1e-6) return null;
   945→    const t = (zPlane - ray.origin.z) / ray.direction.z;
   946→    if (t <= 0) return null;
   947→    return ray.origin.add(ray.direction.scale(t));
   948→  }

   950→  /** Distancia de la pared a lo largo del rayo, para depuracion de raycast. */
   951→  wallDistance(x: number, y: number): number {
   952→    const pick = this.scene.pick(x, y, (m) => m.name === "wall", false, this.camera);
   953→    return pick.hit ? pick.distance : Infinity;
   954→  }

   971→  private static circlePoints(radius: number, segments: number, center?: Vector3): Vector3[] {
   972→    const pts: Vector3[] = [];
   973→    const c = center ?? Vector3.Zero();
   974→    for (let i = 0; i <= segments; i++) {
   975→      const a = (i / segments) * Math.PI * 2;
   976→      pts.push(new Vector3(c.x + Math.cos(a) * radius, c.y + Math.sin(a) * radius, c.z));
   977→    }
   978→    return pts;
   979→  }

   982→function wallMaterial(scene: Scene): StandardMaterial {
   983→  const mat = new StandardMaterial("wallMat", scene);
   984→  const tex = new DynamicTexture("wallTex", { width: 512, height: 512 }, scene, false);
   985→  const ctx = tex.getContext() as CanvasRenderingContext2D;
   986→  ctx.fillStyle = "#5d646e";
   987→  ctx.fillRect(0, 0, 512, 512);
   988→  ctx.strokeStyle = "rgba(30,34,40,0.35)";
   989→  ctx.lineWidth = 2;
   990→  for (let i = 0; i <= 8; i++) {
   991→    const p = (i / 8) * 512;
   992→    ctx.beginPath();
   993→    ctx.moveTo(p, 0);
   994→    ctx.lineTo(p, 512);
   995→    ctx.stroke();
   996→    ctx.beginPath();
   997→    ctx.moveTo(0, p);
   998→    ctx.lineTo(512, p);
   999→    ctx.stroke();
  1000→  }
  1001→  ctx.fillStyle = "rgba(255,255,255,0.05)";
  1002→  for (let i = 0; i < 900; i++) {
  1003→    ctx.fillRect(Math.random() * 512, Math.random() * 512, 2, 2);
  1004→  }
  1005→  tex.update();
  1006→  mat.diffuseTexture = tex;
  1007→  mat.specularColor = new Color3(0.06, 0.06, 0.06);
  1008→  return mat;
  1009→}
