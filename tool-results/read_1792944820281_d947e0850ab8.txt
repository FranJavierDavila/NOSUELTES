   380→   380→    // setTune limpia el preset activo (cualquier ajuste pasa por el); volver a

   391→   391→    this.events.emit({ type: "Message", text: `Balance ${name}: tono ${this.tune.postureTone.toFixed(2)}, límites ${this.tune.jointLimitStiffness.toFixed(2)}, pliegue ${this.tune.foldStrength.toFixed(2)}, estiramiento ×${this.tune.stretchTolerance.toFixed(2)}`, tone: "info" });
   392→   392→  }

   394→   394→  private loadTune(): void {
   395→   395→    try {
   396→   396→      const preset = window.localStorage.getItem(GameApp.PRESET_KEY) as BalancePresetName | null;
   397→   397→      if (preset && preset in PRESET_TUNES) this.presetName = preset;
   398→   398→    } catch {
   399→   399→      this.presetName = null;
   400→   400→    }
   401→   401→    if (this.presetName) {
   402→   402→      this.tune = { ...PRESET_TUNES[this.presetName] };
   403→   403→      return;
   404→   404→    }
   405→   405→    try {
   406→   406→      const raw = window.localStorage.getItem(GameApp.TUNE_KEY);
   407→   407→      if (raw) {
   408→   408→        const t = JSON.parse(raw) as Partial<BalanceTuning>;
   409→   409→        this.tune = {
   410→   410→          jointLimitStiffness: clamp01(t.jointLimitStiffness ?? 1.0),
