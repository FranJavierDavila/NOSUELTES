   283→  async loadChallenge(id: string): Promise<void> {
   284→    this.stopTutorial(false);
   285→    this.challenge = this.repo.getChallenge(id);
   286→    this.character = this.repo.getCharacter(this.challenge.characterId);
   287→    this.profile = this.repo.getBodyProfile(this.character.bodyProfileId);
   288→    this.wall = this.repo.getWall(this.challenge.wallId);
   289→    this.holdPosById = new Map(this.wall.holds.map((h) => [h.id, h.position] as const));
   290→    this.bio = new BiomechanicsSystem(this.character, this.profile);
   291→    this.evaluator = new ChallengeEvaluator(this.challenge);
   292→
   293→    const strengthFactor = lerp(this.profile.biomech.strengthFactorMin, this.profile.biomech.strengthFactorMax, this.character.virtues.strength);
   294→    this.anchorMaxForce = this.character.physical.baseGripForce * strengthFactor;
   295→    this.followForce = this.character.physical.baseGripForce * this.character.control.gripAssist * this.character.control.targetFollowStrength;
   296→
   297→    this.solver = new PbdSolver({
   298→      gravity: v3(...this.profile.sim.gravity),
   299→      damping: this.profile.sim.damping,
   300→      iterations: this.profile.sim.iterations,
   301→      floorY: 0,
   302→      wallFrontZ: this.wall.transform.position[2] - this.wall.dimensions.depth / 2,
   303→      tremorAmplitude: this.profile.sim.tremorAmplitude,
   304→      loadEstimatorGain: this.profile.biomech.loadEstimatorGain
   305→    });
   306→
   307→    this.scene.buildWorld(this.wall);
   308→    this.hud.setChallenge(this.challenge, CHALLENGE_META[id]?.label ?? CHALLENGE_LABELS[this.challenge.type]);
   309→    this.buildInteraction();
   310→    this.spawn();
   311→  }

   313→  /**
   314→   * Recreacion del estado inicial (10.2): libera constraints, destruye y recrea el
   315→   * ragdoll, restaura fatiga/timers/progreso y espera resetSettleSeconds.
   316→   * Idempotente: reconstruye todo desde datos; N reinicios no duplican recursos.
   317→   */
   318→  private spawn(): void {
   319→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   320→    this.interaction?.cancelDrag();
   321→    this.interaction?.cancelAssistAll();
   322→    this.scene.clearHoldHighlights();
   323→    for (const hold of this.wall.holds) this.scene.setHoldOccupied(hold.id, false);
   324→
   325→    this.scene.despawnRagdollVisuals();
   326→    this.rig = spawnRagdoll(this.solver, this.profile, this.character, this.wall, this.challenge, holdById, (limb) => this.anchorMaxForceFor(limb));
   327→    this.scene.spawnRagdollVisuals(this.rig);
   328→
   329→    this.bio.reset();
   330→    this.evaluator?.reset();
   331→    this.gameTime = 0;
   332→    this.accumulator = 0;
   333→    this.guardActive = false;
   334→    this.lastSnapshot = { stability: 0, com: { x: 0, y: 0, z: 0 }, supportCount: 0 };
   335→
   336→    // El combo de flujo se rompe con cada respawn (intento nuevo); el conteo de
   337→    // caidas persiste para la puntuacion del desafio cargado.
   338→    this.flowCount = 0;
   339→    this.flowBonus = 0;
   340→    this.peakFatigue = 0;
   341→    this.timeScale = 1;
   342→    this.slowmoLeft = 0;
   343→    this.fallLandedFx = false;
   344→    this.hud.setFlow(0);
   345→    this.lastWorst = null;
   346→    this.limbOverload.clear();
   347→    this.overloadPressure = 0;
   348→
   349→    // Marcar presas ocupadas por los anclajes iniciales creados en spawnRagdoll.
   350→    for (const req of this.challenge.start.requiredAnchors) {
   351→      this.scene.setHoldOccupied(req.holdId, true);
   352→    }
   353→
   354→    this.settleTimer = this.profile.sim.resetSettleSeconds;
   355→    this.autoStartPending = this.challenge.start.autoStart;
   356→    this.fallReleaseQueue = [];
   357→    this.fallAutoReset = -1;
   358→    this.machine.transition("RESETTING");
   359→    // Despues de la transicion: en FALLEN el objetivo seria el tono flacido de la
   360→    // caida; al re-spawnear el escalador vuelve a estar vivo (tono calibrado).
   361→    this.applyTuneToSolver();
   362→    this.events.emit({ type: "Message", text: `Prueba ${CHALLENGE_META[this.challenge.id]?.label ?? CHALLENGE_LABELS[this.challenge.type]} cargada`, tone: "info" });
   363→  }

   372→  /** Aplica un preset del menu, lo persiste y lo marca como activo en el HUD. */
   373→  applyPreset(name: BalancePresetName): void {
   374→    // setTune limpia el preset activo (cualquier ajuste pasa por el); volver a
   375→    // marcar aqui deja el estado coherente: preset aplicado y persistido.
   376→    this.setTune({ ...PRESET_TUNES[name] });
   377→    this.presetName = name;
   378→    try {
   379→      window.localStorage.setItem(GameApp.PRESET_KEY, name);
   380→    } catch {
   381→      /* sin persistencia */
   382→    }
   383→    this.hud.setPresetActive(name);
   384→    this.debug.setTuneValues({ postureTone: this.tune.postureTone, jointLimitStiffness: this.tune.jointLimitStiffness, foldStrength: this.tune.foldStrength, stretchTolerance: this.tune.stretchTolerance });
   385→    this.events.emit({ type: "Message", text: `Balance ${name}: tono ${this.tune.postureTone.toFixed(2)}, límites ${this.tune.jointLimitStiffness.toFixed(2)}, pliegue ${this.tune.foldStrength.toFixed(2)}, estiramiento ×${this.tune.stretchTolerance.toFixed(2)}`, tone: "info" });
   386→  }
   387→
   388→  private loadTune(): void {
   389→    try {
   390→      const preset = window.localStorage.getItem(GameApp.PRESET_KEY) as BalancePresetName | null;
   391→      if (preset && preset in PRESET_TUNES) this.presetName = preset;
   392→    } catch {
   393→      this.presetName = null;
   394→    }
   395→    if (this.presetName) {
   396→      this.tune = { ...PRESET_TUNES[this.presetName] };
   397→      return;
   398→    }
   399→    try {
   400→      const raw = window.localStorage.getItem(GameApp.TUNE_KEY);
   401→      if (raw) {
   402→        const t = JSON.parse(raw) as Partial<BalanceTuning>;
   403→        this.tune = {
   404→          jointLimitStiffness: clamp01(t.jointLimitStiffness ?? 1.0),
   405→          foldStrength: clamp01(t.foldStrength ?? 0.85),
   406→          postureTone: clamp01(t.postureTone ?? 0.75),
   407→          toneFloor: clamp01(t.toneFloor ?? 0.12),
   408→          stretchTolerance: clampRange(t.stretchTolerance ?? 1.1, 1.02, 1.35)
   409→        };
   410→      }
   411→    } catch {
   412→      /* sin persistencia: defaults */
   413→    }
   414→  }
   415→
   416→  private saveTune(): void {
   417→    try {
   418→      window.localStorage.setItem(GameApp.TUNE_KEY, JSON.stringify(this.tune));
   419→    } catch {
   420→      /* sin persistencia */
   421→    }
   422→  }
   423→
   424→  setTune(patch: Partial<BalanceTuning>): BalanceTuning {
   425→    this.tune = {
   426→      jointLimitStiffness: clamp01(patch.jointLimitStiffness ?? this.tune.jointLimitStiffness),
   427→      foldStrength: clamp01(patch.foldStrength ?? this.tune.foldStrength),
   428→      postureTone: clamp01(patch.postureTone ?? this.tune.postureTone),
   429→      toneFloor: clamp01(patch.toneFloor ?? this.tune.toneFloor),
   430→      stretchTolerance: clampRange(patch.stretchTolerance ?? this.tune.stretchTolerance, 1.02, 1.35)
   431→    };
   432→    // Las reglas de postura imposible siguen al tono: preset realista = estricto.
   433→    this.postureRules = postureRulesForTone(this.tune.postureTone);
   434→    // Un ajuste manual (sliders F3 o __NS.setTune) deja de corresponder al preset.
   435→    this.presetName = null;
   436→    try {
   437→      window.localStorage.removeItem(GameApp.PRESET_KEY);
   438→    } catch {
   439→      /* sin persistencia */
   440→    }
   441→    this.hud?.setPresetActive(null);
   442→    this.applyTuneToSolver();
   443→    this.saveTune();
   444→    return { ...this.tune };
   445→  }

   447→  private applyTuneToSolver(): void {
   448→    if (!this.solver) return;
   449→    this.solver.env.balance.jointLimitStiffness = this.tune.jointLimitStiffness;
   450→    this.solver.env.balance.foldStrength = this.tune.foldStrength;
   451→    this.solver.env.balance.postureTone = this.tune.postureTone;
   452→    this.solver.env.balance.toneFloor = this.tune.toneFloor;
   453→    this.solver.env.balance.stretchTolerance = this.tune.stretchTolerance;
   454→    // En FALLEN el tono objetivo sigue siendo el flacido; en el resto, el calibrado.
   455→    this.solver.setToneTarget(this.machine.state === "FALLEN" ? this.tune.toneFloor : this.tune.postureTone);
   456→  }

   458→  requestReset(): void {

   479→  private saveBest(id: string, rec: BestRecord): boolean {

   483→      window.localStorage.setItem(GameApp.bestKey(id), JSON.stringify(rec));

   492→  private buildInteraction(): void {
   493→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   494→    const world: InteractionWorld = {
   495→      holdById,
   496→      challenge: this.challenge,
   497→      character: this.character,
   498→      profile: this.profile,
   499→      wall: this.wall,
   500→      anchorHoldOf: (limb) => this.solver.getAnchor(limb)?.holdId,
   501→      holdOccupancy: (holdId) => {
   502→        let n = 0;
   503→        for (const limb of LIMB_IDS) {
   504→          if (this.solver.getAnchor(limb)?.holdId === holdId) n++;
   505→          if (this.interaction?.assistHoldOf(limb) === holdId) n++;
   506→        }
   507→        return n;
   508→      },
   509→      chainRootOf: (limb) => {
   510→        const idx = limb === "hand_l" ? P.shL : limb === "hand_r" ? P.shR : limb === "foot_l" ? P.hipL : P.hipR;
   511→        const p = this.solver.getParticle(idx);
   512→        return v3(p.pos.x, p.pos.y, p.pos.z);
   513→      },
   514→      limbPositionOf: (limb) => this.solver.positionOfLimb(limb),
   515→      chainLengthOf: (limb) => {
   516→        if (!this.rig) return 0.6;
   517→        const base = limb.startsWith("hand") ? this.rig.armChain : this.rig.legChain;
   518→        return base * this.character.physical.reachScale;
   519→      },
   520→      postureRules: () => this.postureRules
   521→    };
   522→    this.interaction = new InteractionController(world, {
   523→      onSelectChanged: () => undefined,
   524→      onAccept: (limb, holdId) => {
   525→        const existing = this.solver.getAnchor(limb);
   526→        if (existing) {
   527→          this.solver.releaseAnchor(limb);
   528→          this.events.emit({ type: "GripReleased", limb, holdId: existing.holdId, reason: "PLAYER_RELEASE" });
   529→        }
   530→        const hold = holdById.get(holdId)!;
   531→        this.solver.setTarget(limb, v3(hold.position[0], hold.position[1], hold.position[2]), this.followForce);
   532→        this.refreshHoldOccupancy();
   533→      },
   534→      onReject: (limb, holdId, reason) => {
   535→        this.solver.clearTarget(limb);
   536→        this.events.emit({ type: "HoldRejected", limb, holdId, reason });
   537→        if (holdId) {
   538→          this.scene.setHoldInvalid(holdId, true);
   539→          window.setTimeout(() => this.scene.setHoldInvalid(holdId, false), 450);
   540→        }
   541→      },
   542→      onArrival: (limb, holdId) => {
   543→        this.solver.clearTarget(limb);
   544→        const hold = holdById.get(holdId)!;
   545→        this.solver.createAnchor(limb, holdId, v3(hold.position[0], hold.position[1], hold.position[2]), this.anchorMaxForceFor(limb));
   546→        // ADR-007: cada apoyo consolidado reactiva el impulso de subida (el cuerpo
   547→        // se reorganiza hacia la nueva postura: step-up del pie, pull de brazos).
   548→        this.solver.pulseClimbDrive();
   549→        this.events.emit({ type: "HoldAcquired", limb, holdId });
   550→        this.refreshHoldOccupancy();
   551→      }
   552→    });
   553→  }
   554→
   555→  private refreshHoldOccupancy(): void {
   556→    const anchored = new Set<string>();
   557→    for (const limb of LIMB_IDS) {
   558→      const a = this.solver.getAnchor(limb);
   559→      if (a) anchored.add(a.holdId);
   560→    }
   561→    for (const hold of this.wall.holds) {
   562→      this.scene.setHoldOccupied(hold.id, anchored.has(hold.id));
   563→    }
   564→  }

   566→  // --- Pipeline por paso fijo ---
   567→
   568→  private fixedStep(dt: number): void {
   569→    const result = this.solver.step(dt);
   570→
   571→    // ERROR GUARD (9.2): carga no finita / NaN / energia anomala.
   572→    if ((result.nanError || result.energySpike) && !this.guardActive) {
   573→      this.guardActive = true;
   574→      this.guardCooldown = 0.4;
   575→      this.lastGuardError = result.nanError ? "nan" : "energia";
   576→      this.events.emit({ type: "Message", text: "ERROR GUARD: simulacion no finita; reiniciando prueba", tone: "error" });
   577→      this.machine.transition("ERROR");
   578→    }
   579→    if (this.machine.state === "ERROR") {
   580→      this.guardCooldown -= dt;
   581→      if (this.guardCooldown <= 0) {
   582→        this.spawn();
   583→        return;
   584→      }
   585→      return;
   586→    }
   587→
   588→    const anchors = this.solver.anchorHoldIds();
   589→
   590→    // Biomecanica solo cuando el desafio esta activo (fatiga congelada en pausa/resultado).
   591→    if (this.machine.state === "PLAYING") {
   592→      const bioResult = this.bio.update(dt, result, anchors.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver.tremor);
   593→      // En tutorial la fatiga y las zonas se miden (numeros reales en el paso
   594→      // de vitals) pero la sobrecarga NO libera agarres: el onboarding no puede
   595→      // guiar hacia una caida por mover lento.
   596→      for (const limb of this.tutorial ? [] : bioResult.overloadReleases) {
   597→        const anchor = this.solver.releaseAnchor(limb);
   598→        if (anchor) {
   599→          this.events.emit({ type: "GripReleased", limb, holdId: anchor.holdId, reason: "OVERLOAD" });
   600→          this.refreshHoldOccupancy();
   601→        }
   602→      }
   603→      for (const limb of bioResult.strainStarted) this.events.emit({ type: "StrainStarted", limb });
   604→      for (const limb of bioResult.overloadStarted) this.events.emit({ type: "OverloadStarted", limb });
   605→
   606→      // Asistencia de viaje hacia la presa aceptada (7.2).
   607→      this.interaction?.tick(dt);
   608→
   609→      const anchorsNow = this.solver.anchorHoldIds();
   610→      const stability = computeStability(this.bio, result, anchorsNow.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver, this.profile);
   611→      this.lastSnapshot = {
   612→        stability,
   613→        com: result.com,
   614→        supportCount: anchorsNow.length
   615→      };
   616→
   617→      // Evaluacion del desafio (10.3): caida -> exito -> timeout.
   618→      // El timer no corre durante el tutorial: nadie deberia agotar el tiempo
   619→      // mientras aprende los controles.
   620→      if (!this.tutorial) this.gameTime += dt;
   621→      const outcome = this.evaluator!.tick({
   622→        dt,
   623→        gameTime: this.gameTime,
   624→        anchorsByLimb: Object.fromEntries(anchorsNow.map((a) => [a.limb, a.holdId])),
   625→        supportCount: anchorsNow.length,
   626→        stability,
   627→        pelvisY: this.solver.getParticle(P.pelvisC).pos.y,
   628→        outOfBounds: result.outOfBounds
   629→      });
   630→      if (outcome.kind === "fall") {
   631→        this.events.emit({ type: "FallStarted", reason: outcome.reason });
   632→        this.events.emit({ type: "ChallengeFailed", reason: "FALLEN" });
   633→        this.machine.transition("FALLEN");
   634→        this.beginFallSequence();
   635→      } else if (outcome.kind === "success") {
   636→        this.events.emit({ type: "ChallengeSuccess" });
   637→        this.machine.transition("SUCCESS");
   638→      } else if (outcome.kind === "timeout") {
   639→        this.events.emit({ type: "ChallengeFailed", reason: "TIMEOUT" });
   640→        this.machine.transition("TIMEOUT");
   641→      }
   642→    } else {
   643→      // Fuera de PLAYING la fatiga no cambia; sin tremor.
   644→      for (const limb of LIMB_IDS) this.solver.tremor[limb] = 0;
   645→      this.interaction?.tick(0);
   646→    }
   647→
   648→    // Secuencia de caida (ADR-008): en FALLEN los agarres se abren escalonados y el
   649→    // cuerpo cae al colchon; tras asentarse, reintento automatico (R lo adelanta).
   650→    if (this.machine.state === "FALLEN") {
   651→      this.fallTimer += dt;
   652→      while (this.fallReleaseQueue.length > 0 && this.fallTimer >= this.fallReleaseQueue[0].at) {
   653→        const next = this.fallReleaseQueue.shift()!;
   654→        const anchor = this.solver.releaseAnchor(next.limb);
   655→        if (anchor) {
   656→          this.events.emit({ type: "GripReleased", limb: next.limb, holdId: anchor.holdId, reason: "FALL" });
   657→          this.refreshHoldOccupancy();
   658→        }
   659→      }
   660→      // Impacto contra el colchon: polvo + sacudida (una sola vez por caida).
   661→      if (!this.fallLandedFx && this.fallTimer >= LANDING_FX_AT_S) {
   662→        this.fallLandedFx = true;
   663→        const pv = this.solver.getParticle(P.pelvisC);
   664→        this.scene.burstDust(new Vector3(pv.pos.x, 0.06, pv.pos.z), 42);
   665→        this.scene.shake(0.055);
   666→      }
   667→      if (this.fallAutoReset > 0) {
   668→        this.fallAutoReset -= dt;
   669→        if (this.fallAutoReset <= 0) {
   670→          this.fallAutoReset = -1;
   671→          this.spawn();
   672→        }
   673→      }
   674→    }
   675→  }
   676→
   677→  /**
   678→   * Arranque de la secuencia de caida: cancela interaccion pendiente, abre los
   679→   * agarres con retardos realistas (manos primero) y deja el cuerpo flacido.
   680→   */
   681→  private beginFallSequence(): void {
   682→    this.interaction?.cancelDrag();
   683→    this.interaction?.cancelAssistAll();
   684→    this.fallTimer = 0;
   685→    this.fallReleaseQueue = [];
   686→    this.fallLandedFx = false;
   687→    const hands = ["hand_l", "hand_r"] as const;
   688→    const feet = ["foot_l", "foot_r"] as const;
   689→    let n = 0;
   690→    for (const limb of hands) {
   691→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_HAND_S + n++ * FALL_RELEASE_STAGGER_S });
   692→    }
   693→    for (const limb of feet) {
   694→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_FOOT_S + n++ * FALL_RELEASE_STAGGER_S });
   695→    }
   696→    this.fallReleaseQueue.sort((a, b) => a.at - b.at);
   697→    this.fallAutoReset = FALL_AUTO_RESET_S;
   698→    this.solver.setToneTarget(this.tune.toneFloor);
   699→  }
   700→
   701→  // --- Bucle de render ---
   702→
   703→  private readonly frame = (nowMs: number): void => {
   704→    if (this.disposed) return;
   705→    this.raf = requestAnimationFrame(this.frame);
   706→    this.frameCount++;
   707→    const now = nowMs / 1000;
   708→    const dtFrame = Math.min(Math.max(0, now - this.lastFrame), 0.1);
   709→    this.lastFrame = now;
   710→
   711→    this.fpsFrames++;
   712→    this.fpsElapsed += dtFrame;
   713→    if (this.fpsElapsed >= 0.5) {
   714→      this.fps = this.fpsFrames / this.fpsElapsed;
   715→      this.fpsFrames = 0;
   716→      this.fpsElapsed = 0;
   717→    }
   718→
   719→    const state = this.machine.state;
   720→    const stepping = state !== "PAUSED" && state !== "LOADING" && !this.paused;
   721→    if (stepping) {
   722→      // Slow-motion cinematografico: la escala converge suave hacia el objetivo.
   723→      if (this.slowmoLeft > 0) this.slowmoLeft -= dtFrame;
   724→      const goalScale = this.slowmoLeft > 0 ? SLOWMO_SCALE : 1;
   725→      this.timeScale += (goalScale - this.timeScale) * Math.min(1, dtFrame * 9);
   726→      this.accumulator += dtFrame * this.timeScale;
   727→      let steps = 0;
   728→      const fixed = this.profile.sim.fixedDeltaSeconds;
   729→      while (this.accumulator >= fixed && steps < MAX_STEPS_PER_FRAME) {
   730→        const t0 = performance.now();
   731→        this.fixedStep(fixed);
   732→        this.physicsMs = lerp(this.physicsMs, performance.now() - t0, 0.15);
   733→        this.accumulator -= fixed;
   734→        steps++;
   735→      }
   736→      if (steps >= MAX_STEPS_PER_FRAME) this.accumulator = 0;
   737→    }
   738→
   739→    if (state === "RESETTING") {
   740→      this.settleTimer -= dtFrame;
   741→      if (this.settleTimer <= 0) {
   742→        this.machine.transition(this.autoStartPending ? "PLAYING" : "READY");
   743→        if (this.autoStartPending) {
   744→          this.events.emit({ type: "Message", text: "Recupera dos apoyos antes de perder el agarre", tone: "warn" });
   745→        }
   746→      }
   747→    }
   748→
   749→    const alpha = Math.min(1, this.accumulator / this.profile.sim.fixedDeltaSeconds);
   750→    this.scene.syncRagdoll(this.solver, alpha);
   751→    this.scene.syncJointDebug(this.solver);
   752→
   753→    // Anillo de seleccion + enfasis de etiqueta: extremidad arrastrada (firm) o bajo
   754→    // el cursor (iman sutil). La etiqueta IZQ/DER crece: siempre sabes QUE mueves.
   755→    let emphasisLimb: LimbId | null = null;
   756→    if (this.interaction?.dragging) {
   757→      this.scene.setSelectionRing(this.interaction.selected, this.solver, "grab");
   758→      emphasisLimb = this.interaction.selected;
   759→    } else {
   760→      const hover = this.pointerInside && (state === "READY" || state === "PLAYING")
   761→        ? this.scene.pickLimb(this.pointerX, this.pointerY, this.solver)
   762→        : null;
   763→      this.scene.setSelectionRing(hover, this.solver, "hover");
   764→      emphasisLimb = hover;
   765→    }
   766→    this.scene.setLimbLabelEmphasis(emphasisLimb);
   767→
   768→    // Preview de arrastre y alcance.
   769→    const preview = this.interaction?.getPreview() ?? null;
   770→    this.scene.clearHoldHighlights();
   771→    if (preview?.snappedHoldId) {
   772→      if (preview.validity === "valid") this.scene.setHoldCompatible(preview.snappedHoldId, true);
   773→      else this.scene.setHoldInvalid(preview.snappedHoldId, true);
   774→    }
   775→    this.scene.showDragPreview(this.interaction?.dragging ? preview : null, this.solver);
   776→    if (this.interaction?.dragging && preview) {
   777→      this.scene.setReachRadius(
   778→        (preview.limb.startsWith("hand") ? this.rig!.armChain : this.rig!.legChain) * this.character.physical.reachScale
   779→      );
   780→      this.scene.showReachPreview(preview.limb, this.solver, preview.validity !== "invalid");
   781→    } else {
   782→      this.scene.showReachPreview(null, this.solver, true);
   783→    }
   784→
   785→    // Motivo de rechazo EN VIVO durante el arrastre: el jugador ve POR QUE una presa
   786→    // no es posible antes de soltar (control determinista: nada "al azar").
   787→    if (this.interaction?.dragging) {
   788→      if (preview?.snappedHoldId && preview.validity === "invalid") {
   789→        const reason = this.interaction.pendingRejectReason() ?? this.interaction.rejectReason(preview.limb, preview.snappedHoldId);
   790→        this.hud.setRejectHint(reason ? `✕ ${REJECT_LABELS[reason] ?? reason}` : "Movimiento no posible");
   791→      } else if (preview && !preview.snappedHoldId && !this.challenge.rules.allowFreeTarget) {
   792→        this.hud.setRejectHint("Suelta sobre una presa resaltada");
   793→      } else {
   794→        this.hud.setRejectHint(null);
   795→      }
   796→    } else {
   797→      this.hud.setRejectHint(null);
   798→    }
   799→
   800→    // Tutorial: resaltados y avance de pasos.
   801→    this.tickTutorial();
   802→
   803→    // Retroalimentacion de carga por extremidad (misma paleta que el panel
   804→    // lateral): tinte verde/ambar/rojo de la extremidad y anillo sobre su presa.
   805→    // La sobrecarga ademas lleva CUENTA ATRAS de fallo: urgencia 0..1 (danno
   806→    // acumulado/tolerancia) acelera parpadeo y latidos, y alimenta el chip central.
   807→    const cfgBio = this.profile.biomech;
   808→    this.limbOverload.clear();
   809→    this.overloadPressure = 0;
   810→    for (const limb of LIMB_IDS) {
   811→      const anchored = this.solver.hasAnchor(limb);
   812→      const info = this.bio.infoFor(limb);
   813→      const zone = info?.zone ?? "NORMAL";
   814→      let urgency = 0;
   815→      if (anchored && info && (zone === "OVERLOADED" || zone === "ERROR_GUARD")) {
   816→        const secondsLeft = Math.max(0, (cfgBio.overloadGraceSeconds - info.overloadTimer) / cfgBio.overloadAccumRate);
   817→        const frac = clamp01(1 - (secondsLeft * cfgBio.overloadAccumRate) / Math.max(1e-6, cfgBio.overloadGraceSeconds));
   818→        this.limbOverload.set(limb, { seconds: secondsLeft, frac });
   819→        urgency = frac;
   820→        this.overloadPressure = Math.max(this.overloadPressure, frac);
   821→      }
   822→      this.scene.updateLimbTint(limb, anchored, zone, urgency);
   823→      this.scene.setAnchorRing(limb, this.solver.getAnchor(limb)?.holdId ?? null, zone, urgency);
   824→    }
   825→
   826→    // Voz del escalador: respiracion y quejidos de fondo dirigidos por la fatiga
   827→    // y la carga reales (solo suenan con PLAYING activo; el director se resetea fuera).
   828→    const maxFatigue = Math.max(...LIMB_IDS.map((l) => this.bio.fatigue[l] ?? 0));
   829→    const maxRatio = Math.max(...LIMB_IDS.map((l) => this.bio.infoFor(l)?.ratio ?? 0));
   830→    if (this.machine.state === "PLAYING") this.peakFatigue = Math.max(this.peakFatigue, maxFatigue);
   831→    this.audio.update(dtFrame, {
   832→      playing: this.machine.state === "PLAYING",
   833→      maxFatigue,
   834→      maxRatio,
   835→      overloadRatio: this.profile.biomech.overloadRatio,
   836→      overload: LIMB_IDS.some((l) => this.bio.infoFor(l)?.zone === "OVERLOADED"),
   837→      overloadPressure: this.overloadPressure
   838→    });
   839→
   840→    // Combo de flujo: se rompe si pasa la ventana sin fijar otra presa.
   841→    if (this.flowCount > 0 && performance.now() / 1000 - this.lastGrabAt > FLOW_WINDOW_S) {
   842→      this.flowCount = 0;
   843→      this.hud.setFlow(0);
   844→    }
   845→
   846→    const pelvis = this.solver.getParticle(P.pelvisC);
   847→    this.scene.updateCamera(dtFrame, pelvis.pos, this.recenterOnce);
   848→    this.recenterOnce = false;
   849→
   850→    // HUD y debug a 10 Hz.
   851→    this.hudAccum += dtFrame;
   852→    if (this.hudAccum >= 0.1) {
   853→      this.hudAccum = 0;
   854→      this.updateHud(state);
   855→      this.debug.update(this.buildDebugData(state));
   856→    }
   857→
   858→    this.scene.render();
   859→  };
   860→
   861→  private updateHud(state: GameState): void {
   862→    const running = state === "PLAYING";
   863→    this.hud.setTimer(
   864→      this.evaluator ? Math.max(0, this.challenge.timeLimitSeconds - (running ? this.gameTime : 0)) : null,
   865→      running
   866→    );
   867→    let fatigueMax = 0;
   868→    for (const limb of LIMB_IDS) fatigueMax = Math.max(fatigueMax, this.bio.fatigue[limb]);
   869→    this.hud.setVitals(this.lastSnapshot.stability, fatigueMax);
   870→
   871→    const limbStates: Record<LimbId, { zone: LoadZone; anchored: boolean; label: string; overloadS?: number | null; overloadFrac?: number | null }> = {} as never;
   872→    for (const limb of LIMB_IDS) {
   873→      const anchored = this.solver.hasAnchor(limb);
   874→      const zone = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   875→      const over = this.limbOverload.get(limb);
   876→      limbStates[limb] = {
   877→        zone,
   878→        anchored,
   879→        label: this.interaction?.isAssisting(limb) ? "MOVIENDO" : this.interaction?.dragging && this.interaction.selected === limb ? "OBJETIVO" : "LIBRE",
   880→        overloadS: over ? over.seconds : null,
   881→        overloadFrac: over ? over.frac : null
   882→      };
   883→    }
   884→    this.hud.setLimbStates(limbStates);
   885→    // Chip central: la extremidad mas comprometida (solo jugando y fuera del
   886→    // tutorial: ahi la sobrecarga nunca suelta, el aviso despistaria). Memoria
   887→    // de 0.6 s: los valles del pendulum no hacen parpadear el aviso.
   888→    let worst: { limb: LimbId; seconds: number; frac: number } | null = null;
   889→    for (const [limb, over] of this.limbOverload) {
   890→      if (!worst || over.seconds < worst.seconds) worst = { limb, seconds: over.seconds, frac: over.frac };
   891→    }
   892→    const nowS = performance.now() / 1000;
   893→    if (worst) {

   949→  private onGameEvent(e: GameEvent): void {
   950→    this.audio.play(e);
   951→    switch (e.type) {
   952→      case "StateChanged":
   953→        this.hud.setState(e.to);
   954→        break;
   955→      case "ChallengeSuccess": {
   956→        // Tanda de confeti sobre la presa cima del desafio.
   957→        const topHold = this.challenge.completion.holdIds?.[0] ?? "";
   958→        const topPos = topHold ? this.holdPosById.get(topHold) : undefined;
   959→        if (topPos) this.scene.burstConfetti(new Vector3(topPos[0], topPos[1] + 0.1, topPos[2] - 0.05));
   960→        // Puntuacion, estrellas y record persistente.
   961→        const result = computeScore({
   962→          timeS: this.gameTime,
   963→          timeLimitS: this.challenge.timeLimitSeconds,
   964→          falls: this.fallCount,
   965→          flowBonus: this.flowBonus
   966→        });
   967→        const rec: BestRecord = { score: result.total, timeS: this.gameTime, stars: result.stars };
   968→        const isRecord = this.saveBest(this.challenge.id, rec);
   969→        this.hud.showSuccess({
   970→          stars: result.stars,
   971→          total: result.total,
   972→          timeS: this.gameTime,
   973→          breakdown: result.breakdown,
   974→          isRecord,
   975→          best: this.loadBest(this.challenge.id)
   976→        });
   977→        if (isRecord) this.hud.setBest(this.challenge.id, rec);
   978→        break;
   979→      }
   980→      case "ChallengeFailed":
   981→        this.hud.showBanner(e.reason === "TIMEOUT" ? "TIEMPO AGOTADO" : "CAIDA", "bad");
   982→        break;
   983→      case "HoldAcquired": {
   984→        // Combo de flujo: encadenar fijaciones en menos de FLOW_WINDOW_S suma bonus.
   985→        const now = performance.now() / 1000;
   986→        this.flowCount = now - this.lastGrabAt <= FLOW_WINDOW_S ? Math.min(FLOW_MAX_CHAIN, this.flowCount + 1) : 1;
   987→        this.lastGrabAt = now;
   988→        if (this.flowCount >= 2) this.flowBonus += FLOW_BONUS_PER_STEP * this.flowCount;
   989→        this.hud.setFlow(this.flowCount);
   990→        // Bocada de tiza al fijar una mano.
   991→        if (e.limb.startsWith("hand")) {
   992→          const pos = this.holdPosById.get(e.holdId);
   993→          if (pos) this.scene.burstChalk(new Vector3(pos[0], pos[1], pos[2] - 0.03));
   994→        }
   995→        break;
   996→      }
   997→      case "HoldRejected": {
   998→        const label = REJECT_LABELS[e.reason] ?? e.reason;
   999→        this.hud.toast(`Movimiento rechazado: ${label}`, "warn");
  1000→        break;
  1001→      }
  1002→      case "GripReleased":
  1003→        if (e.reason === "OVERLOAD") {
  1004→          this.hud.toast(`Sobrecarga: el agarre cedió (${e.limb})`, "error");
  1005→          // Sacudida corta + tiza al resbalar el agarre.
  1006→          this.scene.shake(0.02);
  1007→          const pos = this.holdPosById.get(e.holdId);
  1008→          if (pos) this.scene.burstChalk(new Vector3(pos[0], pos[1], pos[2] - 0.03), 14);
  1009→        }
  1010→        break;
  1011→      case "StrainStarted":
  1012→        this.hintOnce("strain", "Agarre en tension: la fatiga de esa extremidad sube mas rapido", "warn");
  1013→        break;
  1014→      case "OverloadStarted":
  1015→        this.hintOnce("overload", "SOBRECARGA: el chip rojo cuenta los segundos que aguanta el agarre (~4,5 s en sobrecarga continua). Alivia esa extremidad ya", "error");
  1016→        break;
  1017→      case "FallStarted":
  1018→        this.hintOnce("fall", "Caída: los agarres se abren y el cuerpo cae. Se reintenta solo (o pulsa R)", "info");
  1019→        // Drama: slow-motion breve + rotura del combo + conteo de caidas.
  1020→        this.slowmoLeft = SLOWMO_DURATION_S;
  1021→        this.fallCount += 1;
  1022→        this.flowCount = 0;
  1023→        this.hud.setFlow(0);
  1024→        break;
  1025→      default:
  1026→        break;
  1027→    }
  1028→  }
