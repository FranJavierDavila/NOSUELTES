   160→  // --- Carga de desafio y reset (10.2) ---

   162→  async loadChallenge(id: string): Promise<void> {
   163→    this.challenge = this.repo.getChallenge(id);
   164→    this.character = this.repo.getCharacter(this.challenge.characterId);
   165→    this.profile = this.repo.getBodyProfile(this.character.bodyProfileId);
   166→    this.wall = this.repo.getWall(this.challenge.wallId);
   167→    this.bio = new BiomechanicsSystem(this.character, this.profile);
   168→    this.evaluator = new ChallengeEvaluator(this.challenge);
   169→
   170→    const strengthFactor = lerp(this.profile.biomech.strengthFactorMin, this.profile.biomech.strengthFactorMax, this.character.virtues.strength);
   171→    this.anchorMaxForce = this.character.physical.baseGripForce * strengthFactor;
   172→    this.followForce = this.character.physical.baseGripForce * this.character.control.gripAssist * this.character.control.targetFollowStrength;
   173→
   174→    this.solver = new PbdSolver({
   175→      gravity: v3(...this.profile.sim.gravity),
   176→      damping: this.profile.sim.damping,
   177→      iterations: this.profile.sim.iterations,
   178→      floorY: 0,
   179→      wallFrontZ: this.wall.transform.position[2] - this.wall.dimensions.depth / 2,
   180→      tremorAmplitude: this.profile.sim.tremorAmplitude,
   181→      loadEstimatorGain: this.profile.biomech.loadEstimatorGain
   182→    });
   183→
   184→    this.scene.buildWorld(this.wall);
   185→    this.hud.setChallenge(this.challenge, CHALLENGE_LABELS[this.challenge.type]);
   186→    this.buildInteraction();
   187→    this.spawn();
   188→  }

   190→  /**
   191→   * Recreacion del estado inicial (10.2): libera constraints, destruye y recrea el
   192→   * ragdoll, restaura fatiga/timers/progreso y espera resetSettleSeconds.
   193→   * Idempotente: reconstruye todo desde datos; N reinicios no duplican recursos.
   194→   */
   195→  private spawn(): void {
   196→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   197→    this.interaction?.cancelDrag();
   198→    this.interaction?.cancelAssistAll();
   199→    this.scene.clearHoldHighlights();
   200→    for (const hold of this.wall.holds) this.scene.setHoldOccupied(hold.id, false);
   201→
   202→    this.scene.despawnRagdollVisuals();
   203→    this.rig = spawnRagdoll(this.solver, this.profile, this.character, this.wall, this.challenge, holdById, this.anchorMaxForce);
   204→    this.scene.spawnRagdollVisuals(this.rig);
   205→
   206→    this.bio.reset();
   207→    this.evaluator?.reset();
   208→    this.gameTime = 0;
   209→    this.accumulator = 0;
   210→    this.guardActive = false;
   211→    this.lastSnapshot = { stability: 0, com: { x: 0, y: 0, z: 0 }, supportCount: 0 };
   212→
   213→    // Marcar presas ocupadas por los anclajes iniciales creados en spawnRagdoll.
   214→    for (const req of this.challenge.start.requiredAnchors) {
   215→      this.scene.setHoldOccupied(req.holdId, true);
   216→    }
   217→
   218→    this.settleTimer = this.profile.sim.resetSettleSeconds;
   219→    this.autoStartPending = this.challenge.start.autoStart;
   220→    this.machine.transition("RESETTING");
   221→    this.events.emit({ type: "Message", text: `Prueba ${CHALLENGE_LABELS[this.challenge.type]} cargada`, tone: "info" });
   222→  }
   223→
   224→  requestReset(): void {
   225→    if (this.machine.state === "LOADING" || this.machine.state === "RESETTING") return;
   226→    this.events.emit({ type: "ResetStarted" });
   227→    this.spawn();
   228→  }

   230→  // --- Interaccion ---
   231→
   232→  private buildInteraction(): void {
   233→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   234→    const world: InteractionWorld = {
   235→      holdById,
   236→      challenge: this.challenge,
   237→      character: this.character,
   238→      profile: this.profile,
   239→      wall: this.wall,
   240→      anchorHoldOf: (limb) => this.solver.getAnchor(limb)?.holdId,
   241→      holdOccupancy: (holdId) => {
   242→        let n = 0;
   243→        for (const limb of LIMB_IDS) {
   244→          if (this.solver.getAnchor(limb)?.holdId === holdId) n++;
   245→          if (this.interaction?.assistHoldOf(limb) === holdId) n++;
   246→        }
   247→        return n;
   248→      },
   249→      chainRootOf: (limb) => {
   250→        const idx = limb === "hand_l" ? P.shL : limb === "hand_r" ? P.shR : limb === "foot_l" ? P.hipL : P.hipR;
   251→        const p = this.solver.getParticle(idx);
   252→        return v3(p.pos.x, p.pos.y, p.pos.z);
   253→      },
   254→      limbPositionOf: (limb) => this.solver.positionOfLimb(limb),
   255→      chainLengthOf: (limb) => {
   256→        if (!this.rig) return 0.6;
   257→        const base = limb.startsWith("hand") ? this.rig.armChain : this.rig.legChain;
   258→        return base * this.character.physical.reachScale;
   259→      }
   260→    };
   261→    this.interaction = new InteractionController(world, {
   262→      onSelectChanged: () => undefined,
   263→      onAccept: (limb, holdId) => {
   264→        const existing = this.solver.getAnchor(limb);
   265→        if (existing) {
   266→          this.solver.releaseAnchor(limb);
   267→          this.events.emit({ type: "GripReleased", limb, holdId: existing.holdId, reason: "PLAYER_RELEASE" });
   268→        }
   269→        const hold = holdById.get(holdId)!;
   270→        this.solver.setTarget(limb, v3(hold.position[0], hold.position[1], hold.position[2]), this.followForce);
   271→        this.refreshHoldOccupancy();
   272→      },
   273→      onReject: (limb, holdId, reason) => {
   274→        this.solver.clearTarget(limb);
   275→        this.events.emit({ type: "HoldRejected", limb, holdId, reason });
   276→        if (holdId) {
   277→          this.scene.setHoldInvalid(holdId, true);
   278→          window.setTimeout(() => this.scene.setHoldInvalid(holdId, false), 450);
   279→        }
   280→      },
   281→      onArrival: (limb, holdId) => {
   282→        this.solver.clearTarget(limb);
   283→        const hold = holdById.get(holdId)!;
   284→        this.solver.createAnchor(limb, holdId, v3(hold.position[0], hold.position[1], hold.position[2]), this.anchorMaxForce);
   285→        this.events.emit({ type: "HoldAcquired", limb, holdId });
   286→        this.refreshHoldOccupancy();
   287→      }
   288→    });
   289→  }
   290→
   291→  private refreshHoldOccupancy(): void {
   292→    const anchored = new Set<string>();
   293→    for (const limb of LIMB_IDS) {
   294→      const a = this.solver.getAnchor(limb);
   295→      if (a) anchored.add(a.holdId);
   296→    }
   297→    for (const hold of this.wall.holds) {
   298→      this.scene.setHoldOccupied(hold.id, anchored.has(hold.id));
   299→    }
   300→  }

   302→  // --- Pipeline por paso fijo ---
   303→
   304→  private fixedStep(dt: number): void {
   305→    const result = this.solver.step(dt);
   306→
   307→    // ERROR GUARD (9.2): carga no finita / NaN / energia anomala.
   308→    if ((result.nanError || result.energySpike) && !this.guardActive) {
   309→      this.guardActive = true;
   310→      this.guardCooldown = 0.4;
   311→      this.lastGuardError = result.nanError ? "nan" : "energia";
   312→      this.events.emit({ type: "Message", text: "ERROR GUARD: simulacion no finita; reiniciando prueba", tone: "error" });
   313→      this.machine.transition("ERROR");
   314→    }
   315→    if (this.machine.state === "ERROR") {
   316→      this.guardCooldown -= dt;
   317→      if (this.guardCooldown <= 0) {
   318→        this.spawn();
   319→        return;
   320→      }
   321→      return;
   322→    }
   323→
   324→    const anchors = this.solver.anchorHoldIds();
   325→
   326→    // Biomecanica solo cuando el desafio esta activo (fatiga congelada en pausa/resultado).
   327→    if (this.machine.state === "PLAYING") {
   328→      const bioResult = this.bio.update(dt, result, anchors.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver.tremor);
   329→      for (const limb of bioResult.overloadReleases) {
   330→        const anchor = this.solver.releaseAnchor(limb);
   331→        if (anchor) {
   332→          this.events.emit({ type: "GripReleased", limb, holdId: anchor.holdId, reason: "OVERLOAD" });
   333→          this.refreshHoldOccupancy();
   334→        }
   335→      }
   336→      for (const limb of bioResult.strainStarted) this.events.emit({ type: "StrainStarted", limb });
   337→      for (const limb of bioResult.overloadStarted) this.events.emit({ type: "OverloadStarted", limb });
   338→
   339→      // Asistencia de viaje hacia la presa aceptada (7.2).
   340→      this.interaction?.tick(dt);
   341→
   342→      const anchorsNow = this.solver.anchorHoldIds();
   343→      const stability = computeStability(this.bio, result, anchorsNow.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver, this.profile);
   344→      this.lastSnapshot = {
   345→        stability,
   346→        com: result.com,
   347→        supportCount: anchorsNow.length
   348→      };
   349→
   350→      // Evaluacion del desafio (10.3): caida -> exito -> timeout.
   351→      this.gameTime += dt;
   352→      const outcome = this.evaluator!.tick({
   353→        dt,
   354→        gameTime: this.gameTime,
   355→        anchorsByLimb: Object.fromEntries(anchorsNow.map((a) => [a.limb, a.holdId])),
   356→        supportCount: anchorsNow.length,
   357→        stability,
   358→        pelvisY: this.solver.getParticle(P.pelvisC).pos.y,
   359→        outOfBounds: result.outOfBounds
   360→      });
   361→      if (outcome.kind === "fall") {
   362→        this.events.emit({ type: "FallStarted", reason: outcome.reason });
   363→        this.events.emit({ type: "ChallengeFailed", reason: "FALLEN" });
   364→        this.machine.transition("FALLEN");
   365→      } else if (outcome.kind === "success") {
   366→        this.events.emit({ type: "ChallengeSuccess" });
   367→        this.machine.transition("SUCCESS");
   368→      } else if (outcome.kind === "timeout") {
   369→        this.events.emit({ type: "ChallengeFailed", reason: "TIMEOUT" });
   370→        this.machine.transition("TIMEOUT");
   371→      }
   372→    } else {
   373→      // Fuera de PLAYING la fatiga no cambia; sin tremor.
   374→      for (const limb of LIMB_IDS) this.solver.tremor[limb] = 0;
   375→      this.interaction?.tick(0);
   376→    }
   377→  }

   379→  // --- Bucle de render ---
   380→
   381→  private readonly frame = (nowMs: number): void => {
   382→    if (this.disposed) return;
   383→    this.raf = requestAnimationFrame(this.frame);
   384→    this.frameCount++;
   385→    const now = nowMs / 1000;
   386→    const dtFrame = Math.min(Math.max(0, now - this.lastFrame), 0.1);
   387→    this.lastFrame = now;
   388→
   389→    this.fpsFrames++;
   390→    this.fpsElapsed += dtFrame;
   391→    if (this.fpsElapsed >= 0.5) {
   392→      this.fps = this.fpsFrames / this.fpsElapsed;
   393→      this.fpsFrames = 0;
   394→      this.fpsElapsed = 0;
   395→    }
   396→
   397→    const state = this.machine.state;
   398→    const stepping = state !== "PAUSED" && state !== "LOADING" && !this.paused;
   399→    if (stepping) {
   400→      this.accumulator += dtFrame;
   401→      let steps = 0;
   402→      const fixed = this.profile.sim.fixedDeltaSeconds;
   403→      while (this.accumulator >= fixed && steps < MAX_STEPS_PER_FRAME) {
   404→        const t0 = performance.now();
   405→        this.fixedStep(fixed);
   406→        this.physicsMs = lerp(this.physicsMs, performance.now() - t0, 0.15);
   407→        this.accumulator -= fixed;
   408→        steps++;
   409→      }
   410→      if (steps >= MAX_STEPS_PER_FRAME) this.accumulator = 0;
   411→    }
   412→
   413→    if (state === "RESETTING") {
   414→      this.settleTimer -= dtFrame;
   415→      if (this.settleTimer <= 0) {
   416→        this.machine.transition(this.autoStartPending ? "PLAYING" : "READY");
   417→        if (this.autoStartPending) {
   418→          this.events.emit({ type: "Message", text: "Recupera dos apoyos antes de perder el agarre", tone: "warn" });
   419→        }
   420→      }
   421→    }
   422→
   423→    const alpha = Math.min(1, this.accumulator / this.profile.sim.fixedDeltaSeconds);
   424→    this.scene.syncRagdoll(this.solver, alpha);
   425→    this.scene.syncJointDebug(this.solver);
   426→    this.scene.setSelectionRing(this.interaction?.dragging ? this.interaction.selected : this.interaction?.selected ?? null, this.solver);
   427→
   428→    // Preview de arrastre y alcance.
   429→    const preview = this.interaction?.getPreview() ?? null;
   430→    this.scene.clearHoldHighlights();
   431→    if (preview?.snappedHoldId) {
   432→      if (preview.validity === "valid") this.scene.setHoldCompatible(preview.snappedHoldId, true);
   433→      else this.scene.setHoldInvalid(preview.snappedHoldId, true);
   434→    }
   435→    this.scene.showDragPreview(this.interaction?.dragging ? preview : null, this.solver);
   436→    if (this.interaction?.dragging && preview) {
   437→      this.scene.setReachRadius(
   438→        (preview.limb.startsWith("hand") ? this.rig!.armChain : this.rig!.legChain) * this.character.physical.reachScale
   439→      );
   440→      this.scene.showReachPreview(preview.limb, this.solver, preview.validity !== "invalid");
   441→    } else {
   442→      this.scene.showReachPreview(null, this.solver, true);
   443→    }
   444→
   445→    // Tinte de esfuerzo por extremidad.
   446→    for (const limb of LIMB_IDS) {
   447→      this.scene.updateLimbTint(limb, this.solver.tremor[limb]);
   448→    }
   449→
   450→    const pelvis = this.solver.getParticle(P.pelvisC);
   451→    this.scene.updateCamera(dtFrame, pelvis.pos, this.recenterOnce);
   452→    this.recenterOnce = false;
   453→
   454→    // HUD y debug a 10 Hz.
   455→    this.hudAccum += dtFrame;
   456→    if (this.hudAccum >= 0.1) {
   457→      this.hudAccum = 0;
   458→      this.updateHud(state);
   459→      this.debug.update(this.buildDebugData(state));
   460→    }
   461→
   462→    this.scene.render();
   463→  };
   464→
   465→  private updateHud(state: GameState): void {
   466→    const running = state === "PLAYING";
   467→    this.hud.setTimer(
   468→      this.evaluator ? Math.max(0, this.challenge.timeLimitSeconds - (running ? this.gameTime : 0)) : null,
   469→      running
   470→    );
   471→    let fatigueMax = 0;
   472→    for (const limb of LIMB_IDS) fatigueMax = Math.max(fatigueMax, this.bio.fatigue[limb]);
   473→    this.hud.setVitals(this.lastSnapshot.stability, fatigueMax);
   474→
   475→    const limbStates: Record<LimbId, { zone: LoadZone; anchored: boolean; label: string }> = {} as never;
   476→    for (const limb of LIMB_IDS) {
   477→      const anchored = this.solver.hasAnchor(limb);
   478→      const zone = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   479→      limbStates[limb] = {
   480→        zone,
   481→        anchored,
   482→        label: this.interaction?.isAssisting(limb) ? "MOVIENDO" : this.interaction?.dragging && this.interaction.selected === limb ? "OBJETIVO" : "LIBRE"
   483→      };
   484→    }
   485→    this.hud.setLimbStates(limbStates);
   486→    this.hud.setProgress(running && this.evaluator && this.evaluator.progress > 0 ? this.evaluator.progress : null);
   487→  }

   489→  private limbDisplayState(limb: LimbId): string {
   490→    if (this.machine.state === "FALLEN") return "FALLING";
   491→    if (this.interaction?.isAssisting(limb)) return "TARGETING";
   492→    if (this.solver.hasAnchor(limb)) {
   493→      const timer = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   494→      return timer === "OVERLOADED" ? "OVERLOADED" : "ANCHORED";
   495→    }
   496→    if (this.interaction?.dragging && this.interaction.selected === limb) return "TARGETING";
   497→    return "FREE";
   498→  }

   500→  private buildDebugData(state: GameState): DebugData {
   501→    const counts = this.solver.counts();
   502→    const limbs = {} as DebugData["limbs"];
   503→    for (const limb of LIMB_IDS) {
   504→      const info = this.bio.infoFor(limb);
   505→      const anchor = this.solver.getAnchor(limb);
   506→      limbs[limb] = {
   507→        state: this.limbDisplayState(limb),
   508→        holdId: anchor?.holdId ?? this.interaction?.assistHoldOf(limb) ?? null,
   509→        loadN: anchor?.loadN ?? 0,
   510→        effectiveN: info?.effectiveForceN ?? 0,
   511→        ratio: info?.ratio ?? 0,
   512→        fatigue: this.bio.fatigue[limb],
   513→        zone: info?.zone ?? "-"
   514→      };
   515→    }
   516→    return {
   517→      fps: Math.round(this.fps),
   518→      backend: this.scene.backend,
   519→      physicsMs: this.physicsMs,
   520→      bodies: counts.particles,
   521→      constraints: counts.sticks + counts.angles,
   522→      anchors: counts.anchors,
   523→      state,
   524→      gameTime: this.gameTime,
   525→      com: this.lastSnapshot.com,
   526→      stability: this.lastSnapshot.stability,
   527→      supportCount: this.lastSnapshot.supportCount,
   528→      limbs,
   529→      flags: this.scene.getDebugFlags()
   530→    };
   531→  }

   535→  private onGameEvent(e: GameEvent): void {
   536→    this.audio.play(e);
   537→    switch (e.type) {
   538→      case "StateChanged":
   539→        this.hud.setState(e.to);
   540→        break;
   541→      case "ChallengeSuccess":
   542→        this.hud.showBanner("EXITO", "ok");
   543→        break;
   544→      case "ChallengeFailed":
   545→        this.hud.showBanner(e.reason === "TIMEOUT" ? "TIEMPO AGOTADO" : "CAIDA", "bad");
   546→        break;
   547→      case "HoldRejected": {
   548→        const label = REJECT_LABELS[e.reason] ?? e.reason;
   549→        this.hud.toast(`Movimiento rechazado: ${label}`, "warn");
   550→        break;
   551→      }
   552→      case "GripReleased":
   553→        if (e.reason === "OVERLOAD") this.hud.toast(`Sobrecarga: se solto el agarre (${e.limb})`, "error");
   554→        break;
   555→      default:
   556→        break;
   557→    }

   568→  private attachInput(): void {
   569→    const canvas = this.canvas;
   570→    const zPlane = () => this.wall.holds[0]?.position[2] ?? -0.18;
   571→
   572→    this.on(canvas, "pointerdown", (e) => {
   573→      this.audio.ensure();
   574→      if (e.button !== 0) return;
   575→      if (!(this.machine.state === "READY" || this.machine.state === "PLAYING")) return;
   576→      const rect = canvas.getBoundingClientRect();
   577→      const limb = this.scene.pickLimb(e.clientX - rect.left, e.clientY - rect.top);
   578→      if (!limb || this.interaction!.isAssisting(limb)) return;
   579→      this.interaction!.beginDrag(limb);
   580→      if (this.machine.state === "READY") {
   581→        // El timer arranca con la primera interaccion (start.autoStart = false).
   582→        this.machine.transition("PLAYING");
   583→      }
   584→      canvas.setPointerCapture(e.pointerId);
   585→    });
   586→
   587→    this.on(canvas, "pointermove", (e) => {
   588→      if (!this.interaction?.dragging) return;
   589→      const rect = canvas.getBoundingClientRect();
   590→      const point = this.scene.wallPlanePoint(e.clientX - rect.left, e.clientY - rect.top, zPlane());
   591→      this.interaction.updateDrag(point ? v3(point.x, point.y, point.z) : null);
   592→      const preview = this.interaction.getPreview();
   593→      if (preview?.snappedHoldId) {
   594→        this.interaction.setRejectReason(this.interaction.rejectReason(preview.limb, preview.snappedHoldId));
   595→      } else {
   596→        this.interaction.setRejectReason(null);
   597→      }
   598→    });
   599→
   600→    this.on(canvas, "pointerup", (e) => {
   601→      if (e.button !== 0) return;
   602→      if (this.interaction?.dragging) {
   603→        this.interaction.endDrag();
   604→        this.scene.clearHoldHighlights();
   605→      }
   606→    });
   607→
   608→    this.on(canvas, "contextmenu", (e) => {
   609→      e.preventDefault();
   610→      this.interaction?.cancelDrag();
   611→    });
   612→
   613→    this.on(window, "keydown", (e) => {
   614→      const ev = e as KeyboardEvent;
   615→      switch (ev.code) {
   616→        case "KeyR":
   617→          this.requestReset();
   618→          break;
   619→        case "KeyH":
   620→          this.hud.toggleHelp();
   621→          break;
   622→        case "F3":
   623→          ev.preventDefault();
   624→          this.debug.toggle();
   625→          break;
   626→        case "Space":
   627→          ev.preventDefault();
   628→          this.togglePause();
   629→          break;
   630→        case "KeyC":
   631→          this.recenterOnce = true;
   632→          break;
   633→        case "F4":
   634→          ev.preventDefault();
   635→          this.scene.toggleOrtho();
   636→          break;
   637→        case "KeyM": {
   638→          const muted = this.audio.toggleMute();
   639→          this.hud.toast(muted ? "Sonido desactivado" : "Sonido activado", "info");
   640→          break;
   641→        }
   642→        case "Escape":
   643→          this.interaction?.cancelDrag();
   644→          break;
   645→        default:
   646→          break;
   647→      }
   648→    });
   649→
   650→    this.on(window, "resize", () => {
   651→      this.scene.resize();
   652→    });
   653→  }

   655→  togglePause(): void {
   656→    if (this.machine.state === "PLAYING") {
   657→      this.machine.transition("PAUSED");
   658→      this.hud.toast("Pausa: fisica y timer congelados", "info");
   659→    } else if (this.machine.state === "PAUSED") {
   660→      this.machine.transition("PLAYING");
   661→    }
   662→  }

   668→  get debugHook(): {
   669→    state: () => GameState;
   670→    frames: () => number;
   671→    settle: () => number;
   672→    lastError: () => string;
   673→    selected: () => LimbId | null;
   674→    dragging: () => boolean;
   675→    preview: () => { limb: LimbId; snap: string | null; validity: string; target: { x: number; y: number } } | null;
   676→    anchors: () => { limb: LimbId; hold: string; load: number }[];
   677→    reject: (limb: LimbId, holdId: string) => string;
   678→    chainInfo: (limb: LimbId) => { root: { x: number; y: number; z: number } };
   679→    pickAt: (x: number, y: number) => LimbId | null;
   680→    pickNearest: (x: number, y: number) => { name: string; dist: number } | null;
   681→    handMeshes: () => { name: string; pickable: boolean; enabled: boolean; pos: { x: number; y: number; z: number } }[];
   682→    renderSize: () => { w: number; h: number; cssW: number; cssH: number };
   683→    project: (x: number, y: number, z: number) => { x: number; y: number };
   684→    limbScreen: () => { limb: string; x: number; y: number }[];
   685→  } {
   686→    return {
   687→      state: () => this.machine.state,
   688→      frames: () => this.frameCount,
   689→      settle: () => this.settleTimer,
   690→      lastError: () => this.lastGuardError,
   691→      selected: () => this.interaction?.selected ?? null,
   692→      dragging: () => this.interaction?.dragging ?? false,
   693→      preview: () => {
   694→        const p = this.interaction?.getPreview();
   695→        return p ? { limb: p.limb, snap: p.snappedHoldId, validity: p.validity, target: { x: Math.round(p.target.x * 100) / 100, y: Math.round(p.target.y * 100) / 100 } } : null;
   696→      },
   697→      pickAt: (x: number, y: number) => this.scene.pickLimb(x, y),
   698→      pickNearest: (x: number, y: number) => this.scene.pickNearestDebug(x, y),
   699→      handMeshes: () => this.scene.handMeshDebug(),
   700→      renderSize: () => this.scene.renderSizeDebug(),
   701→      project: (x: number, y: number, z: number) => this.scene.projectToScreen(x, y, z),
   702→      anchors: () => this.solver.anchorHoldIds().map((a) => ({ limb: a.limb, hold: a.holdId, load: Math.round(a.loadN) })),
   703→      reject: (limb: LimbId, holdId: string) => this.interaction?.rejectReason(limb, holdId) ?? "sin-interaccion",
   704→      chainInfo: (limb: LimbId) => {
   705→        const root = this.interaction ? this.interaction["world" as keyof object] : null;
   706→        const shoulder = limb === "hand_l" ? this.solver.getParticle(2) : limb === "hand_r" ? this.solver.getParticle(3) : limb === "foot_l" ? this.solver.getParticle(5) : this.solver.getParticle(6);
   707→        return { root: { x: +shoulder.pos.x.toFixed(2), y: +shoulder.pos.y.toFixed(2), z: +shoulder.pos.z.toFixed(2) } };
   708→      },
   709→      limbScreen: () => {
   710→        const out: { limb: string; x: number; y: number }[] = [];
   711→        for (const limb of LIMB_IDS) {
   712→          const p = this.solver.positionOfLimb(limb);
   713→          const s = this.scene.projectToScreen(p.x, p.y, p.z);
   714→          out.push({ limb, x: s.x, y: s.y });
   715→        }
   716→        return out;
   717→      }
   718→    };
   719→  }
