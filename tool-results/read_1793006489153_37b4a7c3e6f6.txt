   250→  static async create(canvas: HTMLCanvasElement): Promise<GameScene> {
   251→    // WebGL2 primero: es la ruta verificada en E2E y la universalmente soportada.
   252→    // WebGPU queda como respaldo (y en iframes cross-origin suele estar bloqueado
   253→    // por Permissions Policy, por lo que intentar primero WebGPU hacía fracasar
   254→    // el render en el preview).
   255→    let engine: AbstractEngine;
   256→    let backend = "WebGL 2";
   257→    try {
   258→      engine = new Engine(canvas, true, { stencil: true, preserveDrawingBuffer: false }, true);
   259→    } catch {
   260→      const gpu = new WebGPUEngine(canvas, { antialias: true, stencil: true });
   261→      await gpu.initAsync();
   262→      engine = gpu;
   263→      backend = "WebGPU";
   264→    }
   265→    const scene = new Scene(engine);
   266→    scene.useRightHandedSystem = false;
   267→    return new GameScene(engine, scene, backend, canvas);
   268→  }

   272→  buildWorld(wall: WallDefinition): void {
   273→    for (const hold of this.holdMeshes.values()) hold.dispose();
   274→    this.holdMeshes.clear();
   275→    this.holdBaseColor.clear();
   276→    this.holdRadius.clear();
   277→    for (const m of this.wallMeshes) m.dispose();
   278→    this.wallMeshes = [];
   279→
   280→    const d = wall.dimensions;
   281→    // Marcos por segmento (una sola franja vertical si el JSON no declara segments).
   282→    this.wallFrames = computeWallFrames(wall);
   283→    this.wallTopY = framesTopY(this.wallFrames);
   284→    this.wallHalfWidth = d.width / 2;
   285→    // Camara: mas lejos cuanto mas techo (el escalador avanza HACIA la camara) y
   286→    // cuanto mas alta la pared (se ve mas recorrido de una vez).
   287→    const roof = framesRoofFactor(this.wallFrames);
   288→    this.camBaseDist = Math.min(10.6, 7.8 + 2.3 * roof + Math.max(0, this.wallTopY - 6) * 0.28);
   289→    this.camDist = this.camBaseDist;
   290→    this.camMode = "play";
   291→    this.camModeLeft = 0;
   292→    const wallMat = wallMaterial(this.scene);
   293→    for (const frame of this.wallFrames) {
   294→      const segMesh = MeshBuilder.CreateBox(`wall_seg_${frame.index}`, { width: d.width, height: frame.height, depth: d.depth }, this.scene);
   295→      segMesh.position = new Vector3(frame.center.x, frame.center.y, frame.center.z);
   296→      segMesh.rotation.x = (frame.tiltDeg * Math.PI) / 180;
   297→      segMesh.isPickable = true;
   298→      segMesh.name = "wall"; // el raycast de bloqueo y la distancia de pared lo buscan por nombre
   299→      segMesh.material = wallMat;
   300→      segMesh.receiveShadows = true;
   301→      this.wallMeshes.push(segMesh);
   302→    }
   303→
   304→    const floor = MeshBuilder.CreateGround("floor", { width: 12, height: 12 }, this.scene);
   305→    const floorMat = new StandardMaterial("floorMat", this.scene);
   306→    floorMat.diffuseColor = Color3.FromHexString("#23262d");
   307→    floorMat.specularColor = new Color3(0.05, 0.05, 0.05);
   308→    floor.material = floorMat;
   309→    floor.isPickable = true;
   310→    floor.receiveShadows = true;
   311→
   312→    for (const hold of wall.holds) {
   313→      this.addHoldMesh(hold);
   314→    }
   315→  }

   317→  private addHoldMesh(hold: HoldDefinition): void {
   318→    const mesh = MeshBuilder.CreateSphere(`hold_${hold.id}`, { diameter: hold.radius * 2, segments: 8 }, this.scene);
   319→    mesh.position = new Vector3(hold.position[0], hold.position[1], hold.position[2]);
   320→    mesh.scaling.z = 0.62;
   321→    mesh.isPickable = true;
   322→    mesh.metadata = { holdId: hold.id };
   323→    const color = Color3.FromHexString(HOLD_COLORS[hold.type] ?? "#8a97a8");
   324→    const mat = new StandardMaterial(`holdMat_${hold.id}`, this.scene);
   325→    mat.diffuseColor = color;
   326→    mat.specularColor = new Color3(0.15, 0.15, 0.15);
   327→    mesh.material = mat;
   328→    this.holdMeshes.set(hold.id, mesh);
   329→    this.holdBaseColor.set(hold.id, color);
   330→    this.holdRadius.set(hold.id, hold.radius);
   331→  }

   335→  spawnRagdollVisuals(rig: RagdollRig): void {
   336→    this.despawnRagdollVisuals();
   337→    this.currentRig = rig;
   338→    const scene = this.scene;
   339→    const bodyMat = sharedMaterial("bodyMat", "#4b5563", scene);
   340→    const headMat = sharedMaterial("headMat", "#9aa3af", scene);
   341→    const bagMat = sharedMaterial("bagMat", BAG_COLOR, scene);
   342→    const markMat = emissiveMaterial("markMat", Color3.FromHexString(MARKER_COLOR), scene);
   343→    // Materiales POR EXTREMIDAD: el tinte de zona (verde/ambar/rojo) se aplica
   344→    // de forma independiente a cada brazo y cada pierna.
   345→    const limbMats: Record<LimbId, StandardMaterial> = {
   346→      hand_l: sharedMaterial("limbMat_hand_l", SEG_BASE, scene),
   347→      hand_r: sharedMaterial("limbMat_hand_r", SEG_BASE, scene),
   348→      foot_l: sharedMaterial("limbMat_foot_l", SEG_BASE, scene),
   349→      foot_r: sharedMaterial("limbMat_foot_r", SEG_BASE, scene)
   350→    };
   351→    const segOfLimb: Record<LimbId, Mesh[]> = { hand_l: [], hand_r: [], foot_l: [], foot_r: [] };
   352→
   353→    const P = rig.particles;
   354→    const dist = (ia: number, ib: number): number =>
   355→      Vector3.Distance(
   356→        new Vector3(P[ia].pos.x, P[ia].pos.y, P[ia].pos.z),
   357→        new Vector3(P[ib].pos.x, P[ib].pos.y, P[ib].pos.z)
   358→      );
   359→    const seg = (name: string, ia: number, ib: number, radius: number, mat: StandardMaterial, limb?: LimbId): void => {
   360→      const rest = dist(ia, ib);
   361→      const mesh = MeshBuilder.CreateCapsule(name, { radius, height: rest + radius * 2, tessellation: 10, capSubdivisions: 6 }, scene);
   362→      mesh.material = mat;
   363→      mesh.isPickable = false;
   364→      mesh.rotationQuaternion = Quaternion.Identity();
   365→      this.boneMeshes.push({ a: ia, b: ib, mesh, rest });
   366→      this.ragdollRoot.push(mesh);
   367→      if (limb) segOfLimb[limb].push(mesh);
   368→    };
   369→
   370→    seg("torso", 1, 4, 0.155, bodyMat);
   371→    seg("pelvisBar", 5, 6, 0.095, bodyMat);
   372→    seg("shoulderBar", 2, 3, 0.075, bodyMat);
   373→    seg("neck", 1, 0, 0.055, bodyMat);
   374→    seg("upperArmL", 2, 7, 0.058, limbMats.hand_l, "hand_l");
   375→    seg("forearmL", 7, 9, 0.05, limbMats.hand_l, "hand_l");
   376→    seg("upperArmR", 3, 8, 0.058, limbMats.hand_r, "hand_r");
   377→    seg("forearmR", 8, 10, 0.05, limbMats.hand_r, "hand_r");
   378→    seg("thighL", 5, 11, 0.075, limbMats.foot_l, "foot_l");
   379→    seg("shinL", 11, 13, 0.062, limbMats.foot_l, "foot_l");
   380→    seg("thighR", 6, 12, 0.075, limbMats.foot_r, "foot_r");
   381→    seg("shinR", 12, 14, 0.062, limbMats.foot_r, "foot_r");
   382→
   383→    const head = MeshBuilder.CreateSphere("head", { diameter: 0.24, segments: 12 }, scene);
   384→    head.material = headMat;
   385→    head.isPickable = false;
   386→    this.boneMeshes.push({ a: 0, b: 0, mesh: head, rest: 1 });
   387→    this.ragdollRoot.push(head);
   388→
   389→    // --- Marcas de orientacion frente/espalda ---
   390→    // El cuerpo vive en un plano paralelo a la pared: espalda = -z (hacia la
   391→    // camara), frente = +z (hacia la pared). La mochila (grande, en -z) es la
   392→    // referencia de espalda; el visor facial y los discos de rodilla/codo (+z)
   393→    // marcan el frente. La sincronizacion aplica offZ/offX/offY en el mundo.
   394→    const torsoLen = dist(1, 4);
   395→    const bag = MeshBuilder.CreateBox("backpack", { width: 0.26, height: torsoLen * 0.84, depth: 0.105 }, scene);
   396→    bag.material = bagMat;
   397→    bag.isPickable = false;

   557→  /** Copia posiciones del solver a los meshes con interpolacion entre pasos. */
   558→  syncRagdoll(solver: PbdSolver, alpha: number): void {
   559→    void this.currentRig;
   560→    const tmpA = new Vector3();
   561→    const tmpB = new Vector3();
   562→    for (const bone of this.boneMeshes) {
   563→      const pa = solver.getParticle(bone.a);
   564→      const pb = solver.getParticle(bone.b);
   565→      if (!pa || !pb) continue;
   566→      if (bone.a === bone.b) {
   567→        tmpA.set(pa.pos.x, pa.pos.y, pa.pos.z);
   568→        bone.mesh.position.copyFrom(tmpA);
   569→        // Marcas puntuales (visor, ojos, etiquetas): desplazamiento pared-camara y lateral.
   570→        if (bone.offZ) bone.mesh.position.z += bone.offZ;
   571→        if (bone.offX) bone.mesh.position.x += bone.offX;
   572→        if (bone.offY) bone.mesh.position.y += bone.offY;
   573→        continue;
   574→      }
   575→      const ax = pa.prev.x + (pa.pos.x - pa.prev.x) * alpha;
   576→      const ay = pa.prev.y + (pa.pos.y - pa.prev.y) * alpha;
   577→      const az = pa.prev.z + (pa.pos.z - pa.prev.z) * alpha;
   578→      const bx = pb.prev.x + (pb.pos.x - pb.prev.x) * alpha;
   579→      const by = pb.prev.y + (pb.pos.y - pb.prev.y) * alpha;
   580→      const bz = pb.prev.z + (pb.pos.z - pb.prev.z) * alpha;
   581→      tmpA.set(ax, ay, az);
   582→      tmpB.set(bx, by, bz);
   583→      const mesh = bone.mesh;
   584→      mesh.position.set((ax + bx) / 2 + (bone.offX ?? 0), (ay + by) / 2 + (bone.offY ?? 0), (az + bz) / 2 + (bone.offZ ?? 0));
   585→      const dir = tmpB.subtract(tmpA);
   586→      const len = Math.max(dir.length(), 1e-5);
   587→      const d = dir.scale(1 / len);
   588→      const ref = Math.abs(d.y) < 0.95 ? new Vector3(0, 1, 0) : new Vector3(1, 0, 0);
   589→      const xAxis = Vector3.Cross(ref, d).normalize();
   590→      const zAxis = Vector3.Cross(xAxis, d).normalize();
   591→      if (!mesh.rotationQuaternion) mesh.rotationQuaternion = Quaternion.Identity();
   592→      Quaternion.RotationQuaternionFromAxisToRef(xAxis, d, zAxis, mesh.rotationQuaternion);
   593→      mesh.scaling.y = Math.min(1.35, len / bone.rest);
   594→    }
   595→  }

   703→  /** Tinte de zona por extremidad (9.5 + retroalimentacion del HUD): la
   704→   * extremidad fijada se tine con la paleta del panel lateral (verde normal,
   705→   * ambar tension, rojo sobrecarga con pulso); libre, color base neutro.
   706→   * Aplica al agarre (mano/pie) y a los dos segmentos de su cadena.
   707→   * urgency (0..1): presion de fallo del temporizador de sobrecarga — acelera
   708→   * el parpadeo y sube el brillo a medida que el agarre se acerca a ceder. */
   709→  updateLimbTint(limb: LimbId, anchored: boolean, zone: LoadZone, urgency = 0): void {
   710→    const visual = this.limbVisual[limb];
   711→    if (!visual?.hand || visual.hand.isDisposed()) return;
   712→    const zoneColor = ZONE_COLORS[zone] ?? ZONE_COLORS.NORMAL;
   713→    const overloading = zone === "OVERLOADED" || zone === "ERROR_GUARD";
   714→    const u = overloading ? Math.min(1, Math.max(0, urgency)) : 0;
   715→    const period = 700 - 380 * u;
   716→    const wave = 0.5 + 0.5 * Math.sin((Date.now() % period) / period * Math.PI * 2);
   717→    const glow = anchored ? (overloading ? 0.2 + 0.4 * wave + 0.3 * u : 0.18) : 0;
   718→    const tint = anchored ? Color3.Lerp(visual.base, zoneColor, 0.8) : visual.base;
   719→    const segTint = anchored ? Color3.Lerp(visual.segBase, zoneColor, 0.62) : visual.segBase;
   720→    const handMat = visual.hand.material as StandardMaterial;
   721→    handMat.diffuseColor = tint;
   722→    handMat.emissiveColor = zoneColor.scale(glow);
   723→    for (const segMesh of visual.segments) {
   724→      if (segMesh.isDisposed()) continue;
   725→      const m = segMesh.material as StandardMaterial;
   726→      m.diffuseColor = segTint;
   727→      m.emissiveColor = zoneColor.scale(glow * 0.6);
   728→    }
   729→  }

   731→  /** Anillo sobre la presa anclada de una extremidad: hereda su zona de carga
   732→   * (verde/ambar/rojo) con pulso en sobrecarga, para saber que apoyo esta
   733→   * comprometido sin mirar el panel lateral. La urgencia (0..1) acelera el
   734→   * pulso y lo agranda: la presa "late" mas fuerte cuanto antes cedera. */
   735→  setAnchorRing(limb: LimbId, holdId: string | null, zone: LoadZone, urgency = 0): void {
   736→    const ring = this.anchorRings.get(limb);
   737→    if (!ring) return;
   738→    const holdMesh = holdId ? this.holdMeshes.get(holdId) : undefined;
   739→    if (!holdId || !holdMesh) {
   740→      ring.setEnabled(false);
   741→      return;
   742→    }
   743→    const r = this.holdRadius.get(holdId) ?? 0.14;
   744→    const overloading = zone === "OVERLOADED" || zone === "ERROR_GUARD";
   745→    const u = overloading ? Math.min(1, Math.max(0, urgency)) : 0;
   746→    const period = 700 - 380 * u;
   747→    const wave = 0.5 + 0.5 * Math.sin((Date.now() % period) / period * Math.PI * 2);
   748→    const zoneColor = ZONE_COLORS[zone] ?? ZONE_COLORS.NORMAL;
   749→    const mat = ring.material as StandardMaterial;
   750→    if (overloading) {
   751→      mat.emissiveColor = Color3.Lerp(zoneColor, Color3.FromHexString("#ffffff"), 0.4 * wave + 0.3 * u);
   752→    } else {
   753→      mat.emissiveColor = zoneColor;
   754→    }
   755→    ring.scaling.setAll(((r + 0.05) / 0.18) * (overloading ? 1 + (0.1 + 0.2 * u) * wave : 1));
   756→    // La presa es una esfera aplanada en z (0.62): el anillo flota justo delante
   757→    // de su cara frontal; mano y pie apilados en la misma presa van a cotas distintas.
   758→    const zOff = r * 0.62 + 0.03 + (limb.startsWith("foot") ? 0.035 : 0);
   759→    ring.setEnabled(true);
   760→    ring.position.set(holdMesh.position.x, holdMesh.position.y, holdMesh.position.z - zOff);
   761→  }

   801→  private chainRootPosition(limb: LimbId, solver: PbdSolver): Vector3 {

   815→  private burst(
   816→    name: string,
   817→    emitter: Vector3,
   818→    count: number,
   819→    c1: Color4,
   820→    c2: Color4,
   821→    dir1: Vector3,
   822→    dir2: Vector3,
   823→    opts: { sizeMin: number; sizeMax: number; lifeMin: number; lifeMax: number; gravity?: Vector3; emitTime?: number; scaleGravity?: number }
   824→  ): void {
   825→    const ps = new ParticleSystem(name, Math.max(32, count + 16), this.scene);

   828→    ps.minEmitBox = new Vector3(-0.03, -0.03, -0.03);
   829→    ps.maxEmitBox = new Vector3(0.03, 0.03, 0.03);
   830→    ps.color1 = c1;
   831→    ps.color2 = c2;
   832→    ps.colorDead = new Color4(c1.r, c1.g, c1.b, 0);
   833→    ps.minSize = opts.sizeMin;
   834→    ps.maxSize = opts.sizeMax;
   835→    ps.minLifeTime = opts.lifeMin;
   836→    ps.maxLifeTime = opts.lifeMax;
   837→    ps.emitRate = count / Math.max(0.04, opts.emitTime ?? 0.1);
   838→    ps.direction1 = dir1;
   839→    ps.direction2 = dir2;
   840→    if (opts.gravity) ps.gravity = opts.gravity;
   841→    ps.blendMode = ParticleSystem.BLENDMODE_STANDARD;
   842→    ps.minAngularSpeed = -Math.PI;
   843→    ps.maxAngularSpeed = Math.PI;
   844→    ps.updateSpeed = 0.012;
   845→    ps.targetStopDuration = opts.emitTime ?? 0.1;
   846→    ps.disposeOnStop = true;
   847→    ps.start();
   848→  }

   905→  // --- Camara ---
   906→
   907→  /**
   908→   * Camara de seguimiento cinematografico (peticion del jugador: "una camara que
   909→   * permita un seguimiento muy bueno y espectacular"):
   910→   * - sigue la pelvis con zona muerta (x/y) y mirada adelantada (+0.35 arriba);
   911→   * - distancia ADAPTATIVA: se aleja con la velocidad (pendulos y saltos entran en
   912→   *   cuadro), se acerca en caida (drama, junto al slow-mo) y al completar;
   913→   * - retrocede cuando el escalador avanza bajo un techo hacia la camara (circuito);
   914→   * - respiracion sutil en reposo y sacudida de impacto sobre el punto de mira.
   915→   * Sin balanceo lateral ni roll: el picking en espacio de pantalla debe ser estable.
   916→   */
   917→  updateCamera(dt: number, pelvis: { x: number; y: number; z: number }, recenter: boolean, mode: "play" | "fallen" | "success" = "play", aiming = false): void {
   918→    if (mode !== "play") {
   919→      this.camMode = mode;
   920→      this.camModeLeft = 1.5;
   921→    } else if (this.camModeLeft > 0) {
   922→      this.camModeLeft -= dt;
   923→      if (this.camModeLeft <= 0) this.camMode = "play";
   924→    }
   925→
   926→    const dPelvis = Math.hypot(pelvis.x - this.lastPelvis.x, pelvis.y - this.lastPelvis.y, pelvis.z - this.lastPelvis.z);
   927→    // Guarda finita: una pelvis no finita congela el seguimiento (nunca deriva la
   928→    // camara hacia el vacio); las guardas de mundo de loop.ts reinician la prueba.
   929→    const pelvisFinite = Number.isFinite(pelvis.x + pelvis.y + pelvis.z);
   930→    const speed = pelvisFinite ? dPelvis / Math.max(1e-4, dt) : 0;
   931→    this.lastPelvis = { x: pelvis.x, y: pelvis.y, z: pelvis.z };
   932→
   933→    const deadzoneHalf = 0.55;
   934→    const target = this.camFollow;
   935→    if (recenter) {
   936→      target.copyFrom(this.camTargetBase);
   937→      this.camDist = this.camBaseDist;
   938→    } else if (pelvis && pelvisFinite) {
   939→      const yCap = Math.max(4.2, this.wallTopY + 0.6);
   940→      const xCap = Math.max(1.6, this.wallHalfWidth - 0.45);
   941→      const desiredY = Math.min(yCap, Math.max(1.0, pelvis.y));
   942→      const dy = desiredY - target.y;
   943→      if (Math.abs(dy) > deadzoneHalf) {
   944→        target.y += (dy - Math.sign(dy) * deadzoneHalf) * Math.min(1, dt * 3);
   945→      }
   946→      const desiredX = Math.min(xCap, Math.max(-xCap, pelvis.x));
   947→      const dx = desiredX - target.x;
   948→      if (Math.abs(dx) > deadzoneHalf) {
   949→        target.x += (dx - Math.sign(dx) * deadzoneHalf) * Math.min(1, dt * 3);
   950→      }
   951→      // Bajo techos el cuerpo avanza hacia la camara: el punto de mira lo sigue en z.
   952→      const desiredZ = Math.min(0.3, Math.max(-2.6, pelvis.z * 0.55));
   953→      target.z += (desiredZ - target.z) * Math.min(1, dt * 2.2);
   954→      // TOPE DURO: la zona muerta puede dejar el target a 0,55 del deseado, pero el
   955→      // target JAMAS sale del cuadro razonable de la pared (la camara nunca pierde
   956→      // al personaje, pase lo que pase con la fisica).
   957→      target.y = Math.min(yCap, Math.max(1.0, target.y));
   958→      target.x = Math.min(xCap, Math.max(-xCap, target.x));
   959→    }
   960→    if (!Number.isFinite(target.y)) target.y = this.camTargetBase.y;
   961→    if (!Number.isFinite(target.x)) target.x = this.camTargetBase.x;
   962→    // Clamps de seguridad permanentes (fuera del if: tambien tras recenter).
   963→    target.y = Math.min(Math.max(4.2, this.wallTopY + 0.6), Math.max(1.0, target.y));
   964→    target.x = Math.min(Math.max(1.6, this.wallHalfWidth - 0.45), Math.max(-Math.max(1.6, this.wallHalfWidth - 0.45), target.x));
   965→
   966→    // Distancia adaptativa + posicion (suavizadas). MIENTRAS SE ARRASTRA la
   967→    // posicion queda CONGELADA: el jugador esta apuntando con el cursor y el
   968→    // mapeo cursor->presa no puede moverse bajo sus pies (control determinista).
   969→    const zoomGoal = this.camMode === "fallen" ? -1.4 : this.camMode === "success" ? -0.85 : Math.min(1.15, speed * 0.13);
   970→    const distGoal = Math.min(12, Math.max(5.2, this.camBaseDist + (Number.isFinite(zoomGoal) ? zoomGoal : 0)));
   971→    this.camDist += (distGoal - this.camDist) * Math.min(1, dt * (this.camMode === "play" ? 2.6 : 4.2));
   972→    if (!Number.isFinite(this.camDist)) this.camDist = this.camBaseDist;
   973→    const t = performance.now() / 1000;
   974→    if (!aiming) {
   975→      const roofPull = Math.max(0, -target.z) * 0.85;
   976→      const breathe = Math.sin(t * 0.45) * 0.035;
   977→      const desiredPos = new Vector3(
   978→        target.x * 0.22,
   979→        target.y + 1.12 + breathe,
   980→        -(this.camDist) - roofPull
   981→      );
   982→      this.camera.position.addInPlace(desiredPos.subtract(this.camera.position).scale(Math.min(1, dt * 3.4)));
   983→      // Recuperacion: si la posicion llego a ser no finita (causalidad rota en el
   984→      // render), vuelve al encuadre deseado en vez de dejar el mundo invisible.
   985→      if (!Number.isFinite(this.camera.position.x + this.camera.position.y + this.camera.position.z)) {
   986→        this.camera.position.copyFrom(desiredPos);
   987→      }
   988→    }
   989→
   990→    // La mutacion de camera.target no reorienta a una FreeCamera: hay que llamar a
   991→    // setTarget. Mirada adelantada (ves a donde vas) + sacudida de impacto. El
   992→    // look es un vector NUEVO: lo que setTarget guarde en camera.target nunca
   993→    // alimenta el seguimiento (el estado es camFollow).
   994→    const look = new Vector3(target.x, target.y + 0.35, target.z);
   995→    if (this.shakeAmp > 0.0006) {
   996→      look.x += Math.sin(t * 47) * this.shakeAmp;
   997→      look.y += Math.cos(t * 39) * this.shakeAmp * 0.7;
   998→      this.camera.setTarget(look);
   999→      this.shakeAmp *= Math.exp(-dt * 5.2);
  1000→    } else {
  1001→      this.shakeAmp = 0;
  1002→      this.camera.setTarget(look);
  1003→    }
  1004→
  1005→    if (this.ortho) {
  1006→      const aspect = this.engine.getRenderWidth() / Math.max(1, this.engine.getRenderHeight());
  1007→      const halfH = 3.1;
  1008→      this.camera.orthoLeft = -halfH * aspect;
  1009→      this.camera.orthoRight = halfH * aspect;
  1010→      this.camera.orthoBottom = -halfH;
  1011→      this.camera.orthoTop = halfH;
  1012→    }
  1013→  }
  1014→

  1018→    return this.ortho;

  1118→  // --- Picking ---
  1119→  // (scene.pick/createPickingRay reciben pixeles CSS: el motor ya aplica el DPR.
  1120→  //  Verificado con barrido de factores a DPR 1.5: el factor 1 acierta la presa.)
  1121→
  1122→  /** Radio base de iman en pixeles CSS por tipo de extremidad: los pies, mas grandes
  1123→   * (son los mas dificiles de clicar: queja del usuario de agilidad). El radio real
  1124→   * crece con la VELOCIDAD de la extremidad (abajo): en pendulo se hace mas facil de
  1125→   * coger, no mas dificil. */
  1126→  private static readonly PICK_MAGNET_PX: Record<string, number> = { hand: 54, foot: 68 };
  1127→  /** Adelanto del iman predictivo (s): proyecta tambien donde VA a estar la extremidad. */
  1128→  private static readonly PICK_LEAD_S = 0.12;
  1129→  /** Ganancia del radio por velocidad (m/s): 1.5 m/s => +33% de radio; tope +90%. */
  1130→  private static readonly PICK_SPEED_GAIN = 0.22;
  1131→  private static readonly PICK_SPEED_MAX_BONUS = 0.9;
  1132→
  1133→  /** Posicion de picking/visual de una extremidad: cuando mano y pie comparten presa
  1134→   * (maxOccupants 2) coinciden en el mundo y seria imposible elegir; el pie se marca
  1135→   * 7 cm por debajo y la mano 3 cm por encima (apilado legible y seleccionable). */
  1136→  limbPickPos(limb: LimbId, solver: PbdSolver): { x: number; y: number; z: number } {
  1137→    const p = solver.positionOfLimb(limb);
  1138→    const holdId = solver.getAnchor(limb)?.holdId;
  1139→    if (!holdId) return { x: p.x, y: p.y, z: p.z };
  1140→    const shared = LIMB_IDS.some((other) => other !== limb && solver.getAnchor(other)?.holdId === holdId);
  1141→    if (!shared) return { x: p.x, y: p.y, z: p.z };
  1142→    const dy = limb.startsWith("foot") ? -0.07 : 0.03;
  1143→    return { x: p.x, y: p.y + dy, z: p.z };
  1144→  }
  1145→
  1146→  /** Seleccion de extremidad (7): 1) raycast exacto contra el mesh; 2) iman PREDICTIVO
  1147→   * en espacio de pantalla: por cada extremidad se acepta la MAS CERCANA de dos
  1148→   * candidatas — su posicion actual y donde VA A ESTAR dentro de PICK_LEAD_S (velocidad
  1149→   * del solver) — dentro de un radio que crece con su velocidad. Asi coger una
  1150→   * extremidad en pleno pendulo es tan facil como coger una parada (peticion del
  1151→   * jugador: "complicado seleccionar cuando se mueve mucho"). La pared bloquea el
  1152→   * raycast directo (AT-03); el iman solo alcanza extremidades visibles. */
  1153→  pickLimb(x: number, y: number, solver?: PbdSolver): LimbId | null {
  1154→    const pick = this.scene.pick(
  1155→      x,
  1156→      y,
  1157→      (m) => !!(m.metadata as { limbId?: LimbId } | undefined)?.limbId,
  1158→      false,
  1159→      this.camera
  1160→    );
  1161→    if (pick.hit && pick.pickedMesh) {
  1162→      const wallPick = this.scene.pick(x, y, (m) => m.name === "wall", false, this.camera);
  1163→      if (!(wallPick.hit && wallPick.distance < pick.distance - 0.15)) {
  1164→        const meta = pick.pickedMesh.metadata as { limbId?: LimbId };
  1165→        if (meta.limbId) return meta.limbId;
  1166→      }
  1167→    }
  1168→    if (!solver) return null;
  1169→    // Iman predictivo: proyecta posicion actual y futura de las 4 extremidades y
  1170→    // elige la mas cercana al cursor (px CSS), con radio escalado por velocidad.
  1171→    let best: LimbId | null = null;
  1172→    let bestDist = Infinity;
  1173→    for (const limb of LIMB_IDS) {
  1174→      const pp = this.limbPickPos(limb, solver);
  1175→      const vel = solver.velocityOfLimb(limb);
  1176→      const speed = Math.hypot(vel.x, vel.y, vel.z);
  1177→      const kind = limb.startsWith("hand") ? "hand" : "foot";
  1178→      const radius = GameScene.PICK_MAGNET_PX[kind] * (1 + Math.min(speed * GameScene.PICK_SPEED_GAIN, GameScene.PICK_SPEED_MAX_BONUS));
  1179→      const sNow = this.projectToScreenCss(pp.x, pp.y, pp.z);
  1180→      let d = Math.hypot(sNow.x - x, sNow.y - y);
  1181→      // Si el cursor no alcanza la posicion actual, prueba donde VA a estar
  1182→      // (adelanto corto): un pendulo se "coge" adelantandose a su arco.
  1183→      if (d > radius && speed > 1e-6) {
  1184→        const sNext = this.projectToScreenCss(
  1185→          pp.x + vel.x * GameScene.PICK_LEAD_S,
  1186→          pp.y + vel.y * GameScene.PICK_LEAD_S,
  1187→          pp.z + vel.z * GameScene.PICK_LEAD_S
  1188→        );
  1189→        d = Math.hypot(sNext.x - x, sNext.y - y);
  1190→      }
  1191→      if (d <= radius && d < bestDist) {
  1192→        bestDist = d;
  1193→        best = limb;
  1194→      }

  1199→  /** Proyeccion del cursor sobre el plano paralelo a la pared a la profundidad de agarre (7.1).
  1200→   * NOTA: scene.pick/createPickingRay esperan pixeles CSS del canvas (el motor aplica
  1201→   * internamente el devicePixelRatio con adaptToDeviceRatio=true; verificado con un
  1202→   * barrido de factores a DPR 1.5). NO convertir a pixeles de render aqui. */
  1203→  wallPlanePoint(x: number, y: number, zPlane: number): Vector3 | null {
  1204→    const ray = this.scene.createPickingRay(x, y, Matrix.Identity(), this.camera);
  1205→    if (Math.abs(ray.direction.z) < 1e-6) return null;
  1206→    const t = (zPlane - ray.origin.z) / ray.direction.z;
  1207→    if (t <= 0) return null;
  1208→    return ray.origin.add(ray.direction.scale(t));
  1209→  }

  1211→  /**
  1212→   * Proyeccion del cursor al PLANO DE PRESAS del segmento que gobierna `ref`
  1213→   * (posicion de la raiz de la extremidad arrastrada). En circuito inclinado cada
  1214→   * segmento tiene su propio plano: es lo que hace posible arrastrar bajo un techo.
  1215→   */
  1216→  wallPlanePointNear(x: number, y: number, ref: { x: number; y: number; z: number }): Vector3 | null {
  1217→    if (this.wallFrames.length === 0) return null;
  1218→    const frame = segmentAtPoint(this.wallFrames, ref);
  1219→    const n = frame.normal;
  1220→    const f = facePoint(frame);
  1221→    const holdOff = 0.055; // las presas sobresalen 5,5 cm de la cara (z=-0.18 en vertical)
  1222→    const px = f.x + n.x * holdOff;
  1223→    const py = f.y + n.y * holdOff;
  1224→    const pz = f.z + n.z * holdOff;
  1225→    const ray = this.scene.createPickingRay(x, y, Matrix.Identity(), this.camera);
  1226→    const denom = ray.direction.x * n.x + ray.direction.y * n.y + ray.direction.z * n.z;
  1227→    if (Math.abs(denom) < 1e-5) return null;
  1228→    const t = ((px - ray.origin.x) * n.x + (py - ray.origin.y) * n.y + (pz - ray.origin.z) * n.z) / denom;
  1229→    if (t <= 0.05) return null;
  1230→    return ray.origin.add(ray.direction.scale(t));
  1231→  }

  1276→  /** Integracion de escombros (llamar cada frame de render con dt real). */
  1277→  tickFx(dt: number): void {
  1278→    for (let i = this.debris.length - 1; i >= 0; i--) {
  1279→      const d = this.debris[i];
  1280→      d.life -= dt;
  1281→      // Gravedad suave + drag de papel: el carton planea un poco antes de caer.
  1282→      d.vel.y -= 3.6 * dt;
  1283→      d.vel.scaleInPlace(Math.max(0, 1 - 0.55 * dt));
  1284→      d.mesh.position.addInPlace(d.vel.scale(dt));
  1285→      if (d.mesh.position.y < 0.03) {
  1286→        d.mesh.position.y = 0.03;
  1287→        d.vel.y = Math.abs(d.vel.y) * 0.25;
  1288→        d.vel.scaleInPlace(0.7);
  1289→      }
  1290→      const q = d.mesh.rotationQuaternion ?? Quaternion.Identity();
  1291→      d.rot.addInPlace(d.spin.scale(dt));
  1292→      Quaternion.RotationYawPitchRollToRef(d.rot.x, d.rot.y, d.rot.z, q);
  1293→      d.mesh.rotationQuaternion = q;
  1294→      if (d.life < 0.45) d.mat.alpha = Math.max(0, d.life / 0.45);
  1295→      if (d.life <= 0) {
  1296→        d.mesh.dispose();
  1297→        this.debris.splice(i, 1);
  1298→      }
  1299→    }
  1300→  }

  1323→  private static circlePoints(radius: number, segments: number, center?: Vector3): Vector3[] {
  1324→    const pts: Vector3[] = [];
  1325→    const c = center ?? Vector3.Zero();
  1326→    for (let i = 0; i <= segments; i++) {
  1327→      const a = (i / segments) * Math.PI * 2;
  1328→      pts.push(new Vector3(c.x + Math.cos(a) * radius, c.y + Math.sin(a) * radius, c.z));

  1334→function wallMaterial(scene: Scene): StandardMaterial {
