   320→  async loadChallenge(id: string): Promise<void> {
   321→    this.stopTutorial(false);
   322→    this.challenge = this.repo.getChallenge(id);
   323→    this.character = this.repo.getCharacter(this.challenge.characterId);
   324→    this.profile = this.repo.getBodyProfile(this.character.bodyProfileId);
   325→    this.wall = this.repo.getWall(this.challenge.wallId);
   326→    this.holdPosById = new Map(this.wall.holds.map((h) => [h.id, h.position] as const));
   327→    this.bio = new BiomechanicsSystem(this.character, this.profile);
   328→    this.evaluator = new ChallengeEvaluator(this.challenge);
   329→
   330→    const strengthFactor = lerp(this.profile.biomech.strengthFactorMin, this.profile.biomech.strengthFactorMax, this.character.virtues.strength);
   331→    this.anchorMaxForce = this.character.physical.baseGripForce * strengthFactor;
   332→    this.followForce = this.character.physical.baseGripForce * this.character.control.gripAssist * this.character.control.targetFollowStrength;
   333→
   334→    this.solver = new PbdSolver({
   335→      gravity: v3(...this.profile.sim.gravity),
   336→      damping: this.profile.sim.damping,
   337→      iterations: this.profile.sim.iterations,
   338→      floorY: 0,
   339→      wallFrontZ: this.wall.transform.position[2] - this.wall.dimensions.depth / 2,
   340→      tremorAmplitude: this.profile.sim.tremorAmplitude,
   341→      loadEstimatorGain: this.profile.biomech.loadEstimatorGain
   342→    });
   343→
   344→    this.scene.buildWorld(this.wall);
   345→    this.hud.setChallenge(this.challenge, CHALLENGE_META[id]?.label ?? CHALLENGE_LABELS[this.challenge.type]);
   346→    // Chip permanente de campeonato (panel superior izquierdo) mientras se juega una etapa.
   347→    const stage = stageOf(id);
   348→    this.hud.setChampionshipStatus(
   349→      stage ? `Campeonato · etapa ${stage.index}/5 · ${stage.name} (${stage.heightM} m)` : null
   350→    );
   351→    this.buildInteraction();
   352→    this.spawn();
   353→  }

   356→   * Recreacion del estado inicial (10.2): libera constraints, destruye y recrea el

   360→  private spawn(): void {
   361→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   362→    this.interaction?.cancelDrag();
   363→    this.interaction?.cancelAssistAll();
   364→    this.keyboardLimb = null;
   365→    this.grabOffset = null;
   366→    this.scene.clearHoldHighlights();
   367→    for (const hold of this.wall.holds) this.scene.setHoldOccupied(hold.id, false);
   368→
   369→    this.scene.despawnRagdollVisuals();
   370→    this.rig = spawnRagdoll(this.solver, this.profile, this.character, this.wall, this.challenge, holdById, (limb) => this.anchorMaxForceFor(limb));
   371→    this.scene.spawnRagdollVisuals(this.rig);
   372→
   373→    this.bio.reset();
   374→    this.evaluator?.reset();
   375→    this.gameTime = 0;
   376→    this.accumulator = 0;
   377→    this.guardActive = false;
   378→    this.lastSnapshot = { stability: 0, com: { x: 0, y: 0, z: 0 }, supportCount: 0 };
   379→
   380→    // El combo de flujo se rompe con cada respawn (intento nuevo); el conteo de
   381→    // caidas persiste para la puntuacion del desafio cargado.
   382→    this.flowCount = 0;
   383→    this.flowBonus = 0;
   384→    this.peakFatigue = 0;
   385→    this.timeScale = 1;
   386→    this.slowmoLeft = 0;
   387→    this.fallLandedFx = false;

   389→    this.lastWorst = null;

   391→    this.overloadPressure = 0;
   392→
   393→    // Marcar presas ocupadas por los anclajes iniciales creados en spawnRagdoll.
   394→    for (const req of this.challenge.start.requiredAnchors) {
   395→      this.scene.setHoldOccupied(req.holdId, true);
   396→    }
   397→
   398→    this.settleTimer = this.profile.sim.resetSettleSeconds;
   399→    this.autoStartPending = this.challenge.start.autoStart;

   401→    this.fallAutoReset = -1;

   405→    this.applyTuneToSolver();

   407→  }

   416→  /** Aplica un preset del menu, lo persiste y lo marca como activo en el HUD. */
   417→  applyPreset(name: BalancePresetName): void {
   418→    // setTune limpia el preset activo (cualquier ajuste pasa por el); volver a
   419→    // marcar aqui deja el estado coherente: preset aplicado y persistido.
   420→    this.setTune({ ...PRESET_TUNES[name] });
   421→    this.presetName = name;
   422→    try {
   423→      window.localStorage.setItem(GameApp.PRESET_KEY, name);
   424→    } catch {
   425→      /* sin persistencia */
   426→    }
   427→    this.hud.setPresetActive(name);
   428→    this.debug.setTuneValues({ postureTone: this.tune.postureTone, jointLimitStiffness: this.tune.jointLimitStiffness, foldStrength: this.tune.foldStrength, stretchTolerance: this.tune.stretchTolerance, twistLimitRad: this.tune.twistLimitRad });
   429→    this.events.emit({ type: "Message", text: `Balance ${name}: tono ${this.tune.postureTone.toFixed(2)}, límites ${this.tune.jointLimitStiffness.toFixed(2)}, pliegue ${this.tune.foldStrength.toFixed(2)}, estiramiento ×${this.tune.stretchTolerance.toFixed(2)}`, tone: "info" });
   430→  }

   432→  private loadTune(): void {
   433→    try {
   434→      const preset = window.localStorage.getItem(GameApp.PRESET_KEY) as BalancePresetName | null;
   435→      if (preset && preset in PRESET_TUNES) this.presetName = preset;
   436→    } catch {
   437→      this.presetName = null;
   438→    }
   439→    if (this.presetName) {
   440→      this.tune = { ...PRESET_TUNES[this.presetName] };
   441→      return;
   442→    }
   443→    try {
   444→      const raw = window.localStorage.getItem(GameApp.TUNE_KEY);
   445→      if (raw) {
   446→        const t = JSON.parse(raw) as Partial<BalanceTuning>;
   447→        this.tune = {
   448→          jointLimitStiffness: clamp01(t.jointLimitStiffness ?? 1.0),
   449→          foldStrength: clamp01(t.foldStrength ?? 0.85),
   450→          postureTone: clamp01(t.postureTone ?? 0.75),
   451→          toneFloor: clamp01(t.toneFloor ?? 0.12),
   452→          stretchTolerance: clampRange(t.stretchTolerance ?? 1.1, 1.02, 1.35),
   453→          twistLimitRad: clampRange(t.twistLimitRad ?? 0.45, 0, 1.2)
   454→        };
   455→      }
   456→    } catch {
   457→      /* sin persistencia: defaults */
   458→    }
   459→  }

   466→    }

   472→      foldStrength: clamp01(patch.foldStrength ?? this.tune.foldStrength),
   473→      postureTone: clamp01(patch.postureTone ?? this.tune.postureTone),
   474→      toneFloor: clamp01(patch.toneFloor ?? this.tune.toneFloor),
   475→      stretchTolerance: clampRange(patch.stretchTolerance ?? this.tune.stretchTolerance, 1.02, 1.35),
   476→      twistLimitRad: clampRange(patch.twistLimitRad ?? this.tune.twistLimitRad, 0, 1.2)
   477→    };
   478→    // Las reglas de postura imposible siguen al tono: preset realista = estricto.
   479→    this.postureRules = postureRulesForTone(this.tune.postureTone);
   480→    // Un ajuste manual (sliders F3 o __NS.setTune) deja de corresponder al preset.

   485→      /* sin persistencia */
   486→    }
   487→    this.hud?.setPresetActive(null);

   489→    this.saveTune();

   579→  private buildInteraction(): void {
   580→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   581→    const world: InteractionWorld = {
   582→      holdById,
   583→      challenge: this.challenge,
   584→      character: this.character,
   585→      profile: this.profile,
   586→      wall: this.wall,
   587→      anchorHoldOf: (limb) => this.solver.getAnchor(limb)?.holdId,
   588→      holdOccupancy: (holdId) => {
   589→        let n = 0;
   590→        for (const limb of LIMB_IDS) {
   591→          if (this.solver.getAnchor(limb)?.holdId === holdId) n++;
   592→          if (this.interaction?.assistHoldOf(limb) === holdId) n++;
   593→        }
   594→        return n;
   595→      },
   596→      chainRootOf: (limb) => {
   597→        const idx = limb === "hand_l" ? P.shL : limb === "hand_r" ? P.shR : limb === "foot_l" ? P.hipL : P.hipR;
   598→        const p = this.solver.getParticle(idx);
   599→        return v3(p.pos.x, p.pos.y, p.pos.z);
   600→      },
   601→      limbPositionOf: (limb) => this.solver.positionOfLimb(limb),
   602→      chainLengthOf: (limb) => {
   603→        if (!this.rig) return 0.6;
   604→        const base = limb.startsWith("hand") ? this.rig.armChain : this.rig.legChain;
   605→        return base * this.character.physical.reachScale;
   606→      },
   607→      postureRules: () => this.postureRules
   608→    };
   609→    this.interaction = new InteractionController(world, {
   610→      onSelectChanged: () => undefined,
   611→      onAccept: (limb, holdId) => {
   612→        const existing = this.solver.getAnchor(limb);
   613→        if (existing) {
   614→          this.solver.releaseAnchor(limb);
   615→          this.events.emit({ type: "GripReleased", limb, holdId: existing.holdId, reason: "PLAYER_RELEASE" });
   616→        }
   617→        const hold = holdById.get(holdId)!;
   618→        this.solver.setTarget(limb, v3(hold.position[0], hold.position[1], hold.position[2]), this.followForce);
   619→        this.refreshHoldOccupancy();
   620→      },
   621→      onReject: (limb, holdId, reason) => {
   622→        this.solver.clearTarget(limb);
   623→        this.events.emit({ type: "HoldRejected", limb, holdId, reason });
   624→        if (holdId) {
   625→          this.scene.setHoldInvalid(holdId, true);
   626→          window.setTimeout(() => this.scene.setHoldInvalid(holdId, false), 450);
   627→        }
   628→      },
   629→      onArrival: (limb, holdId) => {
   630→        this.solver.clearTarget(limb);
   631→        const hold = holdById.get(holdId)!;
   632→        this.solver.createAnchor(limb, holdId, v3(hold.position[0], hold.position[1], hold.position[2]), this.anchorMaxForceFor(limb));
   633→        // ADR-007: cada apoyo consolidado reactiva el impulso de subida (el cuerpo
   634→        // se reorganiza hacia la nueva postura: step-up del pie, pull de brazos).
   635→        this.solver.pulseClimbDrive();
   636→        this.events.emit({ type: "HoldAcquired", limb, holdId });
   637→        this.refreshHoldOccupancy();
   638→      }
   639→    });
   640→  }

   642→  private refreshHoldOccupancy(): void {
   643→    const anchored = new Set<string>();
   644→    for (const limb of LIMB_IDS) {
   645→      const a = this.solver.getAnchor(limb);
   646→      if (a) anchored.add(a.holdId);
   647→    }
   648→    for (const hold of this.wall.holds) {
   649→      this.scene.setHoldOccupied(hold.id, anchored.has(hold.id));
   650→    }
   651→  }

   653→  // --- Pipeline por paso fijo ---
   654→
   655→  private fixedStep(dt: number): void {
   656→    const result = this.solver.step(dt);
   657→
   658→    // ERROR GUARD (9.2): carga no finita / NaN / energia anomala.
   659→    if ((result.nanError || result.energySpike) && !this.guardActive) {
   660→      this.guardActive = true;
   661→      this.guardCooldown = 0.4;
   662→      this.lastGuardError = result.nanError ? "nan" : "energia";
   663→      this.events.emit({ type: "Message", text: "ERROR GUARD: simulacion no finita; reiniciando prueba", tone: "error" });
   664→      this.machine.transition("ERROR");
   665→    }
   666→    if (this.machine.state === "ERROR") {
   667→      this.guardCooldown -= dt;
   668→      if (this.guardCooldown <= 0) {
   669→        this.spawn();
   670→        return;
   671→      }
   672→      return;
   673→    }
   674→
   675→    const anchors = this.solver.anchorHoldIds();
   676→
   677→    // Biomecanica solo cuando el desafio esta activo (fatiga congelada en pausa/resultado).
   678→    if (this.machine.state === "PLAYING") {
   679→      const bioResult = this.bio.update(dt, result, anchors.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver.tremor);
   680→      // En tutorial la fatiga y las zonas se miden (numeros reales en el paso
   681→      // de vitals) pero la sobrecarga NO libera agarres: el onboarding no puede
   682→      // guiar hacia una caida por mover lento.
   683→      for (const limb of this.tutorial ? [] : bioResult.overloadReleases) {
   684→        const anchor = this.solver.releaseAnchor(limb);
   685→        if (anchor) {
   686→          this.events.emit({ type: "GripReleased", limb, holdId: anchor.holdId, reason: "OVERLOAD" });
   687→          this.refreshHoldOccupancy();
   688→        }
   689→      }
   690→      for (const limb of bioResult.strainStarted) this.events.emit({ type: "StrainStarted", limb });
   691→      for (const limb of bioResult.overloadStarted) this.events.emit({ type: "OverloadStarted", limb });
   692→
   693→      // Asistencia de viaje hacia la presa aceptada (7.2).
   694→      this.interaction?.tick(dt);
   695→
   696→      const anchorsNow = this.solver.anchorHoldIds();
   697→      const stability = computeStability(this.bio, result, anchorsNow.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver, this.profile);
   698→      this.lastSnapshot = {
   699→        stability,
   700→        com: result.com,
   701→        supportCount: anchorsNow.length
   702→      };
   703→
   704→      // Evaluacion del desafio (10.3): caida -> exito -> timeout.
   705→      // El timer no corre durante el tutorial: nadie deberia agotar el tiempo
   706→      // mientras aprende los controles.
   707→      if (!this.tutorial) this.gameTime += dt;
   708→      const outcome = this.evaluator!.tick({
   709→        dt,
   710→        gameTime: this.gameTime,
   711→        anchorsByLimb: Object.fromEntries(anchorsNow.map((a) => [a.limb, a.holdId])),
   712→        supportCount: anchorsNow.length,
   713→        stability,
   714→        pelvisY: this.solver.getParticle(P.pelvisC).pos.y,
   715→        outOfBounds: result.outOfBounds
   716→      });
   717→      if (outcome.kind === "fall") {
   718→        this.events.emit({ type: "FallStarted", reason: outcome.reason });
   719→        this.events.emit({ type: "ChallengeFailed", reason: "FALLEN" });
   720→        this.machine.transition("FALLEN");
   721→        this.beginFallSequence();
   722→      } else if (outcome.kind === "success") {
   723→        this.events.emit({ type: "ChallengeSuccess" });
   724→        this.machine.transition("SUCCESS");
   725→      } else if (outcome.kind === "timeout") {
   726→        this.events.emit({ type: "ChallengeFailed", reason: "TIMEOUT" });
   727→        this.machine.transition("TIMEOUT");
   728→      }
   729→    } else {
   730→      // Fuera de PLAYING la fatiga no cambia; sin tremor.
   731→      for (const limb of LIMB_IDS) this.solver.tremor[limb] = 0;
   732→      this.interaction?.tick(0);
   733→    }
   734→
   735→    // Secuencia de caida (ADR-008): en FALLEN los agarres se abren escalonados y el
   736→    // cuerpo cae al colchon; tras asentarse, reintento automatico (R lo adelanta).
   737→    if (this.machine.state === "FALLEN") {
   738→      this.fallTimer += dt;
   739→      while (this.fallReleaseQueue.length > 0 && this.fallTimer >= this.fallReleaseQueue[0].at) {
   740→        const next = this.fallReleaseQueue.shift()!;
   741→        const anchor = this.solver.releaseAnchor(next.limb);
   742→        if (anchor) {
   743→          this.events.emit({ type: "GripReleased", limb: next.limb, holdId: anchor.holdId, reason: "FALL" });
   744→          this.refreshHoldOccupancy();
   745→        }
   746→      }
   747→      // Impacto contra el colchon: polvo + sacudida (una sola vez por caida).
   748→      if (!this.fallLandedFx && this.fallTimer >= LANDING_FX_AT_S) {
   749→        this.fallLandedFx = true;
   750→        const pv = this.solver.getParticle(P.pelvisC);
   751→        this.scene.burstDust(new Vector3(pv.pos.x, 0.06, pv.pos.z), 42);
   752→        this.scene.shake(0.055);
   753→      }
   754→      if (this.fallAutoReset > 0) {
   755→        this.fallAutoReset -= dt;
   756→        if (this.fallAutoReset <= 0) {
   757→          this.fallAutoReset = -1;
   758→          this.spawn();
   759→        }
   760→      }
   761→    }
   762→  }
   763→
   764→  /**
   765→   * Arranque de la secuencia de caida: cancela interaccion pendiente, abre los
   766→   * agarres con retardos realistas (manos primero) y deja el cuerpo flacido.
   767→   */
   768→  private beginFallSequence(): void {
   769→    this.interaction?.cancelDrag();
   770→    this.interaction?.cancelAssistAll();
   771→    this.fallTimer = 0;
   772→    this.fallReleaseQueue = [];
   773→    this.fallLandedFx = false;
   774→    const hands = ["hand_l", "hand_r"] as const;
   775→    const feet = ["foot_l", "foot_r"] as const;
   776→    let n = 0;
   777→    for (const limb of hands) {
   778→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_HAND_S + n++ * FALL_RELEASE_STAGGER_S });
   779→    }
   780→    for (const limb of feet) {
   781→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_FOOT_S + n++ * FALL_RELEASE_STAGGER_S });
   782→    }
   783→    this.fallReleaseQueue.sort((a, b) => a.at - b.at);
   784→    this.fallAutoReset = FALL_AUTO_RESET_S;
   785→    this.solver.setToneTarget(this.tune.toneFloor);
   786→  }
   787→
   788→  // --- Bucle de render ---
   789→
   790→  private readonly frame = (nowMs: number): void => {
   791→    if (this.disposed) return;
   792→    this.raf = requestAnimationFrame(this.frame);
   793→    this.frameCount++;
   794→    const now = nowMs / 1000;
   795→    const dtFrame = Math.min(Math.max(0, now - this.lastFrame), 0.1);
   796→    this.lastFrame = now;
   797→
   798→    this.fpsFrames++;
   799→    this.fpsElapsed += dtFrame;
   800→    if (this.fpsElapsed >= 0.5) {
   801→      this.fps = this.fpsFrames / this.fpsElapsed;
   802→      this.fpsFrames = 0;
   803→      this.fpsElapsed = 0;
   804→    }
   805→
   806→    const state = this.machine.state;
   807→    const stepping = state !== "PAUSED" && state !== "LOADING" && !this.paused;
   808→    if (stepping) {
   809→      // Slow-motion cinematografico: la escala converge suave hacia el objetivo.
   810→      if (this.slowmoLeft > 0) this.slowmoLeft -= dtFrame;
   811→      const goalScale = this.slowmoLeft > 0 ? SLOWMO_SCALE : 1;
   812→      this.timeScale += (goalScale - this.timeScale) * Math.min(1, dtFrame * 9);
   813→      this.accumulator += dtFrame * this.timeScale;
   814→      let steps = 0;
   815→      const fixed = this.profile.sim.fixedDeltaSeconds;
   816→      while (this.accumulator >= fixed && steps < MAX_STEPS_PER_FRAME) {
   817→        const t0 = performance.now();
   818→        this.fixedStep(fixed);
   819→        this.physicsMs = lerp(this.physicsMs, performance.now() - t0, 0.15);
   820→        this.accumulator -= fixed;
   821→        steps++;
   822→      }
   823→      if (steps >= MAX_STEPS_PER_FRAME) this.accumulator = 0;
   824→    }
   825→
   826→    if (state === "RESETTING") {
   827→      this.settleTimer -= dtFrame;
   828→      if (this.settleTimer <= 0) {
   829→        this.machine.transition(this.autoStartPending ? "PLAYING" : "READY");
   830→        if (this.autoStartPending) {
   831→          this.events.emit({ type: "Message", text: "Recupera dos apoyos antes de perder el agarre", tone: "warn" });
   832→        }
   833→      }
   834→    }
   835→
   836→    const alpha = Math.min(1, this.accumulator / this.profile.sim.fixedDeltaSeconds);
   837→    this.scene.syncRagdoll(this.solver, alpha);
   838→    this.scene.syncJointDebug(this.solver);
   839→
   840→    // Anillo de seleccion + enfasis de etiqueta: extremidad arrastrada (firm) o bajo
   841→    // el cursor (iman sutil). La etiqueta IZQ/DER crece: siempre sabes QUE mueves.
   842→    let emphasisLimb: LimbId | null = null;
   843→    if (this.interaction?.dragging) {
   844→      this.scene.setSelectionRing(this.interaction.selected, this.solver, "grab");
   845→      emphasisLimb = this.interaction.selected;
   846→    } else {
   847→      const hover = this.pointerInside && (state === "READY" || state === "PLAYING")
   848→        ? this.scene.pickLimb(this.pointerX, this.pointerY, this.solver)
   849→        : null;
   850→      this.scene.setSelectionRing(hover, this.solver, "hover");
   851→      emphasisLimb = hover;
   852→    }
   853→    this.scene.setLimbLabelEmphasis(emphasisLimb);
   854→
   855→    // Preview de arrastre y alcance.
   856→    const preview = this.interaction?.getPreview() ?? null;
   857→    this.scene.clearHoldHighlights();
   858→    if (preview?.snappedHoldId) {
   859→      if (preview.validity === "valid") this.scene.setHoldCompatible(preview.snappedHoldId, true);
   860→      else this.scene.setHoldInvalid(preview.snappedHoldId, true);
   861→    }
   862→    this.scene.showDragPreview(this.interaction?.dragging ? preview : null, this.solver);
   863→    if (this.interaction?.dragging && preview) {
   864→      this.scene.setReachRadius(
   865→        (preview.limb.startsWith("hand") ? this.rig!.armChain : this.rig!.legChain) * this.character.physical.reachScale
   866→      );
   867→      this.scene.showReachPreview(preview.limb, this.solver, preview.validity !== "invalid");
   868→    } else {
   869→      this.scene.showReachPreview(null, this.solver, true);
   870→    }
   871→
   872→    // Motivo de rechazo EN VIVO durante el arrastre: el jugador ve POR QUE una presa
   873→    // no es posible antes de soltar (control determinista: nada "al azar").
   874→    if (this.interaction?.dragging) {
   875→      if (preview?.snappedHoldId && preview.validity === "invalid") {
   876→        const reason = this.interaction.pendingRejectReason() ?? this.interaction.rejectReason(preview.limb, preview.snappedHoldId);
   877→        this.hud.setRejectHint(reason ? `✕ ${REJECT_LABELS[reason] ?? reason}` : "Movimiento no posible");
   878→      } else if (preview && !preview.snappedHoldId && !this.challenge.rules.allowFreeTarget) {
   879→        this.hud.setRejectHint("Suelta sobre una presa resaltada");
   880→      } else {
   881→        this.hud.setRejectHint(null);
   882→      }
   883→    } else {
   884→      this.hud.setRejectHint(null);
   885→    }
   886→
   887→    // Tutorial: resaltados y avance de pasos.
   888→    this.tickTutorial();
   889→
   890→    // Retroalimentacion de carga por extremidad (misma paleta que el panel
   891→    // lateral): tinte verde/ambar/rojo de la extremidad y anillo sobre su presa.
   892→    // La sobrecarga ademas lleva CUENTA ATRAS de fallo: urgencia 0..1 (danno
   893→    // acumulado/tolerancia) acelera parpadeo y latidos, y alimenta el chip central.
   894→    const cfgBio = this.profile.biomech;
   895→    this.limbOverload.clear();
   896→    this.overloadPressure = 0;
   897→    for (const limb of LIMB_IDS) {
   898→      const anchored = this.solver.hasAnchor(limb);
   899→      const info = this.bio.infoFor(limb);
   900→      const zone = info?.zone ?? "NORMAL";
   901→      let urgency = 0;
   902→      if (anchored && info && (zone === "OVERLOADED" || zone === "ERROR_GUARD")) {
   903→        const secondsLeft = Math.max(0, (cfgBio.overloadGraceSeconds - info.overloadTimer) / cfgBio.overloadAccumRate);
   904→        const frac = clamp01(1 - (secondsLeft * cfgBio.overloadAccumRate) / Math.max(1e-6, cfgBio.overloadGraceSeconds));
   905→        this.limbOverload.set(limb, { seconds: secondsLeft, frac });
   906→        urgency = frac;
   907→        this.overloadPressure = Math.max(this.overloadPressure, frac);
   908→      }
   909→      this.scene.updateLimbTint(limb, anchored, zone, urgency);
   910→      this.scene.setAnchorRing(limb, this.solver.getAnchor(limb)?.holdId ?? null, zone, urgency);
   911→    }
   912→
   913→    // Voz del escalador: respiracion y quejidos de fondo dirigidos por la fatiga
   914→    // y la carga reales (solo suenan con PLAYING activo; el director se resetea fuera).
   915→    const maxFatigue = Math.max(...LIMB_IDS.map((l) => this.bio.fatigue[l] ?? 0));
   916→    const maxRatio = Math.max(...LIMB_IDS.map((l) => this.bio.infoFor(l)?.ratio ?? 0));
   917→    if (this.machine.state === "PLAYING") this.peakFatigue = Math.max(this.peakFatigue, maxFatigue);
   918→    this.audio.update(dtFrame, {
   919→      playing: this.machine.state === "PLAYING",
   920→      maxFatigue,
   921→      maxRatio,
   922→      overloadRatio: this.profile.biomech.overloadRatio,
   923→      overload: LIMB_IDS.some((l) => this.bio.infoFor(l)?.zone === "OVERLOADED"),
   924→      overloadPressure: this.overloadPressure
   925→    });
   926→
   927→    // Combo de flujo: se rompe si pasa la ventana sin fijar otra presa.
   928→    if (this.flowCount > 0 && performance.now() / 1000 - this.lastGrabAt > FLOW_WINDOW_S) {
   929→      this.flowCount = 0;
   930→      this.hud.setFlow(0);
   931→    }
   932→
   933→    const pelvis = this.solver.getParticle(P.pelvisC);
   934→    this.scene.updateCamera(dtFrame, pelvis.pos, this.recenterOnce);
   935→    this.recenterOnce = false;
   936→
   937→    // HUD y debug a 10 Hz.
   938→    this.hudAccum += dtFrame;
   939→    if (this.hudAccum >= 0.1) {
   940→      this.hudAccum = 0;
   941→      this.updateHud(state);
   942→      this.debug.update(this.buildDebugData(state));
   943→    }
   944→
   945→    this.scene.render();
   946→  };

   948→  private updateHud(state: GameState): void {
   949→    const running = state === "PLAYING";
   950→    this.hud.setTimer(
   951→      this.evaluator ? Math.max(0, this.challenge.timeLimitSeconds - (running ? this.gameTime : 0)) : null,

   959→    for (const limb of LIMB_IDS) {
   960→      const anchored = this.solver.hasAnchor(limb);
   961→      const zone = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   962→      const over = this.limbOverload.get(limb);

   964→        zone,
   965→        anchored,
   966→        label: this.interaction?.isAssisting(limb) ? "MOVIENDO" : this.interaction?.dragging && this.interaction.selected === limb ? "OBJETIVO" : "LIBRE",
   967→        overloadS: over ? over.seconds : null,

   972→    // Chip central: la extremidad mas comprometida (solo jugando y fuera del
   973→    // tutorial: ahi la sobrecarga nunca suelta, el aviso despistaria). Memoria
   974→    // de 0.6 s: los valles del pendulum no hacen parpadear el aviso.
   975→    let worst: { limb: LimbId; seconds: number; frac: number } | null = null;
   976→    for (const [limb, over] of this.limbOverload) {

   982→      this.lastWorstAt = nowS;

   985→    this.hud.setOverloadWarning(showWarning && this.lastWorst ? this.lastWorst : null);
   986→    this.hud.setProgress(running && this.evaluator && this.evaluator.progress > 0 ? this.evaluator.progress : null);
   987→  }

  1126→        }
  1127→        break;
  1128→      case "StrainStarted":
  1129→        this.hintOnce("strain", "Agarre en tension: la fatiga de esa extremidad sube mas rapido", "warn");
  1130→        break;
  1131→      case "OverloadStarted":

  1140→        this.hud.setFlow(0);

  1142→      default:
  1143→        break;
  1144→    }
  1145→  }

  1284→  private static readonly KEY_OF_LIMB: Record<LimbId, string> = { hand_l: "1", hand_r: "2", foot_l: "3", foot_r: "4" };
  1285→
  1286→  /**
  1287→   * Teclas 1-4: agarrar/soltar la extremidad directamente (peticion del jugador:
  1288→   * "complicado seleccionar una extremidad cuando se mueve mucho"). Al agarrar, el
  1289→   * objetivo nace EN la extremidad (offset respecto al cursor): nada salta. Al
  1290→   * repetir la tecla, se suelta con la validacion normal (asistencia de viaje).
  1291→   */
  1292→  toggleKeyboardGrab(limb: LimbId): void {
  1293→    if (!(this.machine.state === "READY" || this.machine.state === "PLAYING")) return;
  1294→    if (this.interaction?.dragging) {
  1295→      // Solo suelta si es la misma extremidad la agarrada por teclado; un arrastre
  1296→      // de raton en curso no se interrumpe con las teclas.
  1297→      if (this.keyboardLimb === limb) this.endKeyboardDrag();
  1298→      return;
  1299→    }
  1300→    if (this.interaction!.isAssisting(limb)) return;
  1301→    this.interaction!.beginDrag(limb);
  1302→    if (this.machine.state === "READY") this.machine.transition("PLAYING");
  1303→    this.keyboardLimb = limb;
  1304→    // Offset inicial: objetivo = posicion actual de la extremidad (limbPos - cursor),
  1305→    // asi agarrar NO salta: la extremidad solo responde al MOVIMIENTO del cursor.
  1306→    const zPlane = this.wall.holds[0]?.position[2] ?? -0.18;
  1307→    const limbPos = this.solver.positionOfLimb(limb);
  1308→    const cursor = this.scene.wallPlanePoint(this.pointerX, this.pointerY, zPlane);
  1309→    this.grabOffset = cursor ? { x: limbPos.x - cursor.x, y: limbPos.y - cursor.y } : { x: 0, y: 0 };
  1310→    this.applyKeyboardDragTarget(zPlane);
  1311→    this.hud.toast(`${LIMB_LABEL_ES[limb]} agarrada (tecla ${GameApp.KEY_OF_LIMB[limb]}): muévela con el ratón y pulsa ${GameApp.KEY_OF_LIMB[limb]} para soltar`, "info");
  1312→  }

  1328→  /** Re-proyecta el objetivo del agarre por teclado (cursor + offset) al plano. */

  1330→    if (!this.interaction?.dragging || !this.keyboardLimb) return;

  1425→  private frameTutorialUi(step: TutorialStepDef | null): void {
  1426→    if (!this.tutorial) return;
  1427→    const hudRect = this.opts.hudRoot.getBoundingClientRect();
  1428→
  1429→    let spotlight: TutorialSpotlight | null = null;
  1430→    if (step?.highlightLimb) {
  1431→      const p = this.solver.positionOfLimb(step.highlightLimb);
  1432→      const s = this.scene.projectToScreenCss(p.x, p.y, p.z);
  1433→      spotlight = { shape: "circle", x: s.x, y: s.y, r: 48 };
  1434→    } else if (step?.highlightHold) {
  1435→      const pos = this.holdPosById.get(step.highlightHold);
  1436→      if (pos) {
  1437→        const s = this.scene.projectToScreenCss(pos[0], pos[1], pos[2]);
  1438→        spotlight = { shape: "circle", x: s.x, y: s.y, r: 44 };
  1439→      }
  1440→    } else if (step?.highlightHud) {
  1441→      const el = document.getElementById(step.highlightHud);
  1442→      if (el) {
  1443→        const r = el.getBoundingClientRect();
  1444→        if (r.width > 4 && r.height > 4) {
  1445→          spotlight = {
  1446→            shape: "rect",
  1447→            x: r.left - hudRect.left + r.width / 2,
  1448→            y: r.top - hudRect.top + r.height / 2,
  1449→            w: r.width,
  1450→            h: r.height
  1451→          };
  1452→        }
  1453→      }
  1454→    } else if (step?.highlightCharacter) {
  1455→      const pts = LIMB_IDS.map((l) => this.solver.positionOfLimb(l)).map((p) =>
  1456→        this.scene.projectToScreenCss(p.x, p.y, p.z)
  1457→      );
  1458→      if (pts.length > 0) {
  1459→        const xs = pts.map((p) => p.x);
  1460→        const ys = pts.map((p) => p.y);
  1461→        const minX = Math.min(...xs);
  1462→        const maxX = Math.max(...xs);
  1463→        const minY = Math.min(...ys);
  1464→        const maxY = Math.max(...ys);
  1465→        spotlight = {
  1466→          shape: "ellipse",
  1467→          x: (minX + maxX) / 2,
  1468→          y: (minY + maxY) / 2,
  1469→          rx: (maxX - minX) / 2 + 80,
  1470→          ry: (maxY - minY) / 2 + 90
  1471→        };
  1472→      }
  1473→    }
  1474→
  1475→    let gesture: TutorialGesture | null = null;
  1476→    if (step?.gesture) {
  1477→      const limbPos = this.solver.positionOfLimb(step.gesture.from);
  1478→      const from = this.scene.projectToScreenCss(limbPos.x, limbPos.y, limbPos.z);
  1479→      const holdPos = this.holdPosById.get(step.gesture.to);
  1480→      if (holdPos) {
  1481→        const to = this.scene.projectToScreenCss(holdPos[0], holdPos[1], holdPos[2]);
  1482→        gesture = { from, to, labelFrom: "mantén el clic", labelTo: "suelta aquí" };
  1483→      }
  1484→    }
  1485→
  1486→    this.tutorialUi.frame({ spotlight, gesture, time: performance.now() / 1000 });
  1487→  }
