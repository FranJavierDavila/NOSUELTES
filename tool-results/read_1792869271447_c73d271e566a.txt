   175→  async loadChallenge(id: string): Promise<void> {

   182→    this.bio = new BiomechanicsSystem(this.character, this.profile);
   183→    this.evaluator = new ChallengeEvaluator(this.challenge);
   184→
   185→    const strengthFactor = lerp(this.profile.biomech.strengthFactorMin, this.profile.biomech.strengthFactorMax, this.character.virtues.strength);
   186→    this.anchorMaxForce = this.character.physical.baseGripForce * strengthFactor;
   187→    this.followForce = this.character.physical.baseGripForce * this.character.control.gripAssist * this.character.control.targetFollowStrength;
   188→
   189→    this.solver = new PbdSolver({
   190→      gravity: v3(...this.profile.sim.gravity),
   191→      damping: this.profile.sim.damping,
   192→      iterations: this.profile.sim.iterations,
   193→      floorY: 0,
   194→      wallFrontZ: this.wall.transform.position[2] - this.wall.dimensions.depth / 2,
   195→      tremorAmplitude: this.profile.sim.tremorAmplitude,
   196→      loadEstimatorGain: this.profile.biomech.loadEstimatorGain
   197→    });
   198→
   199→    this.scene.buildWorld(this.wall);
   200→    this.hud.setChallenge(this.challenge, CHALLENGE_LABELS[this.challenge.type]);
   201→    this.buildInteraction();
   202→    this.spawn();

   210→  private spawn(): void {
   211→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   212→    this.interaction?.cancelDrag();
   213→    this.interaction?.cancelAssistAll();

   216→
   217→    this.scene.despawnRagdollVisuals();

   219→    this.scene.spawnRagdollVisuals(this.rig);
   220→
   221→    this.bio.reset();
   222→    this.evaluator?.reset();
   223→    this.gameTime = 0;
   224→    this.accumulator = 0;
   225→    this.guardActive = false;
   226→    this.lastSnapshot = { stability: 0, com: { x: 0, y: 0, z: 0 }, supportCount: 0 };
   227→
   228→    // Marcar presas ocupadas por los anclajes iniciales creados en spawnRagdoll.

   230→      this.scene.setHoldOccupied(req.holdId, true);

   236→    this.events.emit({ type: "Message", text: `Prueba ${CHALLENGE_LABELS[this.challenge.type]} cargada`, tone: "info" });

   245→  // --- Interaccion ---
   246→
   247→  private buildInteraction(): void {
   248→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   249→    const world: InteractionWorld = {
   250→      holdById,
   251→      challenge: this.challenge,
   252→      character: this.character,
   253→      profile: this.profile,
   254→      wall: this.wall,
   255→      anchorHoldOf: (limb) => this.solver.getAnchor(limb)?.holdId,
   256→      holdOccupancy: (holdId) => {
   257→        let n = 0;
   258→        for (const limb of LIMB_IDS) {
   259→          if (this.solver.getAnchor(limb)?.holdId === holdId) n++;
   260→          if (this.interaction?.assistHoldOf(limb) === holdId) n++;
   261→        }
   262→        return n;
   263→      },
   264→      chainRootOf: (limb) => {
   265→        const idx = limb === "hand_l" ? P.shL : limb === "hand_r" ? P.shR : limb === "foot_l" ? P.hipL : P.hipR;
   266→        const p = this.solver.getParticle(idx);
   267→        return v3(p.pos.x, p.pos.y, p.pos.z);
   268→      },
   269→      limbPositionOf: (limb) => this.solver.positionOfLimb(limb),
   270→      chainLengthOf: (limb) => {
   271→        if (!this.rig) return 0.6;
   272→        const base = limb.startsWith("hand") ? this.rig.armChain : this.rig.legChain;
   273→        return base * this.character.physical.reachScale;
   274→      }
   275→    };
   276→    this.interaction = new InteractionController(world, {
   277→      onSelectChanged: () => undefined,
   278→      onAccept: (limb, holdId) => {
   279→        const existing = this.solver.getAnchor(limb);
   280→        if (existing) {
   281→          this.solver.releaseAnchor(limb);
   282→          this.events.emit({ type: "GripReleased", limb, holdId: existing.holdId, reason: "PLAYER_RELEASE" });
   283→        }
   284→        const hold = holdById.get(holdId)!;
   285→        this.solver.setTarget(limb, v3(hold.position[0], hold.position[1], hold.position[2]), this.followForce);
   286→        this.refreshHoldOccupancy();
   287→      },
   288→      onReject: (limb, holdId, reason) => {
   289→        this.solver.clearTarget(limb);
   290→        this.events.emit({ type: "HoldRejected", limb, holdId, reason });
   291→        if (holdId) {
   292→          this.scene.setHoldInvalid(holdId, true);
   293→          window.setTimeout(() => this.scene.setHoldInvalid(holdId, false), 450);
   294→        }
   295→      },
   296→      onArrival: (limb, holdId) => {
   297→        this.solver.clearTarget(limb);
   298→        const hold = holdById.get(holdId)!;
   299→        this.solver.createAnchor(limb, holdId, v3(hold.position[0], hold.position[1], hold.position[2]), this.anchorMaxForce);
   300→        this.events.emit({ type: "HoldAcquired", limb, holdId });
   301→        this.refreshHoldOccupancy();
   302→      }
   303→    });
   304→  }
   305→
   306→  private refreshHoldOccupancy(): void {
   307→    const anchored = new Set<string>();
   308→    for (const limb of LIMB_IDS) {
   309→      const a = this.solver.getAnchor(limb);
   310→      if (a) anchored.add(a.holdId);
   311→    }
   312→    for (const hold of this.wall.holds) {
   313→      this.scene.setHoldOccupied(hold.id, anchored.has(hold.id));
   314→    }
   315→  }

   317→  // --- Pipeline por paso fijo ---
   318→
   319→  private fixedStep(dt: number): void {
   320→    const result = this.solver.step(dt);
   321→
   322→    // ERROR GUARD (9.2): carga no finita / NaN / energia anomala.
   323→    if ((result.nanError || result.energySpike) && !this.guardActive) {
   324→      this.guardActive = true;
   325→      this.guardCooldown = 0.4;
   326→      this.lastGuardError = result.nanError ? "nan" : "energia";
   327→      this.events.emit({ type: "Message", text: "ERROR GUARD: simulacion no finita; reiniciando prueba", tone: "error" });
   328→      this.machine.transition("ERROR");
   329→    }
   330→    if (this.machine.state === "ERROR") {
   331→      this.guardCooldown -= dt;
   332→      if (this.guardCooldown <= 0) {
   333→        this.spawn();
   334→        return;
   335→      }
   336→      return;
   337→    }
   338→
   339→    const anchors = this.solver.anchorHoldIds();
   340→
   341→    // Biomecanica solo cuando el desafio esta activo (fatiga congelada en pausa/resultado).
   342→    if (this.machine.state === "PLAYING") {
   343→      const bioResult = this.bio.update(dt, result, anchors.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver.tremor);
   344→      // En tutorial la fatiga y las zonas se miden (numeros reales en el paso
   345→      // de vitals) pero la sobrecarga NO libera agarres: el onboarding no puede
   346→      // guiar hacia una caida por mover lento.
   347→      for (const limb of this.tutorial ? [] : bioResult.overloadReleases) {
   348→        const anchor = this.solver.releaseAnchor(limb);
   349→        if (anchor) {
   350→          this.events.emit({ type: "GripReleased", limb, holdId: anchor.holdId, reason: "OVERLOAD" });
   351→          this.refreshHoldOccupancy();
   352→        }
   353→      }
   354→      for (const limb of bioResult.strainStarted) this.events.emit({ type: "StrainStarted", limb });
   355→      for (const limb of bioResult.overloadStarted) this.events.emit({ type: "OverloadStarted", limb });
   356→
   357→      // Asistencia de viaje hacia la presa aceptada (7.2).
   358→      this.interaction?.tick(dt);
   359→
   360→      const anchorsNow = this.solver.anchorHoldIds();
   361→      const stability = computeStability(this.bio, result, anchorsNow.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver, this.profile);
   362→      this.lastSnapshot = {
   363→        stability,
   364→        com: result.com,
   365→        supportCount: anchorsNow.length
   366→      };
   367→
   368→      // Evaluacion del desafio (10.3): caida -> exito -> timeout.
   369→      // El timer no corre durante el tutorial: nadie deberia agotar el tiempo
   370→      // mientras aprende los controles.
   371→      if (!this.tutorial) this.gameTime += dt;
   372→      const outcome = this.evaluator!.tick({
   373→        dt,
   374→        gameTime: this.gameTime,
   375→        anchorsByLimb: Object.fromEntries(anchorsNow.map((a) => [a.limb, a.holdId])),
   376→        supportCount: anchorsNow.length,
   377→        stability,
   378→        pelvisY: this.solver.getParticle(P.pelvisC).pos.y,
   379→        outOfBounds: result.outOfBounds
   380→      });
   381→      if (outcome.kind === "fall") {
   382→        this.events.emit({ type: "FallStarted", reason: outcome.reason });
   383→        this.events.emit({ type: "ChallengeFailed", reason: "FALLEN" });
   384→        this.machine.transition("FALLEN");
   385→      } else if (outcome.kind === "success") {
   386→        this.events.emit({ type: "ChallengeSuccess" });
   387→        this.machine.transition("SUCCESS");
   388→      } else if (outcome.kind === "timeout") {
   389→        this.events.emit({ type: "ChallengeFailed", reason: "TIMEOUT" });
   390→        this.machine.transition("TIMEOUT");
   391→      }
   392→    } else {
   393→      // Fuera de PLAYING la fatiga no cambia; sin tremor.
   394→      for (const limb of LIMB_IDS) this.solver.tremor[limb] = 0;
   395→      this.interaction?.tick(0);
   396→    }
   397→  }

   399→  // --- Bucle de render ---
   400→
   401→  private readonly frame = (nowMs: number): void => {
   402→    if (this.disposed) return;
   403→    this.raf = requestAnimationFrame(this.frame);
   404→    this.frameCount++;
   405→    const now = nowMs / 1000;
   406→    const dtFrame = Math.min(Math.max(0, now - this.lastFrame), 0.1);
   407→    this.lastFrame = now;
   408→
   409→    this.fpsFrames++;
   410→    this.fpsElapsed += dtFrame;
   411→    if (this.fpsElapsed >= 0.5) {
   412→      this.fps = this.fpsFrames / this.fpsElapsed;
   413→      this.fpsFrames = 0;
   414→      this.fpsElapsed = 0;
   415→    }
   416→
   417→    const state = this.machine.state;
   418→    const stepping = state !== "PAUSED" && state !== "LOADING" && !this.paused;
   419→    if (stepping) {
   420→      this.accumulator += dtFrame;
   421→      let steps = 0;
   422→      const fixed = this.profile.sim.fixedDeltaSeconds;
   423→      while (this.accumulator >= fixed && steps < MAX_STEPS_PER_FRAME) {
   424→        const t0 = performance.now();
   425→        this.fixedStep(fixed);
   426→        this.physicsMs = lerp(this.physicsMs, performance.now() - t0, 0.15);
   427→        this.accumulator -= fixed;
   428→        steps++;
   429→      }
   430→      if (steps >= MAX_STEPS_PER_FRAME) this.accumulator = 0;
   431→    }
   432→
   433→    if (state === "RESETTING") {
   434→      this.settleTimer -= dtFrame;
   435→      if (this.settleTimer <= 0) {
   436→        this.machine.transition(this.autoStartPending ? "PLAYING" : "READY");
   437→        if (this.autoStartPending) {
   438→          this.events.emit({ type: "Message", text: "Recupera dos apoyos antes de perder el agarre", tone: "warn" });
   439→        }
   440→      }
   441→    }
   442→
   443→    const alpha = Math.min(1, this.accumulator / this.profile.sim.fixedDeltaSeconds);
   444→    this.scene.syncRagdoll(this.solver, alpha);
   445→    this.scene.syncJointDebug(this.solver);
   446→    this.scene.setSelectionRing(this.interaction?.dragging ? this.interaction.selected : this.interaction?.selected ?? null, this.solver);
   447→
   448→    // Preview de arrastre y alcance.
   449→    const preview = this.interaction?.getPreview() ?? null;
   450→    this.scene.clearHoldHighlights();
   451→    if (preview?.snappedHoldId) {
   452→      if (preview.validity === "valid") this.scene.setHoldCompatible(preview.snappedHoldId, true);
   453→      else this.scene.setHoldInvalid(preview.snappedHoldId, true);
   454→    }
   455→    this.scene.showDragPreview(this.interaction?.dragging ? preview : null, this.solver);
   456→    if (this.interaction?.dragging && preview) {
   457→      this.scene.setReachRadius(
   458→        (preview.limb.startsWith("hand") ? this.rig!.armChain : this.rig!.legChain) * this.character.physical.reachScale
   459→      );
   460→      this.scene.showReachPreview(preview.limb, this.solver, preview.validity !== "invalid");
   461→    } else {
   462→      this.scene.showReachPreview(null, this.solver, true);
   463→    }
   464→
   465→    // Tutorial: resaltados y avance de pasos.
   466→    this.tickTutorial();
   467→
   468→    // Tinte de esfuerzo por extremidad.
   469→    for (const limb of LIMB_IDS) {
   470→      this.scene.updateLimbTint(limb, this.solver.tremor[limb]);
   471→    }
   472→
   473→    const pelvis = this.solver.getParticle(P.pelvisC);
   474→    this.scene.updateCamera(dtFrame, pelvis.pos, this.recenterOnce);
   475→    this.recenterOnce = false;
   476→
   477→    // HUD y debug a 10 Hz.
   478→    this.hudAccum += dtFrame;
   479→    if (this.hudAccum >= 0.1) {
   480→      this.hudAccum = 0;
   481→      this.updateHud(state);
   482→      this.debug.update(this.buildDebugData(state));
   483→    }
   484→
   485→    this.scene.render();
   486→  };
   487→
   488→  private updateHud(state: GameState): void {
   489→    const running = state === "PLAYING";
   490→    this.hud.setTimer(
   491→      this.evaluator ? Math.max(0, this.challenge.timeLimitSeconds - (running ? this.gameTime : 0)) : null,
   492→      running
   493→    );
   494→    let fatigueMax = 0;
   495→    for (const limb of LIMB_IDS) fatigueMax = Math.max(fatigueMax, this.bio.fatigue[limb]);
   496→    this.hud.setVitals(this.lastSnapshot.stability, fatigueMax);
   497→
   498→    const limbStates: Record<LimbId, { zone: LoadZone; anchored: boolean; label: string }> = {} as never;
   499→    for (const limb of LIMB_IDS) {
   500→      const anchored = this.solver.hasAnchor(limb);
   501→      const zone = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   502→      limbStates[limb] = {
   503→        zone,
   504→        anchored,
   505→        label: this.interaction?.isAssisting(limb) ? "MOVIENDO" : this.interaction?.dragging && this.interaction.selected === limb ? "OBJETIVO" : "LIBRE"
   506→      };
   507→    }
   508→    this.hud.setLimbStates(limbStates);
   509→    this.hud.setProgress(running && this.evaluator && this.evaluator.progress > 0 ? this.evaluator.progress : null);
   510→  }

   512→  private limbDisplayState(limb: LimbId): string {
   513→    if (this.machine.state === "FALLEN") return "FALLING";
   514→    if (this.interaction?.isAssisting(limb)) return "TARGETING";
   515→    if (this.solver.hasAnchor(limb)) {
   516→      const timer = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   517→      return timer === "OVERLOADED" ? "OVERLOADED" : "ANCHORED";
   518→    }
   519→    if (this.interaction?.dragging && this.interaction.selected === limb) return "TARGETING";
   520→    return "FREE";
   521→  }

   558→  private onGameEvent(e: GameEvent): void {
   559→    this.audio.play(e);
   560→    switch (e.type) {
   561→      case "StateChanged":
   562→        this.hud.setState(e.to);
   563→        break;
   564→      case "ChallengeSuccess":
   565→        this.hud.showBanner("EXITO", "ok");
   566→        break;
   567→      case "ChallengeFailed":
   568→        this.hud.showBanner(e.reason === "TIMEOUT" ? "TIEMPO AGOTADO" : "CAIDA", "bad");
   569→        break;
   570→      case "HoldRejected": {
   571→        const label = REJECT_LABELS[e.reason] ?? e.reason;
   572→        this.hud.toast(`Movimiento rechazado: ${label}`, "warn");
   573→        break;
   574→      }
   575→      case "GripReleased":
   576→        if (e.reason === "OVERLOAD") this.hud.toast(`Sobrecarga: se solto el agarre (${e.limb})`, "error");
   577→        break;
   578→      case "StrainStarted":
   579→        this.hintOnce("strain", "Agarre en tension: la fatiga de esa extremidad sube mas rapido", "warn");
   580→        break;
   581→      case "OverloadStarted":
   582→        this.hintOnce("overload", "SOBRECARGA: si no alivias ese brazo, soltara el agarre en segundos", "error");
   583→        break;
   584→      case "FallStarted":
   585→        this.hintOnce("fall", "Caída: el agarre fallo bajo la carga o perdiste los apoyos. R para reintentar", "info");
   586→        break;
   587→      default:
   588→        break;
   589→    }
   590→  }

   600→  private attachInput(): void {
   601→    const canvas = this.canvas;
   602→    const zPlane = () => this.wall.holds[0]?.position[2] ?? -0.18;
   603→
   604→    this.on(canvas, "pointerdown", (e) => {
   605→      this.audio.ensure();
   606→      if (e.button !== 0) return;
   607→      if (!(this.machine.state === "READY" || this.machine.state === "PLAYING")) return;
   608→      const rect = canvas.getBoundingClientRect();
   609→      const limb = this.scene.pickLimb(e.clientX - rect.left, e.clientY - rect.top);
   610→      if (!limb || this.interaction!.isAssisting(limb)) return;
   611→      this.interaction!.beginDrag(limb);
   612→      if (this.machine.state === "READY") {
   613→        // El timer arranca con la primera interaccion (start.autoStart = false).
   614→        this.machine.transition("PLAYING");
   615→      }
   616→      canvas.setPointerCapture(e.pointerId);
   617→    });
   618→
   619→    this.on(canvas, "pointermove", (e) => {
   620→      if (!this.interaction?.dragging) return;
   621→      const rect = canvas.getBoundingClientRect();
   622→      const point = this.scene.wallPlanePoint(e.clientX - rect.left, e.clientY - rect.top, zPlane());
   623→      this.interaction.updateDrag(point ? v3(point.x, point.y, point.z) : null);
   624→      const preview = this.interaction.getPreview();
   625→      if (preview?.snappedHoldId) {
   626→        this.interaction.setRejectReason(this.interaction.rejectReason(preview.limb, preview.snappedHoldId));
   627→      } else {
   628→        this.interaction.setRejectReason(null);
   629→      }
   630→    });
   631→
   632→    this.on(canvas, "pointerup", (e) => {
   633→      if (e.button !== 0) return;
   634→      if (this.interaction?.dragging) {
   635→        this.interaction.endDrag();
   636→        this.scene.clearHoldHighlights();
   637→      }
   638→    });
   639→
   640→    this.on(canvas, "contextmenu", (e) => {
   641→      e.preventDefault();
   642→      this.interaction?.cancelDrag();
   643→    });
   644→
   645→    this.on(window, "keydown", (e) => {
   646→      const ev = e as KeyboardEvent;
   647→      switch (ev.code) {
   648→        case "KeyR":
   649→          this.requestReset();
   650→          break;
   651→        case "KeyH":
   652→          this.hud.toggleHelp();
   653→          break;
   654→        case "F3":
   655→          ev.preventDefault();
   656→          this.debug.toggle();
   657→          break;
   658→        case "Space":
   659→          ev.preventDefault();
   660→          this.togglePause();
   661→          break;
   662→        case "KeyC":
   663→          this.recenterOnce = true;
   664→          break;
   665→        case "F4":
   666→          ev.preventDefault();
   667→          this.scene.toggleOrtho();
   668→          break;
   669→        case "KeyM": {
   670→          const muted = this.audio.toggleMute();
   671→          this.hud.toast(muted ? "Sonido desactivado" : "Sonido activado", "info");
   672→          break;
   673→        }
   674→        case "KeyT":
   675→          this.startTutorial();
   676→          break;
   677→        case "Escape":
   678→          this.interaction?.cancelDrag();
   679→          break;
   680→        default:
   681→          break;
   682→      }
   683→    });
   684→
   685→    this.on(window, "resize", () => {
   686→      this.scene.resize();
   687→    });
   688→  }

   715→  startTutorial(): void {
   716→    // Solo tiene sentido en el desafio de ruta (objetivos fijos validados).

   718→    if (this.machine.state === "LOADING" || this.machine.state === "RESETTING") return;

   721→    this.syncTutorialUi();

   724→  private stopTutorial(markDone: boolean): void {
   725→    if (this.tutorial && markDone) {
   726→      try {
   727→        window.localStorage.setItem(GameApp.TUTORIAL_DONE_KEY, "1");
   728→      } catch {
   729→        /* sin persistencia: el tutorial reaparecera en la proxima sesion */
   730→      }
   731→      this.hud.toast("Tutorial cerrado (T para repetirlo)", "info");
   732→    }
   733→    this.tutorial = null;
   734→    this.tutorialUi.hide();
   735→    this.scene?.setTutorialHoldHighlight(null);
   736→    this.scene?.setTutorialLimbHighlight(null);

   739→  private syncTutorialUi(): void {
   740→    const tut = this.tutorial;
   741→    const step = tut !== null ? tut.current : null;
   742→    if (tut === null || step === null) {
   743→      if (tut?.isFinished) this.stopTutorial(true);
   744→      return;
   745→    }
   746→    this.tutorialUi.update(step, tut.progress, this.buildTutorialCtx(), buildTutorialSteps().length);
   747→  }

   749→  private buildTutorialCtx(): TutorialCtx {
   750→    const anchors: Partial<Record<LimbId, string>> = {};
   751→    const loads: Partial<Record<LimbId, number>> = {};
   752→    for (const a of this.solver.anchorHoldIds()) {
   753→      anchors[a.limb] = a.holdId;
   754→      loads[a.limb] = Math.round(a.loadN);
   755→    }
   756→    const preview = this.interaction?.getPreview() ?? null;
   757→    return {
   758→      state: this.machine.state,
   759→      anchors,
   760→      loads,
   761→      selected: this.interaction?.selected ?? null,
   762→      dragging: this.interaction?.dragging ?? false,
   763→      preview: preview ? { holdId: preview.snappedHoldId, validity: preview.validity } : null,
   764→      supportCount: this.lastSnapshot.supportCount,
   765→      fatigueMax: Math.max(...LIMB_IDS.map((l) => this.bio.fatigue[l] ?? 0))
   766→    };
   767→  }

   769→  private tickTutorial(): void {
   770→    if (!this.tutorial) {
   771→      return;
   772→    }
   773→    const step = this.tutorial.current;
   774→    this.scene.setTutorialHoldHighlight(step?.highlightHold ?? null);
   775→    this.scene.setTutorialLimbHighlight(step?.highlightLimb ?? null);
   776→    this.tutorial.tick(this.buildTutorialCtx());
   777→    this.frameTutorialUi(step);
   778→    // El texto con placeholders vivos (cargas/fatiga) se refresca a 10 Hz.
   779→    if (this.hudAccum >= 0.1) this.syncTutorialUi();

   782→  /**
   783→   * Geometria del tutorial cada frame: proyecta el objetivo del paso (limb,
   784→   * presa, panel del HUD o personaje completo) a px CSS y alimenta al overlay
   785→   * con el spotlight, la demo de arrastre y la posicion de la tarjeta.
   786→   */
   787→  private frameTutorialUi(step: TutorialStepDef | null): void {
   788→    if (!this.tutorial) return;
   789→    const hudRect = this.opts.hudRoot.getBoundingClientRect();
   790→
   791→    let spotlight: TutorialSpotlight | null = null;
   792→    if (step?.highlightLimb) {
   793→      const p = this.solver.positionOfLimb(step.highlightLimb);
   794→      const s = this.scene.projectToScreenCss(p.x, p.y, p.z);
   795→      spotlight = { shape: "circle", x: s.x, y: s.y, r: 48 };
   796→    } else if (step?.highlightHold) {
   797→      const pos = this.holdPosById.get(step.highlightHold);
   798→      if (pos) {
   799→        const s = this.scene.projectToScreenCss(pos[0], pos[1], pos[2]);
   800→        spotlight = { shape: "circle", x: s.x, y: s.y, r: 44 };
   801→      }
   802→    } else if (step?.highlightHud) {
   803→      const el = document.getElementById(step.highlightHud);
   804→      if (el) {
   805→        const r = el.getBoundingClientRect();
   806→        if (r.width > 4 && r.height > 4) {
   807→          spotlight = {
   808→            shape: "rect",
   809→            x: r.left - hudRect.left + r.width / 2,
   810→            y: r.top - hudRect.top + r.height / 2,
   811→            w: r.width,
   812→            h: r.height
   813→          };
   814→        }
   815→      }
   816→    } else if (step?.highlightCharacter) {
   817→      const pts = LIMB_IDS.map((l) => this.solver.positionOfLimb(l)).map((p) =>
   818→        this.scene.projectToScreenCss(p.x, p.y, p.z)
   819→      );
   820→      if (pts.length > 0) {
   821→        const xs = pts.map((p) => p.x);
   822→        const ys = pts.map((p) => p.y);
   823→        const minX = Math.min(...xs);
   824→        const maxX = Math.max(...xs);
   825→        const minY = Math.min(...ys);
   826→        const maxY = Math.max(...ys);
   827→        spotlight = {
   828→          shape: "ellipse",
   829→          x: (minX + maxX) / 2,
   830→          y: (minY + maxY) / 2,
   831→          rx: (maxX - minX) / 2 + 80,
   832→          ry: (maxY - minY) / 2 + 90
   833→        };
   834→      }
   835→    }
   836→
   837→    let gesture: TutorialGesture | null = null;
   838→    if (step?.gesture) {
   839→      const limbPos = this.solver.positionOfLimb(step.gesture.from);
   840→      const from = this.scene.projectToScreenCss(limbPos.x, limbPos.y, limbPos.z);
   841→      const holdPos = this.holdPosById.get(step.gesture.to);
   842→      if (holdPos) {
   843→        const to = this.scene.projectToScreenCss(holdPos[0], holdPos[1], holdPos[2]);
   844→        gesture = { from, to, labelFrom: "mantén el clic", labelTo: "suelta aquí" };
   845→      }
   846→    }
   847→
   848→    this.tutorialUi.frame({ spotlight, gesture, time: performance.now() / 1000 });
   849→  }
