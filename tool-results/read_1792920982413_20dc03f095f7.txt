     1→/**
     2→ * Solver de fisica PBD (Position Based Dynamics) para el ragdoll.
     3→ *
     4→ * Decision documentada (ADR-001 del prototipo): adaptador fisico interno basado en
     5→ * particulas + restricciones de distancia y de angulo. La interfaz PhysicsAdapter se
     6→ * mantiene desacoplada para poder sustituirlo por Havok/Rapier sin tocar gameplay.
     7→ *
     8→ * Modelo de anclaje (estimador documentado, seccion 9.2 del documento): cada anclaje
     9→ * es un resorte amortiguado con fuerza maxima. La carga se mide como la fuerza que el
    10→ * resorte debe aplicar (rigidez x estiramiento, antes de saturar), lo que permite
    11→ * detectar sobrecarga y provoca estiramiento visible cuando el agarre no basta.
    12→ * Nunca se teletransportan cuerpos durante el juego.
    13→ */

    48→/**
    49→ * Restriccion de SENTIDO DE FLEXION (ADR-008): el constraint angular es 2D y no

    51→ * Este constraint mantiene el vertice articular (codo/rodilla) en el lado correcto

   399→  /** Mueve todos los particulas por un delta (solo para la pose inicial y el impulso de arranque). */
   400→  translateRagdoll(delta: V3): void {
   401→    for (const p of this.particles) {
   402→      p.pos = vAdd(p.pos, delta);
   403→      p.prev = vAdd(p.prev, delta);
   404→    }
   405→  }

   407→  /**
   408→   * Relajacion estatica: proyecta restricciones SIN integrar (sin gravedad ni energia
   409→   * cinetica). Convierte una semilla aproximada en una pose consistente antes del primer
   410→   * paso, eliminando las patadas del settle inicial.
   411→   */
   412→  relax(sweeps = 80): void {
   413→    for (let i = 0; i < sweeps; i++) {
   414→      this.solveSticks();
   415→      this.solveAngles();
   416→      this.solveFolds();
   417→      for (const anchor of this.anchors.values()) {
   418→        const p = this.particles[anchor.particle];
   419→        p.pos = vClone(anchor.point);
   420→      }
   421→      this.solveCollisions();
   422→    }
   423→    for (const p of this.particles) {
   424→      p.prev = vClone(p.pos);
   425→    }
   426→  }

   428→  /** Aplica una velocidad inicial (m/s) al cuerpo completo; usado por start.impulse. */
   429→  applyImpulseVelocity(vel: V3, dt: number): void {
   430→    for (const p of this.particles) {
   431→      if (p.invMass > 0) {
   432→        p.prev = vSub(p.pos, vScale(vel, dt));
   433→      }
   434→    }
   435→  }
   436→
   437→  totalMass(): number {
   438→    return this.particles.reduce((acc, p) => acc + p.mass, 0);
   439→  }
   440→
   441→  centerOfMass(): V3 {
   442→    let m = 0;
   443→    const c = v3();
   444→    for (const p of this.particles) {
   445→      c.x += p.pos.x * p.mass;
   446→      c.y += p.pos.y * p.mass;
   447→      c.z += p.pos.z * p.mass;
   448→      m += p.mass;
   449→    }
   450→    return vScale(c, 1 / (m || 1));
   451→  }

   453→  step(dt: number): StepResult {
   454→    const K = this.env.iterations;
   455→    const forces = new Map<number, V3>();
   456→    const anchorLoads: Partial<Record<LimbId, number>> = {};
   457→    const anchorStretch: Partial<Record<LimbId, number>> = {};
   458→    let nanError = false;
   459→    let outOfBounds = false;
   460→    let maxSpeed = 0;
   461→
   462→    // 0) Tono postural (ADR-008): persigue el objetivo con rates asimetricos
   463→    // (la caida relaja rapido; recuperarlo de pie cuesta mas).
   464→    const toneRate = this.toneTarget < this.tone ? 4.5 : 2.0;
   465→    const toneGap = this.toneTarget - this.tone;
   466→    this.tone = Math.abs(toneGap) < 0.01 ? this.toneTarget : this.tone + toneGap * Math.min(1, toneRate * dt);
   467→
   468→    // 1) Fuerza externa de objetivos (resorte amortiguado con tope). Los anclajes NO son
   469→    // resortes: se fijan como pin rigido en la fase de restricciones (ver 4b).
   470→    for (const target of this.targets.values()) {
   471→      const p = this.particles[target.particle];
   472→      const err = vSub(target.target, p.pos);
   473→      const vel = vScale(vSub(p.pos, p.prev), 1 / dt);
   474→      const spring = vAdd(vScale(err, this.env.followStiffness), vScale(vel, -this.env.anchorDamping * 0.5));
   475→      const applied = clampVecMag(spring, target.maxForce);
   476→      forces.set(target.particle, vAdd(forces.get(target.particle) ?? v3(), applied));
   477→    }
   478→
   479→    // 1b) Musculos activos (ADR-007): cuadriceps, step-up, traccion de brazos y tono
   480→    // postural. Sin anclajes el cuerpo es ragdoll puro (caida limpia).
   481→    if (this.env.muscles.musclesOn && this.topology !== null && this.anchors.size > 0) {
   482→      this.solveMuscles(dt, forces);
   483→    }
   484→    this.climbDrive = Math.max(0, this.climbDrive - dt / this.env.muscles.driveSeconds);
   485→
   486→    // 2) Integracion Verlet con paso fijo (con techo de velocidad: los musculos no
   487→    // pueden lanzar el ragdoll a velocidades no humanas; el ERROR GUARD queda para
   488→    // anomalias reales del solver).
   489→    const g = this.env.gravity;
   490→    const vCap = this.env.muscles.vMaxSpeed * dt;
   491→    for (let i = 0; i < this.particles.length; i++) {
   492→      const p = this.particles[i];
   493→      if (p.invMass <= 0) continue;
   494→      const f = forces.get(i);
   495→      const a = f ? vAdd(g, vScale(f, p.invMass)) : g;
   496→      const vel = vScale(vSub(p.pos, p.prev), this.env.damping);
   497→      p.prev = vClone(p.pos);
   498→      let next = vAdd(vAdd(p.pos, vel), vScale(a, dt * dt));
   499→      const newVel = vSub(next, p.prev);
   500→      const newSpeed = vLen(newVel);
   501→      if (newSpeed > vCap) {
   502→        next = vAdd(p.prev, vScale(newVel, vCap / newSpeed));
   503→      }
   504→      p.pos = next;
   505→      // Metrica sobre la velocidad YA acotada (la previa incluye correcciones de pin).
   506→      const measured = Math.min(newSpeed, vCap) / dt;
   507→      if (measured > maxSpeed) maxSpeed = measured;
   508→    }
   509→
   510→    // 3) Tremor de esfuerzo (seccion 9.5): ruido proporcional al estado de tension.
   511→    for (const limb of LIMB_IDS) {
   512→      const t = this.tremor[limb];
   513→      if (t <= 0) continue;
   514→      const idx = this.limbParticle.get(limb);
   515→      if (idx === undefined) continue;
   516→      const p = this.particles[idx];
   517→      const amp = this.env.tremorAmplitude * t;
   518→      p.pos = vAdd(p.pos, v3((Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp));
   519→    }
   520→
   521→    // 4) Proyeccion de restricciones.
   522→    // 4a) Acumulador de reaccion por anclaje: suma de correcciones del pin este paso.
   523→    const pinCorrection = new Map<LimbId, V3>();
   524→    for (let iter = 0; iter < K; iter++) {
   525→      this.solveSticks();
   526→      this.solveAngles();
   527→      this.solveFolds();
   528→      // 4b) Anclajes: pin rigido con tope de fuerza. La carga requerida se mide como la
   529→      // fuerza necesaria para mantener el pin (F = m x correccion / dt^2), un estimador
   530→      // consistente con el impulso aplicado (9.2). En sobrecarga el pin se satura y la
   531→      // extremidad resbala visiblemente de la presa.
   532→      for (const anchor of this.anchors.values()) {
   533→        const p = this.particles[anchor.particle];
   534→        const err = vSub(anchor.point, p.pos);
   535→        const errLen = vLen(err);
   536→        if (errLen < 1e-9) continue;
   537→        const maxCorr = (anchor.maxForce * dt * dt) / p.mass;
   538→        const scaled = errLen > maxCorr ? vScale(err, maxCorr / errLen) : err;
   539→        p.pos = vAdd(p.pos, scaled);
   540→        pinCorrection.set(anchor.limb, vAdd(pinCorrection.get(anchor.limb) ?? v3(), scaled));
   541→      }
   542→      this.solveCollisions();
   543→    }
   544→    // NOTA: la correccion del pin genera momento "fantasma" en Verlet; ya no se
   545→    // neutraliza en prev (cambiaria la calibracion del estimador de carga). El techo
   546→    // de velocidad de la integracion (vMaxSpeed) contiene ese momento residual.
   547→    // Carga suavizada: impulso total demandado por el pin, en newtons, con ganancia
   548→    // calibrada del estimador documentado (las re-proyecciones por iteracion sobre-miden).
   549→    for (const anchor of this.anchors.values()) {
   550→      const net = pinCorrection.get(anchor.limb) ?? v3();
   551→      const measured = this.particles[anchor.particle].mass * (vLen(net) / (dt * dt)) * this.env.loadEstimatorGain;
   552→      anchor.loadN = anchor.loadN === 0 ? measured : anchor.loadN + (measured - anchor.loadN) * 0.3;
   553→      anchor.stretch = vLen(vSub(anchor.point, this.particles[anchor.particle].pos));
   554→      anchorLoads[anchor.limb] = anchor.loadN;
   555→      anchorStretch[anchor.limb] = anchor.stretch;
   556→    }
   557→
   558→    // 5) Guardas: NaN, energia y limites del mundo.
   559→    for (const p of this.particles) {
   560→      if (!isFiniteVec(p.pos) || !isFiniteVec(p.prev)) {
   561→        nanError = true;
   562→        break;
   563→      }
   564→      if (p.pos.x < this.env.bounds.minX || p.pos.x > this.env.bounds.maxX || p.pos.y > this.env.bounds.maxY || p.pos.z < this.env.bounds.minZ || p.pos.z > this.env.bounds.maxZ) {
   565→        outOfBounds = true;
   566→      }
   567→    }
   568→    if (maxSpeed > 18) {
   569→      // Energia anomala: se registra, la capa de juego decide (ERROR GUARD).
   570→      maxSpeed = clamp(maxSpeed, 0, 30);
   571→    }
   572→
   573→    // 6) Lecturas derivadas.
   574→    const floorContacts: { limb: LimbId; point: V3 }[] = [];
   575→    for (const limb of LIMB_IDS) {
   576→      const idx = this.limbParticle.get(limb);
   577→      if (idx === undefined) continue;
   578→      const p = this.particles[idx];
   579→      if (p.pos.y <= this.env.floorY + 0.045) {
   580→        floorContacts.push({ limb, point: v3(p.pos.x, this.env.floorY, p.pos.z) });
   581→      }
   582→    }
   583→
   584→    return {
   585→      nanError,
   586→      energySpike: maxSpeed > 18,
   587→      outOfBounds,
   588→      com: this.centerOfMass(),
   589→      anchorLoads,
   590→      anchorStretch,
   591→      floorContacts,
   592→      jointAngles: this.angles.map((a) => ({
   593→        id: a.id,
   594→        angle: a.currentAngle,
   595→        utilization: jointUtilization(a)
   596→      })),
   597→      maxSpeed
   598→    };
   599→  }
   600→
   601→  /**
   602→   * Fase muscular (ADR-007). Todas las fuerzas se aplican sobre particulas de masa alta
   603→   * (pelvis ~11 kg, pecho ~14 kg) con topes calibrados para un desplazamiento por paso
   604→   * estable; los pines miden la carga resultante y la biomecanica la traduce en fatiga.
   605→   */
   606→  private solveMuscles(dt: number, forces: Map<number, V3>): void {
   607→    const M = this.env.muscles;
   608→    const T = this.topology!;
   609→    const pelvis = this.particles[T.pelvis];
   610→    const chest = this.particles[T.chest];
   611→    const head = this.particles[T.head];
   612→    const addForce = (idx: number, fx: number, fy: number, fz: number): void => {
   613→      const cur = forces.get(idx);
   614→      forces.set(idx, cur ? v3(cur.x + fx, cur.y + fy, cur.z + fz) : v3(fx, fy, fz));
   615→    };
   616→
   617→    // Rampa de activacion por anclaje (evita el tiron al anclar).
   618→    for (const anchor of this.anchors.values()) {
   619→      const rate = anchor.limb.startsWith("foot") ? 1 / 0.35 : 1 / 0.2;
   620→      anchor.activation = Math.min(1, anchor.activation + dt * rate);
   621→    }
   622→
   623→    // Pre-pasada: hay pierna sosteniendo (cuadriceps activo)? Con apoyo de piernas
   624→    // los brazos NO tiran (el pull-up esta para colgar/colocarse, no para despegar
   625→    // al cuerpo que ya esta de pie: evita la suma explosiva de fuerzas).
   626→    let feetSupporting = false;
   627→    // Peso que cargan los brazos ahora mismo (N): modula el balance (un pie que
   628→    // solo descansa no debe volcar el COM hacia si; uno que sostiene, si).
   629→    let handLoadN = 0;
   630→    for (const anchor of this.anchors.values()) {
   631→      if (anchor.limb.startsWith("hand")) handLoadN += Math.max(0, anchor.loadN);
   632→    }
   633→    // Arbitracion de pies: el STEP-UP lo hace solo el pie MAS ALTO (transferencia de
   634→    // peso al escalon, como en escalada real); el balance solo desde el pie activo,
   635→    // para que un apoyo bajo no pelee contra el pie que esta montando.
   636→    let stepUpFoot: LimbId | null = null;
   637→    let stepUpHoldY = -Infinity;
   638→    for (const anchor of this.anchors.values()) {
   639→      if (!anchor.limb.startsWith("foot")) continue;
   640→      const root = this.particles[T.rootOf.get(anchor.limb)!];
   641→      const dyr = root.pos.y - anchor.point.y;
   642→      const dr = Math.hypot(root.pos.x - anchor.point.x, dyr, root.pos.z - anchor.point.z) || 1e-6;
   643→      if (dyr / dr > 0.05) {
   644→        feetSupporting = true;
   645→      }
   646→      if (dyr / dr < 0.02 && anchor.point.y > stepUpHoldY) {
   647→        stepUpHoldY = anchor.point.y;
   648→        stepUpFoot = anchor.limb;
   649→      }
   650→    }
   651→
   652→    let feetAnchored = 0;
   653→    for (const anchor of this.anchors.values()) {
   654→      const isFoot = anchor.limb.startsWith("foot");
   655→      if (isFoot) feetAnchored++;
   656→      const rootIdx = T.rootOf.get(anchor.limb);
   657→      if (rootIdx === undefined) continue;
   658→      const root = this.particles[rootIdx];
   659→      const dx = root.pos.x - anchor.point.x;
   660→      const dy = root.pos.y - anchor.point.y;
   661→      const dz = root.pos.z - anchor.point.z;
   662→      const dist = Math.hypot(dx, dy, dz) || 1e-6;
   663→
   664→      if (isFoot) {
   665→        // CUADRICEPS (apoyo): el pie por DEBAJO de la cadera empuja la pelvis lejos del
   666→        // pie (extension de pierna = ponerse de pie). Proyectado al plano de la pared:
   667→        // la pierna empuja "hacia arriba", no separa del muro.
   668→        if (dy / dist > 0.05) {
   669→          // Servo de pierna: empuja mientras la pierna este flexionada y ceda al
   670→          // acercarse a la extension natural (0.9·cadena = rodilla casi recta).
   671→          // Asi la postura de pie es un equilibrio estable, no un empuje sin tope.
   672→          const targetDist = T.legChain * 0.9;
   673→          const f = clamp((targetDist - dist) * M.quadServoK, 0, M.quadPushMax) * anchor.activation;
   674→          const ul = Math.hypot(dx, dy) || 1e-6;
   675→          // Direccion axial con sesgo vertical (reaccion del suelo ~vertical al de pie).
   676→          const bx = (dx / ul) * (1 - M.quadUpBias);
   677→          const by = (dy / ul) * (1 - M.quadUpBias) + M.quadUpBias;
   678→          const bl = Math.hypot(bx, by) || 1e-6;
   679→          addForce(T.pelvis, (bx / bl) * f, (by / bl) * f, 0);
   680→          // BALANCE: llevar el COM sobre el apoyo (feedback tipo SIMBICON); sin el,
   681→          // un apoyo lateral nunca sostiene de verdad. Solo desde el pie que esta
   682→          // montando (si hay step-up) o desde los apoyos si nadie monta. Escalado por
   683→          // la carga REAL de la pierna frente a los brazos: un pie que descansa no
   684→          // vuelca el cuerpo (evita sobrecargar el brazo contrario en cascada).
   685→          if (stepUpFoot === null || stepUpFoot === anchor.limb) {
   686→            const weightShare = clamp(f / (f + handLoadN + 60), 0.15, 1);
   687→            const fb = clamp((anchor.point.x - pelvis.pos.x) * 400, -M.balanceMax, M.balanceMax) * anchor.activation * weightShare;
   688→            addForce(T.pelvis, fb, 0, 0);
   689→            // 3a ley de Newton: parte de la reaccion entra por el pie al pin, de modo que
   690→            // el apoyo MEDIDO carga de verdad al estar de pie (fatiga de piernas real).
   691→            addForce(anchor.particle, -(bx / bl) * f * 0.5 - fb * 0.4, -(by / bl) * f * 0.5, 0);
   692→          } else {
   693→            addForce(anchor.particle, -(bx / bl) * f * 0.5, -(by / bl) * f * 0.5, 0);
   694→          }
   695→        }
   696→        // PSOAS/STEP-UP: el pie por ENCIMA de la cadera sube el cuerpo hacia la presa
   697→        // (montar el escalon). Es lo que convierte "apoyar un pie" en ascender. La
   698→        // componente vertical del vector hacia la presa regula el empuje solo: al
   699→        // acercarse la pelvis al nivel del pie la fuerza se hace horizontal (COM sobre
   700→        // el apoyo) y el cuadriceps toma el relevo por encima. Solo el pie MAS ALTO
   701→        // monta (arbitracion: transferencia de peso al escalon).
   702→        if (stepUpFoot === anchor.limb && dy / dist < 0.02) {
   703→          const boost = 0.9 + 0.3 * this.climbDrive;
   704→          // Esvaimiento por proximidad: al acercarse la pelvis al nivel del pie la
   705→          // fuerza cede (sin sobregiro mas alla de la postura natural de pie).
   706→          const taper = clamp(-dy / 0.45, 0.55, 1);
   707→          const f = M.stepUpMax * anchor.activation * boost * taper;
   708→          const ux = -dx / dist;
   709→          const uy = -dy / dist;
   710→          const uz = -dz / dist;
   711→          addForce(T.pelvis, ux * f, uy * f, uz * f);
   712→          addForce(anchor.particle, -ux * f * 0.35, -uy * f * 0.35, -uz * f * 0.35);
   713→        }
   714→      } else {
   715→        // TRACCION DE BRAZOS (impulso de subida del jugador): mano por encima del
   716→        // hombro y brazo ESTIRADO (compuerta de holgura) tira del pecho hacia el
   717→        // anclaje. Con piernas de apoyo es el "subirse" que sube los hombros; el
   718→        // servo de pierna limita la postura y evita el pliegue.
   719→        if (this.climbDrive > 0.02 && dy / dist < -0.1) {
   720→          // Holgura medida desde la RAIZ (hombro): en postura plegada el pecho puede
   721→          // estar alto con el hombro colgando bajo y el brazo estirado; el pull debe
   722→          // evaluar el brazo real, no la posicion del pecho.
   723→          const rdx = anchor.point.x - root.pos.x;
   724→          const rdy = anchor.point.y - root.pos.y;
   725→          const rdz = anchor.point.z - root.pos.z;
   726→          const rd = Math.hypot(rdx, rdy, rdz) || 1e-6;
   727→          const slack = rd - T.armChain * 0.9;
   728→          const cdx = anchor.point.x - chest.pos.x;
   729→          const cdy = anchor.point.y - chest.pos.y;
   730→          const cdz = anchor.point.z - chest.pos.z;
   731→          const cd = Math.hypot(cdx, cdy, cdz) || 1e-6;
   732→          if (slack > 0) {
   733→            const gate = clamp(slack / (T.armChain * 0.2), 0.4, 1);
   734→            // Durante un step-up la pierna manda: los brazos asisten al 60%.
   735→            const assist = stepUpFoot === null ? 1 : 0.6;
   736→            // Margen de agarre: no se puede tirar mas fuerte que el agarre restante
   737→            // (un brazo al limite no tira: evita saturar el pin y la sobrecarga).
   738→            const headroom = clamp(1 - anchor.loadN / anchor.maxForce, 0, 1);
   739→            const f = M.armPullMax * anchor.activation * this.climbDrive * gate * assist * headroom;
   740→            addForce(T.chest, (cdx / cd) * f, (cdy / cd) * f, (cdz / cd) * f);
   741→            addForce(rootIdx, (cdx / cd) * f * 0.35, (cdy / cd) * f * 0.35, (cdz / cd) * f * 0.35);
   742→          }
   743→        }
   744→      }
   745→    }
   746→
   747→    // TONO POSTURAL: pecho sobre pelvis y cabeza sobre pecho (anti-muneco de trapo).
   748→    // Escala con el numero de apoyos y con el tono global (ADR-008): cuerpo libre =
   749→    // ragdoll puro; tono bajo (caida) = cuerpo flacido pero sin pliegues imposibles
   750→    // (los limites y folds siguen activos: son anatomia pasiva, no musculo).
   751→    const toneScale = Math.min(1, this.anchors.size / 2) * (0.15 + 0.85 * this.tone);
   752→    const tex = pelvis.pos.x - chest.pos.x;
   753→    const tey = pelvis.pos.y + T.spineLen * 0.97 - chest.pos.y;
   754→    const tez = pelvis.pos.z - chest.pos.z;
   755→    const teLen = Math.hypot(tex, tey, tez) || 1e-6;
   756→    const coreF = Math.min(M.coreMax * toneScale, teLen * 400);
   757→    addForce(T.chest, (tex / teLen) * coreF, (tey / teLen) * coreF, (tez / teLen) * coreF);
   758→    const nex = chest.pos.x - head.pos.x;
   759→    const ney = chest.pos.y + 0.2 - head.pos.y;
   760→    const nez = chest.pos.z - head.pos.z;
   761→    const neLen = Math.hypot(nex, ney, nez) || 1e-6;
   762→    const neckF = Math.min(M.neckMax * toneScale, neLen * 300);
   763→    addForce(T.head, (nex / neLen) * neckF, (ney / neLen) * neckF, (nez / neLen) * neckF);
   764→
   765→    // PEGADO A LA PARED con cualquier anclaje: la colision del cuerpo con la cara del
   766→    // muro impide colgar en la vertical exacta de los pines (las presas sobresalen), y
   767→    // sin fuerza de centrado en z el cuerpo queda 0.1-0.2 m por detras, offset que
   768→    // devora el alcance util de brazos y piernas en CADA movimiento. El escalador real
   769→    // va pegado: cadera y pecho se presionan hacia el plano de presas.
   770→    if (this.anchors.size > 0) {
   771→      const zTarget = this.env.wallFrontZ - 0.26;
   772→      const fzPelvis = clamp((zTarget - pelvis.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   773→      const fzChest = clamp((zTarget - chest.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   774→      addForce(T.pelvis, 0, 0, fzPelvis);
   775→      addForce(T.chest, 0, 0, fzChest);
   776→    }
   777→
   778→    // Tope de fuerza muscular por particula axial: la suma de musculos nunca puede
   779→    // superar el techo (estabilidad de Verlet con las masas de pelvis/pecho).
   780→    const cap = (idx: number, capN: number): void => {
   781→      const f = forces.get(idx);
   782→      if (!f) return;
   783→      const l = vLen(f);
   784→      if (l > capN) forces.set(idx, vScale(f, capN / l));
   785→    };
   786→    cap(T.pelvis, M.muscleForceCapPelvis);
   787→    cap(T.chest, M.muscleForceCapChest);
   788→
   789→    // Instantanea para diagnostico (solo particulas axiales y raices).
   790→    this.lastMuscleForces.clear();
   791→    for (const idx of [T.pelvis, T.chest, T.head]) {
   792→      const f = forces.get(idx);
   793→      if (f) this.lastMuscleForces.set(idx, f);
   794→    }
   795→  }
   796→
   797→  private solveSticks(): void {
   798→    for (const s of this.sticks) {
   799→      const pa = this.particles[s.a];
   800→      const pb = this.particles[s.b];
   801→      const w = pa.invMass + pb.invMass;
   802→      if (w <= 0) continue;
   803→      const d = vSub(pb.pos, pa.pos);
   804→      const len = vLen(d);
   805→      if (len < 1e-9) continue;
   806→      const diff = (len - s.rest) / len;
   807→      const k = s.stiffness;
   808→      pa.pos = vAdd(pa.pos, vScale(d, (diff * pa.invMass * k) / w));
   809→      pb.pos = vSub(pb.pos, vScale(d, (diff * pb.invMass * k) / w));
   810→    }
   811→  }
   812→
   813→  private solveAngles(): void {
   814→    // Limites articulares DUROS (ADR-008): la rigidez de los datos ya no escala la
   815→    // correccion de la violacion; jointLimitStiffness (1 = duro) garantiza que ni
   816→    // colgando ni en caida el angulo cruce el rango anatomico. La flexion DENTRO del
   817→    // rango sigue libre (el constraint solo actua al violar).
   818→    const k = clamp(this.env.balance.jointLimitStiffness, 0, 1);
   819→    for (const c of this.angles) {
   820→      const pj = this.particles[c.j];
   821→      const pa = this.particles[c.a];
   822→      const pb = this.particles[c.b];
   823→      const va = vSub(pa.pos, pj.pos);
   824→      const vb = vSub(pb.pos, pj.pos);
   825→      const la = vLen(va);
   826→      const lb = vLen(vb);
   827→      if (la < 1e-9 || lb < 1e-9) {
   828→        c.currentAngle = Number.isFinite(la) && Number.isFinite(lb) ? c.currentAngle : c.currentAngle;
   829→        continue;
   830→      }
   831→      const ang = angleBetween(va, vb);
   832→      c.currentAngle = ang;
   833→      const target = clamp(ang, c.min, c.max);
   834→      if (Math.abs(target - ang) < 1e-5) continue;
   835→      const axis = vCross(va, vb);
   836→      const axisLen = vLen(axis);
   837→      if (axisLen < 1e-9) continue;
   838→      const rot = rotateAroundAxis(vb, vScale(axis, 1 / axisLen), target - ang);
   839→      const delta = vSub(rot, vb);
   840→      const w = pa.invMass + pb.invMass;
   841→      if (w <= 0) continue;
   842→      pb.pos = vAdd(pb.pos, vScale(delta, (pb.invMass * k) / w));
   843→      pa.pos = vSub(pa.pos, vScale(delta, (pa.invMass * k) / w));
   844→    }
   845→  }
   846→
   847→  /**
   848→   * Sentido de flexion (ADR-008): mantiene el vertice articular en el lado anatomico
   849→   * de la cuerda padre-hijo. El angulo no puede detectarlo (doblada al reves = mismo
   850→   * angulo); esto prohibe rodillas que flexionan hacia atras y codos que se meten en
   851→   * la pared. Correccion dura y repartida: el vertice al lado permitido, los extremos
   852→   * con la contra-mitad (conservacion aproximada).
   853→   */
   854→  private solveFolds(): void {
   855→    const k = clamp(this.env.balance.foldStrength, 0, 1);
   856→    for (const f of this.folds) {
   857→      const pj = this.particles[f.j];
   858→      const pa = this.particles[f.a];
   859→      const pb = this.particles[f.b];
   860→      const mx = (pa.pos.x + pb.pos.x) * 0.5;
   861→      const my = (pa.pos.y + pb.pos.y) * 0.5;
   862→      const mz = (pa.pos.z + pb.pos.z) * 0.5;
   863→      const side = (pj.pos.x - mx) * f.dir.x + (pj.pos.y - my) * f.dir.y + (pj.pos.z - mz) * f.dir.z;
   864→      f.side = side;
   865→      if (side >= f.minSide) continue;
   866→      const corr = (f.minSide - side) * k;
   867→      pj.pos = vAdd(pj.pos, vScale(f.dir, corr * 0.7));
   868→      pa.pos = vSub(pa.pos, vScale(f.dir, corr * 0.15));
   869→      pb.pos = vSub(pb.pos, vScale(f.dir, corr * 0.15));
   870→    }
   871→  }
   872→
   873→  private solveCollisions(): void {
   874→    for (const p of this.particles) {
   875→      // Suelo: los pies y el resto del cuerpo.
   876→      if (p.collideFloor && p.pos.y < this.env.floorY + p.radius) {
   877→        p.pos.y = this.env.floorY + p.radius;
   878→        // Friccion simple: reduce la velocidad tangencial implicita.
   879→        p.prev.x += (p.pos.x - p.prev.x) * 0.25;
   880→        p.prev.z += (p.pos.z - p.prev.z) * 0.25;
   881→      }
   882→      // Pared: los segmentos del cuerpo no atraviesan el plano frontal.
   883→      // (Manos y pies quedan exentos: se apoyan en presas que sobresalen.)
   884→      if (p.collideWall && p.pos.z > this.env.wallFrontZ - p.radius) {
   885→        p.pos.z = this.env.wallFrontZ - p.radius;
   886→      }
   887→    }
   888→  }
   889→}

   891→function clampVecMag(v: V3, max: number): V3 {
   892→  const l = vLen(v);
   893→  if (l <= max || l < 1e-12) return v;
   894→  return vScale(v, max / l);
   895→}

   897→function jointUtilization(c: AngleConstraint): number {
   898→  const mid = (c.min + c.max) / 2;
   899→  const half = (c.max - c.min) / 2;
   900→  if (half <= 1e-9) return 0;
   901→  return clamp(Math.abs(c.currentAngle - mid) / half, 0, 1);
   902→}
