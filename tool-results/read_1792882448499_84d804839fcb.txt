   151→  static async create(canvas: HTMLCanvasElement): Promise<GameScene> {
   152→    // WebGL2 primero: es la ruta verificada en E2E y la universalmente soportada.
   153→    // WebGPU queda como respaldo (y en iframes cross-origin suele estar bloqueado
   154→    // por Permissions Policy, por lo que intentar primero WebGPU hacía fracasar
   155→    // el render en el preview).
   156→    let engine: AbstractEngine;
   157→    let backend = "WebGL 2";
   158→    try {
   159→      engine = new Engine(canvas, true, { stencil: true, preserveDrawingBuffer: false }, true);
   160→    } catch {
   161→      const gpu = new WebGPUEngine(canvas, { antialias: true, stencil: true });
   162→      await gpu.initAsync();
   163→      engine = gpu;
   164→      backend = "WebGPU";
   165→    }
   166→    const scene = new Scene(engine);
   167→    scene.useRightHandedSystem = false;
   168→    return new GameScene(engine, scene, backend, canvas);
   169→  }

   173→  buildWorld(wall: WallDefinition): void {
   174→    for (const hold of this.holdMeshes.values()) hold.dispose();
   175→    this.holdMeshes.clear();
   176→    this.holdBaseColor.clear();
   177→    this.wallMesh?.dispose();
   178→
   179→    const d = wall.dimensions;
   180→    const t = wall.transform.position;
   181→    const wallMesh = MeshBuilder.CreateBox("wall", { width: d.width, height: d.height, depth: d.depth }, this.scene);
   182→    wallMesh.position = new Vector3(t[0], t[1], t[2]);
   183→    wallMesh.isPickable = true;
   184→    wallMesh.material = wallMaterial(this.scene);
   185→    wallMesh.receiveShadows = true;
   186→    this.wallMesh = wallMesh;
   187→
   188→    const floor = MeshBuilder.CreateGround("floor", { width: 12, height: 12 }, this.scene);
   189→    const floorMat = new StandardMaterial("floorMat", this.scene);
   190→    floorMat.diffuseColor = Color3.FromHexString("#23262d");
   191→    floorMat.specularColor = new Color3(0.05, 0.05, 0.05);
   192→    floor.material = floorMat;
   193→    floor.isPickable = true;
   194→    floor.receiveShadows = true;
   195→
   196→    for (const hold of wall.holds) {
   197→      this.addHoldMesh(hold);
   198→    }
   199→  }

   201→  private addHoldMesh(hold: HoldDefinition): void {
   202→    const mesh = MeshBuilder.CreateSphere(`hold_${hold.id}`, { diameter: hold.radius * 2, segments: 8 }, this.scene);
   203→    mesh.position = new Vector3(hold.position[0], hold.position[1], hold.position[2]);
   204→    mesh.scaling.z = 0.62;
   205→    mesh.isPickable = true;
   206→    mesh.metadata = { holdId: hold.id };
   207→    const color = Color3.FromHexString(HOLD_COLORS[hold.type] ?? "#8a97a8");
   208→    const mat = new StandardMaterial(`holdMat_${hold.id}`, this.scene);
   209→    mat.diffuseColor = color;
   210→    mat.specularColor = new Color3(0.15, 0.15, 0.15);
   211→    mesh.material = mat;
   212→    this.holdMeshes.set(hold.id, mesh);
   213→    this.holdBaseColor.set(hold.id, color);
   214→  }

   218→  spawnRagdollVisuals(rig: RagdollRig): void {
   219→    this.despawnRagdollVisuals();
   220→    this.currentRig = rig;
   221→    const scene = this.scene;
   222→    const bodyMat = sharedMaterial("bodyMat", "#4b5563", scene);
   223→    const limbMat = sharedMaterial("limbMat", "#5d6672", scene);
   224→    const gripMat = sharedMaterial("gripMat", "#c7cdd6", scene);
   225→    const headMat = sharedMaterial("headMat", "#9aa3af", scene);
   226→
   227→    const P = rig.particles;
   228→    const p = (i: number) => P[i];
   229→    const seg = (name: string, ia: number, ib: number, radius: number, mat: StandardMaterial): void => {
   230→      const rest = Vector3.Distance(
   231→        new Vector3(P[ia].pos.x, P[ia].pos.y, P[ia].pos.z),
   232→        new Vector3(P[ib].pos.x, P[ib].pos.y, P[ib].pos.z)
   233→      );
   234→      const mesh = MeshBuilder.CreateCapsule(name, { radius, height: rest + radius * 2, tessellation: 10, capSubdivisions: 6 }, scene);
   235→      mesh.material = mat;
   236→      mesh.isPickable = false;
   237→      mesh.rotationQuaternion = Quaternion.Identity();
   238→      this.boneMeshes.push({ a: ia, b: ib, mesh, rest });
   239→      this.ragdollRoot.push(mesh);
   240→    };
   241→
   242→    seg("torso", 1, 4, 0.155, bodyMat);
   243→    seg("pelvisBar", 5, 6, 0.095, bodyMat);
   244→    seg("shoulderBar", 2, 3, 0.075, bodyMat);
   245→    seg("neck", 1, 0, 0.055, limbMat);
   246→    seg("upperArmL", 2, 7, 0.058, limbMat);
   247→    seg("forearmL", 7, 9, 0.05, limbMat);
   248→    seg("upperArmR", 3, 8, 0.058, limbMat);
   249→    seg("forearmR", 8, 10, 0.05, limbMat);
   250→    seg("thighL", 5, 11, 0.075, limbMat);
   251→    seg("shinL", 11, 13, 0.062, limbMat);
   252→    seg("thighR", 6, 12, 0.075, limbMat);
   253→    seg("shinR", 12, 14, 0.062, limbMat);
   254→
   255→    const head = MeshBuilder.CreateSphere("head", { diameter: 0.24, segments: 12 }, scene);
   256→    head.material = headMat;
   257→    head.isPickable = false;
   258→    this.boneMeshes.push({ a: 0, b: 0, mesh: head, rest: 1 });
   259→    this.ragdollRoot.push(head);
   260→
   261→    const grips: [LimbId, number, string, string][] = [
   262→      ["hand_l", 9, "handL", "#dfe4ea"],
   263→      ["hand_r", 10, "handR", "#dfe4ea"],
   264→      ["foot_l", 13, "footL", "#b9c0ca"],
   265→      ["foot_r", 14, "footR", "#b9c0ca"]
   266→    ];
   267→    for (const [limb, idx, name, color] of grips) {
   268→      const mesh = MeshBuilder.CreateSphere(name, { diameter: 0.15, segments: 8 }, scene);
   269→      mesh.material = sharedMaterial(`${name}Mat`, color, scene);
   270→      mesh.isPickable = true;
   271→      mesh.metadata = { limbId: limb };
   272→      mesh.rotationQuaternion = Quaternion.Identity();
   273→      // Sincronizada con el solver como articulacion de un solo punto (a === b).
   274→      this.boneMeshes.push({ a: idx, b: idx, mesh, rest: 1 });
   275→      this.limbVisual[limb] = {
   276→        hand: mesh,
   277→        segments: this.boneMeshes.filter((b) => b.mesh.name.includes(limb === "hand_l" ? "L" : limb === "hand_r" ? "R" : limb === "foot_l" ? "L" : "R")).map((b) => b.mesh)
   278→      };
   279→      this.ragdollRoot.push(mesh);
   280→    }
   281→
   282→    // Esferas de depuracion por particula.
   283→    this.jointSpheres = [];
   284→    for (let i = 0; i < P.length; i++) {
   285→      const s = MeshBuilder.CreateSphere(`joint_${i}`, { diameter: 0.07, segments: 6 }, scene);
   286→      s.material = emissiveMaterial(`jointMat_${i}`, Color3.FromHexString("#e05252"), scene);
   287→      s.isPickable = false;
   288→      s.setEnabled(false);
   289→      this.jointSpheres.push(s);
   290→    }
   291→  }

   294→    for (const b of this.boneMeshes) b.mesh.dispose();
   295→    for (const m of this.jointSpheres) m.dispose();
   296→    this.boneMeshes = [];
   297→    this.jointSpheres = [];
   298→    this.ragdollRoot = [];
   299→    this.currentRig = null;

   302→  /** Copia posiciones del solver a los meshes con interpolacion entre pasos. */
   303→  syncRagdoll(solver: PbdSolver, alpha: number): void {
   304→    void this.currentRig;
   305→    const tmpA = new Vector3();
   306→    const tmpB = new Vector3();
   307→    for (const bone of this.boneMeshes) {
   308→      const pa = solver.getParticle(bone.a);
   309→      const pb = solver.getParticle(bone.b);
   310→      if (!pa || !pb) continue;
   311→      if (bone.a === bone.b) {
   312→        tmpA.set(pa.pos.x, pa.pos.y, pa.pos.z);
   313→        bone.mesh.position.copyFrom(tmpA);
   314→        continue;
   315→      }
   316→      const ax = pa.prev.x + (pa.pos.x - pa.prev.x) * alpha;
   317→      const ay = pa.prev.y + (pa.pos.y - pa.prev.y) * alpha;
   318→      const az = pa.prev.z + (pa.pos.z - pa.prev.z) * alpha;
   319→      const bx = pb.prev.x + (pb.pos.x - pb.prev.x) * alpha;
   320→      const by = pb.prev.y + (pb.pos.y - pb.prev.y) * alpha;
   321→      const bz = pb.prev.z + (pb.pos.z - pb.prev.z) * alpha;
   322→      tmpA.set(ax, ay, az);
   323→      tmpB.set(bx, by, bz);
   324→      const mesh = bone.mesh;
   325→      mesh.position.set((ax + bx) / 2, (ay + by) / 2, (az + bz) / 2);
   326→      const dir = tmpB.subtract(tmpA);
   327→      const len = Math.max(dir.length(), 1e-5);
   328→      const d = dir.scale(1 / len);
   329→      const ref = Math.abs(d.y) < 0.95 ? new Vector3(0, 1, 0) : new Vector3(1, 0, 0);
   330→      const xAxis = Vector3.Cross(ref, d).normalize();
   331→      const zAxis = Vector3.Cross(xAxis, d).normalize();
   332→      if (!mesh.rotationQuaternion) mesh.rotationQuaternion = Quaternion.Identity();
   333→      Quaternion.RotationQuaternionFromAxisToRef(xAxis, d, zAxis, mesh.rotationQuaternion);
   334→      mesh.scaling.y = Math.min(1.35, len / bone.rest);
   335→    }
   336→  }

   343→    }

   365→  setHoldOccupied(holdId: string, occupied: boolean): void {
   366→    if (occupied) this.occupiedHolds.add(holdId);
   367→    else this.occupiedHolds.delete(holdId);
   368→    const mesh = this.holdMeshes.get(holdId);
   369→    const mat = mesh?.material as StandardMaterial | undefined;
   370→    const base = this.holdBaseColor.get(holdId);
   371→    if (mat && base) {
   372→      mat.diffuseColor = occupied ? base.scale(0.45) : base;
   373→    }

   376→  private refreshHoldHighlight(holdId: string): void {
   377→    const mesh = this.holdMeshes.get(holdId);
   378→    if (!mesh) return;
   379→    this.highlight.removeMesh(mesh);
   380→    if (this.highlightedInvalid.has(holdId)) {
   381→      this.highlight.addMesh(mesh, ACCENT_INVALID);
   382→    } else if (this.tutorialHighlightedHolds.has(holdId)) {
   383→      this.highlight.addMesh(mesh, ACCENT_SELECT);
   384→    } else if (this.highlightedCompatible.has(holdId)) {
   385→      this.highlight.addMesh(mesh, ACCENT_VALID);
   386→    }
   387→  }

   389→  // --- Resaltados del tutorial (prioridad media, gestionados cada frame) ---

   391→  setTutorialHoldHighlight(holdId: string | null): void {
   392→    const next = new Set(holdId ? [holdId] : []);
   393→    for (const id of this.tutorialHighlightedHolds) {
   394→      if (!next.has(id)) {
   395→        this.tutorialHighlightedHolds.delete(id);
   396→        this.refreshHoldHighlight(id);
   397→      }
   398→    }
   399→    for (const id of next) {
   400→      if (!this.tutorialHighlightedHolds.has(id)) {
   401→        this.tutorialHighlightedHolds.add(id);
   402→        this.refreshHoldHighlight(id);
   403→      }
   404→    }
   405→  }
   406→
   407→  setTutorialLimbHighlight(limb: LimbId | null): void {
   408→    // La malla se recrea en cada reset: seguimos la IDENTIDAD de la malla, no el id.
   409→    const raw = limb ? this.limbVisual[limb]?.hand ?? null : null;
   410→    const want = raw && !raw.isDisposed() ? raw : null;
   411→    if (want === this.tutorialLimbMesh && limb === this.tutorialLimbId) return;
   412→    if (this.tutorialLimbMesh) {
   413→      this.highlight.removeMesh(this.tutorialLimbMesh);
   414→      this.tutorialLimbMesh = null;
   415→    }
   416→    this.tutorialLimbId = limb;
   417→    if (want) {
   418→      this.highlight.addMesh(want, ACCENT_SELECT);
   419→      this.tutorialLimbMesh = want;
   420→    }
   421→  }

   425→  /** Anillo de extremidad: "grab" (arrastre activo) o "hover" (iman sutil del cursor). */
   426→  setSelectionRing(limb: LimbId | null, solver: PbdSolver, mode: "grab" | "hover" = "grab"): void {
   427→    if (!limb) {
   428→      this.selectionRing.setEnabled(false);
   429→      return;
   430→    }
   431→    const pp = this.limbPickPos(limb, solver);
   432→    const mat = this.selectionRing.material as StandardMaterial;
   433→    if (mode === "hover") {
   434→      this.selectionRing.scaling.setAll(0.86);
   435→      mat.alpha = 0.5;
   436→    } else {
   437→      this.selectionRing.scaling.setAll(1);
   438→      mat.alpha = 1;
   439→    }
   440→    this.selectionRing.setEnabled(true);
   441→    this.selectionRing.position.set(pp.x, pp.y, pp.z - 0.06);
   442→  }

   444→  /** Tinte de esfuerzo por extremidad (9.5): base -> rojo segun tension. */
   445→  updateLimbTint(limb: LimbId, strain: number): void {
   446→    const visual = this.limbVisual[limb];
   447→    if (!visual?.hand) return;
   448→    const base = Color3.FromHexString(limb.startsWith("hand") ? "#dfe4ea" : "#b9c0ca");

   507→  updateCamera(dt: number, pelvis: { x: number; y: number; z: number }, recenter: boolean): void {
   508→    const deadzoneHalf = 0.55;
   509→    const target = this.camera.target;
   510→    if (recenter) {
   511→      target.copyFrom(this.camTargetBase);
   512→    } else if (pelvis) {
   513→      const desiredY = Math.min(4.2, Math.max(1.0, pelvis.y));
   514→      const dy = desiredY - target.y;
   515→      if (Math.abs(dy) > deadzoneHalf) {
   516→        target.y += (dy - Math.sign(dy) * deadzoneHalf) * Math.min(1, dt * 3);
   517→      }
   518→      const desiredX = Math.min(1.6, Math.max(-1.6, pelvis.x));
   519→      const dx = desiredX - target.x;
   520→      if (Math.abs(dx) > deadzoneHalf) {
   521→        target.x += (dx - Math.sign(dx) * deadzoneHalf) * Math.min(1, dt * 3);
   522→      }
   523→    }
   524→    target.z = this.camTargetBase.z;
   525→    if (this.ortho) {
   526→      const aspect = this.engine.getRenderWidth() / Math.max(1, this.engine.getRenderHeight());
   527→      const halfH = 3.1;
   528→      this.camera.orthoLeft = -halfH * aspect;
   529→      this.camera.orthoRight = halfH * aspect;
   530→      this.camera.orthoBottom = -halfH;
   531→      this.camera.orthoTop = halfH;
   532→    }
   533→  }

   538→    return this.ortho;

   623→
   624→  /** Radio de iman en pixeles CSS por tipo de extremidad: los pies, mas grandes
   625→   * (son los mas dificiles de clicar: queja del usuario de agilidad). */
   626→  private static readonly PICK_MAGNET_PX: Record<string, number> = { hand: 46, foot: 60 };
   627→
   628→  /** Posicion de picking/visual de una extremidad: cuando mano y pie comparten presa
   629→   * (maxOccupants 2) coinciden en el mundo y seria imposible elegir; el pie se marca
   630→   * 7 cm por debajo y la mano 3 cm por encima (apilado legible y seleccionable). */
   631→  limbPickPos(limb: LimbId, solver: PbdSolver): { x: number; y: number; z: number } {
   632→    const p = solver.positionOfLimb(limb);
   633→    const holdId = solver.getAnchor(limb)?.holdId;
   634→    if (!holdId) return { x: p.x, y: p.y, z: p.z };
   635→    const shared = LIMB_IDS.some((other) => other !== limb && solver.getAnchor(other)?.holdId === holdId);
   636→    if (!shared) return { x: p.x, y: p.y, z: p.z };
   637→    const dy = limb.startsWith("foot") ? -0.07 : 0.03;
   638→    return { x: p.x, y: p.y + dy, z: p.z };
   639→  }
   640→
   641→  /** Seleccion de extremidad (7): 1) raycast exacto contra el mesh; 2) iman en espacio
   642→   * de pantalla: la extremidad proyectada mas cercana al cursor dentro de su radio.
   643→   * La pared bloquea el raycast directo (AT-03); el iman solo alcanza extremidades
   644→   * visibles delante del plano frontal. */
   645→  pickLimb(x: number, y: number, solver?: PbdSolver): LimbId | null {
   646→    const pick = this.scene.pick(
   647→      x,
   648→      y,
   649→      (m) => !!(m.metadata as { limbId?: LimbId } | undefined)?.limbId,
   650→      false,
   651→      this.camera
   652→    );
   653→    if (pick.hit && pick.pickedMesh) {
   654→      const wallPick = this.scene.pick(x, y, (m) => m.name === "wall", false, this.camera);
   655→      if (!(wallPick.hit && wallPick.distance < pick.distance - 0.15)) {
   656→        const meta = pick.pickedMesh.metadata as { limbId?: LimbId };
   657→        if (meta.limbId) return meta.limbId;
   658→      }
   659→    }
   660→    if (!solver) return null;
   661→    // Iman: proyecta las 4 extremidades y elige la mas cercana al cursor (px CSS).
   662→    let best: LimbId | null = null;
   663→    let bestDist = Infinity;
   664→    for (const limb of LIMB_IDS) {
   665→      const pp = this.limbPickPos(limb, solver);
   666→      const s = this.projectToScreenCss(pp.x, pp.y, pp.z);
   667→      const kind = limb.startsWith("hand") ? "hand" : "foot";
   668→      const radius = GameScene.PICK_MAGNET_PX[kind];
   669→      const d = Math.hypot(s.x - x, s.y - y);
   670→      if (d <= radius && d < bestDist) {
   671→        bestDist = d;
   672→        best = limb;
   673→      }
   674→    }
   675→    return best;
   676→  }

   678→  /** Proyeccion del cursor sobre el plano paralelo a la pared a la profundidad de agarre (7.1). */
   679→  wallPlanePoint(x: number, y: number, zPlane: number): Vector3 | null {
   680→    const ray = this.scene.createPickingRay(x, y, Matrix.Identity(), this.camera);
   681→    if (Math.abs(ray.direction.z) < 1e-6) return null;
   682→    const t = (zPlane - ray.origin.z) / ray.direction.z;
   683→    if (t <= 0) return null;
   684→    return ray.origin.add(ray.direction.scale(t));
   685→  }

   708→  private static circlePoints(radius: number, segments: number, center?: Vector3): Vector3[] {
   709→    const pts: Vector3[] = [];
   710→    const c = center ?? Vector3.Zero();
   711→    for (let i = 0; i <= segments; i++) {
   712→      const a = (i / segments) * Math.PI * 2;
   713→      pts.push(new Vector3(c.x + Math.cos(a) * radius, c.y + Math.sin(a) * radius, c.z));
   714→    }
   715→    return pts;
   716→  }

   719→function wallMaterial(scene: Scene): StandardMaterial {
   720→  const mat = new StandardMaterial("wallMat", scene);
   721→  const tex = new DynamicTexture("wallTex", { width: 512, height: 512 }, scene, false);
   722→  const ctx = tex.getContext() as CanvasRenderingContext2D;
   723→  ctx.fillStyle = "#5d646e";
   724→  ctx.fillRect(0, 0, 512, 512);
   725→  ctx.strokeStyle = "rgba(30,34,40,0.35)";
   726→  ctx.lineWidth = 2;
   727→  for (let i = 0; i <= 8; i++) {
   728→    const p = (i / 8) * 512;
   729→    ctx.beginPath();
   730→    ctx.moveTo(p, 0);
   731→    ctx.lineTo(p, 512);
   732→    ctx.stroke();
   733→    ctx.beginPath();
   734→    ctx.moveTo(0, p);
   735→    ctx.lineTo(512, p);
   736→    ctx.stroke();
   737→  }
   738→  ctx.fillStyle = "rgba(255,255,255,0.05)";
   739→  for (let i = 0; i < 900; i++) {
   740→    ctx.fillRect(Math.random() * 512, Math.random() * 512, 2, 2);
   741→  }
   742→  tex.update();
   743→  mat.diffuseTexture = tex;
   744→  mat.specularColor = new Color3(0.06, 0.06, 0.06);
   745→  return mat;
   746→}

   748→function sharedMaterial(name: string, hex: string, scene: Scene): StandardMaterial {
   749→  const existing = scene.getMaterialByName(name) as StandardMaterial | null;
   750→  if (existing) return existing;
   751→  const mat = new StandardMaterial(name, scene);
   752→  mat.diffuseColor = Color3.FromHexString(hex);
   753→  mat.specularColor = new Color3(0.12, 0.12, 0.12);
   754→  return mat;
   755→}

   757→function emissiveMaterial(name: string, color: Color3, scene: Scene): StandardMaterial {
   758→  const mat = new StandardMaterial(name, scene);
   759→  mat.emissiveColor = color;
   760→  mat.diffuseColor = Color3.Black();
   761→  mat.specularColor = Color3.Black();
   762→  mat.disableLighting = true;
   763→  return mat;
   764→}
