   649→  }

   651→  setProgress(ratio: number | null): void {

   653→      this.progressWrap.classList.add("hidden");

   655→    }

   665→  toast(text: string, tone: "info" | "warn" | "error" | "ok" = "info"): void {
   666→    const el = document.createElement("div");
   667→    const color = tone === "error" ? "border-red-500/50 text-red-300" : tone === "warn" ? "border-amber-500/50 text-amber-300" : tone === "ok" ? "border-emerald-500/50 text-emerald-300" : "border-zinc-600 text-zinc-300";
   668→    el.className = `rounded bg-zinc-950/85 px-3 py-1.5 text-[12px] backdrop-blur border ${color}`;
   669→    el.textContent = text;
   670→    this.toasts.appendChild(el);
   671→    window.setTimeout(() => {
   672→      el.style.transition = "opacity 0.4s";
   673→      el.style.opacity = "0";
   674→      window.setTimeout(() => el.remove(), 450);
   675→    }, 2200);
   676→  }

   678→  showHelp(on: boolean): void {

   693→  /**
