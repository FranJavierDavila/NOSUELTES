   247→   247→  private buildInteraction(): void {
   248→   248→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   249→   249→    const world: InteractionWorld = {
   250→   250→      holdById,
   251→   251→      challenge: this.challenge,
   252→   252→      character: this.character,
   253→   253→      profile: this.profile,
   254→   254→      wall: this.wall,
   255→   255→      anchorHoldOf: (limb) => this.solver.getAnchor(limb)?.holdId,
   256→   256→      holdOccupancy: (holdId) => {
   257→   257→        let n = 0;
   258→   258→        for (const limb of LIMB_IDS) {
   259→   259→          if (this.solver.getAnchor(limb)?.holdId === holdId) n++;
   260→   260→          if (this.interaction?.assistHoldOf(limb) === holdId) n++;
   261→   261→        }
   262→   262→        return n;
   263→   263→      },
   264→   264→      chainRootOf: (limb) => {
   265→   265→        const idx = limb === "hand_l" ? P.shL : limb === "hand_r" ? P.shR : limb === "foot_l" ? P.hipL : P.hipR;
   266→   266→        const p = this.solver.getParticle(idx);
   267→   267→        return v3(p.pos.x, p.pos.y, p.pos.z);
   268→   268→      },
   269→   269→      limbPositionOf: (limb) => this.solver.positionOfLimb(limb),
   270→   270→      chainLengthOf: (limb) => {
   271→   271→        if (!this.rig) return 0.6;
   272→   272→        const base = limb.startsWith("hand") ? this.rig.armChain : this.rig.legChain;
   273→   273→        return base * this.character.physical.reachScale;
   274→   274→      }
   275→   275→    };
   276→   276→    this.interaction = new InteractionController(world, {
   277→   277→      onSelectChanged: () => undefined,
   278→   278→      onAccept: (limb, holdId) => {
   279→   279→        const existing = this.solver.getAnchor(limb);
   280→   280→        if (existing) {
   281→   281→          this.solver.releaseAnchor(limb);
   282→   282→          this.events.emit({ type: "GripReleased", limb, holdId: existing.holdId, reason: "PLAYER_RELEASE" });
   283→   283→        }
   284→   284→        const hold = holdById.get(holdId)!;
   285→   285→        this.solver.setTarget(limb, v3(hold.position[0], hold.position[1], hold.position[2]), this.followForce);
   286→   286→        this.refreshHoldOccupancy();
   287→   287→      },
   288→   288→      onReject: (limb, holdId, reason) => {
   289→   289→        this.solver.clearTarget(limb);
   290→   290→        this.events.emit({ type: "HoldRejected", limb, holdId, reason });
   291→   291→        if (holdId) {
   292→   292→          this.scene.setHoldInvalid(holdId, true);
   293→   293→          window.setTimeout(() => this.scene.setHoldInvalid(holdId, false), 450);
   294→   294→        }
   295→   295→      },
   296→   296→      onArrival: (limb, holdId) => {
   297→   297→        this.solver.clearTarget(limb);
   298→   298→        const hold = holdById.get(holdId)!;
   299→   299→        this.solver.createAnchor(limb, holdId, v3(hold.position[0], hold.position[1], hold.position[2]), this.anchorMaxForce);
   300→   300→        this.events.emit({ type: "HoldAcquired", limb, holdId });
   301→   301→        this.refreshHoldOccupancy();
   302→   302→      }
   303→   303→    });
   304→   304→  }

   306→   306→  private refreshHoldOccupancy(): void {
   307→   307→    const anchored = new Set<string>();
   308→   308→    for (const limb of LIMB_IDS) {
   309→   309→      const a = this.solver.getAnchor(limb);
   310→   310→      if (a) anchored.add(a.holdId);
   311→   311→    }
   312→   312→    for (const hold of this.wall.holds) {
   313→   313→      this.scene.setHoldOccupied(hold.id, anchored.has(hold.id));
   314→   314→    }
   315→   315→  }

   319→   319→  private fixedStep(dt: number): void {
   320→   320→    const result = this.solver.step(dt);
   321→   321→
   322→   322→    // ERROR GUARD (9.2): carga no finita / NaN / energia anomala.
   323→   323→    if ((result.nanError || result.energySpike) && !this.guardActive) {
   324→   324→      this.guardActive = true;
   325→   325→      this.guardCooldown = 0.4;
   326→   326→      this.lastGuardError = result.nanError ? "nan" : "energia";
   327→   327→      this.events.emit({ type: "Message", text: "ERROR GUARD: simulacion no finita; reiniciando prueba", tone: "error" });
   328→   328→      this.machine.transition("ERROR");
   329→   329→    }
   330→   330→    if (this.machine.state === "ERROR") {
   331→   331→      this.guardCooldown -= dt;
   332→   332→      if (this.guardCooldown <= 0) {
   333→   333→        this.spawn();
   334→   334→        return;
   335→   335→      }
   336→   336→      return;
   337→   337→    }
   338→   338→
   339→   339→    const anchors = this.solver.anchorHoldIds();
   340→   340→
   341→   341→    // Biomecanica solo cuando el desafio esta activo (fatiga congelada en pausa/resultado).
   342→   342→    if (this.machine.state === "PLAYING") {
   343→   343→      const bioResult = this.bio.update(dt, result, anchors.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver.tremor);
   344→   344→      // En tutorial la fatiga y las zonas se miden (numeros reales en el paso
   345→   345→      // de vitals) pero la sobrecarga NO libera agarres: el onboarding no puede
   346→   346→      // guiar hacia una caida por mover lento.
   347→   347→      for (const limb of this.tutorial ? [] : bioResult.overloadReleases) {
   348→   348→        const anchor = this.solver.releaseAnchor(limb);
   349→   349→        if (anchor) {
   350→   350→          this.events.emit({ type: "GripReleased", limb, holdId: anchor.holdId, reason: "OVERLOAD" });
   351→   351→          this.refreshHoldOccupancy();
   352→   352→        }
   353→   353→      }
   354→   354→      for (const limb of bioResult.strainStarted) this.events.emit({ type: "StrainStarted", limb });
   355→   355→      for (const limb of bioResult.overloadStarted) this.events.emit({ type: "OverloadStarted", limb });
   356→   356→
   357→   357→      // Asistencia de viaje hacia la presa aceptada (7.2).
   358→   358→      this.interaction?.tick(dt);
   359→   359→
   360→   360→      const anchorsNow = this.solver.anchorHoldIds();
   361→   361→      const stability = computeStability(this.bio, result, anchorsNow.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver, this.profile);
   362→   362→      this.lastSnapshot = {
   363→   363→        stability,
   364→   364→        com: result.com,
   365→   365→        supportCount: anchorsNow.length
   366→   366→      };
   367→   367→
   368→   368→      // Evaluacion del desafio (10.3): caida -> exito -> timeout.
   369→   369→      // El timer no corre durante el tutorial: nadie deberia agotar el tiempo
   370→   370→      // mientras aprende los controles.
   371→   371→      if (!this.tutorial) this.gameTime += dt;
   372→   372→      const outcome = this.evaluator!.tick({
   373→   373→        dt,
   374→   374→        gameTime: this.gameTime,
   375→   375→        anchorsByLimb: Object.fromEntries(anchorsNow.map((a) => [a.limb, a.holdId])),
   376→   376→        supportCount: anchorsNow.length,
   377→   377→        stability,
   378→   378→        pelvisY: this.solver.getParticle(P.pelvisC).pos.y,
   379→   379→        outOfBounds: result.outOfBounds
   380→   380→      });
   381→   381→      if (outcome.kind === "fall") {
   382→   382→        this.events.emit({ type: "FallStarted", reason: outcome.reason });
   383→   383→        this.events.emit({ type: "ChallengeFailed", reason: "FALLEN" });
   384→   384→        this.machine.transition("FALLEN");
   385→   385→      } else if (outcome.kind === "success") {
   386→   386→        this.events.emit({ type: "ChallengeSuccess" });
   387→   387→        this.machine.transition("SUCCESS");
   388→   388→      } else if (outcome.kind === "timeout") {
   389→   389→        this.events.emit({ type: "ChallengeFailed", reason: "TIMEOUT" });
   390→   390→        this.machine.transition("TIMEOUT");
   391→   391→      }
   392→   392→    } else {
   393→   393→      // Fuera de PLAYING la fatiga no cambia; sin tremor.
   394→   394→      for (const limb of LIMB_IDS) this.solver.tremor[limb] = 0;
   395→   395→      this.interaction?.tick(0);
   396→   396→    }
   397→   397→  }

   399→   399→  // --- Bucle de render ---
   400→   400→
   401→   401→  private readonly frame = (nowMs: number): void => {
   402→   402→    if (this.disposed) return;
   403→   403→    this.raf = requestAnimationFrame(this.frame);
   404→   404→    this.frameCount++;
   405→   405→    const now = nowMs / 1000;
   406→   406→    const dtFrame = Math.min(Math.max(0, now - this.lastFrame), 0.1);
   407→   407→    this.lastFrame = now;
   408→   408→
   409→   409→    this.fpsFrames++;
   410→   410→    this.fpsElapsed += dtFrame;
   411→   411→    if (this.fpsElapsed >= 0.5) {
   412→   412→      this.fps = this.fpsFrames / this.fpsElapsed;
   413→   413→      this.fpsFrames = 0;
   414→   414→      this.fpsElapsed = 0;
   415→   415→    }
   416→   416→
   417→   417→    const state = this.machine.state;
   418→   418→    const stepping = state !== "PAUSED" && state !== "LOADING" && !this.paused;
   419→   419→    if (stepping) {
   420→   420→      this.accumulator += dtFrame;
   421→   421→      let steps = 0;
   422→   422→      const fixed = this.profile.sim.fixedDeltaSeconds;
   423→   423→      while (this.accumulator >= fixed && steps < MAX_STEPS_PER_FRAME) {
   424→   424→        const t0 = performance.now();
   425→   425→        this.fixedStep(fixed);
   426→   426→        this.physicsMs = lerp(this.physicsMs, performance.now() - t0, 0.15);
   427→   427→        this.accumulator -= fixed;
   428→   428→        steps++;
   429→   429→      }
   430→   430→      if (steps >= MAX_STEPS_PER_FRAME) this.accumulator = 0;
   431→   431→    }
   432→   432→
   433→   433→    if (state === "RESETTING") {
   434→   434→      this.settleTimer -= dtFrame;
   435→   435→      if (this.settleTimer <= 0) {
   436→   436→        this.machine.transition(this.autoStartPending ? "PLAYING" : "READY");
   437→   437→        if (this.autoStartPending) {
   438→   438→          this.events.emit({ type: "Message", text: "Recupera dos apoyos antes de perder el agarre", tone: "warn" });
   439→   439→        }
   440→   440→      }
   441→   441→    }
   442→   442→
   443→   443→    const alpha = Math.min(1, this.accumulator / this.profile.sim.fixedDeltaSeconds);
   444→   444→    this.scene.syncRagdoll(this.solver, alpha);
   445→   445→    this.scene.syncJointDebug(this.solver);
   446→   446→    this.scene.setSelectionRing(this.interaction?.dragging ? this.interaction.selected : this.interaction?.selected ?? null, this.solver);
   447→   447→
   448→   448→    // Preview de arrastre y alcance.
   449→   449→    const preview = this.interaction?.getPreview() ?? null;
   450→   450→    this.scene.clearHoldHighlights();
   451→   451→    if (preview?.snappedHoldId) {
   452→   452→      if (preview.validity === "valid") this.scene.setHoldCompatible(preview.snappedHoldId, true);
   453→   453→      else this.scene.setHoldInvalid(preview.snappedHoldId, true);
   454→   454→    }
   455→   455→    this.scene.showDragPreview(this.interaction?.dragging ? preview : null, this.solver);
   456→   456→    if (this.interaction?.dragging && preview) {
   457→   457→      this.scene.setReachRadius(
   458→   458→        (preview.limb.startsWith("hand") ? this.rig!.armChain : this.rig!.legChain) * this.character.physical.reachScale
   459→   459→      );
   460→   460→      this.scene.showReachPreview(preview.limb, this.solver, preview.validity !== "invalid");
   461→   461→    } else {
   462→   462→      this.scene.showReachPreview(null, this.solver, true);
   463→   463→    }
   464→   464→
   465→   465→    // Tutorial: resaltados y avance de pasos.
   466→   466→    this.tickTutorial();
   467→   467→
   468→   468→    // Tinte de esfuerzo por extremidad.
   469→   469→    for (const limb of LIMB_IDS) {
   470→   470→      this.scene.updateLimbTint(limb, this.solver.tremor[limb]);
   471→   471→    }
   472→   472→
   473→   473→    const pelvis = this.solver.getParticle(P.pelvisC);
   474→   474→    this.scene.updateCamera(dtFrame, pelvis.pos, this.recenterOnce);
   475→   475→    this.recenterOnce = false;
   476→   476→
   477→   477→    // HUD y debug a 10 Hz.
   478→   478→    this.hudAccum += dtFrame;
   479→   479→    if (this.hudAccum >= 0.1) {
   480→   480→      this.hudAccum = 0;
   481→   481→      this.updateHud(state);
   482→   482→      this.debug.update(this.buildDebugData(state));
   483→   483→    }
   484→   484→
   485→   485→    this.scene.render();
   486→   486→  };

   870→   870→    anchors: () => { limb: LimbId; hold: string; load: number }[];
   871→   871→    reject: (limb: LimbId, holdId: string) => string;
   872→   872→    chainInfo: (limb: LimbId) => { root: { x: number; y: number; z: number } };
   873→   873→    pickAt: (x: number, y: number) => LimbId | null;
   874→   874→    pickNearest: (x: number, y: number) => { name: string; dist: number } | null;
   875→   875→    handMeshes: () => { name: string; pickable: boolean; enabled: boolean; pos: { x: number; y: number; z: number } }[];
   876→   876→    renderSize: () => { w: number; h: number; cssW: number; cssH: number };
   877→   877→    project: (x: number, y: number, z: number) => { x: number; y: number };
   878→   878→    limbScreen: () => { limb: string; x: number; y: number }[];
   879→   879→  } {
   880→   880→    return {
   881→   881→      state: () => this.machine.state,
   882→   882→      frames: () => this.frameCount,
   883→   883→      settle: () => this.settleTimer,
   884→   884→      lastError: () => this.lastGuardError,
   885→   885→      selected: () => this.interaction?.selected ?? null,
   886→   886→      dragging: () => this.interaction?.dragging ?? false,
   887→   887→      preview: () => {
   888→   888→        const p = this.interaction?.getPreview();
   889→   889→        return p ? { limb: p.limb, snap: p.snappedHoldId, validity: p.validity, target: { x: Math.round(p.target.x * 100) / 100, y: Math.round(p.target.y * 100) / 100 } } : null;
   890→   890→      },
   891→   891→      pickAt: (x: number, y: number) => this.scene.pickLimb(x, y),
   892→   892→      pickNearest: (x: number, y: number) => this.scene.pickNearestDebug(x, y),
   893→   893→      handMeshes: () => this.scene.handMeshDebug(),
   894→   894→      renderSize: () => this.scene.renderSizeDebug(),
   895→   895→      project: (x: number, y: number, z: number) => this.scene.projectToScreen(x, y, z),
   896→   896→      anchors: () => this.solver.anchorHoldIds().map((a) => ({ limb: a.limb, hold: a.holdId, load: Math.round(a.loadN) })),
   897→   897→      reject: (limb: LimbId, holdId: string) => this.interaction?.rejectReason(limb, holdId) ?? "sin-interaccion",
   898→   898→      chainInfo: (limb: LimbId) => {
   899→   899→        const root = this.interaction ? this.interaction["world" as keyof object] : null;
   900→   900→        const shoulder = limb === "hand_l" ? this.solver.getParticle(2) : limb === "hand_r" ? this.solver.getParticle(3) : limb === "foot_l" ? this.solver.getParticle(5) : this.solver.getParticle(6);
   901→   901→        return { root: { x: +shoulder.pos.x.toFixed(2), y: +shoulder.pos.y.toFixed(2), z: +shoulder.pos.z.toFixed(2) } };
   902→   902→      },
   903→   903→      limbScreen: () => {
   904→   904→        const out: { limb: string; x: number; y: number }[] = [];

   906→   906→          const p = this.solver.positionOfLimb(limb);
   907→   907→          const s = this.scene.projectToScreen(p.x, p.y, p.z);
   908→   908→          out.push({ limb, x: s.x, y: s.y });
   909→   909→        }
   910→   910→        return out;
   911→   911→      }
   912→   912→    };
   913→   913→  }
