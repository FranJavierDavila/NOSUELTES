   228→  static async create(canvas: HTMLCanvasElement): Promise<GameScene> {
   229→    // WebGL2 primero: es la ruta verificada en E2E y la universalmente soportada.
   230→    // WebGPU queda como respaldo (y en iframes cross-origin suele estar bloqueado
   231→    // por Permissions Policy, por lo que intentar primero WebGPU hacía fracasar
   232→    // el render en el preview).
   233→    let engine: AbstractEngine;
   234→    let backend = "WebGL 2";
   235→    try {
   236→      engine = new Engine(canvas, true, { stencil: true, preserveDrawingBuffer: false }, true);
   237→    } catch {
   238→      const gpu = new WebGPUEngine(canvas, { antialias: true, stencil: true });
   239→      await gpu.initAsync();
   240→      engine = gpu;
   241→      backend = "WebGPU";
   242→    }
   243→    const scene = new Scene(engine);
   244→    scene.useRightHandedSystem = false;
   245→    return new GameScene(engine, scene, backend, canvas);
   246→  }

   250→  buildWorld(wall: WallDefinition): void {
   251→    for (const hold of this.holdMeshes.values()) hold.dispose();
   252→    this.holdMeshes.clear();
   253→    this.holdBaseColor.clear();
   254→    this.holdRadius.clear();
   255→    this.wallMesh?.dispose();
   256→
   257→    const d = wall.dimensions;
   258→    const t = wall.transform.position;
   259→    // La camara debe poder seguir al escalador hasta cerca de la cima (campeonato:
   260→    // paredes de 6 a 10,5 m) y encuadrar travesias hasta el borde de la pared.
   261→    this.wallTopY = t[1] + d.height / 2;
   262→    this.wallHalfWidth = d.width / 2;
   263→    const wallMesh = MeshBuilder.CreateBox("wall", { width: d.width, height: d.height, depth: d.depth }, this.scene);
   264→    wallMesh.position = new Vector3(t[0], t[1], t[2]);
   265→    wallMesh.isPickable = true;
   266→    wallMesh.material = wallMaterial(this.scene);
   267→    wallMesh.receiveShadows = true;
   268→    this.wallMesh = wallMesh;
   269→
   270→    const floor = MeshBuilder.CreateGround("floor", { width: 12, height: 12 }, this.scene);
   271→    const floorMat = new StandardMaterial("floorMat", this.scene);
   272→    floorMat.diffuseColor = Color3.FromHexString("#23262d");
   273→    floorMat.specularColor = new Color3(0.05, 0.05, 0.05);
   274→    floor.material = floorMat;
   275→    floor.isPickable = true;
   276→    floor.receiveShadows = true;
   277→
   278→    for (const hold of wall.holds) {
   279→      this.addHoldMesh(hold);
   280→    }
   281→  }

   283→  private addHoldMesh(hold: HoldDefinition): void {
   284→    const mesh = MeshBuilder.CreateSphere(`hold_${hold.id}`, { diameter: hold.radius * 2, segments: 8 }, this.scene);
   285→    mesh.position = new Vector3(hold.position[0], hold.position[1], hold.position[2]);
   286→    mesh.scaling.z = 0.62;
   287→    mesh.isPickable = true;
   288→    mesh.metadata = { holdId: hold.id };
   289→    const color = Color3.FromHexString(HOLD_COLORS[hold.type] ?? "#8a97a8");
   290→    const mat = new StandardMaterial(`holdMat_${hold.id}`, this.scene);
   291→    mat.diffuseColor = color;
   292→    mat.specularColor = new Color3(0.15, 0.15, 0.15);
   293→    mesh.material = mat;
   294→    this.holdMeshes.set(hold.id, mesh);
   295→    this.holdBaseColor.set(hold.id, color);
   296→    this.holdRadius.set(hold.id, hold.radius);
   297→  }

   301→  spawnRagdollVisuals(rig: RagdollRig): void {
   302→    this.despawnRagdollVisuals();
   303→    this.currentRig = rig;
   304→    const scene = this.scene;
   305→    const bodyMat = sharedMaterial("bodyMat", "#4b5563", scene);
   306→    const headMat = sharedMaterial("headMat", "#9aa3af", scene);
   307→    const bagMat = sharedMaterial("bagMat", BAG_COLOR, scene);
   308→    const markMat = emissiveMaterial("markMat", Color3.FromHexString(MARKER_COLOR), scene);
   309→    // Materiales POR EXTREMIDAD: el tinte de zona (verde/ambar/rojo) se aplica
   310→    // de forma independiente a cada brazo y cada pierna.
   311→    const limbMats: Record<LimbId, StandardMaterial> = {
   312→      hand_l: sharedMaterial("limbMat_hand_l", SEG_BASE, scene),
   313→      hand_r: sharedMaterial("limbMat_hand_r", SEG_BASE, scene),
   314→      foot_l: sharedMaterial("limbMat_foot_l", SEG_BASE, scene),
   315→      foot_r: sharedMaterial("limbMat_foot_r", SEG_BASE, scene)
   316→    };
   317→    const segOfLimb: Record<LimbId, Mesh[]> = { hand_l: [], hand_r: [], foot_l: [], foot_r: [] };
   318→
   319→    const P = rig.particles;
   320→    const dist = (ia: number, ib: number): number =>
   321→      Vector3.Distance(
   322→        new Vector3(P[ia].pos.x, P[ia].pos.y, P[ia].pos.z),
   323→        new Vector3(P[ib].pos.x, P[ib].pos.y, P[ib].pos.z)
   324→      );
   325→    const seg = (name: string, ia: number, ib: number, radius: number, mat: StandardMaterial, limb?: LimbId): void => {
   326→      const rest = dist(ia, ib);
   327→      const mesh = MeshBuilder.CreateCapsule(name, { radius, height: rest + radius * 2, tessellation: 10, capSubdivisions: 6 }, scene);
   328→      mesh.material = mat;
   329→      mesh.isPickable = false;
   330→      mesh.rotationQuaternion = Quaternion.Identity();
   331→      this.boneMeshes.push({ a: ia, b: ib, mesh, rest });
   332→      this.ragdollRoot.push(mesh);
   333→      if (limb) segOfLimb[limb].push(mesh);
   334→    };
   335→
   336→    seg("torso", 1, 4, 0.155, bodyMat);
   337→    seg("pelvisBar", 5, 6, 0.095, bodyMat);
   338→    seg("shoulderBar", 2, 3, 0.075, bodyMat);
   339→    seg("neck", 1, 0, 0.055, bodyMat);
   340→    seg("upperArmL", 2, 7, 0.058, limbMats.hand_l, "hand_l");
   341→    seg("forearmL", 7, 9, 0.05, limbMats.hand_l, "hand_l");
   342→    seg("upperArmR", 3, 8, 0.058, limbMats.hand_r, "hand_r");
   343→    seg("forearmR", 8, 10, 0.05, limbMats.hand_r, "hand_r");
   344→    seg("thighL", 5, 11, 0.075, limbMats.foot_l, "foot_l");
   345→    seg("shinL", 11, 13, 0.062, limbMats.foot_l, "foot_l");
   346→    seg("thighR", 6, 12, 0.075, limbMats.foot_r, "foot_r");
   347→    seg("shinR", 12, 14, 0.062, limbMats.foot_r, "foot_r");
   348→
   349→    const head = MeshBuilder.CreateSphere("head", { diameter: 0.24, segments: 12 }, scene);
   350→    head.material = headMat;
   351→    head.isPickable = false;
   352→    this.boneMeshes.push({ a: 0, b: 0, mesh: head, rest: 1 });
   353→    this.ragdollRoot.push(head);
   354→
   355→    // --- Marcas de orientacion frente/espalda ---
   356→    // El cuerpo vive en un plano paralelo a la pared: espalda = -z (hacia la
   357→    // camara), frente = +z (hacia la pared). La mochila (grande, en -z) es la
   358→    // referencia de espalda; el visor facial y los discos de rodilla/codo (+z)
   359→    // marcan el frente. La sincronizacion aplica offZ/offX/offY en el mundo.
   360→    const torsoLen = dist(1, 4);
   361→    const bag = MeshBuilder.CreateBox("backpack", { width: 0.26, height: torsoLen * 0.84, depth: 0.105 }, scene);
   362→    bag.material = bagMat;
   363→    bag.isPickable = false;
   364→    bag.rotationQuaternion = Quaternion.Identity();
   365→    this.boneMeshes.push({ a: 1, b: 4, mesh: bag, rest: torsoLen, offZ: -0.225 });
   366→    this.ragdollRoot.push(bag);
   367→    // Banda vertical clara sobre la cara externa de la mochila (refleja el turquesa
   368→    // de los marcadores frontales: mismo color = lados opuestos del mismo eje).
   369→    const bagStripe = MeshBuilder.CreateBox("backpackStripe", { width: 0.06, height: torsoLen * 0.78, depth: 0.02 }, scene);
   370→    bagStripe.material = markMat;
   371→    bagStripe.isPickable = false;
   372→    bagStripe.rotationQuaternion = Quaternion.Identity();
   373→    this.boneMeshes.push({ a: 1, b: 4, mesh: bagStripe, rest: torsoLen, offZ: -0.283 });
   374→    this.ragdollRoot.push(bagStripe);
   375→    // Tirantes sobre los hombros: refuerzan la lectura "esto es la espalda" incluso
   376→    // en silueta (se ven desde detras, a diferencia de las marcas frontales).
   377→    const strapMat = sharedMaterial("strapMat", "#33281e", scene);
   378→    for (const [idx, nm] of [[2, "strapL"], [3, "strapR"]] as const) {
   379→      const strap = MeshBuilder.CreateBox(nm, { width: 0.07, height: 0.13, depth: 0.05 }, scene);
   380→      strap.material = strapMat;
   381→      strap.isPickable = false;
   382→      strap.rotationQuaternion = Quaternion.Identity();
   383→      this.boneMeshes.push({ a: idx, b: idx, mesh: strap, rest: 1, offZ: -0.1, offY: 0.015 });
   384→      this.ragdollRoot.push(strap);
   385→    }
   386→
   387→    const eyeMat = sharedMaterial("eyeMat", "#141821", scene);
   388→    const marker = (name: string, idx: number, offZ: number, d: number, zScale: number, offX = 0, offY = 0, mat: StandardMaterial = markMat): void => {
   389→      const disc = MeshBuilder.CreateSphere(name, { diameter: d, segments: 8 }, scene);
   390→      disc.material = mat;
   391→      disc.scaling.z = zScale;
   392→      disc.isPickable = false;
   393→      // Punto anclado (a === b): la sincronizacion solo copia posicion (+offsets).
   394→      this.boneMeshes.push({ a: idx, b: idx, mesh: disc, rest: 1, offZ, offX, offY });
   395→      this.ragdollRoot.push(disc);
