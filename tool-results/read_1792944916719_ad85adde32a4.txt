   300→  setState(state: GameState): void {

   303→    this.stateBadge.className = "rounded px-2 py-0.5 text-[11px] font-semibold " + (
   304→      state === "SUCCESS" ? "bg-emerald-500/20 text-emerald-300"
   305→      : state === "FALLEN" || state === "TIMEOUT" ? "bg-red-500/20 text-red-300"
   306→      : state === "ERROR" ? "bg-red-500/20 text-red-300"
   307→      : state === "PLAYING" ? "bg-amber-500/20 text-amber-300"
   308→      : "bg-zinc-800 text-zinc-300"
   309→    );
   310→    this.pausedBadge.classList.toggle("hidden", state !== "PAUSED");
   311→    if (!terminal) this.showBanner(null);
   312→  }

   336→  setLimbStates(states: Record<LimbId, { zone: LoadZone; anchored: boolean; label: string; overloadS?: number | null; overloadFrac?: number | null }>): void {
   337→    for (const limb of LIMB_IDS) {
   338→      const chip = this.limbChips.get(limb)!;
   339→      const bar = this.limbBars.get(limb)!;
   340→      const s = states[limb];
   341→      const overloaded = s.anchored && s.zone !== "NORMAL" && s.zone !== "STRAINED";
   342→      if (overloaded && typeof s.overloadS === "number") {
   343→        chip.textContent = `SOBRECARGA ${s.overloadS.toFixed(1).replace(".", ",")}s`;
   344→      } else {
   345→        chip.textContent = s.anchored ? (s.zone === "NORMAL" ? "FIJADA" : s.zone === "STRAINED" ? "TENSION" : "SOBRECARGA") : s.label;
   346→      }
   347→      chip.className = "font-mono font-bold " + (s.anchored ? "" : "text-zinc-500");
   348→      chip.style.color = s.anchored ? ZONE_COLOR[s.zone] : "";
   349→      // Barra de cuenta atras de fallo: visible solo en sobrecarga con temporizador.
   350→      if (overloaded && typeof s.overloadFrac === "number") {
   351→        bar.wrap.classList.remove("hidden");
   352→        bar.fill.style.width = `${Math.round(Math.min(1, Math.max(0, s.overloadFrac)) * 100)}%`;
   353→        bar.fill.className = "h-full " + (s.overloadFrac < 0.25 ? "bg-red-400" : "bg-red-500");
   354→      } else {
   355→        bar.wrap.classList.add("hidden");
   356→      }
   357→    }
   358→  }
