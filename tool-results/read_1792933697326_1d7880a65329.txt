   386→   386→  }

   388→   388→  private loadTune(): void {
   389→   389→    try {
   390→   390→      const preset = window.localStorage.getItem(GameApp.PRESET_KEY) as BalancePresetName | null;
   391→   391→      if (preset && preset in PRESET_TUNES) this.presetName = preset;

   393→   393→      this.presetName = null;
   394→   394→    }
   395→   395→    if (this.presetName) {
   396→   396→      this.tune = { ...PRESET_TUNES[this.presetName] };
   397→   397→      return;
   398→   398→    }
   399→   399→    try {
   400→   400→      const raw = window.localStorage.getItem(GameApp.TUNE_KEY);
   401→   401→      if (raw) {
   402→   402→        const t = JSON.parse(raw) as Partial<BalanceTuning>;
   403→   403→        this.tune = {
   404→   404→          jointLimitStiffness: clamp01(t.jointLimitStiffness ?? 1.0),
   405→   405→          foldStrength: clamp01(t.foldStrength ?? 0.85),
   406→   406→          postureTone: clamp01(t.postureTone ?? 0.75),
   407→   407→          toneFloor: clamp01(t.toneFloor ?? 0.12),
   408→   408→          stretchTolerance: clampRange(t.stretchTolerance ?? 1.1, 1.02, 1.35)
   409→   409→        };
   410→   410→      }
   411→   411→    } catch {
   412→   412→      /* sin persistencia: defaults */
   413→   413→    }
   414→   414→  }

   417→   417→    try {
