   277→  async loadChallenge(id: string): Promise<void> {
   278→    this.stopTutorial(false);
   279→    this.challenge = this.repo.getChallenge(id);
   280→    this.character = this.repo.getCharacter(this.challenge.characterId);
   281→    this.profile = this.repo.getBodyProfile(this.character.bodyProfileId);
   282→    this.wall = this.repo.getWall(this.challenge.wallId);
   283→    this.holdPosById = new Map(this.wall.holds.map((h) => [h.id, h.position] as const));
   284→    this.bio = new BiomechanicsSystem(this.character, this.profile);
   285→    this.evaluator = new ChallengeEvaluator(this.challenge);
   286→
   287→    const strengthFactor = lerp(this.profile.biomech.strengthFactorMin, this.profile.biomech.strengthFactorMax, this.character.virtues.strength);
   288→    this.anchorMaxForce = this.character.physical.baseGripForce * strengthFactor;
   289→    this.followForce = this.character.physical.baseGripForce * this.character.control.gripAssist * this.character.control.targetFollowStrength;
   290→
   291→    this.solver = new PbdSolver({
   292→      gravity: v3(...this.profile.sim.gravity),
   293→      damping: this.profile.sim.damping,
   294→      iterations: this.profile.sim.iterations,
   295→      floorY: 0,
   296→      wallFrontZ: this.wall.transform.position[2] - this.wall.dimensions.depth / 2,
   297→      tremorAmplitude: this.profile.sim.tremorAmplitude,
   298→      loadEstimatorGain: this.profile.biomech.loadEstimatorGain
   299→    });
   300→
   301→    this.scene.buildWorld(this.wall);
   302→    this.hud.setChallenge(this.challenge, CHALLENGE_META[id]?.label ?? CHALLENGE_LABELS[this.challenge.type]);
   303→    this.buildInteraction();
   304→    this.spawn();
   305→  }

   330→    // El combo de flujo se rompe con cada respawn (intento nuevo); el conteo de
   331→    // caidas persiste para la puntuacion del desafio cargado.
   332→    this.flowCount = 0;

   382→  private loadTune(): void {
   383→    try {
   384→      const preset = window.localStorage.getItem(GameApp.PRESET_KEY) as BalancePresetName | null;
   385→      if (preset && preset in PRESET_TUNES) this.presetName = preset;
   386→    } catch {
   387→      this.presetName = null;
   388→    }
   389→    if (this.presetName) {
   390→      this.tune = { ...PRESET_TUNES[this.presetName] };
   391→      return;
   392→    }
   393→    try {
   394→      const raw = window.localStorage.getItem(GameApp.TUNE_KEY);
   395→      if (raw) {
   396→        const t = JSON.parse(raw) as Partial<BalanceTuning>;
   397→        this.tune = {
   398→          jointLimitStiffness: clamp01(t.jointLimitStiffness ?? 1.0),
   399→          foldStrength: clamp01(t.foldStrength ?? 0.85),
   400→          postureTone: clamp01(t.postureTone ?? 0.75),
   401→          toneFloor: clamp01(t.toneFloor ?? 0.12),
   402→          stretchTolerance: clampRange(t.stretchTolerance ?? 1.1, 1.02, 1.35)
   403→        };
   404→      }
   405→    } catch {
   406→      /* sin persistencia: defaults */
   407→    }
   408→  }

   418→  setTune(patch: Partial<BalanceTuning>): BalanceTuning {
   419→    this.tune = {
   420→      jointLimitStiffness: clamp01(patch.jointLimitStiffness ?? this.tune.jointLimitStiffness),
   421→      foldStrength: clamp01(patch.foldStrength ?? this.tune.foldStrength),
   422→      postureTone: clamp01(patch.postureTone ?? this.tune.postureTone),
   423→      toneFloor: clamp01(patch.toneFloor ?? this.tune.toneFloor),
   424→      stretchTolerance: clampRange(patch.stretchTolerance ?? this.tune.stretchTolerance, 1.02, 1.35)
   425→    };
   426→    // Un ajuste manual (sliders F3 o __NS.setTune) deja de corresponder al preset.
   427→    this.presetName = null;
   428→    try {
   429→      window.localStorage.removeItem(GameApp.PRESET_KEY);
   430→    } catch {
   431→      /* sin persistencia */
   432→    }
   433→    this.hud?.setPresetActive(null);
   434→    this.applyTuneToSolver();
   435→    this.saveTune();
   436→    return { ...this.tune };
   437→  }

   439→  private applyTuneToSolver(): void {
   440→    if (!this.solver) return;
   441→    this.solver.env.balance.jointLimitStiffness = this.tune.jointLimitStiffness;
   442→    this.solver.env.balance.foldStrength = this.tune.foldStrength;
   443→    this.solver.env.balance.postureTone = this.tune.postureTone;
   444→    this.solver.env.balance.toneFloor = this.tune.toneFloor;
   445→    this.solver.env.balance.stretchTolerance = this.tune.stretchTolerance;
   446→    // En FALLEN el tono objetivo sigue siendo el flacido; en el resto, el calibrado.
   447→    this.solver.setToneTarget(this.machine.state === "FALLEN" ? this.tune.toneFloor : this.tune.postureTone);
   448→  }

   450→  requestReset(): void {
   451→    if (this.machine.state === "LOADING" || this.machine.state === "RESETTING") return;
   452→    this.events.emit({ type: "ResetStarted" });
   453→    this.spawn();
   454→  }

   464→      const raw = window.localStorage.getItem(GameApp.bestKey(id));

   473→    if (prev && !isBetter(rec, prev)) return false;
   474→    try {
   475→      window.localStorage.setItem(GameApp.bestKey(id), JSON.stringify(rec));

   477→      /* sin persistencia */
   478→    }
   479→    return true;

   482→  // --- Interaccion ---

   484→  private buildInteraction(): void {
   485→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   486→    const world: InteractionWorld = {
   487→      holdById,
   488→      challenge: this.challenge,
   489→      character: this.character,
   490→      profile: this.profile,
   491→      wall: this.wall,
   492→      anchorHoldOf: (limb) => this.solver.getAnchor(limb)?.holdId,
   493→      holdOccupancy: (holdId) => {
   494→        let n = 0;
   495→        for (const limb of LIMB_IDS) {
   496→          if (this.solver.getAnchor(limb)?.holdId === holdId) n++;
   497→          if (this.interaction?.assistHoldOf(limb) === holdId) n++;
   498→        }
   499→        return n;
   500→      },
   501→      chainRootOf: (limb) => {
   502→        const idx = limb === "hand_l" ? P.shL : limb === "hand_r" ? P.shR : limb === "foot_l" ? P.hipL : P.hipR;
   503→        const p = this.solver.getParticle(idx);
   504→        return v3(p.pos.x, p.pos.y, p.pos.z);
   505→      },
   506→      limbPositionOf: (limb) => this.solver.positionOfLimb(limb),
   507→      chainLengthOf: (limb) => {
   508→        if (!this.rig) return 0.6;
   509→        const base = limb.startsWith("hand") ? this.rig.armChain : this.rig.legChain;
   510→        return base * this.character.physical.reachScale;
   511→      }
   512→    };
   513→    this.interaction = new InteractionController(world, {
   514→      onSelectChanged: () => undefined,
   515→      onAccept: (limb, holdId) => {
   516→        const existing = this.solver.getAnchor(limb);
   517→        if (existing) {
   518→          this.solver.releaseAnchor(limb);
   519→          this.events.emit({ type: "GripReleased", limb, holdId: existing.holdId, reason: "PLAYER_RELEASE" });
   520→        }
   521→        const hold = holdById.get(holdId)!;
   522→        this.solver.setTarget(limb, v3(hold.position[0], hold.position[1], hold.position[2]), this.followForce);
   523→        this.refreshHoldOccupancy();
   524→      },
   525→      onReject: (limb, holdId, reason) => {
   526→        this.solver.clearTarget(limb);
   527→        this.events.emit({ type: "HoldRejected", limb, holdId, reason });
   528→        if (holdId) {
   529→          this.scene.setHoldInvalid(holdId, true);
   530→          window.setTimeout(() => this.scene.setHoldInvalid(holdId, false), 450);
   531→        }
   532→      },
   533→      onArrival: (limb, holdId) => {
   534→        this.solver.clearTarget(limb);
   535→        const hold = holdById.get(holdId)!;
   536→        this.solver.createAnchor(limb, holdId, v3(hold.position[0], hold.position[1], hold.position[2]), this.anchorMaxForceFor(limb));
   537→        // ADR-007: cada apoyo consolidado reactiva el impulso de subida (el cuerpo
   538→        // se reorganiza hacia la nueva postura: step-up del pie, pull de brazos).
   539→        this.solver.pulseClimbDrive();
   540→        this.events.emit({ type: "HoldAcquired", limb, holdId });
   541→        this.refreshHoldOccupancy();
   542→      }
   543→    });
   544→  }
   545→
   546→  private refreshHoldOccupancy(): void {
   547→    const anchored = new Set<string>();
   548→    for (const limb of LIMB_IDS) {
   549→      const a = this.solver.getAnchor(limb);
   550→      if (a) anchored.add(a.holdId);
   551→    }
   552→    for (const hold of this.wall.holds) {
   553→      this.scene.setHoldOccupied(hold.id, anchored.has(hold.id));
   554→    }
   555→  }

   557→  // --- Pipeline por paso fijo ---
   558→
   559→  private fixedStep(dt: number): void {
   560→    const result = this.solver.step(dt);
   561→
   562→    // ERROR GUARD (9.2): carga no finita / NaN / energia anomala.
   563→    if ((result.nanError || result.energySpike) && !this.guardActive) {
   564→      this.guardActive = true;
   565→      this.guardCooldown = 0.4;
   566→      this.lastGuardError = result.nanError ? "nan" : "energia";
   567→      this.events.emit({ type: "Message", text: "ERROR GUARD: simulacion no finita; reiniciando prueba", tone: "error" });
   568→      this.machine.transition("ERROR");
   569→    }
   570→    if (this.machine.state === "ERROR") {
   571→      this.guardCooldown -= dt;
   572→      if (this.guardCooldown <= 0) {
   573→        this.spawn();
   574→        return;
   575→      }
   576→      return;
   577→    }
   578→
   579→    const anchors = this.solver.anchorHoldIds();
   580→
   581→    // Biomecanica solo cuando el desafio esta activo (fatiga congelada en pausa/resultado).
   582→    if (this.machine.state === "PLAYING") {
   583→      const bioResult = this.bio.update(dt, result, anchors.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver.tremor);
   584→      // En tutorial la fatiga y las zonas se miden (numeros reales en el paso
   585→      // de vitals) pero la sobrecarga NO libera agarres: el onboarding no puede
   586→      // guiar hacia una caida por mover lento.
   587→      for (const limb of this.tutorial ? [] : bioResult.overloadReleases) {
   588→        const anchor = this.solver.releaseAnchor(limb);
   589→        if (anchor) {
   590→          this.events.emit({ type: "GripReleased", limb, holdId: anchor.holdId, reason: "OVERLOAD" });
   591→          this.refreshHoldOccupancy();
   592→        }
   593→      }
   594→      for (const limb of bioResult.strainStarted) this.events.emit({ type: "StrainStarted", limb });
   595→      for (const limb of bioResult.overloadStarted) this.events.emit({ type: "OverloadStarted", limb });
   596→
   597→      // Asistencia de viaje hacia la presa aceptada (7.2).
   598→      this.interaction?.tick(dt);
   599→
   600→      const anchorsNow = this.solver.anchorHoldIds();
   601→      const stability = computeStability(this.bio, result, anchorsNow.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver, this.profile);
   602→      this.lastSnapshot = {
   603→        stability,
   604→        com: result.com,
   605→        supportCount: anchorsNow.length
   606→      };
   607→
   608→      // Evaluacion del desafio (10.3): caida -> exito -> timeout.
   609→      // El timer no corre durante el tutorial: nadie deberia agotar el tiempo
   610→      // mientras aprende los controles.
   611→      if (!this.tutorial) this.gameTime += dt;
   612→      const outcome = this.evaluator!.tick({
   613→        dt,
   614→        gameTime: this.gameTime,
   615→        anchorsByLimb: Object.fromEntries(anchorsNow.map((a) => [a.limb, a.holdId])),
   616→        supportCount: anchorsNow.length,
   617→        stability,
   618→        pelvisY: this.solver.getParticle(P.pelvisC).pos.y,
   619→        outOfBounds: result.outOfBounds
   620→      });
   621→      if (outcome.kind === "fall") {
   622→        this.events.emit({ type: "FallStarted", reason: outcome.reason });
   623→        this.events.emit({ type: "ChallengeFailed", reason: "FALLEN" });
   624→        this.machine.transition("FALLEN");
   625→        this.beginFallSequence();
   626→      } else if (outcome.kind === "success") {
   627→        this.events.emit({ type: "ChallengeSuccess" });
   628→        this.machine.transition("SUCCESS");
   629→      } else if (outcome.kind === "timeout") {
   630→        this.events.emit({ type: "ChallengeFailed", reason: "TIMEOUT" });
   631→        this.machine.transition("TIMEOUT");
   632→      }
   633→    } else {
   634→      // Fuera de PLAYING la fatiga no cambia; sin tremor.
   635→      for (const limb of LIMB_IDS) this.solver.tremor[limb] = 0;
   636→      this.interaction?.tick(0);
   637→    }
   638→
   639→    // Secuencia de caida (ADR-008): en FALLEN los agarres se abren escalonados y el
   640→    // cuerpo cae al colchon; tras asentarse, reintento automatico (R lo adelanta).
   641→    if (this.machine.state === "FALLEN") {
   642→      this.fallTimer += dt;
   643→      while (this.fallReleaseQueue.length > 0 && this.fallTimer >= this.fallReleaseQueue[0].at) {
   644→        const next = this.fallReleaseQueue.shift()!;
   645→        const anchor = this.solver.releaseAnchor(next.limb);
   646→        if (anchor) {
   647→          this.events.emit({ type: "GripReleased", limb: next.limb, holdId: anchor.holdId, reason: "FALL" });
   648→          this.refreshHoldOccupancy();
   649→        }
   650→      }
   651→      // Impacto contra el colchon: polvo + sacudida (una sola vez por caida).
   652→      if (!this.fallLandedFx && this.fallTimer >= LANDING_FX_AT_S) {
   653→        this.fallLandedFx = true;
   654→        const pv = this.solver.getParticle(P.pelvisC);
   655→        this.scene.burstDust(new Vector3(pv.pos.x, 0.06, pv.pos.z), 42);
   656→        this.scene.shake(0.055);
   657→      }
   658→      if (this.fallAutoReset > 0) {
   659→        this.fallAutoReset -= dt;
   660→        if (this.fallAutoReset <= 0) {
   661→          this.fallAutoReset = -1;
   662→          this.spawn();
   663→        }
   664→      }
   665→    }
   666→  }
   667→
   668→  /**
   669→   * Arranque de la secuencia de caida: cancela interaccion pendiente, abre los
   670→   * agarres con retardos realistas (manos primero) y deja el cuerpo flacido.
   671→   */
   672→  private beginFallSequence(): void {
   673→    this.interaction?.cancelDrag();
   674→    this.interaction?.cancelAssistAll();
   675→    this.fallTimer = 0;
   676→    this.fallReleaseQueue = [];
   677→    this.fallLandedFx = false;
   678→    const hands = ["hand_l", "hand_r"] as const;
   679→    const feet = ["foot_l", "foot_r"] as const;
   680→    let n = 0;
   681→    for (const limb of hands) {
   682→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_HAND_S + n++ * FALL_RELEASE_STAGGER_S });
   683→    }
   684→    for (const limb of feet) {
   685→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_FOOT_S + n++ * FALL_RELEASE_STAGGER_S });
   686→    }
   687→    this.fallReleaseQueue.sort((a, b) => a.at - b.at);
   688→    this.fallAutoReset = FALL_AUTO_RESET_S;
   689→    this.solver.setToneTarget(this.tune.toneFloor);
   690→  }

   692→  // --- Bucle de render ---
   693→
   694→  private readonly frame = (nowMs: number): void => {
   695→    if (this.disposed) return;
   696→    this.raf = requestAnimationFrame(this.frame);
   697→    this.frameCount++;
   698→    const now = nowMs / 1000;
   699→    const dtFrame = Math.min(Math.max(0, now - this.lastFrame), 0.1);
   700→    this.lastFrame = now;
   701→
   702→    this.fpsFrames++;
   703→    this.fpsElapsed += dtFrame;
   704→    if (this.fpsElapsed >= 0.5) {
   705→      this.fps = this.fpsFrames / this.fpsElapsed;
   706→      this.fpsFrames = 0;
   707→      this.fpsElapsed = 0;
   708→    }
   709→
   710→    const state = this.machine.state;
   711→    const stepping = state !== "PAUSED" && state !== "LOADING" && !this.paused;
   712→    if (stepping) {
   713→      // Slow-motion cinematografico: la escala converge suave hacia el objetivo.
   714→      if (this.slowmoLeft > 0) this.slowmoLeft -= dtFrame;
   715→      const goalScale = this.slowmoLeft > 0 ? SLOWMO_SCALE : 1;
   716→      this.timeScale += (goalScale - this.timeScale) * Math.min(1, dtFrame * 9);
   717→      this.accumulator += dtFrame * this.timeScale;
   718→      let steps = 0;
   719→      const fixed = this.profile.sim.fixedDeltaSeconds;
   720→      while (this.accumulator >= fixed && steps < MAX_STEPS_PER_FRAME) {
   721→        const t0 = performance.now();
   722→        this.fixedStep(fixed);
   723→        this.physicsMs = lerp(this.physicsMs, performance.now() - t0, 0.15);
   724→        this.accumulator -= fixed;
   725→        steps++;
   726→      }
   727→      if (steps >= MAX_STEPS_PER_FRAME) this.accumulator = 0;
   728→    }
   729→
   730→    if (state === "RESETTING") {
   731→      this.settleTimer -= dtFrame;
   732→      if (this.settleTimer <= 0) {
   733→        this.machine.transition(this.autoStartPending ? "PLAYING" : "READY");
   734→        if (this.autoStartPending) {
   735→          this.events.emit({ type: "Message", text: "Recupera dos apoyos antes de perder el agarre", tone: "warn" });
   736→        }
   737→      }
   738→    }
   739→
   740→    const alpha = Math.min(1, this.accumulator / this.profile.sim.fixedDeltaSeconds);
   741→    this.scene.syncRagdoll(this.solver, alpha);
   742→    this.scene.syncJointDebug(this.solver);
   743→
   744→    // Anillo de seleccion: extremidad arrastrada (firm) o bajo el cursor (iman sutil).
   745→    if (this.interaction?.dragging) {
   746→      this.scene.setSelectionRing(this.interaction.selected, this.solver, "grab");
   747→    } else {
   748→      const hover = this.pointerInside && (state === "READY" || state === "PLAYING")
   749→        ? this.scene.pickLimb(this.pointerX, this.pointerY, this.solver)
   750→        : null;
   751→      this.scene.setSelectionRing(hover, this.solver, "hover");
   752→    }
   753→
   754→    // Preview de arrastre y alcance.
   755→    const preview = this.interaction?.getPreview() ?? null;
   756→    this.scene.clearHoldHighlights();
   757→    if (preview?.snappedHoldId) {
   758→      if (preview.validity === "valid") this.scene.setHoldCompatible(preview.snappedHoldId, true);
   759→      else this.scene.setHoldInvalid(preview.snappedHoldId, true);
   760→    }
   761→    this.scene.showDragPreview(this.interaction?.dragging ? preview : null, this.solver);
   762→    if (this.interaction?.dragging && preview) {
   763→      this.scene.setReachRadius(
   764→        (preview.limb.startsWith("hand") ? this.rig!.armChain : this.rig!.legChain) * this.character.physical.reachScale
   765→      );
   766→      this.scene.showReachPreview(preview.limb, this.solver, preview.validity !== "invalid");
   767→    } else {
   768→      this.scene.showReachPreview(null, this.solver, true);
   769→    }
   770→
   771→    // Tutorial: resaltados y avance de pasos.
   772→    this.tickTutorial();
   773→
   774→    // Retroalimentacion de carga por extremidad (misma paleta que el panel
   775→    // lateral): tinte verde/ambar/rojo de la extremidad y anillo sobre su presa.
   776→    // La sobrecarga ademas lleva CUENTA ATRAS de fallo: urgencia 0..1 (danno
   777→    // acumulado/tolerancia) acelera parpadeo y latidos, y alimenta el chip central.
   778→    const cfgBio = this.profile.biomech;
   779→    this.limbOverload.clear();
   780→    this.overloadPressure = 0;
   781→    for (const limb of LIMB_IDS) {
   782→      const anchored = this.solver.hasAnchor(limb);
   783→      const info = this.bio.infoFor(limb);
   784→      const zone = info?.zone ?? "NORMAL";
   785→      let urgency = 0;
   786→      if (anchored && info && (zone === "OVERLOADED" || zone === "ERROR_GUARD")) {
   787→        const secondsLeft = Math.max(0, (cfgBio.overloadGraceSeconds - info.overloadTimer) / cfgBio.overloadAccumRate);
   788→        const frac = clamp01(1 - (secondsLeft * cfgBio.overloadAccumRate) / Math.max(1e-6, cfgBio.overloadGraceSeconds));
   789→        this.limbOverload.set(limb, { seconds: secondsLeft, frac });
   790→        urgency = frac;
   791→        this.overloadPressure = Math.max(this.overloadPressure, frac);
   792→      }
   793→      this.scene.updateLimbTint(limb, anchored, zone, urgency);
   794→      this.scene.setAnchorRing(limb, this.solver.getAnchor(limb)?.holdId ?? null, zone, urgency);
   795→    }
   796→
   797→    // Voz del escalador: respiracion y quejidos de fondo dirigidos por la fatiga
   798→    // y la carga reales (solo suenan con PLAYING activo; el director se resetea fuera).
   799→    const maxFatigue = Math.max(...LIMB_IDS.map((l) => this.bio.fatigue[l] ?? 0));
   800→    const maxRatio = Math.max(...LIMB_IDS.map((l) => this.bio.infoFor(l)?.ratio ?? 0));
   801→    if (this.machine.state === "PLAYING") this.peakFatigue = Math.max(this.peakFatigue, maxFatigue);
   802→    this.audio.update(dtFrame, {
   803→      playing: this.machine.state === "PLAYING",
   804→      maxFatigue,
   805→      maxRatio,
   806→      overloadRatio: this.profile.biomech.overloadRatio,
   807→      overload: LIMB_IDS.some((l) => this.bio.infoFor(l)?.zone === "OVERLOADED"),
   808→      overloadPressure: this.overloadPressure
   809→    });
   810→
   811→    // Combo de flujo: se rompe si pasa la ventana sin fijar otra presa.
   812→    if (this.flowCount > 0 && performance.now() / 1000 - this.lastGrabAt > FLOW_WINDOW_S) {
   813→      this.flowCount = 0;
   814→      this.hud.setFlow(0);
   815→    }
   816→
   817→    const pelvis = this.solver.getParticle(P.pelvisC);
   818→    this.scene.updateCamera(dtFrame, pelvis.pos, this.recenterOnce);
   819→    this.recenterOnce = false;
   820→
   821→    // HUD y debug a 10 Hz.
   822→    this.hudAccum += dtFrame;
   823→    if (this.hudAccum >= 0.1) {
   824→      this.hudAccum = 0;
   825→      this.updateHud(state);
   826→      this.debug.update(this.buildDebugData(state));
   827→    }
   828→
   829→    this.scene.render();
   830→  };
   831→
   832→  private updateHud(state: GameState): void {
   833→    const running = state === "PLAYING";
   834→    this.hud.setTimer(
   835→      this.evaluator ? Math.max(0, this.challenge.timeLimitSeconds - (running ? this.gameTime : 0)) : null,
   836→      running
   837→    );
   838→    let fatigueMax = 0;
   839→    for (const limb of LIMB_IDS) fatigueMax = Math.max(fatigueMax, this.bio.fatigue[limb]);
   840→    this.hud.setVitals(this.lastSnapshot.stability, fatigueMax);
   841→
   842→    const limbStates: Record<LimbId, { zone: LoadZone; anchored: boolean; label: string; overloadS?: number | null; overloadFrac?: number | null }> = {} as never;
   843→    for (const limb of LIMB_IDS) {
   844→      const anchored = this.solver.hasAnchor(limb);
   845→      const zone = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   846→      const over = this.limbOverload.get(limb);
   847→      limbStates[limb] = {
   848→        zone,
   849→        anchored,
   850→        label: this.interaction?.isAssisting(limb) ? "MOVIENDO" : this.interaction?.dragging && this.interaction.selected === limb ? "OBJETIVO" : "LIBRE",
   851→        overloadS: over ? over.seconds : null,
   852→        overloadFrac: over ? over.frac : null
   853→      };
   854→    }
   855→    this.hud.setLimbStates(limbStates);
   856→    // Chip central: la extremidad mas comprometida (solo jugando y fuera del
   857→    // tutorial: ahi la sobrecarga nunca suelta, el aviso despistaria). Memoria
   858→    // de 0.6 s: los valles del pendulum no hacen parpadear el aviso.
   859→    let worst: { limb: LimbId; seconds: number; frac: number } | null = null;
   860→    for (const [limb, over] of this.limbOverload) {
   861→      if (!worst || over.seconds < worst.seconds) worst = { limb, seconds: over.seconds, frac: over.frac };
   862→    }
   863→    const nowS = performance.now() / 1000;
   864→    if (worst) {
   865→      this.lastWorst = { label: LIMB_LABEL_ES[worst.limb], seconds: worst.seconds, frac: worst.frac };
   866→      this.lastWorstAt = nowS;
   867→    }
   868→    const showWarning = state === "PLAYING" && !this.tutorial && this.lastWorst !== null && (worst !== null || nowS - this.lastWorstAt < 0.6);
   869→    this.hud.setOverloadWarning(showWarning && this.lastWorst ? this.lastWorst : null);
   870→    this.hud.setProgress(running && this.evaluator && this.evaluator.progress > 0 ? this.evaluator.progress : null);
   871→  }

   873→  private limbDisplayState(limb: LimbId): string {
   874→    if (this.machine.state === "FALLEN") return "FALLING";
   875→    if (this.interaction?.isAssisting(limb)) return "TARGETING";
   876→    if (this.solver.hasAnchor(limb)) {
   877→      const timer = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   878→      return timer === "OVERLOADED" ? "OVERLOADED" : "ANCHORED";
   879→    }
   880→    if (this.interaction?.dragging && this.interaction.selected === limb) return "TARGETING";
   881→    return "FREE";
   882→  }

   884→  private buildDebugData(state: GameState): DebugData {
   885→    const counts = this.solver.counts();
   886→    const limbs = {} as DebugData["limbs"];
   887→    for (const limb of LIMB_IDS) {
   888→      const info = this.bio.infoFor(limb);
   889→      const anchor = this.solver.getAnchor(limb);
   890→      limbs[limb] = {
   891→        state: this.limbDisplayState(limb),
   892→        holdId: anchor?.holdId ?? this.interaction?.assistHoldOf(limb) ?? null,
   893→        loadN: anchor?.loadN ?? 0,

   920→  private onGameEvent(e: GameEvent): void {
   921→    this.audio.play(e);
   922→    switch (e.type) {
   923→      case "StateChanged":
   924→        this.hud.setState(e.to);
   925→        break;
   926→      case "ChallengeSuccess": {
   927→        // Tanda de confeti sobre la presa cima del desafio.
   928→        const topHold = this.challenge.completion.holdIds?.[0] ?? "";
   929→        const topPos = topHold ? this.holdPosById.get(topHold) : undefined;
   930→        if (topPos) this.scene.burstConfetti(new Vector3(topPos[0], topPos[1] + 0.1, topPos[2] - 0.05));
   931→        // Puntuacion, estrellas y record persistente.
   932→        const result = computeScore({
   933→          timeS: this.gameTime,
   934→          timeLimitS: this.challenge.timeLimitSeconds,
   935→          falls: this.fallCount,
   936→          flowBonus: this.flowBonus
   937→        });
   938→        const rec: BestRecord = { score: result.total, timeS: this.gameTime, stars: result.stars };
   939→        const isRecord = this.saveBest(this.challenge.id, rec);
   940→        this.hud.showSuccess({
   941→          stars: result.stars,
   942→          total: result.total,
   943→          timeS: this.gameTime,
   944→          breakdown: result.breakdown,
   945→          isRecord,
   946→          best: this.loadBest(this.challenge.id)
   947→        });
   948→        if (isRecord) this.hud.setBest(this.challenge.id, rec);
   949→        break;
   950→      }
   951→      case "ChallengeFailed":
   952→        this.hud.showBanner(e.reason === "TIMEOUT" ? "TIEMPO AGOTADO" : "CAIDA", "bad");
   953→        break;
   954→      case "HoldAcquired": {
   955→        // Combo de flujo: encadenar fijaciones en menos de FLOW_WINDOW_S suma bonus.
   956→        const now = performance.now() / 1000;
   957→        this.flowCount = now - this.lastGrabAt <= FLOW_WINDOW_S ? Math.min(FLOW_MAX_CHAIN, this.flowCount + 1) : 1;
   958→        this.lastGrabAt = now;
   959→        if (this.flowCount >= 2) this.flowBonus += FLOW_BONUS_PER_STEP * this.flowCount;
   960→        this.hud.setFlow(this.flowCount);
   961→        // Bocada de tiza al fijar una mano.
   962→        if (e.limb.startsWith("hand")) {
   963→          const pos = this.holdPosById.get(e.holdId);
   964→          if (pos) this.scene.burstChalk(new Vector3(pos[0], pos[1], pos[2] - 0.03));
   965→        }
   966→        break;
   967→      }
   968→      case "HoldRejected": {
   969→        const label = REJECT_LABELS[e.reason] ?? e.reason;
   970→        this.hud.toast(`Movimiento rechazado: ${label}`, "warn");
   971→        break;
   972→      }
   973→      case "GripReleased":
   974→        if (e.reason === "OVERLOAD") {
   975→          this.hud.toast(`Sobrecarga: el agarre cedió (${e.limb})`, "error");
   976→          // Sacudida corta + tiza al resbalar el agarre.
   977→          this.scene.shake(0.02);
   978→          const pos = this.holdPosById.get(e.holdId);
   979→          if (pos) this.scene.burstChalk(new Vector3(pos[0], pos[1], pos[2] - 0.03), 14);
   980→        }
   981→        break;
   982→      case "StrainStarted":
   983→        this.hintOnce("strain", "Agarre en tension: la fatiga de esa extremidad sube mas rapido", "warn");
   984→        break;
   985→      case "OverloadStarted":
   986→        this.hintOnce("overload", "SOBRECARGA: el chip rojo cuenta los segundos que aguanta el agarre (~4,5 s en sobrecarga continua). Alivia esa extremidad ya", "error");
   987→        break;
   988→      case "FallStarted":
   989→        this.hintOnce("fall", "Caída: los agarres se abren y el cuerpo cae. Se reintenta solo (o pulsa R)", "info");
   990→        // Drama: slow-motion breve + rotura del combo + conteo de caidas.
   991→        this.slowmoLeft = SLOWMO_DURATION_S;
   992→        this.fallCount += 1;
   993→        this.flowCount = 0;
   994→        this.hud.setFlow(0);
   995→        break;
   996→      default:
   997→        break;
   998→    }
   999→  }
