     1→/**
     2→ * Solver de fisica PBD (Position Based Dynamics) para el ragdoll.
     3→ *
     4→ * Decision documentada (ADR-001 del prototipo): adaptador fisico interno basado en
     5→ * particulas + restricciones de distancia y de angulo. La interfaz PhysicsAdapter se
     6→ * mantiene desacoplada para poder sustituirlo por Havok/Rapier sin tocar gameplay.
     7→ *
     8→ * Modelo de anclaje (estimador documentado, seccion 9.2 del documento): cada anclaje
     9→ * es un resorte amortiguado con fuerza maxima. La carga se mide como la fuerza que el
    10→ * resorte debe aplicar (rigidez x estiramiento, antes de saturar), lo que permite
    11→ * detectar sobrecarga y provoca estiramiento visible cuando el agarre no basta.
    12→ * Nunca se teletransportan cuerpos durante el juego.
    13→ */

    48→/**
    49→ * Restriccion de SENTIDO DE FLEXION (ADR-008): el constraint angular es 2D y no
    50→ * distingue doblar de doblar al reves (una rodilla invertida tiene el MISMO angulo).
    51→ * Este constraint mantiene el vertice articular (codo/rodilla) en el lado correcto
    52→ * del plano definido por la cuerda padre-hijo y una direccion de referencia del
    53→ * cuerpo (normal del muro): prohibe la hiperextension/luxacion visual.
    54→ */
    55→export interface FoldConstraint {
    56→  id: string;
    57→  /** Particula del vertice articular (codo/rodilla). */
    58→  j: number;
    59→  /** Extremo padre (hombro/cadera). */
    60→  a: number;
    61→  /** Extremo hijo (mano/pie). */
    62→  b: number;
    63→  /** Direccion unitaria del lado PERMITIDO (mundo). */
    64→  dir: V3;
    65→  /** Separacion minima exigida en esa direccion (m): tolera recto, prohibe el giro. */
    66→  minSide: number;
    67→  strength: number;
    68→  /** Lectura de diagnostico: lado actual (m). */
    69→  side: number;
    70→}
    71→
    72→/**
    73→ * Restriccion de GIRO DE PELVIS (anti-"cadera girada", peticion del jugador):
    74→ * en PBD clasico la pelvis puede rotar libremente alrededor del eje de la columna
    75→ * (todas las riostras hombro-cadera cambian su longitud en SEGUNDO orden al girar),
    76→ * con lo que el mas minimo sesgo del settle la deja cruzada 90 grados y ahi se queda.
    77→ * Este constraint limita el angulo de guinada de la barra de caderas respecto a la
    78→ * barra de hombros, medido alrededor del eje columna (axisFrom->pivot). La correccion
    79→ * rota SOLO las caderas alrededor del eje que pasa por el pivote (pelvisC): conserva
    80→ * exactamente la longitud cadera-cadera y las riostras pelvisC-cadera (correccion
    81→ * minimamente invasiva, anatomia pasiva como los limites articulares).
    82→ */
    83→export interface TwistConstraint {
    84→  id: string;
    85→  /** Barra de referencia (hombros: shL, shR). */
    86→  a1: number;
    87→  a2: number;
    88→  /** Barra corregida (caderas: hipL, hipR). */
    89→  b1: number;
    90→  b2: number;
    91→  /** Pivote de la correccion (pelvisC). */
    92→  pivot: number;
    93→  /** Origen del eje de guinada (pecho: eje columna pecho->pelvis). */
    94→  axisFrom: number;
    95→  /** Tope de guinada (rad); el solver lo sustituye por balance.twistLimitRad en vivo. */
    96→  maxAngle: number;
    97→  stiffness: number;
    98→  /** Lectura de diagnostico: guinada actual (rad). */
    99→  currentAngle: number;
   100→}

   134→/**
   135→ * Constantes musculares (ADR-007). Las fuerzas se aplican sobre particulas AXIALES

   137→ * por paso <= ~2 cm). Captan la carga el estimador de pin existente (9.2), de modo que

   153→  armPullMax: number;

   155→  coreMax: number;

   163→  vMaxSpeed: number;

   165→  muscleForceCapPelvis: number;

   171→/**
   172→ * Rangos del balance realismo/ragdoll (ADR-008), calibrables en vivo (F3).
   173→ * - jointLimitStiffness: dureza de los limites angulares (1 = duro, sin luxaciones).
   174→ * - foldStrength: dureza del sentido de flexion (anti-rodilla/codo invertido).
   175→ * - postureTone: tono postural en juego (1 = erguido y vivo, 0 = muñeco de trapo).
   176→ * - toneFloor: tono residual tras declararse la caida (passivo pero sin pliegues).
   177→ * - stretchTolerance: cuanto puede ALARGARSE una cadena anatomica sobre su longitud
   178→ *   natural antes de que el tope duro la corte (1.05 = realista, 1.2 = muneco de goma).
   179→ */
   180→export interface BalanceTuning {
   181→  jointLimitStiffness: number;
   182→  foldStrength: number;
   183→  postureTone: number;
   184→  toneFloor: number;
   185→  stretchTolerance: number;
   186→  /** Tope de guinada de la pelvis (rad): 0.30 = realista (cadera cuadrada),
   187→   *  0.85 = ragdoll (giro libre casi total). 0 desactiva el tope. */
   188→  twistLimitRad: number;
   189→}
   190→
   191→/**
   192→ * Tope de LONGITUD DE CADENA (anatomia pasiva, como los limites articulares):
   193→ * prohibe que la distancia raiz->extremo (hombro->mano, cadera->pie, pelvis->pecho)
   194→ * supere la longitud natural x tolerancia. Es el anti-"muneco de goma": sin el, las
   195→ * barras elasticas se estiran bajo carga sostenida y aparecen posturas imposibles
   196→ * (brazos de 1.5 m). Desigualdad dura: solo actua al violar el maximo.
   197→ */
   198→interface ChainCap {

   330→  setRagdoll(
   331→    particles: Particle[],
   332→    sticks: Stick[],
   333→    angles: AngleConstraint[],
   334→    limbParticle: Map<LimbId, number>,
   335→    topology?: BodyTopology,
   336→    folds: FoldConstraint[] = [],
   337→    twists: TwistConstraint[] = []
   338→  ): void {
   339→    this.particles = particles;
   340→    this.sticks = sticks;
   341→    this.angles = angles;
   342→    this.folds = folds;
   343→    this.twists = twists;
   344→    this.limbParticle = limbParticle;
   345→    this.topology = topology ?? null;
   346→    this.anchors.clear();
   347→    this.targets.clear();
   348→    this.climbDrive = 0;
   349→    this.rebuildChainCaps();
   350→  }
   351→
   352→  /** Deriva los topes de cadena de la topologia (hombro->mano, cadera->pie, columna). */
   353→  private rebuildChainCaps(): void {
   354→    this.chainCaps = [];
   355→    const t = this.topology;
   356→    if (!t) return;
   357→    for (const limb of LIMB_IDS) {
   358→      const root = t.rootOf.get(limb);
   359→      const end = this.limbParticle.get(limb);
   360→      if (root === undefined || end === undefined) continue;
   361→      const base = limb.startsWith("hand") ? t.armChain : t.legChain;
   362→      this.chainCaps.push({ a: root, b: end, base, current: base, limb });
   363→    }
   364→    // Columna: el torso tampoco se estira (menos tolerancia: es un segmento rigido).
   365→    this.chainCaps.push({ a: t.pelvis, b: t.chest, base: t.spineLen, current: t.spineLen, limb: null });
   366→  }
   367→
   368→  /** Lecturas de diagnostico del estiramiento de cadena (ratio 1 = en el tope). */
   369→  debugChains(): { limb: string; len: number; max: number; ratio: number }[] {
   370→    const tol = this.env.balance.stretchTolerance;
   371→    return this.chainCaps.map((c) => {
   372→      const max = c.limb === null ? c.base * SPINE_STRETCH_TOLERANCE : c.base * tol;
   373→      return { limb: c.limb ?? "spine", len: c.current, max, ratio: max > 1e-9 ? c.current / max : 0 };
   374→    });
   375→  }

   434→  positionOfLimb(limb: LimbId): V3 {
   435→    const idx = this.limbParticle.get(limb);
   436→    return idx === undefined ? v3() : vClone(this.particles[idx].pos);

   439→  /** Velocidad estimada de la extremidad (m/s): para el iman predictivo del picking
   440→   * (coger una extremidad en movimento: apunta a donde VA a estar). */
   441→  velocityOfLimb(limb: LimbId): V3 {
   442→    const idx = this.limbParticle.get(limb);
   443→    if (idx === undefined || this.lastStepDt <= 1e-6) return v3();
   444→    const p = this.particles[idx];
   445→    return vScale(vSub(p.pos, p.prev), 1 / this.lastStepDt);
   446→  }

   465→  createAnchor(limb: LimbId, holdId: string, point: V3, maxForce: number): AnchorState {
   466→    this.releaseAnchor(limb);
   467→    const particle = this.limbParticle.get(limb);
   468→    if (particle === undefined) {
   469→      throw new Error(`No existe particula para la extremidad ${limb}`);
   470→    }
   471→    const anchor: AnchorState = {
   472→      limb,
   473→      particle,
   474→      holdId,
   475→      point: vClone(point),
   476→      maxForce,
   477→      loadN: 0,
   478→      stretch: 0,
   479→      activation: 0
   480→    };
   481→    this.anchors.set(limb, anchor);
   482→    this.targets.delete(limb);
   483→    return anchor;
   484→  }
   485→
   486→  releaseAnchor(limb: LimbId): AnchorState | undefined {
   487→    const a = this.anchors.get(limb);
   488→    this.anchors.delete(limb);
   489→    return a;
   490→  }

   492→  setTarget(limb: LimbId, target: V3, maxForce: number): void {
   493→    const particle = this.limbParticle.get(limb);
   494→    if (particle === undefined) {
   495→      throw new Error(`No existe particula para la extremidad ${limb}`);
   496→    }
   497→    this.targets.set(limb, { limb, particle, target: vClone(target), maxForce });
   498→  }
   499→
   500→  clearTarget(limb: LimbId): void {
   501→    this.targets.delete(limb);
   502→  }
   503→
   504→  /** Mueve todos los particulas por un delta (solo para la pose inicial y el impulso de arranque). */
   505→  translateRagdoll(delta: V3): void {
   506→    for (const p of this.particles) {
   507→      p.pos = vAdd(p.pos, delta);
   508→      p.prev = vAdd(p.prev, delta);
   509→    }
   510→  }
   511→
   512→  /**
   513→   * Relajacion estatica: proyecta restricciones SIN integrar (sin gravedad ni energia
   514→   * cinetica). Convierte una semilla aproximada en una pose consistente antes del primer
   515→   * paso, eliminando las patadas del settle inicial.
   516→   */
   517→  relax(sweeps = 80): void {
   518→    for (let i = 0; i < sweeps; i++) {
   519→      this.solveSticks();
   520→      this.solveTwists();
   521→      this.solveAngles();
   522→      this.solveFolds();
   523→      this.solveChainCaps();
   524→      for (const anchor of this.anchors.values()) {
   525→        const p = this.particles[anchor.particle];
   526→        p.pos = vClone(anchor.point);
   527→      }
   528→      // Segunda pasada de topes: los pines acaban de recolocar las manos en sus
   529→      // presas; si esa recolocacion deja una cadena sobre su maximo, se corta YA
   530→      // (si no, la relajacion convergeria con los brazos estirados).
   531→      this.solveChainCaps();
   532→      // Cierre de la iteracion con los limites articulares: los otros constraints
   533→      // (pliegues, topes, pines) pueden re-violentar un angulo ya corregido; asi el
   534→      // rango anatomico queda garantizado EN EL ESTADO FINAL (el cuello, el mas
   535→      // flexible, era el que se colaba en caidas violentas).
   536→      this.solveAngles();
   537→      this.solveCollisions();
   538→    }
   539→    for (const p of this.particles) {
   540→      p.prev = vClone(p.pos);
   541→    }
   542→  }
   543→
   544→  /** Aplica una velocidad inicial (m/s) al cuerpo completo; usado por start.impulse. */
   545→  applyImpulseVelocity(vel: V3, dt: number): void {
   546→    for (const p of this.particles) {
   547→      if (p.invMass > 0) {
   548→        p.prev = vSub(p.pos, vScale(vel, dt));
   549→      }
   550→    }
   551→  }
   552→
   553→  totalMass(): number {
   554→    return this.particles.reduce((acc, p) => acc + p.mass, 0);
   555→  }
   556→
   557→  centerOfMass(): V3 {
   558→    let m = 0;
   559→    const c = v3();
   560→    for (const p of this.particles) {
   561→      c.x += p.pos.x * p.mass;
   562→      c.y += p.pos.y * p.mass;
   563→      c.z += p.pos.z * p.mass;
   564→      m += p.mass;
   565→    }
   566→    return vScale(c, 1 / (m || 1));
   567→  }
   568→
   569→  step(dt: number): StepResult {
   570→    const K = this.env.iterations;
   571→    if (dt > 1e-6) this.lastStepDt = dt;
   572→    const forces = new Map<number, V3>();
   573→    const anchorLoads: Partial<Record<LimbId, number>> = {};
   574→    const anchorStretch: Partial<Record<LimbId, number>> = {};
   575→    let nanError = false;
   576→    let outOfBounds = false;
   577→    let maxSpeed = 0;
   578→
   579→    // 0) Tono postural (ADR-008): persigue el objetivo con rates asimetricos
   580→    // (la caida relaja rapido; recuperarlo de pie cuesta mas).
   581→    const toneRate = this.toneTarget < this.tone ? 4.5 : 2.0;
   582→    const toneGap = this.toneTarget - this.tone;
   583→    this.tone = Math.abs(toneGap) < 0.01 ? this.toneTarget : this.tone + toneGap * Math.min(1, toneRate * dt);
   584→
   585→    // 1) Fuerza externa de objetivos (resorte amortiguado con tope). Los anclajes NO son
   586→    // resortes: se fijan como pin rigido en la fase de restricciones (ver 4b).
   587→    for (const target of this.targets.values()) {
   588→      const p = this.particles[target.particle];
   589→      const err = vSub(target.target, p.pos);
   590→      const vel = vScale(vSub(p.pos, p.prev), 1 / dt);
   591→      const spring = vAdd(vScale(err, this.env.followStiffness), vScale(vel, -this.env.anchorDamping * 0.5));
   592→      const applied = clampVecMag(spring, target.maxForce);
   593→      forces.set(target.particle, vAdd(forces.get(target.particle) ?? v3(), applied));
   594→    }
   595→
   596→    // 1b) Musculos activos (ADR-007): cuadriceps, step-up, traccion de brazos y tono
   597→    // postural. Sin anclajes el cuerpo es ragdoll puro (caida limpia).
   598→    if (this.env.muscles.musclesOn && this.topology !== null && this.anchors.size > 0) {
   599→      this.solveMuscles(dt, forces);
   600→    }
   601→    this.climbDrive = Math.max(0, this.climbDrive - dt / this.env.muscles.driveSeconds);
   602→
   603→    // 2) Integracion Verlet con paso fijo (con techo de velocidad: los musculos no
   604→    // pueden lanzar el ragdoll a velocidades no humanas; el ERROR GUARD queda para
   605→    // anomalias reales del solver).
   606→    const g = this.env.gravity;
   607→    const vCap = this.env.muscles.vMaxSpeed * dt;
   608→    for (let i = 0; i < this.particles.length; i++) {
   609→      const p = this.particles[i];
   610→      if (p.invMass <= 0) continue;
   611→      const f = forces.get(i);
   612→      const a = f ? vAdd(g, vScale(f, p.invMass)) : g;
   613→      const vel = vScale(vSub(p.pos, p.prev), this.env.damping);
   614→      p.prev = vClone(p.pos);
   615→      let next = vAdd(vAdd(p.pos, vel), vScale(a, dt * dt));
   616→      const newVel = vSub(next, p.prev);
   617→      const newSpeed = vLen(newVel);
   618→      if (newSpeed > vCap) {
   619→        next = vAdd(p.prev, vScale(newVel, vCap / newSpeed));
   620→      }
   621→      p.pos = next;
   622→      // Metrica sobre la velocidad YA acotada (la previa incluye correcciones de pin).
   623→      const measured = Math.min(newSpeed, vCap) / dt;
   624→      if (measured > maxSpeed) maxSpeed = measured;
   625→    }
   626→
   627→    // 3) Tremor de esfuerzo (seccion 9.5): ruido proporcional al estado de tension.
   628→    for (const limb of LIMB_IDS) {
   629→      const t = this.tremor[limb];
   630→      if (t <= 0) continue;
   631→      const idx = this.limbParticle.get(limb);
   632→      if (idx === undefined) continue;
   633→      const p = this.particles[idx];
   634→      const amp = this.env.tremorAmplitude * t;
   635→      p.pos = vAdd(p.pos, v3((Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp));
   636→    }
   637→
   638→    // 4) Proyeccion de restricciones.
   639→    // 4a) Acumulador de reaccion por anclaje: suma de correcciones del pin este paso.
   640→    const pinCorrection = new Map<LimbId, V3>();
   641→    for (let iter = 0; iter < K; iter++) {
   642→      this.solveSticks();
   643→      // El giro ANTES de los limites articulares: los angulos se miden/corrigen sobre
   644→      // la geometria ya cuadrada (lecturas coherentes y sin latigazos al cuello).
   645→      this.solveTwists();
   646→      this.solveAngles();
   647→      this.solveFolds();
   648→      this.solveChainCaps();
   649→      // 4b) Anclajes: pin rigido con tope de fuerza. La carga requerida se mide como la
   650→      // fuerza necesaria para mantener el pin (F = m x correccion / dt^2), un estimador
   651→      // consistente con el impulso aplicado (9.2). En sobrecarga el pin se satura y la
   652→      // extremidad resbala visiblemente de la presa.
   653→      for (const anchor of this.anchors.values()) {
   654→        const p = this.particles[anchor.particle];
   655→        const err = vSub(anchor.point, p.pos);
   656→        const errLen = vLen(err);
   657→        if (errLen < 1e-9) continue;
   658→        const maxCorr = (anchor.maxForce * dt * dt) / p.mass;
   659→        const scaled = errLen > maxCorr ? vScale(err, maxCorr / errLen) : err;
   660→        p.pos = vAdd(p.pos, scaled);
   661→        pinCorrection.set(anchor.limb, vAdd(pinCorrection.get(anchor.limb) ?? v3(), scaled));
   662→      }
   663→      // 4c) Topes de cadena tras los pines: una presa mas alla del alcance anatomico
   664→      // NO estira el brazo (sin "muneco de goma"); el agarre resbala hacia el cuerpo,
   665→      // el cuerpo es arrastrado hacia la presa y la carga del pin se dispara (9.2):
   666→      // la biomecanica lee esa lucha como SOBRECARGA y abre el agarre con cuenta atras.
   667→      this.solveChainCaps();
   668→      // Cierre de iteracion: limites articulares sobre el estado final (ver relax).
   669→      this.solveAngles();
   670→      this.solveCollisions();
   671→    }
   672→    // NOTA: la correccion del pin genera momento "fantasma" en Verlet; ya no se
   673→    // neutraliza en prev (cambiaria la calibracion del estimador de carga). El techo
   674→    // de velocidad de la integracion (vMaxSpeed) contiene ese momento residual.
   675→    // Carga suavizada: impulso total demandado por el pin, en newtons, con ganancia
   676→    // calibrada del estimador documentado (las re-proyecciones por iteracion sobre-miden).
   677→    for (const anchor of this.anchors.values()) {
   678→      const net = pinCorrection.get(anchor.limb) ?? v3();
   679→      const measured = this.particles[anchor.particle].mass * (vLen(net) / (dt * dt)) * this.env.loadEstimatorGain;
   680→      anchor.loadN = anchor.loadN === 0 ? measured : anchor.loadN + (measured - anchor.loadN) * 0.3;
   681→      anchor.stretch = vLen(vSub(anchor.point, this.particles[anchor.particle].pos));
   682→      anchorLoads[anchor.limb] = anchor.loadN;
   683→      anchorStretch[anchor.limb] = anchor.stretch;
   684→    }
   685→
   686→    // 5) Guardas: NaN, energia y limites del mundo.
   687→    for (const p of this.particles) {
   688→      if (!isFiniteVec(p.pos) || !isFiniteVec(p.prev)) {
   689→        nanError = true;
   690→        break;
   691→      }
   692→      if (p.pos.x < this.env.bounds.minX || p.pos.x > this.env.bounds.maxX || p.pos.y > this.env.bounds.maxY || p.pos.z < this.env.bounds.minZ || p.pos.z > this.env.bounds.maxZ) {
   693→        outOfBounds = true;
   694→      }
   695→    }
   696→    if (maxSpeed > 18) {
   697→      // Energia anomala: se registra, la capa de juego decide (ERROR GUARD).
   698→      maxSpeed = clamp(maxSpeed, 0, 30);
   699→    }
   700→
   701→    // 6) Lecturas derivadas.
   702→    const floorContacts: { limb: LimbId; point: V3 }[] = [];
   703→    for (const limb of LIMB_IDS) {
   704→      const idx = this.limbParticle.get(limb);
   705→      if (idx === undefined) continue;
   706→      const p = this.particles[idx];
   707→      if (p.pos.y <= this.env.floorY + 0.045) {
   708→        floorContacts.push({ limb, point: v3(p.pos.x, this.env.floorY, p.pos.z) });
   709→      }
   710→    }
   711→
   712→    return {
   713→      nanError,
   714→      energySpike: maxSpeed > 18,
   715→      outOfBounds,
   716→      com: this.centerOfMass(),
   717→      anchorLoads,
   718→      anchorStretch,
   719→      floorContacts,
   720→      jointAngles: this.angles.map((a) => ({
   721→        id: a.id,
   722→        angle: a.currentAngle,
   723→        utilization: jointUtilization(a)
   724→      })),
   725→      maxSpeed
   726→    };
   727→  }
   728→
   729→  /**
   730→   * Fase muscular (ADR-007). Todas las fuerzas se aplican sobre particulas de masa alta
   731→   * (pelvis ~11 kg, pecho ~14 kg) con topes calibrados para un desplazamiento por paso
   732→   * estable; los pines miden la carga resultante y la biomecanica la traduce en fatiga.
   733→   */
   734→  private solveMuscles(dt: number, forces: Map<number, V3>): void {
   735→    const M = this.env.muscles;
   736→    const T = this.topology!;
   737→    const pelvis = this.particles[T.pelvis];
   738→    const chest = this.particles[T.chest];
   739→    const head = this.particles[T.head];
   740→    const addForce = (idx: number, fx: number, fy: number, fz: number): void => {
   741→      const cur = forces.get(idx);
   742→      forces.set(idx, cur ? v3(cur.x + fx, cur.y + fy, cur.z + fz) : v3(fx, fy, fz));
   743→    };
   744→
   745→    // Rampa de activacion por anclaje (evita el tiron al anclar).
   746→    for (const anchor of this.anchors.values()) {
   747→      const rate = anchor.limb.startsWith("foot") ? 1 / 0.35 : 1 / 0.2;
   748→      anchor.activation = Math.min(1, anchor.activation + dt * rate);
   749→    }
   750→
   751→    // Pre-pasada: hay pierna sosteniendo (cuadriceps activo)? Con apoyo de piernas
   752→    // los brazos NO tiran (el pull-up esta para colgar/colocarse, no para despegar
   753→    // al cuerpo que ya esta de pie: evita la suma explosiva de fuerzas).
   754→    let feetSupporting = false;
   755→    // Peso que cargan los brazos ahora mismo (N): modula el balance (un pie que
   756→    // solo descansa no debe volcar el COM hacia si; uno que sostiene, si).
   757→    let handLoadN = 0;
   758→    for (const anchor of this.anchors.values()) {
   759→      if (anchor.limb.startsWith("hand")) handLoadN += Math.max(0, anchor.loadN);
   760→    }
   761→    // Arbitracion de pies: el STEP-UP lo hace solo el pie MAS ALTO (transferencia de
   762→    // peso al escalon, como en escalada real); el balance solo desde el pie activo,
   763→    // para que un apoyo bajo no pelee contra el pie que esta montando.
   764→    let stepUpFoot: LimbId | null = null;
   765→    let stepUpHoldY = -Infinity;
   766→    for (const anchor of this.anchors.values()) {
   767→      if (!anchor.limb.startsWith("foot")) continue;
   768→      const root = this.particles[T.rootOf.get(anchor.limb)!];
   769→      const dyr = root.pos.y - anchor.point.y;
   770→      const dr = Math.hypot(root.pos.x - anchor.point.x, dyr, root.pos.z - anchor.point.z) || 1e-6;
   771→      if (dyr / dr > 0.05) {
   772→        feetSupporting = true;
   773→      }
   774→      if (dyr / dr < 0.02 && anchor.point.y > stepUpHoldY) {
   775→        stepUpHoldY = anchor.point.y;
   776→        stepUpFoot = anchor.limb;
   777→      }
   778→    }
   779→
   780→    let feetAnchored = 0;
   781→    for (const anchor of this.anchors.values()) {
   782→      const isFoot = anchor.limb.startsWith("foot");
   783→      if (isFoot) feetAnchored++;
   784→      const rootIdx = T.rootOf.get(anchor.limb);
   785→      if (rootIdx === undefined) continue;
   786→      const root = this.particles[rootIdx];
   787→      const dx = root.pos.x - anchor.point.x;
   788→      const dy = root.pos.y - anchor.point.y;
   789→      const dz = root.pos.z - anchor.point.z;
   790→      const dist = Math.hypot(dx, dy, dz) || 1e-6;
   791→
   792→      if (isFoot) {
   793→        // CUADRICEPS (apoyo): el pie por DEBAJO de la cadera empuja la pelvis lejos del
   794→        // pie (extension de pierna = ponerse de pie). Proyectado al plano de la pared:
   795→        // la pierna empuja "hacia arriba", no separa del muro.
   796→        if (dy / dist > 0.05) {
   797→          // Servo de pierna: empuja mientras la pierna este flexionada y ceda al
   798→          // acercarse a la extension natural (0.9·cadena = rodilla casi recta).
   799→          // Asi la postura de pie es un equilibrio estable, no un empuje sin tope.
   800→          const targetDist = T.legChain * 0.9;
   801→          const f = clamp((targetDist - dist) * M.quadServoK, 0, M.quadPushMax) * anchor.activation;
   802→          const ul = Math.hypot(dx, dy) || 1e-6;
   803→          // Direccion axial con sesgo vertical (reaccion del suelo ~vertical al de pie).
   804→          const bx = (dx / ul) * (1 - M.quadUpBias);
   805→          const by = (dy / ul) * (1 - M.quadUpBias) + M.quadUpBias;
   806→          const bl = Math.hypot(bx, by) || 1e-6;
   807→          addForce(T.pelvis, (bx / bl) * f, (by / bl) * f, 0);
   808→          // BALANCE: llevar el COM sobre el apoyo (feedback tipo SIMBICON); sin el,
   809→          // un apoyo lateral nunca sostiene de verdad. Solo desde el pie que esta
   810→          // montando (si hay step-up) o desde los apoyos si nadie monta. Escalado por
   811→          // la carga REAL de la pierna frente a los brazos: un pie que descansa no
   812→          // vuelca el cuerpo (evita sobrecargar el brazo contrario en cascada).
   813→          if (stepUpFoot === null || stepUpFoot === anchor.limb) {
   814→            const weightShare = clamp(f / (f + handLoadN + 60), 0.15, 1);
   815→            const fb = clamp((anchor.point.x - pelvis.pos.x) * 400, -M.balanceMax, M.balanceMax) * anchor.activation * weightShare;
   816→            addForce(T.pelvis, fb, 0, 0);
   817→            // 3a ley de Newton: parte de la reaccion entra por el pie al pin, de modo que
   818→            // el apoyo MEDIDO carga de verdad al estar de pie (fatiga de piernas real).
   819→            addForce(anchor.particle, -(bx / bl) * f * 0.5 - fb * 0.4, -(by / bl) * f * 0.5, 0);
   820→          } else {
   821→            addForce(anchor.particle, -(bx / bl) * f * 0.5, -(by / bl) * f * 0.5, 0);
   822→          }
   823→        }
   824→        // PSOAS/STEP-UP: el pie por ENCIMA de la cadera sube el cuerpo hacia la presa
   825→        // (montar el escalon). Es lo que convierte "apoyar un pie" en ascender. La
   826→        // componente vertical del vector hacia la presa regula el empuje solo: al
   827→        // acercarse la pelvis al nivel del pie la fuerza se hace horizontal (COM sobre
   828→        // el apoyo) y el cuadriceps toma el relevo por encima. Solo el pie MAS ALTO
   829→        // monta (arbitracion: transferencia de peso al escalon).
   830→        if (stepUpFoot === anchor.limb && dy / dist < 0.02) {
   831→          const boost = 0.9 + 0.3 * this.climbDrive;
   832→          // Esvaimiento por proximidad: al acercarse la pelvis al nivel del pie la
   833→          // fuerza cede (sin sobregiro mas alla de la postura natural de pie).
   834→          const taper = clamp(-dy / 0.45, 0.55, 1);
   835→          const f = M.stepUpMax * anchor.activation * boost * taper;
   836→          const ux = -dx / dist;
   837→          const uy = -dy / dist;
   838→          const uz = -dz / dist;
   839→          addForce(T.pelvis, ux * f, uy * f, uz * f);
   840→          addForce(anchor.particle, -ux * f * 0.35, -uy * f * 0.35, -uz * f * 0.35);
   841→        }
   842→      } else {
   843→        // TRACCION DE BRAZOS (impulso de subida del jugador): mano por encima del
   844→        // hombro y brazo ESTIRADO (compuerta de holgura) tira del pecho hacia el
   845→        // anclaje. Con piernas de apoyo es el "subirse" que sube los hombros; el
   846→        // servo de pierna limita la postura y evita el pliegue.
   847→        if (this.climbDrive > 0.02 && dy / dist < -0.1) {
   848→          // Holgura medida desde la RAIZ (hombro): en postura plegada el pecho puede
   849→          // estar alto con el hombro colgando bajo y el brazo estirado; el pull debe
   850→          // evaluar el brazo real, no la posicion del pecho.
   851→          const rdx = anchor.point.x - root.pos.x;
   852→          const rdy = anchor.point.y - root.pos.y;
   853→          const rdz = anchor.point.z - root.pos.z;
   854→          const rd = Math.hypot(rdx, rdy, rdz) || 1e-6;
   855→          const slack = rd - T.armChain * 0.9;
   856→          const cdx = anchor.point.x - chest.pos.x;
   857→          const cdy = anchor.point.y - chest.pos.y;
   858→          const cdz = anchor.point.z - chest.pos.z;
   859→          const cd = Math.hypot(cdx, cdy, cdz) || 1e-6;
   860→          if (slack > 0) {
   861→            const gate = clamp(slack / (T.armChain * 0.2), 0.4, 1);
   862→            // Durante un step-up la pierna manda: los brazos asisten al 60%.
   863→            const assist = stepUpFoot === null ? 1 : 0.6;
   864→            // Margen de agarre: no se puede tirar mas fuerte que el agarre restante
   865→            // (un brazo al limite no tira: evita saturar el pin y la sobrecarga).
   866→            const headroom = clamp(1 - anchor.loadN / anchor.maxForce, 0, 1);
   867→            const f = M.armPullMax * anchor.activation * this.climbDrive * gate * assist * headroom;
   868→            addForce(T.chest, (cdx / cd) * f, (cdy / cd) * f, (cdz / cd) * f);
   869→            addForce(rootIdx, (cdx / cd) * f * 0.35, (cdy / cd) * f * 0.35, (cdz / cd) * f * 0.35);
   870→          }
   871→        }
   872→      }
   873→    }
   874→
   875→    // TONO POSTURAL: pecho sobre pelvis y cabeza sobre pecho (anti-muneco de trapo).
   876→    // Escala con el numero de apoyos y con el tono global (ADR-008): cuerpo libre =
   877→    // ragdoll puro; tono bajo (caida) = cuerpo flacido pero sin pliegues imposibles
   878→    // (los limites y folds siguen activos: son anatomia pasiva, no musculo).
   879→    const toneScale = Math.min(1, this.anchors.size / 2) * (0.15 + 0.85 * this.tone);
   880→    const tex = pelvis.pos.x - chest.pos.x;
   881→    const tey = pelvis.pos.y + T.spineLen * 0.97 - chest.pos.y;
   882→    const tez = pelvis.pos.z - chest.pos.z;
   883→    const teLen = Math.hypot(tex, tey, tez) || 1e-6;
   884→    const coreF = Math.min(M.coreMax * toneScale, teLen * 400);
   885→    addForce(T.chest, (tex / teLen) * coreF, (tey / teLen) * coreF, (tez / teLen) * coreF);
   886→    const nex = chest.pos.x - head.pos.x;
   887→    const ney = chest.pos.y + 0.2 - head.pos.y;
   888→    const nez = chest.pos.z - head.pos.z;
   889→    const neLen = Math.hypot(nex, ney, nez) || 1e-6;
   890→    const neckF = Math.min(M.neckMax * toneScale, neLen * 300);
   891→    addForce(T.head, (nex / neLen) * neckF, (ney / neLen) * neckF, (nez / neLen) * neckF);
   892→
   893→    // PEGADO A LA PARED con cualquier anclaje: la colision del cuerpo con la cara del
   894→    // muro impide colgar en la vertical exacta de los pines (las presas sobresalen), y
   895→    // sin fuerza de centrado en z el cuerpo queda 0.1-0.2 m por detras, offset que
   896→    // devora el alcance util de brazos y piernas en CADA movimiento. El escalador real
   897→    // va pegado: cadera y pecho se presionan hacia el plano de presas.
   898→    if (this.anchors.size > 0) {
   899→      const zTarget = this.env.wallFrontZ - 0.26;
   900→      const fzPelvis = clamp((zTarget - pelvis.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   901→      const fzChest = clamp((zTarget - chest.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   902→      addForce(T.pelvis, 0, 0, fzPelvis);
   903→      addForce(T.chest, 0, 0, fzChest);
   904→    }
   905→
   906→    // Tope de fuerza muscular por particula axial: la suma de musculos nunca puede
   907→    // superar el techo (estabilidad de Verlet con las masas de pelvis/pecho).
   908→    const cap = (idx: number, capN: number): void => {
   909→      const f = forces.get(idx);
   910→      if (!f) return;
   911→      const l = vLen(f);
   912→      if (l > capN) forces.set(idx, vScale(f, capN / l));
   913→    };
   914→    cap(T.pelvis, M.muscleForceCapPelvis);
   915→    cap(T.chest, M.muscleForceCapChest);
   916→
   917→    // Instantanea para diagnostico (solo particulas axiales y raices).
   918→    this.lastMuscleForces.clear();
   919→    for (const idx of [T.pelvis, T.chest, T.head]) {
   920→      const f = forces.get(idx);
   921→      if (f) this.lastMuscleForces.set(idx, f);
   922→    }
   923→  }
   924→
   925→  private solveSticks(): void {
   926→    for (const s of this.sticks) {
   927→      const pa = this.particles[s.a];
   928→      const pb = this.particles[s.b];
   929→      const w = pa.invMass + pb.invMass;
   930→      if (w <= 0) continue;
   931→      const d = vSub(pb.pos, pa.pos);
   932→      const len = vLen(d);
   933→      if (len < 1e-9) continue;
   934→      const diff = (len - s.rest) / len;
   935→      const k = s.stiffness;
   936→      pa.pos = vAdd(pa.pos, vScale(d, (diff * pa.invMass * k) / w));
   937→      pb.pos = vSub(pb.pos, vScale(d, (diff * pb.invMass * k) / w));
   938→    }
   939→  }
   940→
   941→  private solveAngles(): void {
   942→    // Limites articulares DUROS (ADR-008): la rigidez de los datos ya no escala la
   943→    // correccion de la violacion; jointLimitStiffness (1 = duro) garantiza que ni
   944→    // colgando ni en caida el angulo cruce el rango anatomico. La flexion DENTRO del
   945→    // rango sigue libre (el constraint solo actua al violar).
   946→    const k = clamp(this.env.balance.jointLimitStiffness, 0, 1);
   947→    for (const c of this.angles) {
   948→      const pj = this.particles[c.j];
   949→      const pa = this.particles[c.a];
   950→      const pb = this.particles[c.b];
   951→      const va = vSub(pa.pos, pj.pos);
   952→      const vb = vSub(pb.pos, pj.pos);
   953→      const la = vLen(va);
   954→      const lb = vLen(vb);
   955→      if (la < 1e-9 || lb < 1e-9) {
   956→        c.currentAngle = Number.isFinite(la) && Number.isFinite(lb) ? c.currentAngle : c.currentAngle;
   957→        continue;
   958→      }
   959→      const ang = angleBetween(va, vb);
   960→      c.currentAngle = ang;
   961→      const target = clamp(ang, c.min, c.max);
   962→      if (Math.abs(target - ang) < 1e-5) continue;
   963→      const axis = vCross(va, vb);
   964→      const axisLen = vLen(axis);
   965→      if (axisLen < 1e-9) continue;
   966→      const rot = rotateAroundAxis(vb, vScale(axis, 1 / axisLen), target - ang);
   967→      const delta = vSub(rot, vb);
   968→      const w = pa.invMass + pb.invMass;
   969→      if (w <= 0) continue;
   970→      pb.pos = vAdd(pb.pos, vScale(delta, (pb.invMass * k) / w));
   971→      pa.pos = vSub(pa.pos, vScale(delta, (pa.invMass * k) / w));
   972→    }
   973→  }
   974→
   975→  /**
   976→   * Sentido de flexion (ADR-008): mantiene el vertice articular en el lado anatomico
   977→   * de la cuerda padre-hijo. El angulo no puede detectarlo (doblada al reves = mismo
   978→   * angulo); esto prohibe rodillas que flexionan hacia atras y codos que se meten en
   979→   * la pared. Correccion dura y repartida: el vertice al lado permitido, los extremos
   980→   * con la contra-mitad (conservacion aproximada).
   981→   */
   982→  private solveFolds(): void {
   983→    const k = clamp(this.env.balance.foldStrength, 0, 1);
   984→    for (const f of this.folds) {
   985→      const pj = this.particles[f.j];
   986→      const pa = this.particles[f.a];
   987→      const pb = this.particles[f.b];
   988→      const mx = (pa.pos.x + pb.pos.x) * 0.5;
   989→      const my = (pa.pos.y + pb.pos.y) * 0.5;
   990→      const mz = (pa.pos.z + pb.pos.z) * 0.5;
   991→      const side = (pj.pos.x - mx) * f.dir.x + (pj.pos.y - my) * f.dir.y + (pj.pos.z - mz) * f.dir.z;
   992→      f.side = side;
   993→      if (side >= f.minSide) continue;
   994→      const corr = (f.minSide - side) * k;
   995→      pj.pos = vAdd(pj.pos, vScale(f.dir, corr * 0.7));
   996→      pa.pos = vSub(pa.pos, vScale(f.dir, corr * 0.15));
   997→      pb.pos = vSub(pb.pos, vScale(f.dir, corr * 0.15));
   998→    }
   999→  }
  1000→
  1001→  /**
  1002→   * Guinada de pelvis limitada (anti-"cadera girada"). Anatomia pasiva: la dureza
  1003→   * sigue jointLimitStiffness. Mide el angulo de la barra de caderas respecto a la
  1004→   * de hombros alrededor del eje columna y, si excede el tope, rota las caderas de
  1005→   * vuelta alrededor del eje que pasa por pelvisC (sin tocar longitudes al pivote).
  1006→   */
  1007→  private solveTwists(): void {
  1008→    const max = Math.max(0, this.env.balance.twistLimitRad);
  1009→    if (max <= 0) return;
  1010→    // Ganancia moderada por iteracion: convergencia geometrica rapida ((1-g)^12 ~ 0)
  1011→    // sin inyectar un latigazo de momentum en caidas violentas (el tope es anatomia
  1012→    // pasiva, no un servo duro).
  1013→    const g = 0.6 * clamp(this.env.balance.jointLimitStiffness, 0, 1);
  1014→    for (const t of this.twists) {
  1015→      const pivot = this.particles[t.pivot];
  1016→      const origin = this.particles[t.axisFrom];
  1017→      const axisRaw = vSub(pivot.pos, origin.pos);
  1018→      const alen = vLen(axisRaw);
  1019→      if (alen < 1e-9) continue;
  1020→      const axis = vScale(axisRaw, 1 / alen);
  1021→      const p1 = this.particles[t.a1];
  1022→      const p2 = this.particles[t.a2];
  1023→      const q1 = this.particles[t.b1];
  1024→      const q2 = this.particles[t.b2];
  1025→      const rd = vSub(p2.pos, p1.pos);
  1026→      const cd = vSub(q2.pos, q1.pos);
  1027→      // Ambas barras proyectadas en el plano perpendicular al eje de la columna.
  1028→      const rdPerp = vSub(rd, vScale(axis, vDot(rd, axis)));
  1029→      const cdPerp = vSub(cd, vScale(axis, vDot(cd, axis)));
  1030→      const rl = vLen(rdPerp);
  1031→      const cl = vLen(cdPerp);
  1032→      if (rl < 1e-9 || cl < 1e-9) continue;
  1033→      const rn = vScale(rdPerp, 1 / rl);
  1034→      const cn = vScale(cdPerp, 1 / cl);
  1035→      const sinA = clamp(vDot(vCross(rn, cn), axis), -1, 1);
  1036→      const cosA = clamp(vDot(rn, cn), -1, 1);
  1037→      const ang = Math.atan2(sinA, cosA);
  1038→      t.currentAngle = ang;
  1039→      // El tope VIGENTE es el del tune en vivo (presets/F3), no el almacenado en el rig.
  1040→      const over = ang - clamp(ang, -max, max);
  1041→      if (Math.abs(over) < 1e-5) continue;
  1042→      const corr = -over * t.stiffness * g;
  1043→      const rotate = (idx: number): void => {
  1044→        const p = this.particles[idx];
  1045→        const r = vSub(p.pos, pivot.pos);
  1046→        p.pos = vAdd(pivot.pos, rotateAroundAxis(r, axis, corr));
  1047→      };
  1048→      rotate(t.b1);
  1049→      rotate(t.b2);
  1050→    }
  1051→  }
  1052→
  1053→  /**
  1054→   * Topes de longitud de cadena (anti-muneco de goma). Dureza 1.0 SIEMPRE: es
  1055→   * anatomia pasiva, no musculo (como los limites articulares duros de ADR-008).
  1056→   * La tolerancia si escala con el balance: realista corta antes, ragdoll permite
  1057→   * mas estiramiento antes de cortar. Al violarse, arrastra la raiz hacia el extremo:
  1058→   * una presa "imposible" no alarga el brazo, sube el cuerpo (y dispara la carga).
  1059→   */
  1060→  private solveChainCaps(): void {
  1061→    const tol = Math.max(1.0, this.env.balance.stretchTolerance);
  1062→    for (const c of this.chainCaps) {
  1063→      const max = c.limb === null ? c.base * SPINE_STRETCH_TOLERANCE : c.base * tol;
  1064→      const pa = this.particles[c.a];
  1065→      const pb = this.particles[c.b];
  1066→      const w = pa.invMass + pb.invMass;
  1067→      const d = vSub(pb.pos, pa.pos);
  1068→      const len = vLen(d);
  1069→      c.current = len;
  1070→      if (w <= 0 || len <= max || len < 1e-9) continue;
  1071→      const diff = (len - max) / len;
  1072→      pa.pos = vAdd(pa.pos, vScale(d, (diff * pa.invMass) / w));
  1073→      pb.pos = vSub(pb.pos, vScale(d, (diff * pb.invMass) / w));
  1074→    }
  1075→  }
  1076→
  1077→  private solveCollisions(): void {
  1078→    for (const p of this.particles) {
  1079→      // Suelo: los pies y el resto del cuerpo.
  1080→      if (p.collideFloor && p.pos.y < this.env.floorY + p.radius) {
  1081→        p.pos.y = this.env.floorY + p.radius;
  1082→        // Friccion simple: reduce la velocidad tangencial implicita.
  1083→        p.prev.x += (p.pos.x - p.prev.x) * 0.25;
  1084→        p.prev.z += (p.pos.z - p.prev.z) * 0.25;
  1085→      }
  1086→      // Pared: los segmentos del cuerpo no atraviesan el plano frontal.
  1087→      // (Manos y pies quedan exentos: se apoyan en presas que sobresalen.)
  1088→      if (p.collideWall && p.pos.z > this.env.wallFrontZ - p.radius) {
  1089→        p.pos.z = this.env.wallFrontZ - p.radius;
  1090→      }
  1091→    }
  1092→  }
  1093→}

  1095→function clampVecMag(v: V3, max: number): V3 {
  1096→  const l = vLen(v);
  1097→  if (l <= max || l < 1e-12) return v;
  1098→  return vScale(v, max / l);
  1099→}

  1101→function jointUtilization(c: AngleConstraint): number {
  1102→  const mid = (c.min + c.max) / 2;
  1103→  const half = (c.max - c.min) / 2;
  1104→  if (half <= 1e-9) return 0;
  1105→  return clamp(Math.abs(c.currentAngle - mid) / half, 0, 1);
  1106→}
