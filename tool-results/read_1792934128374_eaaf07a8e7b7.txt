     1→/**
     2→ * Solver de fisica PBD (Position Based Dynamics) para el ragdoll.
     3→ *
     4→ * Decision documentada (ADR-001 del prototipo): adaptador fisico interno basado en
     5→ * particulas + restricciones de distancia y de angulo. La interfaz PhysicsAdapter se
     6→ * mantiene desacoplada para poder sustituirlo por Havok/Rapier sin tocar gameplay.
     7→ *
     8→ * Modelo de anclaje (estimador documentado, seccion 9.2 del documento): cada anclaje
     9→ * es un resorte amortiguado con fuerza maxima. La carga se mide como la fuerza que el
    10→ * resorte debe aplicar (rigidez x estiramiento, antes de saturar), lo que permite
    11→ * detectar sobrecarga y provoca estiramiento visible cuando el agarre no basta.
    12→ * Nunca se teletransportan cuerpos durante el juego.
    13→ */

    48→/**
    49→ * Restriccion de SENTIDO DE FLEXION (ADR-008): el constraint angular es 2D y no
    50→ * distingue doblar de doblar al reves (una rodilla invertida tiene el MISMO angulo).
    51→ * Este constraint mantiene el vertice articular (codo/rodilla) en el lado correcto
    52→ * del plano definido por la cuerda padre-hijo y una direccion de referencia del
    53→ * cuerpo (normal del muro): prohibe la hiperextension/luxacion visual.
    54→ */
    55→export interface FoldConstraint {
    56→  id: string;
    57→  /** Particula del vertice articular (codo/rodilla). */
    58→  j: number;
    59→  /** Extremo padre (hombro/cadera). */
    60→  a: number;
    61→  /** Extremo hijo (mano/pie). */
    62→  b: number;
    63→  /** Direccion unitaria del lado PERMITIDO (mundo). */
    64→  dir: V3;
    65→  /** Separacion minima exigida en esa direccion (m): tolera recto, prohibe el giro. */
    66→  minSide: number;
    67→  strength: number;
    68→  /** Lectura de diagnostico: lado actual (m). */
    69→  side: number;
    70→}

   141→/**
   142→ * Rangos del balance realismo/ragdoll (ADR-008), calibrables en vivo (F3).
   143→ * - jointLimitStiffness: dureza de los limites angulares (1 = duro, sin luxaciones).
   144→ * - foldStrength: dureza del sentido de flexion (anti-rodilla/codo invertido).
   145→ * - postureTone: tono postural en juego (1 = erguido y vivo, 0 = muñeco de trapo).

   147→ * - stretchTolerance: cuanto puede ALARGARSE una cadena anatomica sobre su longitud

   158→/**
   159→ * Tope de LONGITUD DE CADENA (anatomia pasiva, como los limites articulares):
   160→ * prohibe que la distancia raiz->extremo (hombro->mano, cadera->pie, pelvis->pecho)

   162→ * barras elasticas se estiran bajo carga sostenida y aparecen posturas imposibles
   163→ * (brazos de 1.5 m). Desigualdad dura: solo actua al violar el maximo.

   293→  setRagdoll(
   294→    particles: Particle[],
   295→    sticks: Stick[],
   296→    angles: AngleConstraint[],
   297→    limbParticle: Map<LimbId, number>,
   298→    topology?: BodyTopology,
   299→    folds: FoldConstraint[] = []
   300→  ): void {
   301→    this.particles = particles;
   302→    this.sticks = sticks;
   303→    this.angles = angles;
   304→    this.folds = folds;
   305→    this.limbParticle = limbParticle;
   306→    this.topology = topology ?? null;
   307→    this.anchors.clear();
   308→    this.targets.clear();
   309→    this.climbDrive = 0;
   310→    this.rebuildChainCaps();
   311→  }
   312→
   313→  /** Deriva los topes de cadena de la topologia (hombro->mano, cadera->pie, columna). */
   314→  private rebuildChainCaps(): void {
   315→    this.chainCaps = [];
   316→    const t = this.topology;
   317→    if (!t) return;
   318→    for (const limb of LIMB_IDS) {
   319→      const root = t.rootOf.get(limb);
   320→      const end = this.limbParticle.get(limb);
   321→      if (root === undefined || end === undefined) continue;
   322→      const base = limb.startsWith("hand") ? t.armChain : t.legChain;
   323→      this.chainCaps.push({ a: root, b: end, base, current: base, limb });
   324→    }
   325→    // Columna: el torso tampoco se estira (menos tolerancia: es un segmento rigido).
   326→    this.chainCaps.push({ a: t.pelvis, b: t.chest, base: t.spineLen, current: t.spineLen, limb: null });
   327→  }
   328→
   329→  /** Lecturas de diagnostico del estiramiento de cadena (ratio 1 = en el tope). */
   330→  debugChains(): { limb: string; len: number; max: number; ratio: number }[] {
   331→    const tol = this.env.balance.stretchTolerance;
   332→    return this.chainCaps.map((c) => {
   333→      const max = c.limb === null ? c.base * SPINE_STRETCH_TOLERANCE : c.base * tol;
   334→      return { limb: c.limb ?? "spine", len: c.current, max, ratio: max > 1e-9 ? c.current / max : 0 };
   335→    });
   336→  }

   411→  createAnchor(limb: LimbId, holdId: string, point: V3, maxForce: number): AnchorState {
   412→    this.releaseAnchor(limb);
   413→    const particle = this.limbParticle.get(limb);

   415→      throw new Error(`No existe particula para la extremidad ${limb}`);
   416→    }

   418→      limb,
   419→      particle,
   420→      holdId,
   421→      point: vClone(point),
   422→      maxForce,
   423→      loadN: 0,
   424→      stretch: 0,
   425→      activation: 0
   426→    };
   427→    this.anchors.set(limb, anchor);
   428→    this.targets.delete(limb);
   429→    return anchor;

   450→  /** Mueve todos los particulas por un delta (solo para la pose inicial y el impulso de arranque). */
   451→  translateRagdoll(delta: V3): void {
   452→    for (const p of this.particles) {
   453→      p.pos = vAdd(p.pos, delta);
   454→      p.prev = vAdd(p.prev, delta);
   455→    }
   456→  }
   457→
   458→  /**
   459→   * Relajacion estatica: proyecta restricciones SIN integrar (sin gravedad ni energia
   460→   * cinetica). Convierte una semilla aproximada en una pose consistente antes del primer
   461→   * paso, eliminando las patadas del settle inicial.
   462→   */
   463→  relax(sweeps = 80): void {
   464→    for (let i = 0; i < sweeps; i++) {
   465→      this.solveSticks();
   466→      this.solveAngles();
   467→      this.solveFolds();
   468→      this.solveChainCaps();
   469→      for (const anchor of this.anchors.values()) {
   470→        const p = this.particles[anchor.particle];
   471→        p.pos = vClone(anchor.point);
   472→      }
   473→      // Segunda pasada de topes: los pines acaban de recolocar las manos en sus
   474→      // presas; si esa recolocacion deja una cadena sobre su maximo, se corta YA
   475→      // (si no, la relajacion convergeria con los brazos estirados).
   476→      this.solveChainCaps();
   477→      this.solveCollisions();
   478→    }
   479→    for (const p of this.particles) {
   480→      p.prev = vClone(p.pos);
   481→    }
   482→  }
   483→
   484→  /** Aplica una velocidad inicial (m/s) al cuerpo completo; usado por start.impulse. */
   485→  applyImpulseVelocity(vel: V3, dt: number): void {
   486→    for (const p of this.particles) {
   487→      if (p.invMass > 0) {
   488→        p.prev = vSub(p.pos, vScale(vel, dt));
   489→      }
   490→    }
   491→  }
   492→
   493→  totalMass(): number {
   494→    return this.particles.reduce((acc, p) => acc + p.mass, 0);
   495→  }
   496→
   497→  centerOfMass(): V3 {
   498→    let m = 0;
   499→    const c = v3();
   500→    for (const p of this.particles) {
   501→      c.x += p.pos.x * p.mass;
   502→      c.y += p.pos.y * p.mass;
   503→      c.z += p.pos.z * p.mass;
   504→      m += p.mass;
   505→    }
   506→    return vScale(c, 1 / (m || 1));
   507→  }
   508→
   509→  step(dt: number): StepResult {
   510→    const K = this.env.iterations;
   511→    const forces = new Map<number, V3>();
   512→    const anchorLoads: Partial<Record<LimbId, number>> = {};
   513→    const anchorStretch: Partial<Record<LimbId, number>> = {};
   514→    let nanError = false;
   515→    let outOfBounds = false;
   516→    let maxSpeed = 0;
   517→
   518→    // 0) Tono postural (ADR-008): persigue el objetivo con rates asimetricos
   519→    // (la caida relaja rapido; recuperarlo de pie cuesta mas).
   520→    const toneRate = this.toneTarget < this.tone ? 4.5 : 2.0;
   521→    const toneGap = this.toneTarget - this.tone;
   522→    this.tone = Math.abs(toneGap) < 0.01 ? this.toneTarget : this.tone + toneGap * Math.min(1, toneRate * dt);
   523→
   524→    // 1) Fuerza externa de objetivos (resorte amortiguado con tope). Los anclajes NO son
   525→    // resortes: se fijan como pin rigido en la fase de restricciones (ver 4b).
   526→    for (const target of this.targets.values()) {
   527→      const p = this.particles[target.particle];
   528→      const err = vSub(target.target, p.pos);
   529→      const vel = vScale(vSub(p.pos, p.prev), 1 / dt);
   530→      const spring = vAdd(vScale(err, this.env.followStiffness), vScale(vel, -this.env.anchorDamping * 0.5));
   531→      const applied = clampVecMag(spring, target.maxForce);
   532→      forces.set(target.particle, vAdd(forces.get(target.particle) ?? v3(), applied));
   533→    }
   534→
   535→    // 1b) Musculos activos (ADR-007): cuadriceps, step-up, traccion de brazos y tono
   536→    // postural. Sin anclajes el cuerpo es ragdoll puro (caida limpia).
   537→    if (this.env.muscles.musclesOn && this.topology !== null && this.anchors.size > 0) {
   538→      this.solveMuscles(dt, forces);
   539→    }
   540→    this.climbDrive = Math.max(0, this.climbDrive - dt / this.env.muscles.driveSeconds);
   541→
   542→    // 2) Integracion Verlet con paso fijo (con techo de velocidad: los musculos no
   543→    // pueden lanzar el ragdoll a velocidades no humanas; el ERROR GUARD queda para
   544→    // anomalias reales del solver).
   545→    const g = this.env.gravity;
   546→    const vCap = this.env.muscles.vMaxSpeed * dt;
   547→    for (let i = 0; i < this.particles.length; i++) {
   548→      const p = this.particles[i];
   549→      if (p.invMass <= 0) continue;
   550→      const f = forces.get(i);
   551→      const a = f ? vAdd(g, vScale(f, p.invMass)) : g;
   552→      const vel = vScale(vSub(p.pos, p.prev), this.env.damping);
   553→      p.prev = vClone(p.pos);
   554→      let next = vAdd(vAdd(p.pos, vel), vScale(a, dt * dt));
   555→      const newVel = vSub(next, p.prev);
   556→      const newSpeed = vLen(newVel);
   557→      if (newSpeed > vCap) {
   558→        next = vAdd(p.prev, vScale(newVel, vCap / newSpeed));
   559→      }
   560→      p.pos = next;
   561→      // Metrica sobre la velocidad YA acotada (la previa incluye correcciones de pin).
   562→      const measured = Math.min(newSpeed, vCap) / dt;
   563→      if (measured > maxSpeed) maxSpeed = measured;
   564→    }
   565→
   566→    // 3) Tremor de esfuerzo (seccion 9.5): ruido proporcional al estado de tension.
   567→    for (const limb of LIMB_IDS) {
   568→      const t = this.tremor[limb];
   569→      if (t <= 0) continue;
   570→      const idx = this.limbParticle.get(limb);
   571→      if (idx === undefined) continue;
   572→      const p = this.particles[idx];
   573→      const amp = this.env.tremorAmplitude * t;
   574→      p.pos = vAdd(p.pos, v3((Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp));
   575→    }
   576→
   577→    // 4) Proyeccion de restricciones.
   578→    // 4a) Acumulador de reaccion por anclaje: suma de correcciones del pin este paso.
   579→    const pinCorrection = new Map<LimbId, V3>();
   580→    for (let iter = 0; iter < K; iter++) {
   581→      this.solveSticks();
   582→      this.solveAngles();
   583→      this.solveFolds();
   584→      this.solveChainCaps();
   585→      // 4b) Anclajes: pin rigido con tope de fuerza. La carga requerida se mide como la
   586→      // fuerza necesaria para mantener el pin (F = m x correccion / dt^2), un estimador
   587→      // consistente con el impulso aplicado (9.2). En sobrecarga el pin se satura y la
   588→      // extremidad resbala visiblemente de la presa.
   589→      for (const anchor of this.anchors.values()) {
   590→        const p = this.particles[anchor.particle];
   591→        const err = vSub(anchor.point, p.pos);
   592→        const errLen = vLen(err);
   593→        if (errLen < 1e-9) continue;
   594→        const maxCorr = (anchor.maxForce * dt * dt) / p.mass;
   595→        const scaled = errLen > maxCorr ? vScale(err, maxCorr / errLen) : err;
   596→        p.pos = vAdd(p.pos, scaled);
   597→        pinCorrection.set(anchor.limb, vAdd(pinCorrection.get(anchor.limb) ?? v3(), scaled));
   598→      }
   599→      // 4c) Topes de cadena tras los pines: una presa mas alla del alcance anatomico
   600→      // NO estira el brazo (sin "muneco de goma"); el agarre resbala hacia el cuerpo,
   601→      // el cuerpo es arrastrado hacia la presa y la carga del pin se dispara (9.2):
   602→      // la biomecanica lee esa lucha como SOBRECARGA y abre el agarre con cuenta atras.
   603→      this.solveChainCaps();
   604→      this.solveCollisions();
   605→    }
   606→    // NOTA: la correccion del pin genera momento "fantasma" en Verlet; ya no se
   607→    // neutraliza en prev (cambiaria la calibracion del estimador de carga). El techo
   608→    // de velocidad de la integracion (vMaxSpeed) contiene ese momento residual.
   609→    // Carga suavizada: impulso total demandado por el pin, en newtons, con ganancia
   610→    // calibrada del estimador documentado (las re-proyecciones por iteracion sobre-miden).
   611→    for (const anchor of this.anchors.values()) {
   612→      const net = pinCorrection.get(anchor.limb) ?? v3();
   613→      const measured = this.particles[anchor.particle].mass * (vLen(net) / (dt * dt)) * this.env.loadEstimatorGain;
   614→      anchor.loadN = anchor.loadN === 0 ? measured : anchor.loadN + (measured - anchor.loadN) * 0.3;
   615→      anchor.stretch = vLen(vSub(anchor.point, this.particles[anchor.particle].pos));
   616→      anchorLoads[anchor.limb] = anchor.loadN;
   617→      anchorStretch[anchor.limb] = anchor.stretch;
   618→    }
   619→
   620→    // 5) Guardas: NaN, energia y limites del mundo.
   621→    for (const p of this.particles) {
   622→      if (!isFiniteVec(p.pos) || !isFiniteVec(p.prev)) {
   623→        nanError = true;
   624→        break;
   625→      }
   626→      if (p.pos.x < this.env.bounds.minX || p.pos.x > this.env.bounds.maxX || p.pos.y > this.env.bounds.maxY || p.pos.z < this.env.bounds.minZ || p.pos.z > this.env.bounds.maxZ) {
   627→        outOfBounds = true;
   628→      }
   629→    }
   630→    if (maxSpeed > 18) {
   631→      // Energia anomala: se registra, la capa de juego decide (ERROR GUARD).
   632→      maxSpeed = clamp(maxSpeed, 0, 30);
   633→    }
   634→
   635→    // 6) Lecturas derivadas.
   636→    const floorContacts: { limb: LimbId; point: V3 }[] = [];
   637→    for (const limb of LIMB_IDS) {
   638→      const idx = this.limbParticle.get(limb);
   639→      if (idx === undefined) continue;
   640→      const p = this.particles[idx];
   641→      if (p.pos.y <= this.env.floorY + 0.045) {
   642→        floorContacts.push({ limb, point: v3(p.pos.x, this.env.floorY, p.pos.z) });
   643→      }
   644→    }
   645→
   646→    return {
   647→      nanError,
   648→      energySpike: maxSpeed > 18,
   649→      outOfBounds,
   650→      com: this.centerOfMass(),
   651→      anchorLoads,
   652→      anchorStretch,
   653→      floorContacts,
   654→      jointAngles: this.angles.map((a) => ({
   655→        id: a.id,
   656→        angle: a.currentAngle,
   657→        utilization: jointUtilization(a)
   658→      })),
   659→      maxSpeed
   660→    };
   661→  }
   662→
   663→  /**
   664→   * Fase muscular (ADR-007). Todas las fuerzas se aplican sobre particulas de masa alta
   665→   * (pelvis ~11 kg, pecho ~14 kg) con topes calibrados para un desplazamiento por paso
   666→   * estable; los pines miden la carga resultante y la biomecanica la traduce en fatiga.
   667→   */
   668→  private solveMuscles(dt: number, forces: Map<number, V3>): void {
   669→    const M = this.env.muscles;
   670→    const T = this.topology!;
   671→    const pelvis = this.particles[T.pelvis];
   672→    const chest = this.particles[T.chest];
   673→    const head = this.particles[T.head];
   674→    const addForce = (idx: number, fx: number, fy: number, fz: number): void => {
   675→      const cur = forces.get(idx);
   676→      forces.set(idx, cur ? v3(cur.x + fx, cur.y + fy, cur.z + fz) : v3(fx, fy, fz));
   677→    };
   678→
   679→    // Rampa de activacion por anclaje (evita el tiron al anclar).
   680→    for (const anchor of this.anchors.values()) {
   681→      const rate = anchor.limb.startsWith("foot") ? 1 / 0.35 : 1 / 0.2;
   682→      anchor.activation = Math.min(1, anchor.activation + dt * rate);
   683→    }
   684→
   685→    // Pre-pasada: hay pierna sosteniendo (cuadriceps activo)? Con apoyo de piernas
   686→    // los brazos NO tiran (el pull-up esta para colgar/colocarse, no para despegar
   687→    // al cuerpo que ya esta de pie: evita la suma explosiva de fuerzas).
   688→    let feetSupporting = false;
   689→    // Peso que cargan los brazos ahora mismo (N): modula el balance (un pie que
   690→    // solo descansa no debe volcar el COM hacia si; uno que sostiene, si).
   691→    let handLoadN = 0;
   692→    for (const anchor of this.anchors.values()) {
   693→      if (anchor.limb.startsWith("hand")) handLoadN += Math.max(0, anchor.loadN);
   694→    }
   695→    // Arbitracion de pies: el STEP-UP lo hace solo el pie MAS ALTO (transferencia de
   696→    // peso al escalon, como en escalada real); el balance solo desde el pie activo,
   697→    // para que un apoyo bajo no pelee contra el pie que esta montando.
   698→    let stepUpFoot: LimbId | null = null;
   699→    let stepUpHoldY = -Infinity;
   700→    for (const anchor of this.anchors.values()) {
   701→      if (!anchor.limb.startsWith("foot")) continue;
   702→      const root = this.particles[T.rootOf.get(anchor.limb)!];
   703→      const dyr = root.pos.y - anchor.point.y;
   704→      const dr = Math.hypot(root.pos.x - anchor.point.x, dyr, root.pos.z - anchor.point.z) || 1e-6;
   705→      if (dyr / dr > 0.05) {
   706→        feetSupporting = true;
   707→      }
   708→      if (dyr / dr < 0.02 && anchor.point.y > stepUpHoldY) {
   709→        stepUpHoldY = anchor.point.y;
   710→        stepUpFoot = anchor.limb;
   711→      }
   712→    }
   713→
   714→    let feetAnchored = 0;
   715→    for (const anchor of this.anchors.values()) {
   716→      const isFoot = anchor.limb.startsWith("foot");
   717→      if (isFoot) feetAnchored++;
   718→      const rootIdx = T.rootOf.get(anchor.limb);
   719→      if (rootIdx === undefined) continue;
   720→      const root = this.particles[rootIdx];
   721→      const dx = root.pos.x - anchor.point.x;
   722→      const dy = root.pos.y - anchor.point.y;
   723→      const dz = root.pos.z - anchor.point.z;
   724→      const dist = Math.hypot(dx, dy, dz) || 1e-6;
   725→
   726→      if (isFoot) {
   727→        // CUADRICEPS (apoyo): el pie por DEBAJO de la cadera empuja la pelvis lejos del
   728→        // pie (extension de pierna = ponerse de pie). Proyectado al plano de la pared:
   729→        // la pierna empuja "hacia arriba", no separa del muro.
   730→        if (dy / dist > 0.05) {
   731→          // Servo de pierna: empuja mientras la pierna este flexionada y ceda al
   732→          // acercarse a la extension natural (0.9·cadena = rodilla casi recta).
   733→          // Asi la postura de pie es un equilibrio estable, no un empuje sin tope.
   734→          const targetDist = T.legChain * 0.9;
   735→          const f = clamp((targetDist - dist) * M.quadServoK, 0, M.quadPushMax) * anchor.activation;
   736→          const ul = Math.hypot(dx, dy) || 1e-6;
   737→          // Direccion axial con sesgo vertical (reaccion del suelo ~vertical al de pie).
   738→          const bx = (dx / ul) * (1 - M.quadUpBias);
   739→          const by = (dy / ul) * (1 - M.quadUpBias) + M.quadUpBias;
   740→          const bl = Math.hypot(bx, by) || 1e-6;
   741→          addForce(T.pelvis, (bx / bl) * f, (by / bl) * f, 0);
   742→          // BALANCE: llevar el COM sobre el apoyo (feedback tipo SIMBICON); sin el,
   743→          // un apoyo lateral nunca sostiene de verdad. Solo desde el pie que esta
   744→          // montando (si hay step-up) o desde los apoyos si nadie monta. Escalado por
   745→          // la carga REAL de la pierna frente a los brazos: un pie que descansa no
   746→          // vuelca el cuerpo (evita sobrecargar el brazo contrario en cascada).
   747→          if (stepUpFoot === null || stepUpFoot === anchor.limb) {
   748→            const weightShare = clamp(f / (f + handLoadN + 60), 0.15, 1);
   749→            const fb = clamp((anchor.point.x - pelvis.pos.x) * 400, -M.balanceMax, M.balanceMax) * anchor.activation * weightShare;
   750→            addForce(T.pelvis, fb, 0, 0);
   751→            // 3a ley de Newton: parte de la reaccion entra por el pie al pin, de modo que
   752→            // el apoyo MEDIDO carga de verdad al estar de pie (fatiga de piernas real).
   753→            addForce(anchor.particle, -(bx / bl) * f * 0.5 - fb * 0.4, -(by / bl) * f * 0.5, 0);
   754→          } else {
   755→            addForce(anchor.particle, -(bx / bl) * f * 0.5, -(by / bl) * f * 0.5, 0);
   756→          }
   757→        }
   758→        // PSOAS/STEP-UP: el pie por ENCIMA de la cadera sube el cuerpo hacia la presa
   759→        // (montar el escalon). Es lo que convierte "apoyar un pie" en ascender. La
   760→        // componente vertical del vector hacia la presa regula el empuje solo: al
   761→        // acercarse la pelvis al nivel del pie la fuerza se hace horizontal (COM sobre
   762→        // el apoyo) y el cuadriceps toma el relevo por encima. Solo el pie MAS ALTO
   763→        // monta (arbitracion: transferencia de peso al escalon).
   764→        if (stepUpFoot === anchor.limb && dy / dist < 0.02) {
   765→          const boost = 0.9 + 0.3 * this.climbDrive;
   766→          // Esvaimiento por proximidad: al acercarse la pelvis al nivel del pie la
   767→          // fuerza cede (sin sobregiro mas alla de la postura natural de pie).
   768→          const taper = clamp(-dy / 0.45, 0.55, 1);
   769→          const f = M.stepUpMax * anchor.activation * boost * taper;
   770→          const ux = -dx / dist;
   771→          const uy = -dy / dist;
   772→          const uz = -dz / dist;
   773→          addForce(T.pelvis, ux * f, uy * f, uz * f);
   774→          addForce(anchor.particle, -ux * f * 0.35, -uy * f * 0.35, -uz * f * 0.35);
   775→        }
   776→      } else {
   777→        // TRACCION DE BRAZOS (impulso de subida del jugador): mano por encima del
   778→        // hombro y brazo ESTIRADO (compuerta de holgura) tira del pecho hacia el
   779→        // anclaje. Con piernas de apoyo es el "subirse" que sube los hombros; el
   780→        // servo de pierna limita la postura y evita el pliegue.
   781→        if (this.climbDrive > 0.02 && dy / dist < -0.1) {
   782→          // Holgura medida desde la RAIZ (hombro): en postura plegada el pecho puede
   783→          // estar alto con el hombro colgando bajo y el brazo estirado; el pull debe
   784→          // evaluar el brazo real, no la posicion del pecho.
   785→          const rdx = anchor.point.x - root.pos.x;
   786→          const rdy = anchor.point.y - root.pos.y;
   787→          const rdz = anchor.point.z - root.pos.z;
   788→          const rd = Math.hypot(rdx, rdy, rdz) || 1e-6;
   789→          const slack = rd - T.armChain * 0.9;
   790→          const cdx = anchor.point.x - chest.pos.x;
   791→          const cdy = anchor.point.y - chest.pos.y;
   792→          const cdz = anchor.point.z - chest.pos.z;
   793→          const cd = Math.hypot(cdx, cdy, cdz) || 1e-6;
   794→          if (slack > 0) {
   795→            const gate = clamp(slack / (T.armChain * 0.2), 0.4, 1);
   796→            // Durante un step-up la pierna manda: los brazos asisten al 60%.
   797→            const assist = stepUpFoot === null ? 1 : 0.6;
   798→            // Margen de agarre: no se puede tirar mas fuerte que el agarre restante
   799→            // (un brazo al limite no tira: evita saturar el pin y la sobrecarga).
   800→            const headroom = clamp(1 - anchor.loadN / anchor.maxForce, 0, 1);
   801→            const f = M.armPullMax * anchor.activation * this.climbDrive * gate * assist * headroom;
   802→            addForce(T.chest, (cdx / cd) * f, (cdy / cd) * f, (cdz / cd) * f);
   803→            addForce(rootIdx, (cdx / cd) * f * 0.35, (cdy / cd) * f * 0.35, (cdz / cd) * f * 0.35);
   804→          }
   805→        }
   806→      }
   807→    }
   808→
   809→    // TONO POSTURAL: pecho sobre pelvis y cabeza sobre pecho (anti-muneco de trapo).
   810→    // Escala con el numero de apoyos y con el tono global (ADR-008): cuerpo libre =
   811→    // ragdoll puro; tono bajo (caida) = cuerpo flacido pero sin pliegues imposibles
   812→    // (los limites y folds siguen activos: son anatomia pasiva, no musculo).
   813→    const toneScale = Math.min(1, this.anchors.size / 2) * (0.15 + 0.85 * this.tone);
   814→    const tex = pelvis.pos.x - chest.pos.x;
   815→    const tey = pelvis.pos.y + T.spineLen * 0.97 - chest.pos.y;
   816→    const tez = pelvis.pos.z - chest.pos.z;
   817→    const teLen = Math.hypot(tex, tey, tez) || 1e-6;
   818→    const coreF = Math.min(M.coreMax * toneScale, teLen * 400);
   819→    addForce(T.chest, (tex / teLen) * coreF, (tey / teLen) * coreF, (tez / teLen) * coreF);
   820→    const nex = chest.pos.x - head.pos.x;
   821→    const ney = chest.pos.y + 0.2 - head.pos.y;
   822→    const nez = chest.pos.z - head.pos.z;
   823→    const neLen = Math.hypot(nex, ney, nez) || 1e-6;
   824→    const neckF = Math.min(M.neckMax * toneScale, neLen * 300);
   825→    addForce(T.head, (nex / neLen) * neckF, (ney / neLen) * neckF, (nez / neLen) * neckF);
   826→
   827→    // PEGADO A LA PARED con cualquier anclaje: la colision del cuerpo con la cara del
   828→    // muro impide colgar en la vertical exacta de los pines (las presas sobresalen), y
   829→    // sin fuerza de centrado en z el cuerpo queda 0.1-0.2 m por detras, offset que
   830→    // devora el alcance util de brazos y piernas en CADA movimiento. El escalador real
   831→    // va pegado: cadera y pecho se presionan hacia el plano de presas.
   832→    if (this.anchors.size > 0) {
   833→      const zTarget = this.env.wallFrontZ - 0.26;
   834→      const fzPelvis = clamp((zTarget - pelvis.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   835→      const fzChest = clamp((zTarget - chest.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   836→      addForce(T.pelvis, 0, 0, fzPelvis);
   837→      addForce(T.chest, 0, 0, fzChest);
   838→    }
   839→
   840→    // Tope de fuerza muscular por particula axial: la suma de musculos nunca puede
   841→    // superar el techo (estabilidad de Verlet con las masas de pelvis/pecho).
   842→    const cap = (idx: number, capN: number): void => {
   843→      const f = forces.get(idx);
   844→      if (!f) return;
   845→      const l = vLen(f);
   846→      if (l > capN) forces.set(idx, vScale(f, capN / l));
   847→    };
   848→    cap(T.pelvis, M.muscleForceCapPelvis);
   849→    cap(T.chest, M.muscleForceCapChest);
   850→
   851→    // Instantanea para diagnostico (solo particulas axiales y raices).
   852→    this.lastMuscleForces.clear();
   853→    for (const idx of [T.pelvis, T.chest, T.head]) {
   854→      const f = forces.get(idx);
   855→      if (f) this.lastMuscleForces.set(idx, f);
   856→    }
   857→  }
   858→
   859→  private solveSticks(): void {
   860→    for (const s of this.sticks) {
   861→      const pa = this.particles[s.a];
   862→      const pb = this.particles[s.b];
   863→      const w = pa.invMass + pb.invMass;
   864→      if (w <= 0) continue;
   865→      const d = vSub(pb.pos, pa.pos);
   866→      const len = vLen(d);
   867→      if (len < 1e-9) continue;
   868→      const diff = (len - s.rest) / len;
   869→      const k = s.stiffness;
   870→      pa.pos = vAdd(pa.pos, vScale(d, (diff * pa.invMass * k) / w));
   871→      pb.pos = vSub(pb.pos, vScale(d, (diff * pb.invMass * k) / w));
   872→    }
   873→  }
   874→
   875→  private solveAngles(): void {
   876→    // Limites articulares DUROS (ADR-008): la rigidez de los datos ya no escala la
   877→    // correccion de la violacion; jointLimitStiffness (1 = duro) garantiza que ni
   878→    // colgando ni en caida el angulo cruce el rango anatomico. La flexion DENTRO del
   879→    // rango sigue libre (el constraint solo actua al violar).
   880→    const k = clamp(this.env.balance.jointLimitStiffness, 0, 1);
   881→    for (const c of this.angles) {
   882→      const pj = this.particles[c.j];
   883→      const pa = this.particles[c.a];
   884→      const pb = this.particles[c.b];
   885→      const va = vSub(pa.pos, pj.pos);
   886→      const vb = vSub(pb.pos, pj.pos);
   887→      const la = vLen(va);
   888→      const lb = vLen(vb);
   889→      if (la < 1e-9 || lb < 1e-9) {
   890→        c.currentAngle = Number.isFinite(la) && Number.isFinite(lb) ? c.currentAngle : c.currentAngle;
   891→        continue;
   892→      }
   893→      const ang = angleBetween(va, vb);
   894→      c.currentAngle = ang;
   895→      const target = clamp(ang, c.min, c.max);
   896→      if (Math.abs(target - ang) < 1e-5) continue;
   897→      const axis = vCross(va, vb);
   898→      const axisLen = vLen(axis);
   899→      if (axisLen < 1e-9) continue;
   900→      const rot = rotateAroundAxis(vb, vScale(axis, 1 / axisLen), target - ang);
   901→      const delta = vSub(rot, vb);
   902→      const w = pa.invMass + pb.invMass;
   903→      if (w <= 0) continue;
   904→      pb.pos = vAdd(pb.pos, vScale(delta, (pb.invMass * k) / w));
   905→      pa.pos = vSub(pa.pos, vScale(delta, (pa.invMass * k) / w));
   906→    }
   907→  }
   908→
   909→  /**
   910→   * Sentido de flexion (ADR-008): mantiene el vertice articular en el lado anatomico
   911→   * de la cuerda padre-hijo. El angulo no puede detectarlo (doblada al reves = mismo
   912→   * angulo); esto prohibe rodillas que flexionan hacia atras y codos que se meten en
   913→   * la pared. Correccion dura y repartida: el vertice al lado permitido, los extremos
   914→   * con la contra-mitad (conservacion aproximada).
   915→   */
   916→  private solveFolds(): void {
   917→    const k = clamp(this.env.balance.foldStrength, 0, 1);
   918→    for (const f of this.folds) {
   919→      const pj = this.particles[f.j];
   920→      const pa = this.particles[f.a];
   921→      const pb = this.particles[f.b];
   922→      const mx = (pa.pos.x + pb.pos.x) * 0.5;
   923→      const my = (pa.pos.y + pb.pos.y) * 0.5;
   924→      const mz = (pa.pos.z + pb.pos.z) * 0.5;
   925→      const side = (pj.pos.x - mx) * f.dir.x + (pj.pos.y - my) * f.dir.y + (pj.pos.z - mz) * f.dir.z;
   926→      f.side = side;
   927→      if (side >= f.minSide) continue;
   928→      const corr = (f.minSide - side) * k;
   929→      pj.pos = vAdd(pj.pos, vScale(f.dir, corr * 0.7));
   930→      pa.pos = vSub(pa.pos, vScale(f.dir, corr * 0.15));
   931→      pb.pos = vSub(pb.pos, vScale(f.dir, corr * 0.15));
   932→    }
   933→  }
   934→
   935→  /**
   936→   * Topes de longitud de cadena (anti-muneco de goma). Dureza 1.0 SIEMPRE: es
   937→   * anatomia pasiva, no musculo (como los limites articulares duros de ADR-008).
   938→   * La tolerancia si escala con el balance: realista corta antes, ragdoll permite
   939→   * mas estiramiento antes de cortar. Al violarse, arrastra la raiz hacia el extremo:
   940→   * una presa "imposible" no alarga el brazo, sube el cuerpo (y dispara la carga).
   941→   */
   942→  private solveChainCaps(): void {
   943→    const tol = Math.max(1.0, this.env.balance.stretchTolerance);
   944→    for (const c of this.chainCaps) {
   945→      const max = c.limb === null ? c.base * SPINE_STRETCH_TOLERANCE : c.base * tol;
   946→      const pa = this.particles[c.a];
   947→      const pb = this.particles[c.b];
   948→      const w = pa.invMass + pb.invMass;
   949→      const d = vSub(pb.pos, pa.pos);
   950→      const len = vLen(d);
   951→      c.current = len;
   952→      if (w <= 0 || len <= max || len < 1e-9) continue;
   953→      const diff = (len - max) / len;
   954→      pa.pos = vAdd(pa.pos, vScale(d, (diff * pa.invMass) / w));
   955→      pb.pos = vSub(pb.pos, vScale(d, (diff * pb.invMass) / w));
   956→    }
   957→  }
   958→
   959→  private solveCollisions(): void {
   960→    for (const p of this.particles) {
   961→      // Suelo: los pies y el resto del cuerpo.
   962→      if (p.collideFloor && p.pos.y < this.env.floorY + p.radius) {
   963→        p.pos.y = this.env.floorY + p.radius;
   964→        // Friccion simple: reduce la velocidad tangencial implicita.
   965→        p.prev.x += (p.pos.x - p.prev.x) * 0.25;
   966→        p.prev.z += (p.pos.z - p.prev.z) * 0.25;
   967→      }
   968→      // Pared: los segmentos del cuerpo no atraviesan el plano frontal.
   969→      // (Manos y pies quedan exentos: se apoyan en presas que sobresalen.)
   970→      if (p.collideWall && p.pos.z > this.env.wallFrontZ - p.radius) {
   971→        p.pos.z = this.env.wallFrontZ - p.radius;
   972→      }
   973→    }
   974→  }
   975→}

   977→function clampVecMag(v: V3, max: number): V3 {
   978→  const l = vLen(v);
   979→  if (l <= max || l < 1e-12) return v;
   980→  return vScale(v, max / l);
   981→}
   982→
   983→function jointUtilization(c: AngleConstraint): number {
   984→  const mid = (c.min + c.max) / 2;
   985→  const half = (c.max - c.min) / 2;
   986→  if (half <= 1e-9) return 0;
   987→  return clamp(Math.abs(c.currentAngle - mid) / half, 0, 1);
   988→}
