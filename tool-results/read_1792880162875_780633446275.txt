     1→     1→/**
     2→     2→ * Solver de fisica PBD (Position Based Dynamics) para el ragdoll.
     3→     3→ *
     4→     4→ * Decision documentada (ADR-001 del prototipo): adaptador fisico interno basado en
     5→     5→ * particulas + restricciones de distancia y de angulo. La interfaz PhysicsAdapter se

   319→   319→  /** Mueve todos los particulas por un delta (solo para la pose inicial y el impulso de arranque). */
   320→   320→  translateRagdoll(delta: V3): void {
   321→   321→    for (const p of this.particles) {
   322→   322→      p.pos = vAdd(p.pos, delta);
   323→   323→      p.prev = vAdd(p.prev, delta);

   327→   327→  /**
   328→   328→   * Relajacion estatica: proyecta restricciones SIN integrar (sin gravedad ni energia

   332→   332→  relax(sweeps = 80): void {
   333→   333→    for (let i = 0; i < sweeps; i++) {
   334→   334→      this.solveSticks();
   335→   335→      this.solveAngles();
   336→   336→      for (const anchor of this.anchors.values()) {
   337→   337→        const p = this.particles[anchor.particle];
   338→   338→        p.pos = vClone(anchor.point);
   339→   339→      }
   340→   340→      this.solveCollisions();
   341→   341→    }
   342→   342→    for (const p of this.particles) {
   343→   343→      p.prev = vClone(p.pos);
   344→   344→    }
   345→   345→  }

   347→   347→  /** Aplica una velocidad inicial (m/s) al cuerpo completo; usado por start.impulse. */
   348→   348→  applyImpulseVelocity(vel: V3, dt: number): void {
   349→   349→    for (const p of this.particles) {
   350→   350→      if (p.invMass > 0) {
   351→   351→        p.prev = vSub(p.pos, vScale(vel, dt));
   352→   352→      }
   353→   353→    }
   354→   354→  }

   356→   356→  totalMass(): number {
   357→   357→    return this.particles.reduce((acc, p) => acc + p.mass, 0);

   360→   360→  centerOfMass(): V3 {
   361→   361→    let m = 0;
   362→   362→    const c = v3();
   363→   363→    for (const p of this.particles) {
   364→   364→      c.x += p.pos.x * p.mass;
   365→   365→      c.y += p.pos.y * p.mass;
   366→   366→      c.z += p.pos.z * p.mass;
   367→   367→      m += p.mass;
   368→   368→    }
   369→   369→    return vScale(c, 1 / (m || 1));
   370→   370→  }

   372→   372→  step(dt: number): StepResult {
   373→   373→    const K = this.env.iterations;
   374→   374→    const forces = new Map<number, V3>();
   375→   375→    const anchorLoads: Partial<Record<LimbId, number>> = {};
   376→   376→    const anchorStretch: Partial<Record<LimbId, number>> = {};
   377→   377→    let nanError = false;
   378→   378→    let outOfBounds = false;
   379→   379→    let maxSpeed = 0;
   380→   380→
   381→   381→    // 1) Fuerza externa de objetivos (resorte amortiguado con tope). Los anclajes NO son
   382→   382→    // resortes: se fijan como pin rigido en la fase de restricciones (ver 4b).
   383→   383→    for (const target of this.targets.values()) {
   384→   384→      const p = this.particles[target.particle];
   385→   385→      const err = vSub(target.target, p.pos);
   386→   386→      const vel = vScale(vSub(p.pos, p.prev), 1 / dt);
   387→   387→      const spring = vAdd(vScale(err, this.env.followStiffness), vScale(vel, -this.env.anchorDamping * 0.5));
   388→   388→      const applied = clampVecMag(spring, target.maxForce);
   389→   389→      forces.set(target.particle, vAdd(forces.get(target.particle) ?? v3(), applied));
   390→   390→    }
   391→   391→
   392→   392→    // 1b) Musculos activos (ADR-007): cuadriceps, step-up, traccion de brazos y tono
   393→   393→    // postural. Sin anclajes el cuerpo es ragdoll puro (caida limpia).
   394→   394→    if (this.env.muscles.musclesOn && this.topology !== null && this.anchors.size > 0) {
   395→   395→      this.solveMuscles(dt, forces);
   396→   396→    }
   397→   397→    this.climbDrive = Math.max(0, this.climbDrive - dt / this.env.muscles.driveSeconds);
   398→   398→
   399→   399→    // 2) Integracion Verlet con paso fijo (con techo de velocidad: los musculos no
   400→   400→    // pueden lanzar el ragdoll a velocidades no humanas; el ERROR GUARD queda para
   401→   401→    // anomalias reales del solver).
   402→   402→    const g = this.env.gravity;
   403→   403→    const vCap = this.env.muscles.vMaxSpeed * dt;
   404→   404→    for (let i = 0; i < this.particles.length; i++) {
   405→   405→      const p = this.particles[i];
   406→   406→      if (p.invMass <= 0) continue;
   407→   407→      const f = forces.get(i);
   408→   408→      const a = f ? vAdd(g, vScale(f, p.invMass)) : g;
   409→   409→      const vel = vScale(vSub(p.pos, p.prev), this.env.damping);
   410→   410→      p.prev = vClone(p.pos);
   411→   411→      let next = vAdd(vAdd(p.pos, vel), vScale(a, dt * dt));
   412→   412→      const newVel = vSub(next, p.prev);
   413→   413→      const newSpeed = vLen(newVel);
   414→   414→      if (newSpeed > vCap) {
   415→   415→        next = vAdd(p.prev, vScale(newVel, vCap / newSpeed));
   416→   416→      }
   417→   417→      p.pos = next;
   418→   418→      // Metrica sobre la velocidad YA acotada (la previa incluye correcciones de pin).
   419→   419→      const measured = Math.min(newSpeed, vCap) / dt;
   420→   420→      if (measured > maxSpeed) maxSpeed = measured;
   421→   421→    }
   422→   422→
   423→   423→    // 3) Tremor de esfuerzo (seccion 9.5): ruido proporcional al estado de tension.
   424→   424→    for (const limb of LIMB_IDS) {
   425→   425→      const t = this.tremor[limb];
   426→   426→      if (t <= 0) continue;
   427→   427→      const idx = this.limbParticle.get(limb);
   428→   428→      if (idx === undefined) continue;
   429→   429→      const p = this.particles[idx];
   430→   430→      const amp = this.env.tremorAmplitude * t;
   431→   431→      p.pos = vAdd(p.pos, v3((Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp, (Math.random() * 2 - 1) * amp));
   432→   432→    }
   433→   433→
   434→   434→    // 4) Proyeccion de restricciones.
   435→   435→    // 4a) Acumulador de reaccion por anclaje: suma de correcciones del pin este paso.
   436→   436→    const pinCorrection = new Map<LimbId, V3>();
   437→   437→    for (let iter = 0; iter < K; iter++) {
   438→   438→      this.solveSticks();
   439→   439→      this.solveAngles();
   440→   440→      // 4b) Anclajes: pin rigido con tope de fuerza. La carga requerida se mide como la
   441→   441→      // fuerza necesaria para mantener el pin (F = m x correccion / dt^2), un estimador
   442→   442→      // consistente con el impulso aplicado (9.2). En sobrecarga el pin se satura y la
   443→   443→      // extremidad resbala visiblemente de la presa.
   444→   444→      for (const anchor of this.anchors.values()) {
   445→   445→        const p = this.particles[anchor.particle];
   446→   446→        const err = vSub(anchor.point, p.pos);
   447→   447→        const errLen = vLen(err);
   448→   448→        if (errLen < 1e-9) continue;
   449→   449→        const maxCorr = (anchor.maxForce * dt * dt) / p.mass;
   450→   450→        const scaled = errLen > maxCorr ? vScale(err, maxCorr / errLen) : err;
   451→   451→        p.pos = vAdd(p.pos, scaled);
   452→   452→        pinCorrection.set(anchor.limb, vAdd(pinCorrection.get(anchor.limb) ?? v3(), scaled));
   453→   453→      }
   454→   454→      this.solveCollisions();
   455→   455→    }
   456→   456→    // NOTA: la correccion del pin genera momento "fantasma" en Verlet; ya no se
   457→   457→    // neutraliza en prev (cambiaria la calibracion del estimador de carga). El techo
   458→   458→    // de velocidad de la integracion (vMaxSpeed) contiene ese momento residual.
   459→   459→    // Carga suavizada: impulso total demandado por el pin, en newtons, con ganancia
   460→   460→    // calibrada del estimador documentado (las re-proyecciones por iteracion sobre-miden).
   461→   461→    for (const anchor of this.anchors.values()) {
   462→   462→      const net = pinCorrection.get(anchor.limb) ?? v3();
   463→   463→      const measured = this.particles[anchor.particle].mass * (vLen(net) / (dt * dt)) * this.env.loadEstimatorGain;
   464→   464→      anchor.loadN = anchor.loadN === 0 ? measured : anchor.loadN + (measured - anchor.loadN) * 0.3;
   465→   465→      anchor.stretch = vLen(vSub(anchor.point, this.particles[anchor.particle].pos));
   466→   466→      anchorLoads[anchor.limb] = anchor.loadN;
   467→   467→      anchorStretch[anchor.limb] = anchor.stretch;
   468→   468→    }
   469→   469→
   470→   470→    // 5) Guardas: NaN, energia y limites del mundo.
   471→   471→    for (const p of this.particles) {
   472→   472→      if (!isFiniteVec(p.pos) || !isFiniteVec(p.prev)) {
   473→   473→        nanError = true;
   474→   474→        break;
   475→   475→      }
   476→   476→      if (p.pos.x < this.env.bounds.minX || p.pos.x > this.env.bounds.maxX || p.pos.y > this.env.bounds.maxY || p.pos.z < this.env.bounds.minZ || p.pos.z > this.env.bounds.maxZ) {
   477→   477→        outOfBounds = true;
   478→   478→      }
   479→   479→    }
   480→   480→    if (maxSpeed > 18) {
   481→   481→      // Energia anomala: se registra, la capa de juego decide (ERROR GUARD).
   482→   482→      maxSpeed = clamp(maxSpeed, 0, 30);
   483→   483→    }
   484→   484→
   485→   485→    // 6) Lecturas derivadas.
   486→   486→    const floorContacts: { limb: LimbId; point: V3 }[] = [];
   487→   487→    for (const limb of LIMB_IDS) {
   488→   488→      const idx = this.limbParticle.get(limb);
   489→   489→      if (idx === undefined) continue;
   490→   490→      const p = this.particles[idx];
   491→   491→      if (p.pos.y <= this.env.floorY + 0.045) {
   492→   492→        floorContacts.push({ limb, point: v3(p.pos.x, this.env.floorY, p.pos.z) });
   493→   493→      }
   494→   494→    }
   495→   495→
   496→   496→    return {
   497→   497→      nanError,
   498→   498→      energySpike: maxSpeed > 18,
   499→   499→      outOfBounds,
   500→   500→      com: this.centerOfMass(),
   501→   501→      anchorLoads,
   502→   502→      anchorStretch,
   503→   503→      floorContacts,
   504→   504→      jointAngles: this.angles.map((a) => ({
   505→   505→        id: a.id,
   506→   506→        angle: a.currentAngle,
   507→   507→        utilization: jointUtilization(a)
   508→   508→      })),
   509→   509→      maxSpeed
   510→   510→    };
   511→   511→  }
   512→   512→
   513→   513→  /**
   514→   514→   * Fase muscular (ADR-007). Todas las fuerzas se aplican sobre particulas de masa alta
   515→   515→   * (pelvis ~11 kg, pecho ~14 kg) con topes calibrados para un desplazamiento por paso
   516→   516→   * estable; los pines miden la carga resultante y la biomecanica la traduce en fatiga.
   517→   517→   */
   518→   518→  private solveMuscles(dt: number, forces: Map<number, V3>): void {
   519→   519→    const M = this.env.muscles;
   520→   520→    const T = this.topology!;
   521→   521→    const pelvis = this.particles[T.pelvis];
   522→   522→    const chest = this.particles[T.chest];
   523→   523→    const head = this.particles[T.head];
   524→   524→    const addForce = (idx: number, fx: number, fy: number, fz: number): void => {
   525→   525→      const cur = forces.get(idx);
   526→   526→      forces.set(idx, cur ? v3(cur.x + fx, cur.y + fy, cur.z + fz) : v3(fx, fy, fz));
   527→   527→    };
   528→   528→
   529→   529→    // Rampa de activacion por anclaje (evita el tiron al anclar).
   530→   530→    for (const anchor of this.anchors.values()) {
   531→   531→      const rate = anchor.limb.startsWith("foot") ? 1 / 0.35 : 1 / 0.2;
   532→   532→      anchor.activation = Math.min(1, anchor.activation + dt * rate);
   533→   533→    }
   534→   534→
   535→   535→    // Pre-pasada: hay pierna sosteniendo (cuadriceps activo)? Con apoyo de piernas
   536→   536→    // los brazos NO tiran (el pull-up esta para colgar/colocarse, no para despegar
   537→   537→    // al cuerpo que ya esta de pie: evita la suma explosiva de fuerzas).
   538→   538→    let feetSupporting = false;
   539→   539→    // Peso que cargan los brazos ahora mismo (N): modula el balance (un pie que
   540→   540→    // solo descansa no debe volcar el COM hacia si; uno que sostiene, si).
   541→   541→    let handLoadN = 0;
   542→   542→    for (const anchor of this.anchors.values()) {
   543→   543→      if (anchor.limb.startsWith("hand")) handLoadN += Math.max(0, anchor.loadN);
   544→   544→    }
   545→   545→    // Arbitracion de pies: el STEP-UP lo hace solo el pie MAS ALTO (transferencia de
   546→   546→    // peso al escalon, como en escalada real); el balance solo desde el pie activo,
   547→   547→    // para que un apoyo bajo no pelee contra el pie que esta montando.
   548→   548→    let stepUpFoot: LimbId | null = null;
   549→   549→    let stepUpHoldY = -Infinity;
   550→   550→    for (const anchor of this.anchors.values()) {
   551→   551→      if (!anchor.limb.startsWith("foot")) continue;
   552→   552→      const root = this.particles[T.rootOf.get(anchor.limb)!];
   553→   553→      const dyr = root.pos.y - anchor.point.y;
   554→   554→      const dr = Math.hypot(root.pos.x - anchor.point.x, dyr, root.pos.z - anchor.point.z) || 1e-6;
   555→   555→      if (dyr / dr > 0.05) {
   556→   556→        feetSupporting = true;
   557→   557→      }
   558→   558→      if (dyr / dr < 0.02 && anchor.point.y > stepUpHoldY) {
   559→   559→        stepUpHoldY = anchor.point.y;
   560→   560→        stepUpFoot = anchor.limb;
   561→   561→      }
   562→   562→    }
   563→   563→
   564→   564→    let feetAnchored = 0;
   565→   565→    for (const anchor of this.anchors.values()) {
   566→   566→      const isFoot = anchor.limb.startsWith("foot");
   567→   567→      if (isFoot) feetAnchored++;
   568→   568→      const rootIdx = T.rootOf.get(anchor.limb);
   569→   569→      if (rootIdx === undefined) continue;
   570→   570→      const root = this.particles[rootIdx];
   571→   571→      const dx = root.pos.x - anchor.point.x;
   572→   572→      const dy = root.pos.y - anchor.point.y;
   573→   573→      const dz = root.pos.z - anchor.point.z;
   574→   574→      const dist = Math.hypot(dx, dy, dz) || 1e-6;
   575→   575→
   576→   576→      if (isFoot) {
   577→   577→        // CUADRICEPS (apoyo): el pie por DEBAJO de la cadera empuja la pelvis lejos del
   578→   578→        // pie (extension de pierna = ponerse de pie). Proyectado al plano de la pared:
   579→   579→        // la pierna empuja "hacia arriba", no separa del muro.
   580→   580→        if (dy / dist > 0.05) {
   581→   581→          // Servo de pierna: empuja mientras la pierna este flexionada y ceda al
   582→   582→          // acercarse a la extension natural (0.9·cadena = rodilla casi recta).
   583→   583→          // Asi la postura de pie es un equilibrio estable, no un empuje sin tope.
   584→   584→          const targetDist = T.legChain * 0.9;
   585→   585→          const f = clamp((targetDist - dist) * M.quadServoK, 0, M.quadPushMax) * anchor.activation;
   586→   586→          const ul = Math.hypot(dx, dy) || 1e-6;
   587→   587→          // Direccion axial con sesgo vertical (reaccion del suelo ~vertical al de pie).
   588→   588→          const bx = (dx / ul) * (1 - M.quadUpBias);
   589→   589→          const by = (dy / ul) * (1 - M.quadUpBias) + M.quadUpBias;
   590→   590→          const bl = Math.hypot(bx, by) || 1e-6;
   591→   591→          addForce(T.pelvis, (bx / bl) * f, (by / bl) * f, 0);
   592→   592→          // BALANCE: llevar el COM sobre el apoyo (feedback tipo SIMBICON); sin el,
   593→   593→          // un apoyo lateral nunca sostiene de verdad. Solo desde el pie que esta
   594→   594→          // montando (si hay step-up) o desde los apoyos si nadie monta. Escalado por
   595→   595→          // la carga REAL de la pierna frente a los brazos: un pie que descansa no
   596→   596→          // vuelca el cuerpo (evita sobrecargar el brazo contrario en cascada).
   597→   597→          if (stepUpFoot === null || stepUpFoot === anchor.limb) {
   598→   598→            const weightShare = clamp(f / (f + handLoadN + 60), 0.15, 1);
   599→   599→            const fb = clamp((anchor.point.x - pelvis.pos.x) * 400, -M.balanceMax, M.balanceMax) * anchor.activation * weightShare;
   600→   600→            addForce(T.pelvis, fb, 0, 0);
   601→   601→            // 3a ley de Newton: parte de la reaccion entra por el pie al pin, de modo que
   602→   602→            // el apoyo MEDIDO carga de verdad al estar de pie (fatiga de piernas real).
   603→   603→            addForce(anchor.particle, -(bx / bl) * f * 0.5 - fb * 0.4, -(by / bl) * f * 0.5, 0);
   604→   604→          } else {
   605→   605→            addForce(anchor.particle, -(bx / bl) * f * 0.5, -(by / bl) * f * 0.5, 0);
   606→   606→          }
   607→   607→        }
   608→   608→        // PSOAS/STEP-UP: el pie por ENCIMA de la cadera sube el cuerpo hacia la presa
   609→   609→        // (montar el escalon). Es lo que convierte "apoyar un pie" en ascender. La
   610→   610→        // componente vertical del vector hacia la presa regula el empuje solo: al
   611→   611→        // acercarse la pelvis al nivel del pie la fuerza se hace horizontal (COM sobre
   612→   612→        // el apoyo) y el cuadriceps toma el relevo por encima. Solo el pie MAS ALTO
   613→   613→        // monta (arbitracion: transferencia de peso al escalon).
   614→   614→        if (stepUpFoot === anchor.limb && dy / dist < 0.02) {
   615→   615→          const boost = 0.9 + 0.3 * this.climbDrive;
   616→   616→          // Esvaimiento por proximidad: al acercarse la pelvis al nivel del pie la
   617→   617→          // fuerza cede (sin sobregiro mas alla de la postura natural de pie).
   618→   618→          const taper = clamp(-dy / 0.45, 0.55, 1);
   619→   619→          const f = M.stepUpMax * anchor.activation * boost * taper;
   620→   620→          const ux = -dx / dist;
   621→   621→          const uy = -dy / dist;
   622→   622→          const uz = -dz / dist;
   623→   623→          addForce(T.pelvis, ux * f, uy * f, uz * f);
   624→   624→          addForce(anchor.particle, -ux * f * 0.35, -uy * f * 0.35, -uz * f * 0.35);
   625→   625→        }
   626→   626→      } else {
   627→   627→        // TRACCION DE BRAZOS (impulso de subida del jugador): mano por encima del
   628→   628→        // hombro y brazo ESTIRADO (compuerta de holgura) tira del pecho hacia el
   629→   629→        // anclaje. Con piernas de apoyo es el "subirse" que sube los hombros; el
   630→   630→        // servo de pierna limita la postura y evita el pliegue.
   631→   631→        if (this.climbDrive > 0.02 && dy / dist < -0.1) {
   632→   632→          // Holgura medida desde la RAIZ (hombro): en postura plegada el pecho puede
   633→   633→          // estar alto con el hombro colgando bajo y el brazo estirado; el pull debe
   634→   634→          // evaluar el brazo real, no la posicion del pecho.
   635→   635→          const rdx = anchor.point.x - root.pos.x;
   636→   636→          const rdy = anchor.point.y - root.pos.y;
   637→   637→          const rdz = anchor.point.z - root.pos.z;
   638→   638→          const rd = Math.hypot(rdx, rdy, rdz) || 1e-6;
   639→   639→          const slack = rd - T.armChain * 0.9;
   640→   640→          const cdx = anchor.point.x - chest.pos.x;
   641→   641→          const cdy = anchor.point.y - chest.pos.y;
   642→   642→          const cdz = anchor.point.z - chest.pos.z;
   643→   643→          const cd = Math.hypot(cdx, cdy, cdz) || 1e-6;
   644→   644→          if (slack > 0) {
   645→   645→            const gate = clamp(slack / (T.armChain * 0.2), 0.4, 1);
   646→   646→            // Durante un step-up la pierna manda: los brazos asisten al 60%.
   647→   647→            const assist = stepUpFoot === null ? 1 : 0.6;
   648→   648→            // Margen de agarre: no se puede tirar mas fuerte que el agarre restante
   649→   649→            // (un brazo al limite no tira: evita saturar el pin y la sobrecarga).
   650→   650→            const headroom = clamp(1 - anchor.loadN / anchor.maxForce, 0, 1);
   651→   651→            const f = M.armPullMax * anchor.activation * this.climbDrive * gate * assist * headroom;
   652→   652→            addForce(T.chest, (cdx / cd) * f, (cdy / cd) * f, (cdz / cd) * f);
   653→   653→            addForce(rootIdx, (cdx / cd) * f * 0.35, (cdy / cd) * f * 0.35, (cdz / cd) * f * 0.35);
   654→   654→          }
   655→   655→        }
   656→   656→      }
   657→   657→    }
   658→   658→
   659→   659→    // TONO POSTURAL: pecho sobre pelvis y cabeza sobre pecho (anti-muneco de trapo).
   660→   660→    // Escala con el numero de apoyos: cuerpo libre = ragdoll puro.
   661→   661→    const toneScale = Math.min(1, this.anchors.size / 2);
   662→   662→    const tex = pelvis.pos.x - chest.pos.x;
   663→   663→    const tey = pelvis.pos.y + T.spineLen * 0.97 - chest.pos.y;
   664→   664→    const tez = pelvis.pos.z - chest.pos.z;
   665→   665→    const teLen = Math.hypot(tex, tey, tez) || 1e-6;
   666→   666→    const coreF = Math.min(M.coreMax * toneScale, teLen * 400);
   667→   667→    addForce(T.chest, (tex / teLen) * coreF, (tey / teLen) * coreF, (tez / teLen) * coreF);
   668→   668→    const nex = chest.pos.x - head.pos.x;
   669→   669→    const ney = chest.pos.y + 0.2 - head.pos.y;
   670→   670→    const nez = chest.pos.z - head.pos.z;
   671→   671→    const neLen = Math.hypot(nex, ney, nez) || 1e-6;
   672→   672→    const neckF = Math.min(M.neckMax * toneScale, neLen * 300);
   673→   673→    addForce(T.head, (nex / neLen) * neckF, (ney / neLen) * neckF, (nez / neLen) * neckF);
   674→   674→
   675→   675→    // PEGADO A LA PARED con cualquier anclaje: la colision del cuerpo con la cara del
   676→   676→    // muro impide colgar en la vertical exacta de los pines (las presas sobresalen), y
   677→   677→    // sin fuerza de centrado en z el cuerpo queda 0.1-0.2 m por detras, offset que
   678→   678→    // devora el alcance util de brazos y piernas en CADA movimiento. El escalador real
   679→   679→    // va pegado: cadera y pecho se presionan hacia el plano de presas.
   680→   680→    if (this.anchors.size > 0) {
   681→   681→      const zTarget = this.env.wallFrontZ - 0.26;
   682→   682→      const fzPelvis = clamp((zTarget - pelvis.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   683→   683→      const fzChest = clamp((zTarget - chest.pos.z) * 400, -M.wallLeanMax, M.wallLeanMax);
   684→   684→      addForce(T.pelvis, 0, 0, fzPelvis);
   685→   685→      addForce(T.chest, 0, 0, fzChest);
   686→   686→    }
   687→   687→
   688→   688→    // Tope de fuerza muscular por particula axial: la suma de musculos nunca puede
   689→   689→    // superar el techo (estabilidad de Verlet con las masas de pelvis/pecho).
   690→   690→    const cap = (idx: number, capN: number): void => {
   691→   691→      const f = forces.get(idx);
   692→   692→      if (!f) return;
   693→   693→      const l = vLen(f);
   694→   694→      if (l > capN) forces.set(idx, vScale(f, capN / l));
   695→   695→    };
   696→   696→    cap(T.pelvis, M.muscleForceCapPelvis);
   697→   697→    cap(T.chest, M.muscleForceCapChest);
   698→   698→
   699→   699→    // Instantanea para diagnostico (solo particulas axiales y raices).
   700→   700→    this.lastMuscleForces.clear();
   701→   701→    for (const idx of [T.pelvis, T.chest, T.head]) {
   702→   702→      const f = forces.get(idx);
   703→   703→      if (f) this.lastMuscleForces.set(idx, f);
   704→   704→    }
   705→   705→  }
   706→   706→
   707→   707→  private solveSticks(): void {
   708→   708→    for (const s of this.sticks) {
   709→   709→      const pa = this.particles[s.a];
   710→   710→      const pb = this.particles[s.b];
   711→   711→      const w = pa.invMass + pb.invMass;
   712→   712→      if (w <= 0) continue;
   713→   713→      const d = vSub(pb.pos, pa.pos);
   714→   714→      const len = vLen(d);
   715→   715→      if (len < 1e-9) continue;
   716→   716→      const diff = (len - s.rest) / len;
   717→   717→      const k = s.stiffness;
   718→   718→      pa.pos = vAdd(pa.pos, vScale(d, (diff * pa.invMass * k) / w));
   719→   719→      pb.pos = vSub(pb.pos, vScale(d, (diff * pb.invMass * k) / w));
   720→   720→    }
   721→   721→  }
   722→   722→
   723→   723→  private solveAngles(): void {
   724→   724→    for (const c of this.angles) {
   725→   725→      const pj = this.particles[c.j];
   726→   726→      const pa = this.particles[c.a];
   727→   727→      const pb = this.particles[c.b];
   728→   728→      const va = vSub(pa.pos, pj.pos);
   729→   729→      const vb = vSub(pb.pos, pj.pos);
   730→   730→      const la = vLen(va);
   731→   731→      const lb = vLen(vb);
   732→   732→      if (la < 1e-9 || lb < 1e-9) {
   733→   733→        c.currentAngle = Number.isFinite(la) && Number.isFinite(lb) ? c.currentAngle : c.currentAngle;
   734→   734→        continue;
   735→   735→      }
   736→   736→      const ang = angleBetween(va, vb);
   737→   737→      c.currentAngle = ang;
   738→   738→      const target = clamp(ang, c.min, c.max);
   739→   739→      if (Math.abs(target - ang) < 1e-5) continue;
   740→   740→      const axis = vCross(va, vb);
   741→   741→      const axisLen = vLen(axis);
   742→   742→      if (axisLen < 1e-9) continue;
   743→   743→      const rot = rotateAroundAxis(vb, vScale(axis, 1 / axisLen), target - ang);
   744→   744→      const delta = vSub(rot, vb);
   745→   745→      const w = pa.invMass + pb.invMass;
   746→   746→      if (w <= 0) continue;
   747→   747→      const k = c.stiffness;
   748→   748→      pb.pos = vAdd(pb.pos, vScale(delta, (pb.invMass * k) / w));
   749→   749→      pa.pos = vSub(pa.pos, vScale(delta, (pa.invMass * k) / w));
   750→   750→    }
   751→   751→  }
   752→   752→
   753→   753→  private solveCollisions(): void {
   754→   754→    for (const p of this.particles) {
   755→   755→      // Suelo: los pies y el resto del cuerpo.
   756→   756→      if (p.collideFloor && p.pos.y < this.env.floorY + p.radius) {
   757→   757→        p.pos.y = this.env.floorY + p.radius;
   758→   758→        // Friccion simple: reduce la velocidad tangencial implicita.
   759→   759→        p.prev.x += (p.pos.x - p.prev.x) * 0.25;
   760→   760→        p.prev.z += (p.pos.z - p.prev.z) * 0.25;
   761→   761→      }
   762→   762→      // Pared: los segmentos del cuerpo no atraviesan el plano frontal.

   771→   771→function clampVecMag(v: V3, max: number): V3 {
   772→   772→  const l = vLen(v);
   773→   773→  if (l <= max || l < 1e-12) return v;

   777→   777→function jointUtilization(c: AngleConstraint): number {
   778→   778→  const mid = (c.min + c.max) / 2;
   779→   779→  const half = (c.max - c.min) / 2;
   780→   780→  if (half <= 1e-9) return 0;
   781→   781→  return clamp(Math.abs(c.currentAngle - mid) / half, 0, 1);
   782→   782→}
