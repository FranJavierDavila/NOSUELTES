   287→  async loadChallenge(id: string): Promise<void> {
   288→    this.stopTutorial(false);
   289→    this.challenge = this.repo.getChallenge(id);
   290→    this.character = this.repo.getCharacter(this.challenge.characterId);
   291→    this.profile = this.repo.getBodyProfile(this.character.bodyProfileId);
   292→    this.wall = this.repo.getWall(this.challenge.wallId);
   293→    this.holdPosById = new Map(this.wall.holds.map((h) => [h.id, h.position] as const));
   294→    this.bio = new BiomechanicsSystem(this.character, this.profile);
   295→    this.evaluator = new ChallengeEvaluator(this.challenge);
   296→
   297→    const strengthFactor = lerp(this.profile.biomech.strengthFactorMin, this.profile.biomech.strengthFactorMax, this.character.virtues.strength);
   298→    this.anchorMaxForce = this.character.physical.baseGripForce * strengthFactor;
   299→    this.followForce = this.character.physical.baseGripForce * this.character.control.gripAssist * this.character.control.targetFollowStrength;
   300→
   301→    this.solver = new PbdSolver({
   302→      gravity: v3(...this.profile.sim.gravity),
   303→      damping: this.profile.sim.damping,
   304→      iterations: this.profile.sim.iterations,
   305→      floorY: 0,
   306→      wallFrontZ: this.wall.transform.position[2] - this.wall.dimensions.depth / 2,
   307→      tremorAmplitude: this.profile.sim.tremorAmplitude,
   308→      loadEstimatorGain: this.profile.biomech.loadEstimatorGain
   309→    });
   310→
   311→    this.scene.buildWorld(this.wall);
   312→    this.hud.setChallenge(this.challenge, CHALLENGE_META[id]?.label ?? CHALLENGE_LABELS[this.challenge.type]);
   313→    this.buildInteraction();
   314→    this.spawn();
   315→  }

   317→  /**
   318→   * Recreacion del estado inicial (10.2): libera constraints, destruye y recrea el
   319→   * ragdoll, restaura fatiga/timers/progreso y espera resetSettleSeconds.

   322→  private spawn(): void {
   323→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   324→    this.interaction?.cancelDrag();
   325→    this.interaction?.cancelAssistAll();
   326→    this.keyboardLimb = null;
   327→    this.grabOffset = null;
   328→    this.scene.clearHoldHighlights();
   329→    for (const hold of this.wall.holds) this.scene.setHoldOccupied(hold.id, false);
   330→
   331→    this.scene.despawnRagdollVisuals();
   332→    this.rig = spawnRagdoll(this.solver, this.profile, this.character, this.wall, this.challenge, holdById, (limb) => this.anchorMaxForceFor(limb));
   333→    this.scene.spawnRagdollVisuals(this.rig);
   334→
   335→    this.bio.reset();
   336→    this.evaluator?.reset();
   337→    this.gameTime = 0;
   338→    this.accumulator = 0;
   339→    this.guardActive = false;
   340→    this.lastSnapshot = { stability: 0, com: { x: 0, y: 0, z: 0 }, supportCount: 0 };
   341→
   342→    // El combo de flujo se rompe con cada respawn (intento nuevo); el conteo de
   343→    // caidas persiste para la puntuacion del desafio cargado.
   344→    this.flowCount = 0;
   345→    this.flowBonus = 0;
   346→    this.peakFatigue = 0;
   347→    this.timeScale = 1;
   348→    this.slowmoLeft = 0;
   349→    this.fallLandedFx = false;
   350→    this.hud.setFlow(0);
   351→    this.lastWorst = null;
   352→    this.limbOverload.clear();
   353→    this.overloadPressure = 0;
   354→
   355→    // Marcar presas ocupadas por los anclajes iniciales creados en spawnRagdoll.
   356→    for (const req of this.challenge.start.requiredAnchors) {
   357→      this.scene.setHoldOccupied(req.holdId, true);
   358→    }
   359→
   360→    this.settleTimer = this.profile.sim.resetSettleSeconds;
   361→    this.autoStartPending = this.challenge.start.autoStart;
   362→    this.fallReleaseQueue = [];
   363→    this.fallAutoReset = -1;
   364→    this.machine.transition("RESETTING");
   365→    // Despues de la transicion: en FALLEN el objetivo seria el tono flacido de la
   366→    // caida; al re-spawnear el escalador vuelve a estar vivo (tono calibrado).
   367→    this.applyTuneToSolver();
   368→    this.events.emit({ type: "Message", text: `Prueba ${CHALLENGE_META[this.challenge.id]?.label ?? CHALLENGE_LABELS[this.challenge.type]} cargada`, tone: "info" });
   369→  }

   391→    this.events.emit({ type: "Message", text: `Balance ${name}: tono ${this.tune.postureTone.toFixed(2)}, límites ${this.tune.jointLimitStiffness.toFixed(2)}, pliegue ${this.tune.foldStrength.toFixed(2)}, estiramiento ×${this.tune.stretchTolerance.toFixed(2)}`, tone: "info" });
   392→  }

   394→  private loadTune(): void {
   395→    try {
   396→      const preset = window.localStorage.getItem(GameApp.PRESET_KEY) as BalancePresetName | null;
   397→      if (preset && preset in PRESET_TUNES) this.presetName = preset;
   398→    } catch {
   399→      this.presetName = null;
   400→    }
   401→    if (this.presetName) {
   402→      this.tune = { ...PRESET_TUNES[this.presetName] };
   403→      return;
   404→    }
   405→    try {
   406→      const raw = window.localStorage.getItem(GameApp.TUNE_KEY);
   407→      if (raw) {
   408→        const t = JSON.parse(raw) as Partial<BalanceTuning>;
   409→        this.tune = {
   410→          jointLimitStiffness: clamp01(t.jointLimitStiffness ?? 1.0),
   411→          foldStrength: clamp01(t.foldStrength ?? 0.85),
   412→          postureTone: clamp01(t.postureTone ?? 0.75),
   413→          toneFloor: clamp01(t.toneFloor ?? 0.12),
   414→          stretchTolerance: clampRange(t.stretchTolerance ?? 1.1, 1.02, 1.35),
   415→          twistLimitRad: clampRange(t.twistLimitRad ?? 0.45, 0, 1.2)
   416→        };
   417→      }
   418→    } catch {
   419→      /* sin persistencia: defaults */
   420→    }
   421→  }

   436→      toneFloor: clamp01(patch.toneFloor ?? this.tune.toneFloor),
   437→      stretchTolerance: clampRange(patch.stretchTolerance ?? this.tune.stretchTolerance, 1.02, 1.35),
   438→      twistLimitRad: clampRange(patch.twistLimitRad ?? this.tune.twistLimitRad, 0, 1.2)
   439→    };
   440→    // Las reglas de postura imposible siguen al tono: preset realista = estricto.
   441→    this.postureRules = postureRulesForTone(this.tune.postureTone);
   442→    // Un ajuste manual (sliders F3 o __NS.setTune) deja de corresponder al preset.
   443→    this.presetName = null;
   444→    try {
   445→      window.localStorage.removeItem(GameApp.PRESET_KEY);
   446→    } catch {
   447→      /* sin persistencia */
   448→    }
   449→    this.hud?.setPresetActive(null);
   450→    this.applyTuneToSolver();
   451→    this.saveTune();
   452→    return { ...this.tune };
   453→  }
   454→
   455→  private applyTuneToSolver(): void {
   456→    if (!this.solver) return;
   457→    this.solver.env.balance.jointLimitStiffness = this.tune.jointLimitStiffness;
   458→    this.solver.env.balance.foldStrength = this.tune.foldStrength;
   459→    this.solver.env.balance.postureTone = this.tune.postureTone;
   460→    this.solver.env.balance.toneFloor = this.tune.toneFloor;
   461→    this.solver.env.balance.stretchTolerance = this.tune.stretchTolerance;
   462→    this.solver.env.balance.twistLimitRad = this.tune.twistLimitRad;
   463→    // En FALLEN el tono objetivo sigue siendo el flacido; en el resto, el calibrado.
   464→    this.solver.setToneTarget(this.machine.state === "FALLEN" ? this.tune.toneFloor : this.tune.postureTone);
   465→  }

   467→  requestReset(): void {
   468→    if (this.machine.state === "LOADING" || this.machine.state === "RESETTING") return;
   469→    this.events.emit({ type: "ResetStarted" });
   470→    this.spawn();
   471→  }

   481→      const raw = window.localStorage.getItem(GameApp.bestKey(id));

   492→      window.localStorage.setItem(GameApp.bestKey(id), JSON.stringify(rec));

   501→  private buildInteraction(): void {
   502→    const holdById = new Map(this.wall.holds.map((h) => [h.id, h]));
   503→    const world: InteractionWorld = {
   504→      holdById,
   505→      challenge: this.challenge,
   506→      character: this.character,
   507→      profile: this.profile,
   508→      wall: this.wall,
   509→      anchorHoldOf: (limb) => this.solver.getAnchor(limb)?.holdId,
   510→      holdOccupancy: (holdId) => {
   511→        let n = 0;
   512→        for (const limb of LIMB_IDS) {
   513→          if (this.solver.getAnchor(limb)?.holdId === holdId) n++;
   514→          if (this.interaction?.assistHoldOf(limb) === holdId) n++;
   515→        }
   516→        return n;
   517→      },
   518→      chainRootOf: (limb) => {
   519→        const idx = limb === "hand_l" ? P.shL : limb === "hand_r" ? P.shR : limb === "foot_l" ? P.hipL : P.hipR;
   520→        const p = this.solver.getParticle(idx);
   521→        return v3(p.pos.x, p.pos.y, p.pos.z);
   522→      },
   523→      limbPositionOf: (limb) => this.solver.positionOfLimb(limb),
   524→      chainLengthOf: (limb) => {
   525→        if (!this.rig) return 0.6;
   526→        const base = limb.startsWith("hand") ? this.rig.armChain : this.rig.legChain;
   527→        return base * this.character.physical.reachScale;
   528→      },
   529→      postureRules: () => this.postureRules
   530→    };
   531→    this.interaction = new InteractionController(world, {
   532→      onSelectChanged: () => undefined,
   533→      onAccept: (limb, holdId) => {
   534→        const existing = this.solver.getAnchor(limb);
   535→        if (existing) {
   536→          this.solver.releaseAnchor(limb);
   537→          this.events.emit({ type: "GripReleased", limb, holdId: existing.holdId, reason: "PLAYER_RELEASE" });
   538→        }
   539→        const hold = holdById.get(holdId)!;
   540→        this.solver.setTarget(limb, v3(hold.position[0], hold.position[1], hold.position[2]), this.followForce);
   541→        this.refreshHoldOccupancy();
   542→      },
   543→      onReject: (limb, holdId, reason) => {
   544→        this.solver.clearTarget(limb);
   545→        this.events.emit({ type: "HoldRejected", limb, holdId, reason });
   546→        if (holdId) {
   547→          this.scene.setHoldInvalid(holdId, true);
   548→          window.setTimeout(() => this.scene.setHoldInvalid(holdId, false), 450);
   549→        }
   550→      },
   551→      onArrival: (limb, holdId) => {
   552→        this.solver.clearTarget(limb);
   553→        const hold = holdById.get(holdId)!;
   554→        this.solver.createAnchor(limb, holdId, v3(hold.position[0], hold.position[1], hold.position[2]), this.anchorMaxForceFor(limb));
   555→        // ADR-007: cada apoyo consolidado reactiva el impulso de subida (el cuerpo
   556→        // se reorganiza hacia la nueva postura: step-up del pie, pull de brazos).
   557→        this.solver.pulseClimbDrive();
   558→        this.events.emit({ type: "HoldAcquired", limb, holdId });
   559→        this.refreshHoldOccupancy();
   560→      }
   561→    });
   562→  }
   563→
   564→  private refreshHoldOccupancy(): void {
   565→    const anchored = new Set<string>();
   566→    for (const limb of LIMB_IDS) {
   567→      const a = this.solver.getAnchor(limb);
   568→      if (a) anchored.add(a.holdId);
   569→    }
   570→    for (const hold of this.wall.holds) {
   571→      this.scene.setHoldOccupied(hold.id, anchored.has(hold.id));
   572→    }
   573→  }

   575→  // --- Pipeline por paso fijo ---
   576→
   577→  private fixedStep(dt: number): void {
   578→    const result = this.solver.step(dt);
   579→
   580→    // ERROR GUARD (9.2): carga no finita / NaN / energia anomala.
   581→    if ((result.nanError || result.energySpike) && !this.guardActive) {
   582→      this.guardActive = true;
   583→      this.guardCooldown = 0.4;
   584→      this.lastGuardError = result.nanError ? "nan" : "energia";
   585→      this.events.emit({ type: "Message", text: "ERROR GUARD: simulacion no finita; reiniciando prueba", tone: "error" });
   586→      this.machine.transition("ERROR");
   587→    }
   588→    if (this.machine.state === "ERROR") {
   589→      this.guardCooldown -= dt;
   590→      if (this.guardCooldown <= 0) {
   591→        this.spawn();
   592→        return;
   593→      }
   594→      return;
   595→    }
   596→
   597→    const anchors = this.solver.anchorHoldIds();
   598→
   599→    // Biomecanica solo cuando el desafio esta activo (fatiga congelada en pausa/resultado).
   600→    if (this.machine.state === "PLAYING") {
   601→      const bioResult = this.bio.update(dt, result, anchors.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver.tremor);
   602→      // En tutorial la fatiga y las zonas se miden (numeros reales en el paso
   603→      // de vitals) pero la sobrecarga NO libera agarres: el onboarding no puede
   604→      // guiar hacia una caida por mover lento.
   605→      for (const limb of this.tutorial ? [] : bioResult.overloadReleases) {
   606→        const anchor = this.solver.releaseAnchor(limb);
   607→        if (anchor) {
   608→          this.events.emit({ type: "GripReleased", limb, holdId: anchor.holdId, reason: "OVERLOAD" });
   609→          this.refreshHoldOccupancy();
   610→        }
   611→      }
   612→      for (const limb of bioResult.strainStarted) this.events.emit({ type: "StrainStarted", limb });
   613→      for (const limb of bioResult.overloadStarted) this.events.emit({ type: "OverloadStarted", limb });
   614→
   615→      // Asistencia de viaje hacia la presa aceptada (7.2).
   616→      this.interaction?.tick(dt);
   617→
   618→      const anchorsNow = this.solver.anchorHoldIds();
   619→      const stability = computeStability(this.bio, result, anchorsNow.map((a) => ({ limb: a.limb, holdId: a.holdId })), this.solver, this.profile);
   620→      this.lastSnapshot = {
   621→        stability,
   622→        com: result.com,
   623→        supportCount: anchorsNow.length
   624→      };
   625→
   626→      // Evaluacion del desafio (10.3): caida -> exito -> timeout.
   627→      // El timer no corre durante el tutorial: nadie deberia agotar el tiempo
   628→      // mientras aprende los controles.
   629→      if (!this.tutorial) this.gameTime += dt;
   630→      const outcome = this.evaluator!.tick({
   631→        dt,
   632→        gameTime: this.gameTime,
   633→        anchorsByLimb: Object.fromEntries(anchorsNow.map((a) => [a.limb, a.holdId])),
   634→        supportCount: anchorsNow.length,
   635→        stability,
   636→        pelvisY: this.solver.getParticle(P.pelvisC).pos.y,
   637→        outOfBounds: result.outOfBounds
   638→      });
   639→      if (outcome.kind === "fall") {
   640→        this.events.emit({ type: "FallStarted", reason: outcome.reason });
   641→        this.events.emit({ type: "ChallengeFailed", reason: "FALLEN" });
   642→        this.machine.transition("FALLEN");
   643→        this.beginFallSequence();
   644→      } else if (outcome.kind === "success") {
   645→        this.events.emit({ type: "ChallengeSuccess" });
   646→        this.machine.transition("SUCCESS");
   647→      } else if (outcome.kind === "timeout") {
   648→        this.events.emit({ type: "ChallengeFailed", reason: "TIMEOUT" });
   649→        this.machine.transition("TIMEOUT");
   650→      }
   651→    } else {
   652→      // Fuera de PLAYING la fatiga no cambia; sin tremor.
   653→      for (const limb of LIMB_IDS) this.solver.tremor[limb] = 0;
   654→      this.interaction?.tick(0);
   655→    }
   656→
   657→    // Secuencia de caida (ADR-008): en FALLEN los agarres se abren escalonados y el
   658→    // cuerpo cae al colchon; tras asentarse, reintento automatico (R lo adelanta).
   659→    if (this.machine.state === "FALLEN") {
   660→      this.fallTimer += dt;
   661→      while (this.fallReleaseQueue.length > 0 && this.fallTimer >= this.fallReleaseQueue[0].at) {
   662→        const next = this.fallReleaseQueue.shift()!;
   663→        const anchor = this.solver.releaseAnchor(next.limb);
   664→        if (anchor) {
   665→          this.events.emit({ type: "GripReleased", limb: next.limb, holdId: anchor.holdId, reason: "FALL" });
   666→          this.refreshHoldOccupancy();
   667→        }
   668→      }
   669→      // Impacto contra el colchon: polvo + sacudida (una sola vez por caida).
   670→      if (!this.fallLandedFx && this.fallTimer >= LANDING_FX_AT_S) {
   671→        this.fallLandedFx = true;
   672→        const pv = this.solver.getParticle(P.pelvisC);
   673→        this.scene.burstDust(new Vector3(pv.pos.x, 0.06, pv.pos.z), 42);
   674→        this.scene.shake(0.055);
   675→      }
   676→      if (this.fallAutoReset > 0) {
   677→        this.fallAutoReset -= dt;
   678→        if (this.fallAutoReset <= 0) {
   679→          this.fallAutoReset = -1;
   680→          this.spawn();
   681→        }
   682→      }
   683→    }
   684→  }
   685→
   686→  /**
   687→   * Arranque de la secuencia de caida: cancela interaccion pendiente, abre los
   688→   * agarres con retardos realistas (manos primero) y deja el cuerpo flacido.
   689→   */
   690→  private beginFallSequence(): void {
   691→    this.interaction?.cancelDrag();
   692→    this.interaction?.cancelAssistAll();
   693→    this.fallTimer = 0;
   694→    this.fallReleaseQueue = [];
   695→    this.fallLandedFx = false;
   696→    const hands = ["hand_l", "hand_r"] as const;
   697→    const feet = ["foot_l", "foot_r"] as const;
   698→    let n = 0;
   699→    for (const limb of hands) {
   700→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_HAND_S + n++ * FALL_RELEASE_STAGGER_S });
   701→    }
   702→    for (const limb of feet) {
   703→      if (this.solver.hasAnchor(limb)) this.fallReleaseQueue.push({ limb, at: FALL_RELEASE_FOOT_S + n++ * FALL_RELEASE_STAGGER_S });
   704→    }
   705→    this.fallReleaseQueue.sort((a, b) => a.at - b.at);
   706→    this.fallAutoReset = FALL_AUTO_RESET_S;
   707→    this.solver.setToneTarget(this.tune.toneFloor);
   708→  }
   709→
   710→  // --- Bucle de render ---
   711→
   712→  private readonly frame = (nowMs: number): void => {
   713→    if (this.disposed) return;
   714→    this.raf = requestAnimationFrame(this.frame);
   715→    this.frameCount++;
   716→    const now = nowMs / 1000;
   717→    const dtFrame = Math.min(Math.max(0, now - this.lastFrame), 0.1);
   718→    this.lastFrame = now;
   719→
   720→    this.fpsFrames++;
   721→    this.fpsElapsed += dtFrame;
   722→    if (this.fpsElapsed >= 0.5) {
   723→      this.fps = this.fpsFrames / this.fpsElapsed;
   724→      this.fpsFrames = 0;
   725→      this.fpsElapsed = 0;
   726→    }
   727→
   728→    const state = this.machine.state;
   729→    const stepping = state !== "PAUSED" && state !== "LOADING" && !this.paused;
   730→    if (stepping) {
   731→      // Slow-motion cinematografico: la escala converge suave hacia el objetivo.
   732→      if (this.slowmoLeft > 0) this.slowmoLeft -= dtFrame;
   733→      const goalScale = this.slowmoLeft > 0 ? SLOWMO_SCALE : 1;
   734→      this.timeScale += (goalScale - this.timeScale) * Math.min(1, dtFrame * 9);
   735→      this.accumulator += dtFrame * this.timeScale;
   736→      let steps = 0;
   737→      const fixed = this.profile.sim.fixedDeltaSeconds;
   738→      while (this.accumulator >= fixed && steps < MAX_STEPS_PER_FRAME) {
   739→        const t0 = performance.now();
   740→        this.fixedStep(fixed);
   741→        this.physicsMs = lerp(this.physicsMs, performance.now() - t0, 0.15);
   742→        this.accumulator -= fixed;
   743→        steps++;
   744→      }
   745→      if (steps >= MAX_STEPS_PER_FRAME) this.accumulator = 0;
   746→    }
   747→
   748→    if (state === "RESETTING") {
   749→      this.settleTimer -= dtFrame;
   750→      if (this.settleTimer <= 0) {
   751→        this.machine.transition(this.autoStartPending ? "PLAYING" : "READY");
   752→        if (this.autoStartPending) {
   753→          this.events.emit({ type: "Message", text: "Recupera dos apoyos antes de perder el agarre", tone: "warn" });
   754→        }
   755→      }
   756→    }
   757→
   758→    const alpha = Math.min(1, this.accumulator / this.profile.sim.fixedDeltaSeconds);
   759→    this.scene.syncRagdoll(this.solver, alpha);
   760→    this.scene.syncJointDebug(this.solver);
   761→
   762→    // Anillo de seleccion + enfasis de etiqueta: extremidad arrastrada (firm) o bajo
   763→    // el cursor (iman sutil). La etiqueta IZQ/DER crece: siempre sabes QUE mueves.
   764→    let emphasisLimb: LimbId | null = null;
   765→    if (this.interaction?.dragging) {
   766→      this.scene.setSelectionRing(this.interaction.selected, this.solver, "grab");
   767→      emphasisLimb = this.interaction.selected;
   768→    } else {
   769→      const hover = this.pointerInside && (state === "READY" || state === "PLAYING")
   770→        ? this.scene.pickLimb(this.pointerX, this.pointerY, this.solver)
   771→        : null;
   772→      this.scene.setSelectionRing(hover, this.solver, "hover");
   773→      emphasisLimb = hover;
   774→    }
   775→    this.scene.setLimbLabelEmphasis(emphasisLimb);
   776→
   777→    // Preview de arrastre y alcance.
   778→    const preview = this.interaction?.getPreview() ?? null;
   779→    this.scene.clearHoldHighlights();
   780→    if (preview?.snappedHoldId) {
   781→      if (preview.validity === "valid") this.scene.setHoldCompatible(preview.snappedHoldId, true);
   782→      else this.scene.setHoldInvalid(preview.snappedHoldId, true);
   783→    }
   784→    this.scene.showDragPreview(this.interaction?.dragging ? preview : null, this.solver);
   785→    if (this.interaction?.dragging && preview) {
   786→      this.scene.setReachRadius(
   787→        (preview.limb.startsWith("hand") ? this.rig!.armChain : this.rig!.legChain) * this.character.physical.reachScale
   788→      );
   789→      this.scene.showReachPreview(preview.limb, this.solver, preview.validity !== "invalid");
   790→    } else {
   791→      this.scene.showReachPreview(null, this.solver, true);
   792→    }
   793→
   794→    // Motivo de rechazo EN VIVO durante el arrastre: el jugador ve POR QUE una presa
   795→    // no es posible antes de soltar (control determinista: nada "al azar").
   796→    if (this.interaction?.dragging) {
   797→      if (preview?.snappedHoldId && preview.validity === "invalid") {
   798→        const reason = this.interaction.pendingRejectReason() ?? this.interaction.rejectReason(preview.limb, preview.snappedHoldId);
   799→        this.hud.setRejectHint(reason ? `✕ ${REJECT_LABELS[reason] ?? reason}` : "Movimiento no posible");
   800→      } else if (preview && !preview.snappedHoldId && !this.challenge.rules.allowFreeTarget) {
   801→        this.hud.setRejectHint("Suelta sobre una presa resaltada");
   802→      } else {
   803→        this.hud.setRejectHint(null);
   804→      }
   805→    } else {
   806→      this.hud.setRejectHint(null);
   807→    }
   808→
   809→    // Tutorial: resaltados y avance de pasos.
   810→    this.tickTutorial();
   811→
   812→    // Retroalimentacion de carga por extremidad (misma paleta que el panel
   813→    // lateral): tinte verde/ambar/rojo de la extremidad y anillo sobre su presa.
   814→    // La sobrecarga ademas lleva CUENTA ATRAS de fallo: urgencia 0..1 (danno
   815→    // acumulado/tolerancia) acelera parpadeo y latidos, y alimenta el chip central.
   816→    const cfgBio = this.profile.biomech;
   817→    this.limbOverload.clear();
   818→    this.overloadPressure = 0;
   819→    for (const limb of LIMB_IDS) {
   820→      const anchored = this.solver.hasAnchor(limb);
   821→      const info = this.bio.infoFor(limb);
   822→      const zone = info?.zone ?? "NORMAL";
   823→      let urgency = 0;
   824→      if (anchored && info && (zone === "OVERLOADED" || zone === "ERROR_GUARD")) {
   825→        const secondsLeft = Math.max(0, (cfgBio.overloadGraceSeconds - info.overloadTimer) / cfgBio.overloadAccumRate);
   826→        const frac = clamp01(1 - (secondsLeft * cfgBio.overloadAccumRate) / Math.max(1e-6, cfgBio.overloadGraceSeconds));
   827→        this.limbOverload.set(limb, { seconds: secondsLeft, frac });
   828→        urgency = frac;
   829→        this.overloadPressure = Math.max(this.overloadPressure, frac);
   830→      }
   831→      this.scene.updateLimbTint(limb, anchored, zone, urgency);
   832→      this.scene.setAnchorRing(limb, this.solver.getAnchor(limb)?.holdId ?? null, zone, urgency);
   833→    }
   834→
   835→    // Voz del escalador: respiracion y quejidos de fondo dirigidos por la fatiga
   836→    // y la carga reales (solo suenan con PLAYING activo; el director se resetea fuera).
   837→    const maxFatigue = Math.max(...LIMB_IDS.map((l) => this.bio.fatigue[l] ?? 0));
   838→    const maxRatio = Math.max(...LIMB_IDS.map((l) => this.bio.infoFor(l)?.ratio ?? 0));
   839→    if (this.machine.state === "PLAYING") this.peakFatigue = Math.max(this.peakFatigue, maxFatigue);
   840→    this.audio.update(dtFrame, {
   841→      playing: this.machine.state === "PLAYING",
   842→      maxFatigue,
   843→      maxRatio,
   844→      overloadRatio: this.profile.biomech.overloadRatio,
   845→      overload: LIMB_IDS.some((l) => this.bio.infoFor(l)?.zone === "OVERLOADED"),
   846→      overloadPressure: this.overloadPressure
   847→    });
   848→
   849→    // Combo de flujo: se rompe si pasa la ventana sin fijar otra presa.
   850→    if (this.flowCount > 0 && performance.now() / 1000 - this.lastGrabAt > FLOW_WINDOW_S) {
   851→      this.flowCount = 0;
   852→      this.hud.setFlow(0);
   853→    }
   854→
   855→    const pelvis = this.solver.getParticle(P.pelvisC);
   856→    this.scene.updateCamera(dtFrame, pelvis.pos, this.recenterOnce);
   857→    this.recenterOnce = false;
   858→
   859→    // HUD y debug a 10 Hz.
   860→    this.hudAccum += dtFrame;
   861→    if (this.hudAccum >= 0.1) {
   862→      this.hudAccum = 0;
   863→      this.updateHud(state);
   864→      this.debug.update(this.buildDebugData(state));
   865→    }
   866→
   867→    this.scene.render();
   868→  };
   869→
   870→  private updateHud(state: GameState): void {
   871→    const running = state === "PLAYING";
   872→    this.hud.setTimer(
   873→      this.evaluator ? Math.max(0, this.challenge.timeLimitSeconds - (running ? this.gameTime : 0)) : null,
   874→      running
   875→    );
   876→    let fatigueMax = 0;
   877→    for (const limb of LIMB_IDS) fatigueMax = Math.max(fatigueMax, this.bio.fatigue[limb]);
   878→    this.hud.setVitals(this.lastSnapshot.stability, fatigueMax);
   879→
   880→    const limbStates: Record<LimbId, { zone: LoadZone; anchored: boolean; label: string; overloadS?: number | null; overloadFrac?: number | null }> = {} as never;
   881→    for (const limb of LIMB_IDS) {
   882→      const anchored = this.solver.hasAnchor(limb);
   883→      const zone = this.bio.infoFor(limb)?.zone ?? "NORMAL";
   884→      const over = this.limbOverload.get(limb);
   885→      limbStates[limb] = {
   886→        zone,
   887→        anchored,
   888→        label: this.interaction?.isAssisting(limb) ? "MOVIENDO" : this.interaction?.dragging && this.interaction.selected === limb ? "OBJETIVO" : "LIBRE",
   889→        overloadS: over ? over.seconds : null,
   890→        overloadFrac: over ? over.frac : null
   891→      };
   892→    }
   893→    this.hud.setLimbStates(limbStates);
   894→    // Chip central: la extremidad mas comprometida (solo jugando y fuera del
   895→    // tutorial: ahi la sobrecarga nunca suelta, el aviso despistaria). Memoria
   896→    // de 0.6 s: los valles del pendulum no hacen parpadear el aviso.
   897→    let worst: { limb: LimbId; seconds: number; frac: number } | null = null;
   898→    for (const [limb, over] of this.limbOverload) {
   899→      if (!worst || over.seconds < worst.seconds) worst = { limb, seconds: over.seconds, frac: over.frac };
   900→    }
   901→    const nowS = performance.now() / 1000;
   902→    if (worst) {
   903→      this.lastWorst = { label: LIMB_LABEL_ES[worst.limb], seconds: worst.seconds, frac: worst.frac };
   904→      this.lastWorstAt = nowS;
   905→    }
   906→    const showWarning = state === "PLAYING" && !this.tutorial && this.lastWorst !== null && (worst !== null || nowS - this.lastWorstAt < 0.6);
   907→    this.hud.setOverloadWarning(showWarning && this.lastWorst ? this.lastWorst : null);
   908→    this.hud.setProgress(running && this.evaluator && this.evaluator.progress > 0 ? this.evaluator.progress : null);
   909→  }

   958→  private onGameEvent(e: GameEvent): void {
   959→    this.audio.play(e);
   960→    switch (e.type) {
   961→      case "StateChanged":
   962→        this.hud.setState(e.to);
   963→        break;
   964→      case "ChallengeSuccess": {
   965→        // Tanda de confeti sobre la presa cima del desafio.
   966→        const topHold = this.challenge.completion.holdIds?.[0] ?? "";
   967→        const topPos = topHold ? this.holdPosById.get(topHold) : undefined;
   968→        if (topPos) this.scene.burstConfetti(new Vector3(topPos[0], topPos[1] + 0.1, topPos[2] - 0.05));
   969→        // Puntuacion, estrellas y record persistente.
   970→        const result = computeScore({
   971→          timeS: this.gameTime,
   972→          timeLimitS: this.challenge.timeLimitSeconds,
   973→          falls: this.fallCount,
   974→          flowBonus: this.flowBonus
   975→        });
   976→        const rec: BestRecord = { score: result.total, timeS: this.gameTime, stars: result.stars };
   977→        const isRecord = this.saveBest(this.challenge.id, rec);
   978→        this.hud.showSuccess({
   979→          stars: result.stars,
   980→          total: result.total,
   981→          timeS: this.gameTime,
   982→          breakdown: result.breakdown,
   983→          isRecord,
   984→          best: this.loadBest(this.challenge.id)
   985→        });
   986→        if (isRecord) this.hud.setBest(this.challenge.id, rec);
   987→        break;
   988→      }
   989→      case "ChallengeFailed":
   990→        this.hud.showBanner(e.reason === "TIMEOUT" ? "TIEMPO AGOTADO" : "CAIDA", "bad");
   991→        break;
   992→      case "HoldAcquired": {
   993→        // Combo de flujo: encadenar fijaciones en menos de FLOW_WINDOW_S suma bonus.
   994→        const now = performance.now() / 1000;
   995→        this.flowCount = now - this.lastGrabAt <= FLOW_WINDOW_S ? Math.min(FLOW_MAX_CHAIN, this.flowCount + 1) : 1;
   996→        this.lastGrabAt = now;
   997→        if (this.flowCount >= 2) this.flowBonus += FLOW_BONUS_PER_STEP * this.flowCount;
   998→        this.hud.setFlow(this.flowCount);
   999→        // Bocada de tiza al fijar una mano.
  1000→        if (e.limb.startsWith("hand")) {
  1001→          const pos = this.holdPosById.get(e.holdId);
  1002→          if (pos) this.scene.burstChalk(new Vector3(pos[0], pos[1], pos[2] - 0.03));
  1003→        }
  1004→        break;
  1005→      }
  1006→      case "HoldRejected": {
  1007→        const label = REJECT_LABELS[e.reason] ?? e.reason;
  1008→        this.hud.toast(`Movimiento rechazado: ${label}`, "warn");
  1009→        break;
  1010→      }
  1011→      case "GripReleased":
  1012→        if (e.reason === "OVERLOAD") {
  1013→          this.hud.toast(`Sobrecarga: el agarre cedió (${e.limb})`, "error");
  1014→          // Sacudida corta + tiza al resbalar el agarre.
  1015→          this.scene.shake(0.02);
  1016→          const pos = this.holdPosById.get(e.holdId);
  1017→          if (pos) this.scene.burstChalk(new Vector3(pos[0], pos[1], pos[2] - 0.03), 14);
  1018→        }
  1019→        break;
  1020→      case "StrainStarted":
  1021→        this.hintOnce("strain", "Agarre en tension: la fatiga de esa extremidad sube mas rapido", "warn");
  1022→        break;
  1023→      case "OverloadStarted":
  1024→        this.hintOnce("overload", "SOBRECARGA: el chip rojo cuenta los segundos que aguanta el agarre (~4,5 s en sobrecarga continua). Alivia esa extremidad ya", "error");
  1025→        break;
  1026→      case "FallStarted":
  1027→        this.hintOnce("fall", "Caída: los agarres se abren y el cuerpo cae. Se reintenta solo (o pulsa R)", "info");
  1028→        // Drama: slow-motion breve + rotura del combo + conteo de caidas.
  1029→        this.slowmoLeft = SLOWMO_DURATION_S;
  1030→        this.fallCount += 1;
  1031→        this.flowCount = 0;
  1032→        this.hud.setFlow(0);
  1033→        break;
  1034→      default:
  1035→        break;
  1036→    }
  1037→  }

  1178→  /**
  1179→   * Teclas 1-4: agarrar/soltar la extremidad directamente (peticion del jugador:
  1180→   * "complicado seleccionar una extremidad cuando se mueve mucho"). Al agarrar, el
  1181→   * objetivo nace EN la extremidad (offset respecto al cursor): nada salta. Al
  1182→   * repetir la tecla, se suelta con la validacion normal (asistencia de viaje).
  1183→   */
  1184→  toggleKeyboardGrab(limb: LimbId): void {
  1185→    if (!(this.machine.state === "READY" || this.machine.state === "PLAYING")) return;
  1186→    if (this.interaction?.dragging) {
  1187→      // Solo suelta si es la misma extremidad la agarrada por teclado; un arrastre
  1188→      // de raton en curso no se interrumpe con las teclas.
  1189→      if (this.keyboardLimb === limb) this.endKeyboardDrag();
  1190→      return;
  1191→    }
  1192→    if (this.interaction!.isAssisting(limb)) return;
  1193→    this.interaction!.beginDrag(limb);
  1194→    if (this.machine.state === "READY") this.machine.transition("PLAYING");
  1195→    this.keyboardLimb = limb;
  1196→    // Offset inicial: objetivo = posicion actual de la extremidad (limbPos - cursor),
  1197→    // asi agarrar NO salta: la extremidad solo responde al MOVIMIENTO del cursor.
  1198→    const zPlane = this.wall.holds[0]?.position[2] ?? -0.18;
  1199→    const limbPos = this.solver.positionOfLimb(limb);
  1200→    const cursor = this.scene.wallPlanePoint(this.pointerX, this.pointerY, zPlane);
  1201→    this.grabOffset = cursor ? { x: limbPos.x - cursor.x, y: limbPos.y - cursor.y } : { x: 0, y: 0 };
  1202→    this.applyKeyboardDragTarget(zPlane);
  1203→    this.hud.toast(`${LIMB_LABEL_ES[limb]} agarrada (tecla ${GameApp.KEY_OF_LIMB[limb]}): muévela con el ratón y pulsa ${GameApp.KEY_OF_LIMB[limb]} para soltar`, "info");
